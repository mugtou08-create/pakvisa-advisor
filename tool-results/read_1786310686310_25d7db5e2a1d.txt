    47→function BudgetPieCustomTooltip({ active, payload }: BudgetPieTooltipProps) {
    48→  if (!active || !payload || !payload.length) return null;
    49→  const item = payload[0].payload;
    50→  const total = payload.reduce((sum, p) => sum + p.payload.value, 0);
    51→  return (
    52→    <div className="rounded-lg border bg-popover px-3 py-2 shadow-md text-xs">
    53→      <div className="flex items-center gap-2 mb-0.5">
    54→        <div className="w-2.5 h-2.5 rounded-full shrink-0" style={{ backgroundColor: item.color }} />
    55→        <span className="font-medium">{item.name}</span>
    56→      </div>
    57→      <div className="text-muted-foreground">
    58→        ${item.value.toLocaleString('en-US', { minimumFractionDigits: 0, maximumFractionDigits: 0 })}
    59→        <span className="ml-1 text-[10px]">({Math.round((item.value / total) * 100)}%)</span>
    60→      </div>
    61→    </div>
    62→  );
    63→}

    65→/** Mini radar chart for country cards (3-axis) */
    66→export function MiniRadarChart({ safety, cost, ease }: { safety: number; cost: number; ease: number }) {
    67→  const cx = 28, cy = 28, r = 22;
    68→  const axes = 3;
    69→  const angleStep = (2 * Math.PI) / axes;
    70→  const startAngle = -Math.PI / 2;
    71→  const s = safety / 10;
    72→  const c = Math.max(0.05, 1 - cost / 3000);
    73→  const e = ease;
    74→  const values = [s, c, e];
    75→  const points = values.map((val, i) => {
    76→    const angle = startAngle + i * angleStep;
    77→    return `${cx + val * r * Math.cos(angle)},${cy + val * r * Math.sin(angle)}`;
    78→  }).join(' ');
    79→  const gridLines = [0.33, 0.66, 1].map(level => {
    80→    const pts = Array.from({ length: axes }, (_, i) => {
    81→      const angle = startAngle + i * angleStep;
    82→      return `${cx + level * r * Math.cos(angle)},${cy + level * r * Math.sin(angle)}`;
    83→    }).join(' ');
    84→    return <polygon key={level} points={pts} fill="none" stroke="currentColor" strokeWidth="0.3" className="text-white/20" />;
    85→  });
    86→
    87→  return (
    88→    <svg width="56" height="56" viewBox="0 0 56 56" className="inline-block opacity-80">
    89→      {gridLines}
    90→      {Array.from({ length: axes }, (_, i) => {
    91→        const angle = startAngle + i * angleStep;
    92→        return <line key={i} x1={cx} y1={cy} x2={cx + r * Math.cos(angle)} y2={cy + r * Math.sin(angle)} stroke="currentColor" strokeWidth="0.3" className="text-white/20" />;
    93→      })}
    94→      <polygon points={points} fill="rgba(249,115,22,0.15)" stroke="#fbbf24" strokeWidth="1" />
    95→      {values.map((val, i) => {
    96→        const angle = startAngle + i * angleStep;
    97→        return <circle key={i} cx={cx + val * r * Math.cos(angle)} cy={cy + val * r * Math.sin(angle)} r="1.5" fill="#fbbf24" />;
    98→      })}
    99→    </svg>
   100→  );
   101→}

   118→export function InteractiveWorldMap({ countries, onSelectCountry }: { countries: CountryData[]; onSelectCountry: (c: CountryData) => void }) {
   119→  const [svgContent, setSvgContent] = useState<string | null>(null);
   120→  const [hoveredCountry, setHoveredCountry] = useState<string | null>(null);
   121→  const [tooltipPos, setTooltipPos] = useState<{ x: number; y: number; name: string; status: { fill: string; label: string } } | null>(null);
   122→  const containerRef = useRef<HTMLDivElement>(null);
   123→
   124→  // Build lookup: SVG country name -> CountryData
   125→  const countryLookup = useMemo(() => {
   126→    const map = new globalThis.Map<string, CountryData>();
   127→    countries.forEach(c => {
   128→      const normalized = c.name.replace(/\s+/g, '').toUpperCase();
   129→      map.set(normalized, c);
   130→      map.set(c.name.toUpperCase(), c);
   131→    });
   132→    Object.entries(COUNTRY_NAME_ALIASES).forEach(([svgName, aliases]) => {
   133→      aliases.forEach(alias => {
   134→        const found = countries.find(co => {
   135→          const coNorm = co.name.replace(/\s+/g, '').toUpperCase();
   136→          const aliasNorm = alias.replace(/\s+/g, '').toUpperCase();
   137→          return coNorm === aliasNorm;
   138→        });
   139→        if (found) map.set(svgName.toUpperCase(), found);
   140→      });
   141→    });
   142→    return map;
   143→  }, [countries]);
   144→
   145→  const getStatus = (countryName: string): { fill: string; label: string } => {
   146→    const c = countryLookup.get(countryName.toUpperCase())
   147→      || countryLookup.get(countryName.replace(/\s+/g, '').toUpperCase());
   148→    if (!c) return VISA_CATEGORY_COLORS.unknown;
   149→    if (c.visaFree) return VISA_CATEGORY_COLORS.visaFree;
   150→    if (c.visaOnArrival) return VISA_CATEGORY_COLORS.visaOnArrival;
   151→    if (c.etaAvailable) return VISA_CATEGORY_COLORS.eVisa;
   152→    return VISA_CATEGORY_COLORS.embassy;
   153→  };
   154→
   155→  // Load SVG and pre-process colors into the string itself
   156→  useEffect(() => {
   157→    fetch('/world-map.svg')
   158→      .then(res => res.text())
   159→      .then(text => {
   160→        let processed = text
   161→          .replace(/<defs>[\s\S]*?<\/defs>/, '')
   162→          .replace(/<rect[^>]*fill="[^"]*"[^>]*\/>/, '');
   163→
   164→        // Inject inline style into path elements with data-map-country
   165→        processed = processed.replace(
   166→          /<path\s+([^>]*?)data-map-country="([^"]+)"([^>]*)\s*\/>/g,
   167→          (_full, before, name, after) => {
   168→            const status = getStatus(name);
   169→            // Inject style with !important fill to override any CSS
   170→            return `<path ${before}data-map-country="${name}"${after} style="fill:${status.fill};stroke:#fff;stroke-width:0.8;cursor:pointer;transition:all 0.2s ease;opacity:0.85" />`;
   171→          }
   172→        );
   173→
   174→        // Style non-country paths (generic landmasses)
   175→        processed = processed.replace(
   176→          /<path\s+(?!([^>]*?)data-map-country)([^>]*?)\s*\/>/g,
   177→          (_full, _before, after) => {
   178→            // Only style paths that don't already have a style from above
   179→            if (_full.includes('style=')) return _full;
   180→            return `<path style="fill:#d4d4d8;stroke:#fff;stroke-width:0.5" ${after.replace(/<\/?path[^>]*>/g, '')} />`;
   181→          }
   182→        );
   183→
   184→        // Add ocean background rect as first child after <svg>
   185→        processed = processed.replace(
   186→          /<svg[^>]*>/,
   187→          (match) => `${match}<rect width="1000" height="500" fill="#e8f4f8" />`
   188→        );
   189→
   190→        setSvgContent(processed);
   191→      })
   192→      .catch(() => setSvgContent(''));
   193→  }, []);
   194→
   195→  const handleMouseMove = useCallback((e: React.MouseEvent) => {
   196→    if (!e.target || !(e.target instanceof SVGElement)) {
   197→      setTooltipPos(null);
   198→      return;
   199→    }
   200→    const path = (e.target as SVGElement).closest('path[data-map-country]') as SVGPathElement | null;
   201→    if (path) {
   202→      const name = path.getAttribute('data-map-country') || '';
   203→      const status = getStatus(name);
   204→      if (hoveredCountry !== name) {
   205→        setHoveredCountry(name);
   206→        const allPaths = containerRef.current?.querySelectorAll('path[data-map-country]');
   207→        allPaths?.forEach(p => {
   208→          if (p === path) {
   209→            p.style.opacity = '1';
   210→            p.style.strokeWidth = '1.5';
   211→          } else {
   212→            p.style.opacity = '0.5';
   213→            p.style.strokeWidth = '0.8';
   214→          }
   215→        });
   216→      }
   217→      const rect = containerRef.current?.getBoundingClientRect();
   218→      if (rect) {
   219→        setTooltipPos({
   220→          x: e.clientX - rect.left,
   221→          y: e.clientY - rect.top,
   222→          name,
   223→          status,
   224→        });
   225→      }
   226→    } else {
   227→      if (hoveredCountry) {
   228→        setHoveredCountry(null);
   229→        const allPaths = containerRef.current?.querySelectorAll('path[data-map-country]');
   230→        allPaths?.forEach(p => {
   231→          p.style.opacity = '0.85';
   232→          p.style.strokeWidth = '0.8';
   233→        });
   234→        setTooltipPos(null);
   235→      }
   236→    }
   237→  }, [hoveredCountry, getStatus]);
   238→
   239→  const handleClick = useCallback((e: React.MouseEvent) => {
   240→    if (!e.target || !(e.target instanceof SVGElement)) return;
   241→    const path = (e.target as SVGElement).closest('path[data-map-country]') as SVGPathElement | null;
   242→    if (!path) return;
   243→    const name = path.getAttribute('data-map-country') || '';
   244→    const c = countryLookup.get(name.toUpperCase())
   245→      || countryLookup.get(name.replace(/\s+/g, '').toUpperCase());
   246→    if (c) onSelectCountry(c);
   247→  }, [countryLookup, onSelectCountry]);
   248→
   249→  const handleMouseLeave = useCallback(() => {
   250→    setHoveredCountry(null);
   251→    setTooltipPos(null);
   252→    const allPaths = containerRef.current?.querySelectorAll('path[data-map-country]');
   253→    allPaths?.forEach(p => {
   254→      p.style.opacity = '0.85';
   255→      p.style.strokeWidth = '0.8';
   256→    });
   257→  }, []);
   258→
   259→  // Count stats for the legend
   260→  const legendStats = useMemo(() => {
   261→    let visaFree = 0, voa = 0, eVisa = 0, embassy = 0, noData = 0;
   262→    const svgCountries = svgContent?.match(/data-map-country="([^"]+)"/g) || [];
   263→    svgCountries.forEach(match => {
   264→      const name = match.match(/"([^"]+)"/)?.[1] || '';
   265→      const status = getStatus(name);
   266→      if (status.fill === VISA_CATEGORY_COLORS.visaFree.fill) visaFree++;
   267→      else if (status.fill === VISA_CATEGORY_COLORS.visaOnArrival.fill) voa++;
   268→      else if (status.fill === VISA_CATEGORY_COLORS.eVisa.fill) eVisa++;
   269→      else if (status.fill === VISA_CATEGORY_COLORS.embassy.fill) embassy++;
   270→      else noData++;
   271→    });
   272→    return { visaFree, voa, eVisa, embassy, noData };
   273→  }, [svgContent, getStatus]);
   274→
   275→  return (
   276→    <div className="bg-gradient-to-br from-slate-50 to-slate-100 dark:from-slate-900 dark:to-slate-800 rounded-2xl p-4 md:p-6 border border-border/50">
   277→      <div className="flex items-center justify-between mb-4">
   278→        <h3 className="text-sm font-semibold flex items-center gap-2">
   279→          <Globe className="w-4 h-4 text-amber-500" />
   280→          Visa Requirements World Map
   281→        </h3>
   282→        <span className="text-[10px] text-muted-foreground bg-muted px-2 py-0.5 rounded-full">Hover &amp; click countries</span>
   283→      </div>
   284→
   285→      <div
   286→        ref={containerRef}
   287→        className="relative w-full rounded-lg overflow-hidden bg-[#e8f4f8] dark:bg-slate-800"
   288→        style={{ paddingBottom: '50%' }}
   289→        onMouseMove={handleMouseMove}
   290→        onClick={handleClick}
   291→        onMouseLeave={handleMouseLeave}
   292→      >
   293→        {svgContent ? (
   294→          <svg
   295→            viewBox="0 0 1000 500"
   296→            className="absolute inset-0 w-full h-full"
   297→            preserveAspectRatio="xMidYMid meet"
   298→            dangerouslySetInnerHTML={{ __html: svgContent }}
   299→          />
   300→        ) : (
   301→          <div className="absolute inset-0 flex items-center justify-center">
   302→            <div className="flex flex-col items-center gap-3">
   303→              <div className="w-8 h-8 border-2 border-amber-500 border-t-transparent rounded-full animate-spin" />
   304→              <span className="text-xs text-muted-foreground">Loading world map...</span>
   305→            </div>
   306→          </div>
   307→        )}
   308→
   309→        {/* Floating tooltip */}
   310→        {tooltipPos && (
   311→          <div
   312→            className="absolute z-50 pointer-events-none px-3 py-2 bg-white dark:bg-slate-900 rounded-lg shadow-xl border border-border/50 text-xs whitespace-nowrap"
   313→            style={{
   314→              left: `${Math.min(tooltipPos.x + 12, 90)}%`,
   315→              top: `${Math.max(tooltipPos.y - 10, 2)}%`,
   316→            }}
   317→          >
   318→            <div className="font-semibold">{tooltipPos.name}</div>
   319→            <div className="flex items-center gap-1.5 mt-0.5">
   320→              <span className="inline-block w-2 h-2 rounded-full" style={{ backgroundColor: tooltipPos.status.fill }} />
   321→              <span style={{ color: tooltipPos.status.fill }}>{tooltipPos.status.label}</span>
   322→            </div>
   323→          </div>
   324→        )}
   325→      </div>
   326→
   327→      {/* Legend with counts */}
   328→      <div className="flex flex-wrap gap-x-4 gap-y-2 mt-4 justify-center">
   329→        {Object.entries(VISA_CATEGORY_COLORS).map(([key, item]) => {
   330→          const count = key === 'visaFree' ? legendStats.visaFree
   331→            : key === 'visaOnArrival' ? legendStats.voa
   332→            : key === 'eVisa' ? legendStats.eVisa
   333→            : key === 'embassy' ? legendStats.embassy
   334→            : legendStats.noData;
   335→          return (
   336→            <div key={key} className="flex items-center gap-1.5 text-xs text-muted-foreground">
   337→              <div className="w-3 h-3 rounded-sm shrink-0" style={{ backgroundColor: item.fill }} />
   338→              <span className="whitespace-nowrap">{item.label}</span>
   339→              <span className="text-[10px] bg-muted/50 px-1.5 py-0.5 rounded-full font-medium">{count}</span>
   340→            </div>
   341→          );
   342→        })}
   343→      </div>
   344→    </div>
   345→  );
   346→}

   393→export function BestMatchRecommendations({ countries, scoreResults, onSelectCountry }: { countries: CountryData[]; scoreResults: ScoreBreakdown[]; onSelectCountry: (c: CountryData) => void }) {
   394→  const [loaded, setLoaded] = useState(false);
   395→  const { userProfile } = useAppStore();
   396→
   397→  useEffect(() => {
   398→    const timer = setTimeout(() => setLoaded(true), 2000);
   399→    return () => clearTimeout(timer);
   400→  }, []);
   401→
   402→  const recommendations = useMemo(() => {
   403→    if (scoreResults.length > 0) {
   404→      return scoreResults
   405→        .sort((a, b) => b.finalScore - a.finalScore)
   406→        .slice(0, 3)
   407→        .map((s) => {
   408→          const c = countries.find((co) => co.code === s.countryCode);
   409→          if (!c) return null;
   410→          const reason = s.finalScore >= 70 ? 'Excellent match based on your profile' : s.finalScore >= 40 ? 'Good match with minor improvements needed' : 'Consider improving your profile first';
   411→          return { country: c, matchPct: Math.round(s.finalScore), reason };
   412→        })
   413→        .filter(Boolean) as { country: CountryData; matchPct: number; reason: string }[];
   414→    }
   415→    // Default recommendations
   416→    const defaults = [
   417→      { code: 'MY', reason: 'Visa-free entry for 30 days with beautiful destinations' },
   418→      { code: 'AE', reason: 'Visa on arrival with world-class tourism infrastructure' },
   419→      { code: 'TR', reason: 'Easy e-Visa process with rich cultural experiences' },
   420→    ];
   421→    return defaults.map((d) => {
   422→      const c = countries.find((co) => co.code === d.code);
   423→      return c ? { country: c, matchPct: c.visaFree ? 95 : c.visaOnArrival ? 88 : 75, reason: d.reason } : null;
   424→    }).filter(Boolean) as { country: CountryData; matchPct: number; reason: string }[];
   425→  }, [scoreResults, countries]);
   426→
   427→  if (!loaded) {
   428→    return (
   429→      <div>
   430→        <div className="scroll-section-title">
   431→          <h3 className="text-sm font-semibold flex items-center gap-2 whitespace-nowrap">
   432→            <Target className="w-4 h-4 text-amber-500" />
   433→            Recommended For You
   434→          </h3>
   435→        </div>
   436→        <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">

   506→// ============ ANIMATED COUNTER ============
   507→export function AnimatedCounter({ target, suffix = '', prefix = '' }: { target: number; suffix?: string; prefix?: string }) {
   508→  const [count, setCount] = useState(0);
   509→  const hasAnimated = useRef(false);
   510→
   511→  useEffect(() => {
   512→    if (hasAnimated.current) return;
   513→    hasAnimated.current = true;
   514→    const duration = 2000;
   515→    const start = performance.now();
   516→    const animate = (now: number) => {
   517→      const elapsed = now - start;
   518→      const progress = Math.min(elapsed / duration, 1);
   519→      const eased = 1 - Math.pow(1 - progress, 3);
   520→      setCount(Math.round(eased * target));
   521→      if (progress < 1) requestAnimationFrame(animate);
   522→    };
   523→    requestAnimationFrame(animate);
   524→  }, [target]);
   525→
   526→  return <span>{prefix}{count}{suffix}</span>;
   527→}

   605→    const formatter = new Intl.DateTimeFormat('en-US', { timeZone: timezone, hour: 'numeric', hour12: false });

   607→    if (hour >= 6 && hour < 18) return { label: 'Day', icon: <Sun className="w-3 h-3" />, color: 'text-amber-500' };
   608→    if (hour >= 18 && hour < 22) return { label: 'Evening', icon: <Sunset className="w-3 h-3" />, color: 'text-orange-500' };
   609→    return { label: 'Night', icon: <Moon className="w-3 h-3" />, color: 'text-slate-400' };
   610→  } catch {
   611→    return { label: 'Unknown', icon: <Clock className="w-3 h-3" />, color: 'text-muted-foreground' };

   616→export function useCountryTime(timezone: string) {
   617→  const [time, setTime] = useState('');
   618→  useEffect(() => {
   619→    const tick = () => {
   620→      try {
   621→        const now = new Date();
   622→        setTime(now.toLocaleTimeString('en-US', { timeZone: timezone, hour: 'numeric', minute: '2-digit', hour12: true }));
   623→      } catch { setTime(''); }
   624→    };
   625→    tick();
   626→    const id = setInterval(tick, 30000);

   628→  }, [timezone]);

   951→// ============ VISA REQUIREMENT COMPARISON TABLE (TASK 10 - B4) ============
   952→export function ComparisonTable({ results }: { results: ScoreBreakdown[] }) {
   953→  const [countryDetails, setCountryDetails] = useState<Record<string, CountryData>>({});
   954→  const [loading, setLoading] = useState(false);
   955→
   956→  useEffect(() => {
   957→    if (results.length === 0) return;
   958→    let cancelled = false;
   959→    setLoading(true);
   960→    void (async () => {
   961→      try {
   962→        const details = await Promise.all(results.map(r =>
   963→          fetch(`/api/countries/${r.countryCode}`).then(res => res.json()).then(data => data.data).catch(() => null)
   964→        ));
   965→        if (cancelled) return;
   966→        const map: Record<string, CountryData> = {};
   967→        details.forEach((d: CountryData | null, i: number) => {
   968→          if (d) map[results[i].countryCode] = d;
   969→        });
   970→        setCountryDetails(map);
   971→      } finally {
   972→        if (!cancelled) setLoading(false);
   973→      }
   974→    })();
   975→    return () => { cancelled = true; };
   976→  }, [results]);
   977→
   978→  if (results.length < 2 || loading) return null;
   979→
   980→  const rows = [
   981→    { label: 'Visa Type', key: 'visaType' },
   982→    { label: 'Eligibility', key: 'eligibility', isScore: true },
   983→    { label: 'Visa Likelihood', key: 'visaLikelihood', isScore: true },
   984→    { label: 'Cost Suitability', key: 'costSuitability', isScore: true },
   985→    { label: 'Final Score', key: 'finalScore', isScore: true, bold: true },
   986→    { label: 'Confidence', key: 'confidence', isPercent: true },
   987→    { label: 'Processing Time', key: 'processing', isText: true },
   988→    { label: 'Visa Fee', key: 'fee', isText: true },
   989→    { label: 'Monthly Cost', key: 'monthly', isText: true },
   990→    { label: 'Safety Rating', key: 'safety', isText: true },
   991→  ];
   992→
   993→  const getCellValue = (row: typeof rows[0], result: ScoreBreakdown) => {
   994→    if (row.isScore) {
   995→      const val = result[row.key as keyof ScoreBreakdown] as number;
   996→      return { text: `${Math.round(val)}%`, value: val };
   997→    }
   998→    if (row.isPercent) {
   999→      return { text: `${Math.round(result.confidence * 100)}%`, value: result.confidence };
  1000→    }
  1001→    if (row.isText) {
  1002→      const detail = countryDetails[result.countryCode];
  1003→      switch (row.key) {
  1004→        case 'processing': return { text: `${detail?.processingDaysMin || '?'}-${detail?.processingDaysMax || '?'} days`, value: 0 };
  1005→        case 'fee': return { text: `$${detail?.costProfile?.visaFeeUSD || 0}`, value: -(detail?.costProfile?.visaFeeUSD || 0) };
  1006→        case 'monthly': return { text: `$${detail?.costProfile?.totalMonthlyUSD || 0}/mo`, value: -(detail?.costProfile?.totalMonthlyUSD || 0) };
  1007→        case 'safety': return { text: `${detail?.safetyRating || '?'}/10`, value: detail?.safetyRating || 0 };
  1008→        default: return { text: 'N/A', value: 0 };
  1009→      }
  1010→    }
  1011→    return { text: String(result[row.key as keyof ScoreBreakdown] || 'N/A'), value: 0 };
  1012→  };
  1013→
  1014→  const getBestIdx = (row: typeof rows[0]) => {
  1015→    if (row.key === 'visaType') return -1;
  1016→    const values = results.map(r => getCellValue(row, r));
  1017→    // For scores and safety: higher is better; for costs/fees: lower absolute value is best
  1018→    if (row.isScore || row.key === 'confidence' || row.key === 'safety') {
  1019→      return values.reduce((best, v, i) => v.value > values[best].value ? i : best, 0);
  1020→    }
  1021→    return values.reduce((best, v, i) => Math.abs(v.value) < Math.abs(values[best].value) ? i : best, 0);
  1022→  };
  1023→
  1024→  return (
  1025→    <Card className="premium-card overflow-x-auto">
  1026→      <CardHeader>
  1027→        <CardTitle className="text-base flex items-center gap-2">
  1028→          <Gavel className="w-5 h-5 text-amber-500" />
  1029→          Detailed Comparison Table
  1030→        </CardTitle>
  1031→        <CardDescription>Side-by-side analysis with best values highlighted</CardDescription>
  1032→      </CardHeader>
  1033→      <CardContent className="overflow-x-auto">
  1034→        <Table>
  1035→          <TableHeader>
  1036→            <TableRow>
  1037→              <TableHead className="w-36">Metric</TableHead>
  1038→              {results.map(r => (
  1039→                <TableHead key={r.countryCode} className="text-center">
  1040→                  <div className="flex flex-col items-center gap-0.5">
  1041→                    <span className="text-sm font-semibold">{r.country}</span>
  1042→                    <Badge variant="outline" className="text-[10px]">{r.countryCode}</Badge>
  1043→                  </div>
  1044→                </TableHead>
  1045→              ))}
  1046→            </TableRow>
  1047→          </TableHeader>
  1048→          <TableBody>
  1049→            {rows.map(row => {
  1050→              const bestIdx = getBestIdx(row);
  1051→              return (
  1052→                <TableRow key={row.key}>
  1053→                  <TableCell className={`font-medium text-sm ${row.bold ? 'font-bold' : ''}`}>{row.label}</TableCell>
  1054→                  {results.map((r, i) => {
  1055→                    const cell = getCellValue(row, r);
  1056→                    const isBest = i === bestIdx;
  1057→                    return (
  1058→                      <TableCell key={r.countryCode} className={`text-center text-sm ${isBest ? 'print-best bg-amber-50 dark:bg-amber-900/20 font-semibold text-amber-700 dark:text-amber-400' : ''}`}>
  1059→                        {cell.text}
  1060→                        {isBest && <span className="ml-1">✓</span>}
  1061→                      </TableCell>
  1062→                    );
  1063→                  })}
  1064→                </TableRow>
  1065→              );
  1066→            })}
  1067→          </TableBody>
  1068→        </Table>
  1069→      </CardContent>
  1070→    </Card>
  1071→  );
  1072→}
