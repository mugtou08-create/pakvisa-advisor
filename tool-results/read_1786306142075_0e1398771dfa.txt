   659→        <p className="text-xs text-amber-600 flex items-center gap-1">
   660→          <CheckCircle2 className="w-3 h-3" /> Profile is complete! Ready for scoring.

   745→// ============ TYPING TEXT ANIMATION ============
   746→export function TypingText({ phrases, className = '' }: { phrases: string[]; className?: string }) {
   747→  const [current, setCurrent] = useState(0);
   748→  const [text, setText] = useState('');
   749→  const [isDeleting, setIsDeleting] = useState(false);
   750→
   751→  useEffect(() => {
   752→    const phrase = phrases[current];
   753→    const timeout = setTimeout(() => {
   754→      if (!isDeleting) {
   755→        setText(phrase.substring(0, text.length + 1));
   756→        if (text.length === phrase.length) {
   757→          setTimeout(() => setIsDeleting(true), 2200);
   758→        }
   759→      } else {
   760→        setText(phrase.substring(0, text.length - 1));
   761→        if (text.length === 0) {
   762→          setIsDeleting(false);
   763→          setCurrent((current + 1) % phrases.length);
   764→        }
   765→      }
   766→    }, isDeleting ? 25 : 55);
   767→    return () => clearTimeout(timeout);
   768→  }, [text, isDeleting, current, phrases]);
   769→
   770→  return (
   771→    <span className={className}>
   772→      {text}
   773→      <span className="inline-block w-0 h-[1.1em] ml-0.5 align-middle typing-text-cursor" />
   774→    </span>
   775→  );
   776→}

   828→// ============ MINI RADAR CHART (3-axis for country cards) ============
   829→export function MiniRadarChart({ safety, cost, ease }: { safety: number; cost: number; ease: number }) {
   830→  const cx = 28, cy = 28, r = 22;
   831→  const axes = 3;
   832→  const angleStep = (2 * Math.PI) / axes;
   833→  const startAngle = -Math.PI / 2;
   834→  const s = safety / 10;
   835→  const c = Math.max(0.05, 1 - cost / 3000);
   836→  const e = ease;
   837→  const values = [s, c, e];
   838→  const points = values.map((val, i) => {
   839→    const angle = startAngle + i * angleStep;
   840→    return `${cx + val * r * Math.cos(angle)},${cy + val * r * Math.sin(angle)}`;
   841→  }).join(' ');
   842→  const gridLines = [0.33, 0.66, 1].map(level => {
   843→    const pts = Array.from({ length: axes }, (_, i) => {
   844→      const angle = startAngle + i * angleStep;
   845→      return `${cx + level * r * Math.cos(angle)},${cy + level * r * Math.sin(angle)}`;
   846→    }).join(' ');
   847→    return <polygon key={level} points={pts} fill="none" stroke="currentColor" strokeWidth="0.3" className="text-white/20" />;
   848→  });
   849→
   850→  return (
   851→    <svg width="56" height="56" viewBox="0 0 56 56" className="inline-block opacity-80">
   852→      {gridLines}
   853→      {Array.from({ length: axes }, (_, i) => {
   854→        const angle = startAngle + i * angleStep;
   855→        return <line key={i} x1={cx} y1={cy} x2={cx + r * Math.cos(angle)} y2={cy + r * Math.sin(angle)} stroke="currentColor" strokeWidth="0.3" className="text-white/20" />;
   856→      })}
   857→      <polygon points={points} fill="rgba(249,115,22,0.15)" stroke="#fbbf24" strokeWidth="1" />
   858→      {values.map((val, i) => {
   859→        const angle = startAngle + i * angleStep;
   860→        return <circle key={i} cx={cx + val * r * Math.cos(angle)} cy={cy + val * r * Math.sin(angle)} r="1.5" fill="#fbbf24" />;
   861→      })}
   862→    </svg>
   863→  );
   864→}

   866→// ============ RADIAL GAUGE (SEMICIRCLE) ============
   867→export function RadialGauge({ score, size = 140 }: { score: number; size?: number }) {
   868→  const strokeWidth = 12;
   869→  const r = (size - strokeWidth) / 2;
   870→  const cx = size / 2;
   871→  const cy = size - 15;
   872→  const startX = cx - r;
   873→  const endX = cx + r;
   874→  const arcLength = Math.PI * r;
   875→  const filledLength = (score / 100) * arcLength;
   876→  const color = score >= 70 ? '#f59e0b' : score >= 40 ? '#f59e0b' : '#ef4444';
   877→
   878→  return (
   879→    <div className="flex flex-col items-center">
   880→      <svg width={size} height={size / 2 + 25} viewBox={`0 0 ${size} ${size / 2 + 25}`}>
   881→        <path d={`M ${startX} ${cy} A ${r} ${r} 0 0 1 ${endX} ${cy}`} fill="none" stroke="currentColor" strokeWidth={strokeWidth} className="text-muted/20" strokeLinecap="round" />
   882→        <path
   883→          d={`M ${startX} ${cy} A ${r} ${r} 0 0 1 ${endX} ${cy}`}
   884→          fill="none" stroke={color} strokeWidth={strokeWidth} strokeLinecap="round"
   885→          strokeDasharray={`${arcLength}`}
   886→          strokeDashoffset={arcLength - filledLength}
   887→          className="transition-all duration-1000 ease-out"
   888→        />
   889→        <text x={cx} y={cy - 12} textAnchor="middle" dominantBaseline="middle" style={{ fill: color, fontSize: '28px', fontWeight: 'bold' }}>{Math.round(score)}</text>

   896→// ============ ANIMATED SCORE NUMBER ============
   897→export function AnimatedScoreNumber({ value, delay = 0 }: { value: number; delay?: number }) {
   898→  const [displayed, setDisplayed] = useState(0);
   899→  const hasAnimated = useRef(false);
   900→
   901→  useEffect(() => {
   902→    if (hasAnimated.current) return;
   903→    hasAnimated.current = true;
   904→    const start = performance.now() + delay;
   905→    const duration = 1200;
   906→    const animate = (now: number) => {
   907→      const elapsed = now - start;
   908→      if (elapsed < 0) { requestAnimationFrame(animate); return; }
   909→      const progress = Math.min(elapsed / duration, 1);
   910→      const eased = 1 - Math.pow(1 - progress, 3);
   911→      setDisplayed(Math.round(eased * value));
   912→      if (progress < 1) requestAnimationFrame(animate);
   913→    };
   914→    requestAnimationFrame(animate);
   915→  }, [value, delay]);
   916→
   917→  return <span>{displayed}</span>;
   918→}

  1341→// ============ ANIMATED SCORE RING ============
  1342→export function AnimatedScoreRing({ score, size = 80, label }: { score: number; size?: number; label?: string }) {
  1343→  const radius = (size - 8) / 2;
  1344→  const circumference = 2 * Math.PI * radius;
  1345→  const offset = circumference - (score / 100) * circumference;
  1346→  const color = score >= 70 ? '#f59e0b' : score >= 40 ? '#f59e0b' : '#ef4444';
  1347→  const [animatedOffset, setAnimatedOffset] = useState(circumference);
  1348→  const isHighScore = score > 80;
  1349→
  1350→  useEffect(() => {
  1351→    const timer = setTimeout(() => setAnimatedOffset(offset), 100);
  1352→    return () => clearTimeout(timer);
  1353→  }, [offset, circumference]);
  1354→
  1355→  return (
  1356→    <div className="flex flex-col items-center gap-1 relative">
  1357→      <div className={isHighScore ? 'score-pulse rounded-full' : ''}>

  1377→// ============ MICRO SPARKLINE ============
  1378→export function MicroSparkline({ data, color, width = 60, height = 20 }: { data: number[]; color?: string; width?: number; height?: number }) {
  1379→  if (!data || data.length === 0) return null;
  1380→  const maxVal = Math.max(...data);
  1381→  const minVal = Math.min(...data);
  1382→  const range = maxVal - minVal || 1;
  1383→  const points = data.map((v, i) => {
  1384→    const x = (i / (data.length - 1)) * width;
  1385→    const y = height - ((v - minVal) / range) * (height - 4) - 2;
  1386→    return `${x},${y}`;
  1387→  }).join(' ');
  1388→  const lineColor = color || (data[data.length - 1] >= data[0] ? '#f59e0b' : '#ef4444');
  1389→  return (
  1390→    <svg width={width} height={height} className="inline-block" aria-hidden="true">
  1391→      <polyline points={points} fill="none" stroke={lineColor} strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
  1392→    </svg>
  1393→  );
  1394→}
