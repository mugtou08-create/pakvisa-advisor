     1→     1→'use client';
     2→     2→

    41→    41→  const [token, setToken] = useState<string | null>(null);

   149→   149→      toast.error('Connection error');

'bg-amber-100 dark:bg-amber-900/30 text-amber-700 dark:text-amber-400 font-medium' : 'hover:bg-muted text-muted-foreground hover:text-foreground'}`}

   317→   317→                    Feature Toggles

   322→   322→                    <CardHeader className="pb-3">
