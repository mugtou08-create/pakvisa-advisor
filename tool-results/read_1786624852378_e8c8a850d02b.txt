    93→function generatePageNumbers(current: number, total: number): (number | 'ellipsis')[] {
    94→  if (total <= 7) {
    95→    return Array.from({ length: total }, (_, i) => i + 1);
    96→  }
    97→  const pages: (number | 'ellipsis')[] = [1];
    98→  if (current > 3) pages.push('ellipsis');
    99→  const start = Math.max(2, current - 1);
   100→  const end = Math.min(total - 1, current + 1);
   101→  for (let i = start; i <= end; i++) pages.push(i);
   102→  if (current < total - 2) pages.push('ellipsis');
   103→  if (total > 1) pages.push(total);
   104→  return pages;
   105→}

   110→function getVisaType(c: CountryData) {

   114→  return { label: 'Embassy Required', color: 'bg-gray-100 text-gray-600 dark:bg-gray-800 dark:text-gray-400', dot: 'bg-gray-400' };
