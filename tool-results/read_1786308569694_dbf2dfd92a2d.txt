   175→// ============ SCORE HISTORY CHART (SVG) ============
   176→export function ScoreHistoryChart() {
   177→  // eslint-disable-next-line react-hooks/preserve-manual-memoization
   178→  const history = useMemo(() => {
   179→    try {
   180→      const raw = localStorage.getItem(SCORING_HISTORY_KEY);
   181→      if (raw) {
   182→        const entries = JSON.parse(raw);
   183→        return entries.slice(-8).map((e: any) => ({
   184→          date: new Date(e.date).toLocaleDateString('en-US', { month: 'short', day: 'numeric' }),
   185→          topScore: e.topScore || (e.scores && e.scores[0] ? e.scores[0].finalScore : 0),
   186→          countryCount: e.countryCount || (e.scores ? e.scores.length : 0),
   187→        }));
   188→      }
   189→    } catch {}
   190→    return [];
   191→  }, []);
   192→  
   193→  if (history.length < 2) return null;
   194→  
   195→  const chartW = 400, chartH = 120, pad = 30;
   196→  const maxScore = Math.max(...history.map(e => e.topScore), 1);
   197→  const pts = history.map((e, i) => ({
   198→    x: pad + (i / (history.length - 1)) * (chartW - 2 * pad),
   199→    y: chartH - pad - (e.topScore / maxScore) * (chartH - 2 * pad),
   200→  }));
   201→  
   202→  return (
   203→    <Card className="p-4">
   204→      <h4 className="text-sm font-semibold mb-3 flex items-center gap-1.5">
   205→        <TrendingUp className="w-4 h-4 text-amber-500" />
   206→        Score Trend
   207→      </h4>
   208→      <svg viewBox={`0 0 ${chartW} ${chartH}`} className="w-full">
   209→        {/* Grid lines */}
   210→        {[0, 0.5, 1].map(v => (
   211→          <line key={v} x1={pad} y1={chartH - pad - v * (chartH - 2 * pad)} x2={chartW - pad} y2={chartH - pad - v * (chartH - 2 * pad)} stroke="var(--border)" strokeWidth="0.5" strokeDasharray="4 4" />
   212→        ))}
   213→        {/* Y-axis labels */}
   214→        <text x={pad - 5} y={chartH - pad + 4} textAnchor="end" fill="var(--muted-foreground)" fontSize="9">0</text>
   215→        <text x={pad - 5} y={pad + 4} textAnchor="end" fill="var(--muted-foreground)" fontSize="9">{maxScore}</text>
   216→        {/* Area fill */}
   217→        <path d={`M ${pts[0].x} ${chartH - pad} L ${pts.map(p => `${p.x} ${p.y}`).join(' L')} L ${pts[pts.length - 1].x} ${chartH - pad} Z`} fill="url(#chartGradient)" opacity="0.3" />
   218→        {/* Line */}
   219→        <polyline points={pts.map(p => `${p.x},${p.y}`).join(' ')} fill="none" stroke="#f59e0b" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" />
   220→        {/* Data points */}
   221→        {pts.map((p, i) => (
   222→          <g key={i}>
   223→            <circle cx={p.x} cy={p.y} r="4" fill="white" stroke="#f59e0b" strokeWidth="2" />
   224→            <text x={p.x} y={chartH - pad + 14} textAnchor="middle" fill="var(--muted-foreground)" fontSize="8">{history[i].date}</text>
   225→          </g>
   226→        ))}
   227→        <defs>
   228→          <linearGradient id="chartGradient" x1="0" y1="0" x2="0" y2="1">
   229→            <stop offset="0%" stopColor="#f59e0b" />
   230→            <stop offset="100%" stopColor="#f59e0b" stopOpacity="0" />
   231→          </linearGradient>
   232→        </defs>
   233→      </svg>
   234→    </Card>
   235→  );
   236→}

   238→// ============ PASSPORT POWER INDEX WIDGET ============
   239→export function PassportPowerIndex({ countries, stats }: { countries: CountryData[]; stats: { totalCountries: number; visaFreeCount: number; visaOnArrivalCount: number; eVisaCount: number; embassyRequiredCount: number; avgCostUSD?: number; cheapestCountry?: { name: string; code: string; flagEmoji: string; visaFeeUSD: number }; fastestProcessing?: { name: string; code: string; flagEmoji: string; processingDaysMin: number } } | null }) {
   240→  // Exclusive classification: visaFree > voa > evisa > embassy (same logic as API stats)
   241→  const visaFreeCount = stats?.visaFreeCount || countries.filter(c => c.visaFree).length;
   242→  const voaCount = stats?.visaOnArrivalCount || countries.filter(c => !c.visaFree && c.visaOnArrival).length;
   243→  const eVisaCount = stats?.eVisaCount || countries.filter(c => !c.visaFree && !c.visaOnArrival && c.etaAvailable).length;
   244→  const embassyCount = stats?.embassyRequiredCount || countries.filter(c => !c.visaFree && !c.visaOnArrival && !c.etaAvailable).length;
   245→  const total = 199;
   246→  const rank = 106;
   247→  const accessible = visaFreeCount + voaCount + eVisaCount;
   248→  const powerScore = Math.round((accessible / total) * 100);
   249→  const segments = [
   250→    { label: 'Visa Free', count: visaFreeCount, color: '#f59e0b' },
   251→    { label: 'Visa on Arrival', count: voaCount, color: '#f59e0b' },
   252→    { label: 'e-Visa', count: eVisaCount, color: '#f97316' },
   253→    { label: 'Embassy Required', count: embassyCount, color: '#ef4444' },
   254→  ];
   255→
   256→  // Circular gauge animation state
   257→  const [animatedScore, setAnimatedScore] = useState(0);
   258→  const [animatedRank, setAnimatedRank] = useState(0);
   259→  const [animatedAccessible, setAnimatedAccessible] = useState(0);
   260→  const [showCalcInfo, setShowCalcInfo] = useState(false);
   261→  const gaugeRef = useRef<HTMLDivElement>(null);
   262→  const hasAnimated = useRef(false);
   263→
   264→  useEffect(() => {
   265→    const el = gaugeRef.current;
   266→    if (!el) return;
   267→    const observer = new IntersectionObserver(
   268→      ([entry]) => {
   269→        if (entry.isIntersecting && !hasAnimated.current) {
   270→          hasAnimated.current = true;
   271→          const duration = 1500;
   272→          const start = performance.now();
   273→          const animate = (currentTime: number) => {
   274→            const elapsed = currentTime - start;
   275→            const progress = Math.min(elapsed / duration, 1);
   276→            const eased = 1 - Math.pow(1 - progress, 3);
   277→            setAnimatedScore(Math.round(eased * powerScore));
   278→            setAnimatedRank(Math.round(eased * rank));
   279→            setAnimatedAccessible(Math.round(eased * accessible));
   280→            if (progress < 1) requestAnimationFrame(animate);
   281→          };
   282→          requestAnimationFrame(animate);
   283→          observer.disconnect();
   284→        }
   285→      },
   286→      { threshold: 0.2 }
   287→    );
   288→    observer.observe(el);
   289→    return () => observer.disconnect();
   290→  }, [powerScore, rank, accessible]);
   291→
   292→  // SVG circular gauge params
   293→  const radius = 54;
   294→  const strokeWidth = 8;
   295→  const circumference = 2 * Math.PI * radius;
   296→  const gaugeProgress = (animatedScore / 100) * circumference;

   311→    { flag: '\u{1F1E7}\u{1F1E9}', name: 'Bangladesh', rank: 101 },
   312→    { flag: '\u{1F1E6}\u{1F1EB}', name: 'Afghanistan', rank: 111 },
   313→  ];
   314→
   315→  return (
   316→    <div className="glass-section p-5" ref={gaugeRef}>
   317→      <div className="flex items-center justify-between mb-4">
   318→        <h3 className="text-sm font-semibold flex items-center gap-2">
   319→          <Globe className="w-4 h-4 text-amber-500" />
   320→          Passport Power Index
   321→        </h3>
   322→        <Badge variant="secondary" className="text-[10px] bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-400">
   323→          {animatedAccessible} accessible

   327→      <div className="flex flex-col sm:flex-row items-center gap-6 mb-4">

   329→        <div className="relative shrink-0">
   330→          <svg width={140} height={140} className="-rotate-90">
   331→            {/* Background track */}
   332→            <circle
   333→              cx={70} cy={70} r={radius}

   353→            {/* Animated progress arc */}
   354→            <circle
   355→              cx={70} cy={70} r={radius}
   356→              fill="none"
   357→              stroke={gaugeColor}
   358→              strokeWidth={strokeWidth}
   359→              strokeLinecap="round"

   779→export function TravelWeatherWidget({ country }: { country: CountryData }) {
   780→  const [useFahrenheit, setUseFahrenheit] = useState(false);
   781→  const currentMonthIndex = new Date().getMonth();
   782→
   783→  let temps: Record<string, number> = {};
   784→  try {
   785→    temps = typeof country.monthlyTemps === 'string'
   786→      ? JSON.parse(country.monthlyTemps)
   787→      : (country.monthlyTemps || {});
   788→  } catch { temps = {}; }
   789→
   790→  const bestMonthsLower = country.bestTravelMonths?.toLowerCase() || '';
   791→
   792→  const convertTemp = (c: number) => useFahrenheit ? Math.round(c * 9 / 5 + 32) : c;
   793→  const unit = useFahrenheit ? '°F' : '°C';
   794→
   795→  // Find global min/max across all months for color scaling
   796→  const allTemps = MONTH_NAMES.map(m => temps[m] || 0).filter(t => t > 0);
   797→  const globalMin = allTemps.length > 0 ? Math.min(...allTemps) : 0;
   798→  const globalMax = allTemps.length > 0 ? Math.max(...allTemps) : 40;
   799→  const tempRange = globalMax - globalMin || 1;
   800→
   801→  // Color: green for cold, amber for warm, orange-red for hot
   802→  const getTempColor = (c: number) => {
   803→    const ratio = Math.min(1, Math.max(0, (c - globalMin) / tempRange));
   804→    if (ratio < 0.33) {
   805→      // Green range
   806→      const t = ratio / 0.33;
   807→      return `hsl(${120 - t * 30}, 70%, ${40 + t * 10}%)`;
   808→    } else if (ratio < 0.66) {
   809→      // Amber range
