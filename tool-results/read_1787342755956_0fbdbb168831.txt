   118→function getVisaLabel(country: { visaFree: boolean; visaOnArrival: boolean; etaAvailable: boolean }): string {
   119→  if (country.visaFree) return 'Visa Free';
   120→  if (country.visaOnArrival) return 'Visa on Arrival';
   121→  if (country.etaAvailable) return 'e-Visa';
   122→  return 'Embassy Required';

   143→function safetyStars(rating: number): string {
   144→  const clamped = Math.min(Math.max(rating, 0), 5);
   145→  return '\u2605'.repeat(clamped) + '\u2606'.repeat(5 - clamped);

   209→  });

   211→  if (!country) return { title: 'Country Not Found | PakVisa Advisor' };
   212→
   213→  const visaLabel = getVisaLabel(country);

   809→function generateFAQs(
   810→  country: {
   811→    name: string; code: string; visaFree: boolean; visaOnArrival: boolean; etaAvailable: boolean;
   812→    processingDaysMin: number; processingDaysMax: number; safetyRating: number; safetySummary: string;
   813→    bestTravelMonths: string; avgTempC: string; currency: string; currencyCode: string;
   814→  },
   815→  costProfile: CostProfileData | undefined,
   816→  visaLabel: string,
   817→  isEmbassyRequired: boolean,
   818→): { q: string; a: string }[] {
   819→  const faqs: { q: string; a: string }[] = [];
   820→
   821→  // Q1: Do Pakistani citizens need a visa?
   822→  faqs.push({
   823→    q: `Do Pakistani citizens need a visa for ${country.name}?`,
   824→    a: country.visaFree
   825→      ? `No, Pakistani citizens do not need a visa to visit ${country.name}. You can travel visa-free with a valid Pakistani passport. Make sure your passport has at least 6 months of validity remaining.`
   826→      : country.visaOnArrival
   827→        ? `Pakistani citizens can get a visa on arrival at ${country.name}. Simply present your valid Pakistani passport at the immigration counter upon arrival. Processing is usually instant.`
   828→        : country.etaAvailable
   829→          ? `Yes, Pakistani citizens need an e-Visa to visit ${country.name}. The e-Visa can be applied for online through the official government portal. Processing typically takes ${country.processingDaysMin}-${country.processingDaysMax} business days.`
   830→          : `Yes, Pakistani citizens need a visa to visit ${country.name}. You must apply at the embassy or consulate. The application process typically takes ${country.processingDaysMin}-${country.processingDaysMax} business days.`,
   831→  });
   832→
   833→  // Q2: What is the visa fee?
   834→  const feeText = costProfile && costProfile.visaFeeUSD > 0
   835→    ? `The visa fee for ${country.name} is approximately $${costProfile.visaFeeUSD} USD (\u2248 PKR ${usdToPkr(costProfile.visaFeeUSD)}). Additional service fees of $${costProfile.serviceFeeUSD} may apply.`
   836→    : `There is no visa fee for Pakistani citizens visiting ${country.name} as it is ${visaLabel.toLowerCase()}.`
   837→  ;
   838→  faqs.push({
   839→    q: `What is the visa fee for ${country.name} for Pakistani citizens?`,
   840→    a: feeText,
   841→  });
   842→
   843→  // Q3: Processing time
   844→  faqs.push({
   845→    q: `How long does it take to get a ${country.name} visa?`,
   846→    a: country.visaFree
   847→      ? `No processing time is needed as ${country.name} offers visa-free entry to Pakistani passport holders. You can travel immediately.`
   848→      : country.visaOnArrival
   849→        ? `The visa on arrival for ${country.name} is processed instantly at the airport or border crossing. The entire process typically takes 5-15 minutes.`
   850→        : `${country.name} visa processing typically takes ${country.processingDaysMin}-${country.processingDaysMax} business days for Pakistani citizens. It is recommended to apply at least 2-3 weeks before your planned travel date.`,
   851→  });
   852→
   853→  // Q4: Safety
   854→  faqs.push({
   855→    q: `Is ${country.name} safe for Pakistani tourists?`,
   856→    a: country.safetySummary
   857→      ? `${country.name} has a safety rating of ${Math.min(country.safetyRating, 5)}/5. ${country.safetySummary}`
   858→      : `${country.name} has a safety rating of ${Math.min(country.safetyRating, 5)}/5. As with any travel destination, exercise standard precautions, keep your belongings secure, and follow local laws and customs.`,
   859→  });
   860→
   861→  // Q5: Best time to visit or documents required
   862→  if (country.bestTravelMonths) {
   863→    faqs.push({
   864→      q: `What is the best time to visit ${country.name} from Pakistan?`,
   865→      a: `The best months to visit ${country.name} are ${country.bestTravelMonths}. The average temperature is around ${country.avgTempC}\u00B0C. The local currency is ${country.currency} (${country.currencyCode}). Plan your trip during these months for the most comfortable weather conditions.`,
   866→    });
   867→  } else {
   868→    faqs.push({
   869→      q: `What documents do I need for a ${country.name} visa?`,
   870→      a: `Common requirements for a ${country.name} visa include: a valid Pakistani passport with 6+ months validity, passport-sized photographs, proof of accommodation, return flight ticket, travel insurance, and proof of sufficient funds. For the complete and most up-to-date list, check the official embassy website or use PakVisa Advisor.`,
   871→    });
   872→  }
   873→
   874→  return faqs;
   875→}
