   118→function getVisaLabel(country: { visaFree: boolean; visaOnArrival: boolean; etaAvailable: boolean }): string {
   119→  if (country.visaFree) return 'Visa Free';

   139→function usdToPkr(usd: number): string {

   143→function safetyStars(rating: number): string {
   144→  const clamped = Math.min(Math.max(rating, 0), 5);
   145→  return '\u2605'.repeat(clamped) + '\u2606'.repeat(5 - clamped);

   201→export async function generateMetadata({ params }: { params: Promise<{ slug: string }> }): Promise<Metadata> {
   202→  const { slug: rawSlug } = await params;
   203→  const slug = rawSlug.toLowerCase();
   204→  const code = SLUG_TO_CODE[slug];
   205→  if (!code) return { title: 'Country Not Found | PakVisa Advisor' };
   206→
   207→  const country = await db.country.findUnique({
   208→    where: { code },
   209→    include: { costProfiles: true, visaTypes: true },
   210→  });
   211→
   212→  if (!country) return { title: 'Country Not Found | PakVisa Advisor' };
   213→
   214→  const visaLabel = getVisaLabel(country);
   215→  const costProfile = country.costProfiles[0];
   216→  const visaFee = costProfile ? `$${costProfile.visaFeeUSD} (\u2248 PKR ${usdToPkr(costProfile.visaFeeUSD)})` : 'Free';
   217→  const processing = country.visaFree ? 'No processing required' : `${country.processingDaysMin}-${country.processingDaysMax} business days`;
   218→  const title = `${country.name} Visa for Pakistani Citizens - Requirements, Fees & Guide 2026`;
   219→  const description = `Complete ${country.name} visa guide for Pakistani passport holders. ${visaLabel} access, ${visaFee} fee, ${processing} processing. Updated for 2026 with requirements, costs, and embassy details.`;
   220→
   221→  return {
   222→    title,
   223→    description,
   224→    keywords: [
   225→      `${country.name} visa Pakistan`,
   226→      `${country.name} visa Pakistani passport`,
   227→      `${country.name} ${visaLabel.toLowerCase()} Pakistan`,
   228→      `Pakistani passport ${country.name}`,
   229→      `${country.name} embassy Islamabad`,
   230→      `${country.name} visa requirements 2026`,
   231→      `how to get ${country.name} visa from Pakistan`,
   232→      `${country.name} visa fee PKR`,
   233→    ],
   234→    openGraph: {
   235→      type: 'article',
   236→      locale: 'en_US',
   237→      url: `https://pakvisaadvisor.com/${slug}`,
   238→      siteName: 'PakVisa Advisor',
   239→      title,
   240→      description,
   241→      images: [{ url: '/og-image.png', width: 1200, height: 630, alt: `${country.name} Visa Guide - PakVisa Advisor` }],
   242→    },
   243→    twitter: {
   244→      card: 'summary_large_image',
   245→      title,
   246→      description,
   247→      images: ['/og-image.png'],
   248→    },
   249→    alternates: { canonical: `https://pakvisaadvisor.com/${slug}` },

   917→function generateFAQs(
   918→  country: {
   919→    name: string; code: string; visaFree: boolean; visaOnArrival: boolean; etaAvailable: boolean;
   920→    processingDaysMin: number; processingDaysMax: number; safetyRating: number; safetySummary: string;
   921→    bestTravelMonths: string; avgTempC: string; currency: string; currencyCode: string;
   922→  },
   923→  costProfile: CostProfileData | undefined,
   924→  visaLabel: string,
   925→  isEmbassyRequired: boolean,
   926→): { q: string; a: string }[] {
   927→  const faqs: { q: string; a: string }[] = [];
   928→
   929→  // Q1: Do Pakistani citizens need a visa?
   930→  faqs.push({
   931→    q: `Do Pakistani citizens need a visa for ${country.name}?`,
   932→    a: country.visaFree
   933→      ? `No, Pakistani citizens do not need a visa to visit ${country.name}. You can travel visa-free with a valid Pakistani passport. Make sure your passport has at least 6 months of validity remaining.`
   934→      : country.visaOnArrival
   935→        ? `Pakistani citizens can get a visa on arrival at ${country.name}. Simply present your valid Pakistani passport at the immigration counter upon arrival. Processing is usually instant.`
   936→        : country.etaAvailable
   937→          ? `Yes, Pakistani citizens need an e-Visa to visit ${country.name}. The e-Visa can be applied for online through the official government portal. Processing typically takes ${country.processingDaysMin}-${country.processingDaysMax} business days.`
   938→          : `Yes, Pakistani citizens need a visa to visit ${country.name}. You must apply at the embassy or consulate. The application process typically takes ${country.processingDaysMin}-${country.processingDaysMax} business days.`,
   939→  });
   940→
   941→  // Q2: What is the visa fee?
   942→  const feeText = costProfile && costProfile.visaFeeUSD > 0
   943→    ? `The visa fee for ${country.name} is approximately $${costProfile.visaFeeUSD} USD (\u2248 PKR ${usdToPkr(costProfile.visaFeeUSD)}). Additional service fees of $${costProfile.serviceFeeUSD} may apply.`
   944→    : `There is no visa fee for Pakistani citizens visiting ${country.name} as it is ${visaLabel.toLowerCase()}.`
   945→  ;
   946→  faqs.push({
   947→    q: `What is the visa fee for ${country.name} for Pakistani citizens?`,
   948→    a: feeText,
   949→  });
   950→
   951→  // Q3: Processing time
   952→  faqs.push({
   953→    q: `How long does it take to get a ${country.name} visa?`,
   954→    a: country.visaFree
   955→      ? `No processing time is needed as ${country.name} offers visa-free entry to Pakistani passport holders. You can travel immediately.`
   956→      : country.visaOnArrival
   957→        ? `The visa on arrival for ${country.name} is processed instantly at the airport or border crossing. The entire process typically takes 5-15 minutes.`
   958→        : `${country.name} visa processing typically takes ${country.processingDaysMin}-${country.processingDaysMax} business days for Pakistani citizens. It is recommended to apply at least 2-3 weeks before your planned travel date.`,
   959→  });
   960→
   961→  // Q4: Safety
   962→  faqs.push({
   963→    q: `Is ${country.name} safe for Pakistani tourists?`,
   964→    a: country.safetySummary
   965→      ? `${country.name} has a safety rating of ${Math.min(country.safetyRating, 5)}/5. ${country.safetySummary}`
   966→      : `${country.name} has a safety rating of ${Math.min(country.safetyRating, 5)}/5. As with any travel destination, exercise standard precautions, keep your belongings secure, and follow local laws and customs.`,
   967→  });
   968→
   969→  // Q5: Best time to visit or documents required
   970→  if (country.bestTravelMonths) {
   971→    faqs.push({
   972→      q: `What is the best time to visit ${country.name} from Pakistan?`,
   973→      a: `The best months to visit ${country.name} are ${country.bestTravelMonths}. The average temperature is around ${country.avgTempC}\u00B0C. The local currency is ${country.currency} (${country.currencyCode}). Plan your trip during these months for the most comfortable weather conditions.`,
   974→    });
   975→  } else {
   976→    faqs.push({
   977→      q: `What documents do I need for a ${country.name} visa?`,
   978→      a: `Common requirements for a ${country.name} visa include: a valid Pakistani passport with 6+ months validity, passport-sized photographs, proof of accommodation, return flight ticket, travel insurance, and proof of sufficient funds. For the complete and most up-to-date list, check the official embassy website or use PakVisa Advisor.`,
   979→    });
   980→  }
   981→
   982→  return faqs;
   983→}
