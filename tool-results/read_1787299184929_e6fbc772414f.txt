    34→// ============================================================
    35→function useAnimatedCounter(target: number, duration: number = 1200, enabled: boolean = true) {
    36→  const [count, setCount] = useState(0);
    37→  useEffect(() => {
    38→    if (!enabled || target === 0) return;
    39→    let start = 0;
    40→    let startTime: number | null = null;
    41→    let raf: number;
    42→    const step = (timestamp: number) => {
    43→      if (!startTime) startTime = timestamp;
    44→      const progress = Math.min((timestamp - startTime) / duration, 1);
    45→      // Ease-out cubic
    46→      const eased = 1 - Math.pow(1 - progress, 3);
    47→      const current = Math.round(eased * target);
    48→      setCount(current);
    49→      if (progress < 1) {
    50→        raf = requestAnimationFrame(step);
    51→      }
    52→    };
    53→    raf = requestAnimationFrame(step);
    54→    return () => cancelAnimationFrame(raf);
    55→  }, [target, duration, enabled]);
    56→  return count;
    57→}

   127→// Pagination Helper
   128→// ============================================================
   129→function generatePageNumbers(current: number, total: number): (number | 'ellipsis')[] {
   130→  if (total <= 7) {
   131→    return Array.from({ length: total }, (_, i) => i + 1);
   132→  }
   133→  const pages: (number | 'ellipsis')[] = [1];
   134→  if (current > 3) pages.push('ellipsis');
   135→  const start = Math.max(2, current - 1);
   136→  const end = Math.min(total - 1, current + 1);
   137→  for (let i = start; i <= end; i++) pages.push(i);
   138→  if (current < total - 2) pages.push('ellipsis');
   139→  if (total > 1) pages.push(total);
   140→  return pages;
   141→}

   143→// ============================================================
   144→// Visa Type Helpers (single source of truth for access type)
   145→// Priority: visaFree > visaOnArrival > etaAvailable > embassy
   146→// ============================================================
   147→function getPrimaryAccess(c: CountryData): string {
   148→  if (c.visaFree) return 'visa-free';
   149→  if (c.visaOnArrival) return 'visa-on-arrival';
   150→  if (c.etaAvailable) return 'e-visa';
   151→  return 'embassy';
   152→}
   153→function getVisaType(c: CountryData) {
   154→  const access = getPrimaryAccess(c);
   155→  switch (access) {
   156→    case 'visa-free':     return { label: 'Visa Free', color: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-400', dot: 'bg-emerald-500' };
   157→    case 'visa-on-arrival': return { label: 'Visa on Arrival', color: 'bg-amber-100 text-amber-700 dark:bg-amber-900/30 dark:text-amber-400', dot: 'bg-amber-500' };
   158→    case 'e-visa':       return { label: 'e-Visa', color: 'bg-sky-100 text-sky-700 dark:bg-sky-900/30 dark:text-sky-400', dot: 'bg-sky-500' };
   159→    default:             return { label: 'Embassy Required', color: 'bg-gray-100 text-gray-600 dark:bg-gray-800 dark:text-gray-400', dot: 'bg-gray-400' };
   160→  }
   161→}
