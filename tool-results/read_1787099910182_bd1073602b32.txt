    20→    20→function LiveClock({ timezone, cityName }: { timezone: string; cityName: string }) {
    21→    21→  const [time, setTime] = useState('');
    22→    22→  const [date, setDate] = useState('');
    23→    23→
    24→    24→  useEffect(() => {
    25→    25→    function updateTime() {
    26→    26→      try {
    27→    27→        const now = new Date();
    28→    28→        const formatter = new Intl.DateTimeFormat('en-US', {
    29→    29→          timeZone: timezone,
    30→    30→          hour: '2-digit',
    31→    31→          minute: '2-digit',
    32→    32→          second: '2-digit',
    33→    33→          hour12: true,
    34→    34→        });
    35→    35→        const dateFormatter = new Intl.DateTimeFormat('en-US', {
    36→    36→          timeZone: timezone,
    37→    37→          weekday: 'short',
    38→    38→          month: 'short',
    39→    39→          day: 'numeric',
    40→    40→        });
    41→    41→        setTime(formatter.format(now));
    42→    42→        setDate(dateFormatter.format(now));
    43→    43→      } catch {
    44→    44→        setTime('—');
    45→    45→        setDate('—');
    46→    46→      }
    47→    47→    }
    48→    48→    updateTime();
    49→    49→    const interval = setInterval(updateTime, 1000);
    50→    50→    return () => clearInterval(interval);
    51→    51→  }, [timezone]);
    52→    52→
    53→    53→  // Calculate time difference from PKT (Asia/Karachi)
    54→    54→  const timeDiff = useMemo(() => {
    55→    55→    try {
    56→    56→      const now = new Date();
    57→    57→      // Get UTC offsets in minutes for both timezones
    58→    58→      const targetParts = new Intl.DateTimeFormat('en-US', {
    59→    59→        timeZone: timezone,
    60→    60→        timeZoneName: 'shortOffset',
    61→    61→      }).formatToParts(now);
    62→    62→      const pktParts = new Intl.DateTimeFormat('en-US', {
    63→    63→        timeZone: 'Asia/Karachi',
    64→    64→        timeZoneName: 'shortOffset',
    65→    65→      }).formatToParts(now);
    66→    66→
    67→    67→      const parseOffset = (parts: Intl.DateTimeFormatPart[]) => {
    68→    68→        const tzPart = parts.find(p => p.type === 'timeZoneName');
    69→    69→        if (!tzPart) return 0;
    70→    70→        const match = tzPart.value.match(/GMT([+-]?)(\d{1,2})(?::(\d{2}))?/);
    71→    71→        if (!match) return 0;
    72→    72→        const sign = match[1] === '-' ? -1 : 1;
    73→    73→        const hours = parseInt(match[2]);
    74→    74→        const minutes = match[3] ? parseInt(match[3]) : 0;
    75→    75→        return sign * (hours * 60 + minutes);
    76→    76→      };
    77→    77→
    78→    78→      const targetOffset = parseOffset(targetParts);
    79→    79→      const pktOffset = parseOffset(pktParts);
    80→    80→      const diffHours = (targetOffset - pktOffset) / 60;
    81→    81→
    82→    82→      if (diffHours === 0) return 'Same as PKT';
    83→    83→      if (diffHours > 0) return `+${diffHours}h from PKT`;
    84→    84→      return `${diffHours}h from PKT`;
    85→    85→    } catch {
    86→    86→      return '';
    87→    87→    }
    88→    88→  }, [timezone]);
    89→    89→
    90→    90→  return (
    91→    91→    <div className="flex items-center gap-3 rounded-lg bg-sky-50 dark:bg-sky-950/20 border border-sky-200 dark:border-sky-800 p-3">
    92→    92→      <Clock className="w-5 h-5 text-sky-600 shrink-0" />
    93→    93→      <div>
    94→    94→        <p className="text-sm font-semibold text-sky-900 dark:text-sky-200">{time}</p>
    95→    95→        <p className="text-xs text-sky-700 dark:text-sky-400">{date} {timeDiff && `· ${timeDiff}`}</p>
    96→    96→      </div>
    97→    97→    </div>
    98→    98→  );
    99→    99→}

   104→   104→function CurrencyConverter({ currencyCode, currencyName }: { currencyCode: string; currencyName: string }) {
   105→   105→  const [amount, setAmount] = useState<string>('100');
   106→   106→  const [direction, setDirection] = useState<'topkr' | 'frompkr'>('topkr');
   107→   107→
   108→   108→  const rate = EXCHANGE_RATES[currencyCode];
   109→   109→  const [topkr, frompkr] = useMemo(() => {
   110→   110→    if (!rate) return ['—', '—'];
   111→   111→    return [
   112→   112→      (parseFloat(amount || '0') * rate).toLocaleString('en-PK', { maximumFractionDigits: 2 }),
   113→   113→      (parseFloat(amount || '0') / rate).toLocaleString(undefined, { maximumFractionDigits: 2 }),
   114→   114→    ];
   115→   115→  }, [amount, rate]);
   116→   116→
   117→   117→  return (
   118→   118→    <div className="rounded-lg bg-emerald-50 dark:bg-emerald-950/20 border border-emerald-200 dark:border-emerald-800 p-3 space-y-2">
   119→   119→      <div className="flex items-center justify-between">
   120→   120→        <div className="flex items-center gap-2">
   121→   121→          <Banknote className="w-4 h-4 text-emerald-600" />
   122→   122→          <span className="text-xs font-semibold text-emerald-900 dark:text-emerald-200">Currency Converter</span>
   123→   123→        </div>
   124→   124→        <Badge variant="outline" className="text-[10px] border-emerald-300 dark:border-emerald-700 text-emerald-700 dark:text-emerald-400">
   125→   125→          {currencyCode} · {currencyName}
   126→   126→        </Badge>
   127→   127→      </div>
   128→   128→      <div className="flex items-center gap-2">
   129→   129→        <input
   130→   130→          type="number"
   131→   131→          value={amount}
   132→   132→          onChange={(e) => setAmount(e.target.value)}
   133→   133→          className="w-full h-8 px-2 rounded-md border bg-white dark:bg-background text-sm focus:outline-none focus:ring-1 focus:ring-emerald-500"
   134→   134→          min="0"
   135→   135→        />
   136→   136→        <button
   137→   137→          onClick={() => setDirection((d) => d === 'topkr' ? 'frompkr' : 'topkr')}
   138→   138→          className="p-1.5 rounded-md border bg-white dark:bg-background hover:bg-muted transition-colors"
   139→   139→          title="Swap direction"
   140→   140→        >
   141→   141→          <ArrowRightLeft className="w-3.5 h-3.5 text-emerald-600" />
   142→   142→        </button>
   143→   143→      </div>
   144→   144→      {rate ? (
   145→   145→        <div className="flex items-center justify-between text-xs">
   146→   146→          <span className="text-muted-foreground">
   147→   147→            {direction === 'topkr' ? `${amount || '0'} ${currencyCode} =` : `PKR ${amount || '0'} =`}
   148→   148→          </span>
   149→   149→          <span className="font-semibold text-emerald-800 dark:text-emerald-300">
   150→   150→            {direction === 'topkr' ? `PKR ${topkr}` : `${frompkr} ${currencyCode}`}
   151→   151→          </span>
   152→   152→        </div>
   153→   153→      ) : (
   154→   154→        <p className="text-xs text-muted-foreground">Exchange rate not available</p>
   155→   155→      )}
   156→   156→      <p className="text-[10px] text-emerald-700/70 dark:text-emerald-400/70">
   157→   157→        Rate: 1 {currencyCode} ≈ PKR {rate?.toFixed(2) || '—'} · Approximate, verify before transaction
   158→   158→      </p>
   159→   159→    </div>
   160→   160→  );
   161→   161→}

   371→   371→function WeatherInfo({ avgTemp, monthlyTemps }: { avgTemp: string; monthlyTemps: Record<string, number> }) {
   372→   372→  const temps = Object.entries(monthlyTemps);
   373→   373→  if (temps.length === 0 && !avgTemp) return null;
   374→   374→
   375→   375→  const maxTemp = temps.length > 0 ? Math.max(...temps.map(([, v]) => v)) : parseInt(avgTemp) || 0;
   376→   376→  const minTemp = temps.length > 0 ? Math.min(...temps.map(([, v]) => v)) : parseInt(avgTemp) || 0;
   377→   377→
   378→   378→  return (
   379→   379→    <div>
   380→   380→      <h5 className="text-xs font-semibold text-muted-foreground uppercase tracking-wide mb-2">Climate & Weather</h5>
   381→   381→      <div className="rounded-lg bg-muted/50 p-3 space-y-3">
   382→   382→        <div className="flex items-center justify-between text-sm">
   383→   383→          <span className="flex items-center gap-1.5">
   384→   384→            <Thermometer className="w-4 h-4 text-red-500" />
   385→   385→            <span className="font-medium">Avg Temperature</span>
   386→   386→          </span>
   387→   387→          <span className="font-semibold">{avgTemp}°C</span>
   388→   388→        </div>
   389→   389→        <div className="flex items-center gap-3 text-xs text-muted-foreground">
   390→   390→          <span>🔥 High: {maxTemp}°C</span>
   391→   391→          <span>❄️ Low: {minTemp}°C</span>
   392→   392→        </div>
   393→   393→        {temps.length > 0 && (
   394→   394→          <div className="flex items-end gap-0.5 h-8">
   395→   395→            {temps.map(([month, temp], idx) => {
   396→   396→              const height = maxTemp > minTemp ? ((temp - minTemp) / (maxTemp - minTemp)) * 100 : 50;
   397→   397→              return (
   398→   398→                <div key={month} className="flex-1 flex flex-col items-center gap-0.5">
   399→   399→                  <div
   400→   400→                    className="w-full rounded-t-sm min-h-[2px] transition-all"
   401→   401→                    style={{
   402→   402→                      height: `${Math.max(height, 5)}%`,
   403→   403→                      backgroundColor: temp > 30 ? '#ef4444' : temp > 20 ? '#f59e0b' : temp > 10 ? '#3b82f6' : '#6366f1',
