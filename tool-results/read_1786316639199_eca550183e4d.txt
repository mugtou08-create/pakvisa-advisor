  1577→function AnimatedNumber({ value, suffix }: { value: number; suffix: string }) {
  1578→  const [display, setDisplay] = useState(0);
  1579→  useEffect(() => {
  1580→    const dur = 1500;
  1581→    const start = Date.now();
  1582→    const step = () => {
  1583→      const elapsed = Date.now() - start;
  1584→      const progress = Math.min(elapsed / dur, 1);
  1585→      const eased = 1 - Math.pow(1 - progress, 3);
  1586→      setDisplay(Math.round(value * eased));
  1587→      if (progress < 1) requestAnimationFrame(step);
  1588→    };
  1589→    requestAnimationFrame(step);
  1590→  }, [value]);
  1591→  return <>{display.toLocaleString()}{suffix}</>;
  1592→}
