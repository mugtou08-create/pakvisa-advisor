    39→/** Mini radar chart for country cards (3-axis) */
    40→export function MiniRadarChart({ safety, cost, ease }: { safety: number; cost: number; ease: number }) {
    41→  const cx = 28, cy = 28, r = 22;
    42→  const axes = 3;
    43→  const angleStep = (2 * Math.PI) / axes;
    44→  const startAngle = -Math.PI / 2;
    45→  const s = safety / 10;
    46→  const c = Math.max(0.05, 1 - cost / 3000);
    47→  const e = ease;
    48→  const values = [s, c, e];
    49→  const points = values.map((val, i) => {
    50→    const angle = startAngle + i * angleStep;
    51→    return `${cx + val * r * Math.cos(angle)},${cy + val * r * Math.sin(angle)}`;
    52→  }).join(' ');
    53→  const gridLines = [0.33, 0.66, 1].map(level => {
    54→    const pts = Array.from({ length: axes }, (_, i) => {
    55→      const angle = startAngle + i * angleStep;
    56→      return `${cx + level * r * Math.cos(angle)},${cy + level * r * Math.sin(angle)}`;
    57→    }).join(' ');
    58→    return <polygon key={level} points={pts} fill="none" stroke="currentColor" strokeWidth="0.3" className="text-white/20" />;
    59→  });
    60→
    61→  return (
    62→    <svg width="56" height="56" viewBox="0 0 56 56" className="inline-block opacity-80">
    63→      {gridLines}
    64→      {Array.from({ length: axes }, (_, i) => {
    65→        const angle = startAngle + i * angleStep;
    66→        return <line key={i} x1={cx} y1={cy} x2={cx + r * Math.cos(angle)} y2={cy + r * Math.sin(angle)} stroke="currentColor" strokeWidth="0.3" className="text-white/20" />;
    67→      })}
    68→      <polygon points={points} fill="rgba(249,115,22,0.15)" stroke="#fbbf24" strokeWidth="1" />
    69→      {values.map((val, i) => {
    70→        const angle = startAngle + i * angleStep;
    71→        return <circle key={i} cx={cx + val * r * Math.cos(angle)} cy={cy + val * r * Math.sin(angle)} r="1.5" fill="#fbbf24" />;
    72→      })}
    73→    </svg>
    74→  );
    75→}

    78→export function FlagImage({ code, flagUrl, size = 24, className = '', emoji }: { code: string; flagUrl?: string; size?: number; className?: string; emoji?: string }) {
    79→  const src = flagUrl || getFlagUrl(code);
    80→  if (!src) return <span className={className} style={{ fontSize: size * 0.8 }}>{emoji || '\u{1F3F3}\uFE0F'}</span>;
    81→  return (
    82→    <Image
    83→      src={src}
    84→      alt={`Flag`}
    85→      width={size}
    86→      height={Math.round(size * 0.67)}
    87→      className={`object-cover rounded-sm ${className}`}
    88→      unoptimized
    89→    />
    90→  );
    91→}
    92→export function InteractiveWorldMap({ countries, onSelectCountry }: { countries: CountryData[]; onSelectCountry: (c: CountryData) => void }) {
    93→  const [svgContent, setSvgContent] = useState<string | null>(null);
    94→  const [hoveredCountry, setHoveredCountry] = useState<string | null>(null);
    95→  const [tooltipPos, setTooltipPos] = useState<{ x: number; y: number; name: string; status: { fill: string; label: string } } | null>(null);
    96→  const containerRef = useRef<HTMLDivElement>(null);
    97→
    98→  // Build lookup: SVG country name -> CountryData
    99→  const countryLookup = useMemo(() => {
   100→    const map = new globalThis.Map<string, CountryData>();
   101→    countries.forEach(c => {
   102→      const normalized = c.name.replace(/\s+/g, '').toUpperCase();
   103→      map.set(normalized, c);
   104→      map.set(c.name.toUpperCase(), c);
   105→    });
   106→    Object.entries(COUNTRY_NAME_ALIASES).forEach(([svgName, aliases]) => {
   107→      aliases.forEach(alias => {
   108→        const found = countries.find(co => {
   109→          const coNorm = co.name.replace(/\s+/g, '').toUpperCase();
   110→          const aliasNorm = alias.replace(/\s+/g, '').toUpperCase();
   111→          return coNorm === aliasNorm;
   112→        });
   113→        if (found) map.set(svgName.toUpperCase(), found);
   114→      });
   115→    });
   116→    return map;
   117→  }, [countries]);
   118→
   119→  const getStatus = (countryName: string): { fill: string; label: string } => {
   120→    const c = countryLookup.get(countryName.toUpperCase())
   121→      || countryLookup.get(countryName.replace(/\s+/g, '').toUpperCase());
   122→    if (!c) return VISA_CATEGORY_COLORS.unknown;
   123→    if (c.visaFree) return VISA_CATEGORY_COLORS.visaFree;
   124→    if (c.visaOnArrival) return VISA_CATEGORY_COLORS.visaOnArrival;
   125→    if (c.etaAvailable) return VISA_CATEGORY_COLORS.eVisa;
   126→    return VISA_CATEGORY_COLORS.embassy;
   127→  };
   128→
   129→  // Load SVG and pre-process colors into the string itself
   130→  useEffect(() => {
   131→    fetch('/world-map.svg')
   132→      .then(res => res.text())
   133→      .then(text => {
   134→        let processed = text
   135→          .replace(/<defs>[\s\S]*?<\/defs>/, '')
   136→          .replace(/<rect[^>]*fill="[^"]*"[^>]*\/>/, '');
   137→
   138→        // Inject inline style into path elements with data-map-country
   139→        processed = processed.replace(
   140→          /<path\s+([^>]*?)data-map-country="([^"]+)"([^>]*)\s*\/>/g,
   141→          (_full, before, name, after) => {
   142→            const status = getStatus(name);
   143→            // Inject style with !important fill to override any CSS
   144→            return `<path ${before}data-map-country="${name}"${after} style="fill:${status.fill};stroke:#fff;stroke-width:0.8;cursor:pointer;transition:all 0.2s ease;opacity:0.85" />`;
   145→          }
   146→        );
   147→
   148→        // Style non-country paths (generic landmasses)
   149→        processed = processed.replace(
   150→          /<path\s+(?!([^>]*?)data-map-country)([^>]*?)\s*\/>/g,
   151→          (_full, _before, after) => {
   152→            // Only style paths that don't already have a style from above
   153→            if (_full.includes('style=')) return _full;
   154→            return `<path style="fill:#d4d4d8;stroke:#fff;stroke-width:0.5" ${after.replace(/<\/?path[^>]*>/g, '')} />`;
   155→          }
   156→        );
   157→
   158→        // Add ocean background rect as first child after <svg>
   159→        processed = processed.replace(
   160→          /<svg[^>]*>/,
   161→          (match) => `${match}<rect width="1000" height="500" fill="#e8f4f8" />`
   162→        );
   163→
   164→        setSvgContent(processed);
   165→      })
   166→      .catch(() => setSvgContent(''));
   167→  }, []);
   168→
   169→  const handleMouseMove = useCallback((e: React.MouseEvent) => {
   170→    if (!e.target || !(e.target instanceof SVGElement)) {
   171→      setTooltipPos(null);
   172→      return;
   173→    }
   174→    const path = (e.target as SVGElement).closest('path[data-map-country]') as SVGPathElement | null;
   175→    if (path) {
   176→      const name = path.getAttribute('data-map-country') || '';
   177→      const status = getStatus(name);
   178→      if (hoveredCountry !== name) {
   179→        setHoveredCountry(name);
   180→        const allPaths = containerRef.current?.querySelectorAll('path[data-map-country]');
   181→        allPaths?.forEach(p => {
   182→          if (p === path) {
   183→            p.style.opacity = '1';
   184→            p.style.strokeWidth = '1.5';
   185→          } else {
   186→            p.style.opacity = '0.5';
   187→            p.style.strokeWidth = '0.8';
   188→          }
   189→        });
   190→      }
   191→      const rect = containerRef.current?.getBoundingClientRect();
   192→      if (rect) {
   193→        setTooltipPos({
   194→          x: e.clientX - rect.left,
   195→          y: e.clientY - rect.top,
   196→          name,
   197→          status,
   198→        });
   199→      }
   200→    } else {
   201→      if (hoveredCountry) {
   202→        setHoveredCountry(null);
   203→        const allPaths = containerRef.current?.querySelectorAll('path[data-map-country]');
   204→        allPaths?.forEach(p => {
   205→          p.style.opacity = '0.85';
   206→          p.style.strokeWidth = '0.8';
   207→        });
   208→        setTooltipPos(null);
   209→      }
   210→    }
   211→  }, [hoveredCountry, getStatus]);
   212→
   213→  const handleClick = useCallback((e: React.MouseEvent) => {
   214→    if (!e.target || !(e.target instanceof SVGElement)) return;
   215→    const path = (e.target as SVGElement).closest('path[data-map-country]') as SVGPathElement | null;
   216→    if (!path) return;
   217→    const name = path.getAttribute('data-map-country') || '';
   218→    const c = countryLookup.get(name.toUpperCase())
   219→      || countryLookup.get(name.replace(/\s+/g, '').toUpperCase());
   220→    if (c) onSelectCountry(c);
   221→  }, [countryLookup, onSelectCountry]);
   222→
   223→  const handleMouseLeave = useCallback(() => {
   224→    setHoveredCountry(null);
   225→    setTooltipPos(null);
   226→    const allPaths = containerRef.current?.querySelectorAll('path[data-map-country]');
   227→    allPaths?.forEach(p => {
   228→      p.style.opacity = '0.85';
   229→      p.style.strokeWidth = '0.8';
   230→    });
   231→  }, []);
   232→
   233→  // Count stats for the legend
   234→  const legendStats = useMemo(() => {
   235→    let visaFree = 0, voa = 0, eVisa = 0, embassy = 0, noData = 0;
   236→    const svgCountries = svgContent?.match(/data-map-country="([^"]+)"/g) || [];
   237→    svgCountries.forEach(match => {
   238→      const name = match.match(/"([^"]+)"/)?.[1] || '';
   239→      const status = getStatus(name);
   240→      if (status.fill === VISA_CATEGORY_COLORS.visaFree.fill) visaFree++;
   241→      else if (status.fill === VISA_CATEGORY_COLORS.visaOnArrival.fill) voa++;
   242→      else if (status.fill === VISA_CATEGORY_COLORS.eVisa.fill) eVisa++;
   243→      else if (status.fill === VISA_CATEGORY_COLORS.embassy.fill) embassy++;
   244→      else noData++;
   245→    });
   246→    return { visaFree, voa, eVisa, embassy, noData };
   247→  }, [svgContent, getStatus]);
   248→
   249→  return (
   250→    <div className="bg-gradient-to-br from-slate-50 to-slate-100 dark:from-slate-900 dark:to-slate-800 rounded-2xl p-4 md:p-6 border border-border/50">
   251→      <div className="flex items-center justify-between mb-4">
   252→        <h3 className="text-sm font-semibold flex items-center gap-2">
   253→          <Globe className="w-4 h-4 text-amber-500" />
   254→          Visa Requirements World Map
   255→        </h3>
   256→        <span className="text-[10px] text-muted-foreground bg-muted px-2 py-0.5 rounded-full">Hover &amp; click countries</span>
   257→      </div>
   258→
   259→      <div
   260→        ref={containerRef}
   261→        className="relative w-full rounded-lg overflow-hidden bg-[#e8f4f8] dark:bg-slate-800"
   262→        style={{ paddingBottom: '50%' }}
   263→        onMouseMove={handleMouseMove}
   264→        onClick={handleClick}
   265→        onMouseLeave={handleMouseLeave}
   266→      >
   267→        {svgContent ? (
   268→          <svg
   269→            viewBox="0 0 1000 500"
   270→            className="absolute inset-0 w-full h-full"
   271→            preserveAspectRatio="xMidYMid meet"
   272→            dangerouslySetInnerHTML={{ __html: svgContent }}
   273→          />
   274→        ) : (
   275→          <div className="absolute inset-0 flex items-center justify-center">
   276→            <div className="flex flex-col items-center gap-3">
   277→              <div className="w-8 h-8 border-2 border-amber-500 border-t-transparent rounded-full animate-spin" />
   278→              <span className="text-xs text-muted-foreground">Loading world map...</span>
   279→            </div>
   280→          </div>
   281→        )}
   282→
   283→        {/* Floating tooltip */}
   284→        {tooltipPos && (
   285→          <div
   286→            className="absolute z-50 pointer-events-none px-3 py-2 bg-white dark:bg-slate-900 rounded-lg shadow-xl border border-border/50 text-xs whitespace-nowrap"
   287→            style={{
   288→              left: `${Math.min(tooltipPos.x + 12, 90)}%`,
   289→              top: `${Math.max(tooltipPos.y - 10, 2)}%`,
   290→            }}
   291→          >
   292→            <div className="font-semibold">{tooltipPos.name}</div>
   293→            <div className="flex items-center gap-1.5 mt-0.5">
   294→              <span className="inline-block w-2 h-2 rounded-full" style={{ backgroundColor: tooltipPos.status.fill }} />
   295→              <span style={{ color: tooltipPos.status.fill }}>{tooltipPos.status.label}</span>
   296→            </div>
   297→          </div>
   298→        )}
   299→      </div>
   300→
   301→      {/* Legend with counts */}
   302→      <div className="flex flex-wrap gap-x-4 gap-y-2 mt-4 justify-center">
   303→        {Object.entries(VISA_CATEGORY_COLORS).map(([key, item]) => {
   304→          const count = key === 'visaFree' ? legendStats.visaFree
   305→            : key === 'visaOnArrival' ? legendStats.voa
   306→            : key === 'eVisa' ? legendStats.eVisa
   307→            : key === 'embassy' ? legendStats.embassy
   308→            : legendStats.noData;
   309→          return (
   310→            <div key={key} className="flex items-center gap-1.5 text-xs text-muted-foreground">
   311→              <div className="w-3 h-3 rounded-sm shrink-0" style={{ backgroundColor: item.fill }} />

   313→              <span className="text-[10px] bg-muted/50 px-1.5 py-0.5 rounded-full font-medium">{count}</span>
   314→            </div>
   315→          );
   316→        })}
   317→      </div>
   318→    </div>
   319→  );
   320→}

   366→// ============ BEST MATCH RECOMMENDATIONS (B3) ============
   367→export function BestMatchRecommendations({ countries, scoreResults, onSelectCountry }: { countries: CountryData[]; scoreResults: ScoreBreakdown[]; onSelectCountry: (c: CountryData) => void }) {
   368→  const [loaded, setLoaded] = useState(false);
   369→  const { userProfile } = useAppStore();
   370→
   371→  useEffect(() => {
   372→    const timer = setTimeout(() => setLoaded(true), 2000);
   373→    return () => clearTimeout(timer);
   374→  }, []);
   375→
   376→  const recommendations = useMemo(() => {
   377→    if (scoreResults.length > 0) {
   378→      return scoreResults
   379→        .sort((a, b) => b.finalScore - a.finalScore)
   380→        .slice(0, 3)
   381→        .map((s) => {
   382→          const c = countries.find((co) => co.code === s.countryCode);
   383→          if (!c) return null;
   384→          const reason = s.finalScore >= 70 ? 'Excellent match based on your profile' : s.finalScore >= 40 ? 'Good match with minor improvements needed' : 'Consider improving your profile first';
   385→          return { country: c, matchPct: Math.round(s.finalScore), reason };
   386→        })
   387→        .filter(Boolean) as { country: CountryData; matchPct: number; reason: string }[];
   388→    }
   389→    // Default recommendations
   390→    const defaults = [
   391→      { code: 'MY', reason: 'Visa-free entry for 30 days with beautiful destinations' },
   392→      { code: 'AE', reason: 'Visa on arrival with world-class tourism infrastructure' },
   393→      { code: 'TR', reason: 'Easy e-Visa process with rich cultural experiences' },
   394→    ];
   395→    return defaults.map((d) => {
   396→      const c = countries.find((co) => co.code === d.code);
   397→      return c ? { country: c, matchPct: c.visaFree ? 95 : c.visaOnArrival ? 88 : 75, reason: d.reason } : null;
   398→    }).filter(Boolean) as { country: CountryData; matchPct: number; reason: string }[];
   399→  }, [scoreResults, countries]);
   400→
   401→  if (!loaded) {
   402→    return (
   403→      <div>
   404→        <div className="scroll-section-title">
   405→          <h3 className="text-sm font-semibold flex items-center gap-2 whitespace-nowrap">
   406→            <Target className="w-4 h-4 text-amber-500" />
   407→            Recommended For You
   408→          </h3>
   409→        </div>
   410→        <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
   411→          {[1, 2, 3].map((i) => (
   412→            <Skeleton key={i} className="h-28 rounded-xl" />
   413→          ))}
   414→        </div>
   415→      </div>
   416→    );
   417→  }
   418→
   419→  return (
   420→    <div>
   421→      <div className="scroll-section-title">
   422→        <h3 className="text-sm font-semibold flex items-center gap-2 whitespace-nowrap">
   423→          <Target className="w-4 h-4 text-amber-500" />
   424→          Recommended For You
   425→          {!userProfile && <Badge variant="secondary" className="text-[10px]">Based on popularity</Badge>}
   426→          {userProfile && <Badge variant="secondary" className="text-[10px] bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-400">Personalized</Badge>}
   427→        </h3>
   428→      </div>
   429→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
   430→        {recommendations.map((rec, idx) => (
   431→          <motion.div
   432→            key={rec.country.code}
   433→            initial={{ opacity: 0, y: 20 }}
   434→            animate={{ opacity: 1, y: 0 }}
   435→            transition={{ delay: idx * 0.1 }}
   436→            className="glass-section p-4 hover:border-amber-500/30 transition-all cursor-pointer group"
   437→            onClick={() => onSelectCountry(rec.country)}
   438→          >
   439→            <div className="flex items-center gap-2 mb-2">
   440→              <FlagImage code={rec.country.code} flagUrl={rec.country.flagUrl} size={28} />

   480→// ============ ANIMATED COUNTER ============
   481→export function AnimatedCounter({ target, suffix = '', prefix = '' }: { target: number; suffix?: string; prefix?: string }) {
   482→  const [count, setCount] = useState(0);
   483→  const hasAnimated = useRef(false);
   484→
   485→  useEffect(() => {
   486→    if (hasAnimated.current) return;
   487→    hasAnimated.current = true;
   488→    const duration = 2000;
   489→    const start = performance.now();
   490→    const animate = (now: number) => {
   491→      const elapsed = now - start;
   492→      const progress = Math.min(elapsed / duration, 1);
   493→      const eased = 1 - Math.pow(1 - progress, 3);
   494→      setCount(Math.round(eased * target));
   495→      if (progress < 1) requestAnimationFrame(animate);
   496→    };
   497→    requestAnimationFrame(animate);
   498→  }, [target]);
   499→
   500→  return <span>{prefix}{count}{suffix}</span>;
   501→}

   576→export function getTimeOfDay(timezone: string): { label: string; icon: React.ReactNode; color: string } {
   577→  try {
   578→    const now = new Date();
   579→    const formatter = new Intl.DateTimeFormat('en-US', { timeZone: timezone, hour: 'numeric', hour12: false });
   580→    const hour = parseInt(formatter.format(now));
   581→    if (hour >= 6 && hour < 18) return { label: 'Day', icon: <Sun className="w-3 h-3" />, color: 'text-amber-500' };
   582→    if (hour >= 18 && hour < 22) return { label: 'Evening', icon: <Sunset className="w-3 h-3" />, color: 'text-orange-500' };
   583→    return { label: 'Night', icon: <Moon className="w-3 h-3" />, color: 'text-slate-400' };
   584→  } catch {
   585→    return { label: 'Unknown', icon: <Clock className="w-3 h-3" />, color: 'text-muted-foreground' };
   586→  }

   589→/** Hook: returns the current time string (e.g. "3:45 PM") in the given timezone, updates every 60s */
   590→export function useCountryTime(timezone: string) {
   591→  const [time, setTime] = useState('');
   592→  useEffect(() => {
   593→    const tick = () => {
   594→      try {
   595→        const now = new Date();
   596→        setTime(now.toLocaleTimeString('en-US', { timeZone: timezone, hour: 'numeric', minute: '2-digit', hour12: true }));
   597→      } catch { setTime(''); }
   598→    };
   599→    tick();
   600→    const id = setInterval(tick, 30000);
   601→    return () => clearInterval(id);
   602→  }, [timezone]);
   603→  return time;
   604→}

   926→export function ComparisonTable({ results }: { results: ScoreBreakdown[] }) {
   927→  const [countryDetails, setCountryDetails] = useState<Record<string, CountryData>>({});
   928→  const [loading, setLoading] = useState(false);
   929→
   930→  useEffect(() => {
   931→    if (results.length === 0) return;
   932→    let cancelled = false;
   933→    setLoading(true);
   934→    void (async () => {
   935→      try {
   936→        const details = await Promise.all(results.map(r =>
   937→          fetch(`/api/countries/${r.countryCode}`).then(res => res.json()).then(data => data.data).catch(() => null)
   938→        ));
   939→        if (cancelled) return;
   940→        const map: Record<string, CountryData> = {};
   941→        details.forEach((d: CountryData | null, i: number) => {
   942→          if (d) map[results[i].countryCode] = d;
   943→        });
   944→        setCountryDetails(map);
   945→      } finally {
   946→        if (!cancelled) setLoading(false);
   947→      }
   948→    })();
   949→    return () => { cancelled = true; };
   950→  }, [results]);
   951→
   952→  if (results.length < 2 || loading) return null;
   953→
   954→  const rows = [
   955→    { label: 'Visa Type', key: 'visaType' },
   956→    { label: 'Eligibility', key: 'eligibility', isScore: true },
   957→    { label: 'Visa Likelihood', key: 'visaLikelihood', isScore: true },
   958→    { label: 'Cost Suitability', key: 'costSuitability', isScore: true },
   959→    { label: 'Final Score', key: 'finalScore', isScore: true, bold: true },
   960→    { label: 'Confidence', key: 'confidence', isPercent: true },
   961→    { label: 'Processing Time', key: 'processing', isText: true },
   962→    { label: 'Visa Fee', key: 'fee', isText: true },
   963→    { label: 'Monthly Cost', key: 'monthly', isText: true },
   964→    { label: 'Safety Rating', key: 'safety', isText: true },
   965→  ];
   966→
   967→  const getCellValue = (row: typeof rows[0], result: ScoreBreakdown) => {
   968→    if (row.isScore) {
   969→      const val = result[row.key as keyof ScoreBreakdown] as number;
   970→      return { text: `${Math.round(val)}%`, value: val };
   971→    }
   972→    if (row.isPercent) {
   973→      return { text: `${Math.round(result.confidence * 100)}%`, value: result.confidence };
   974→    }
   975→    if (row.isText) {
   976→      const detail = countryDetails[result.countryCode];
   977→      switch (row.key) {
   978→        case 'processing': return { text: `${detail?.processingDaysMin || '?'}-${detail?.processingDaysMax || '?'} days`, value: 0 };
   979→        case 'fee': return { text: `$${detail?.costProfile?.visaFeeUSD || 0}`, value: -(detail?.costProfile?.visaFeeUSD || 0) };
   980→        case 'monthly': return { text: `$${detail?.costProfile?.totalMonthlyUSD || 0}/mo`, value: -(detail?.costProfile?.totalMonthlyUSD || 0) };
   981→        case 'safety': return { text: `${detail?.safetyRating || '?'}/10`, value: detail?.safetyRating || 0 };
   982→        default: return { text: 'N/A', value: 0 };
   983→      }
   984→    }
   985→    return { text: String(result[row.key as keyof ScoreBreakdown] || 'N/A'), value: 0 };
   986→  };
   987→
   988→  const getBestIdx = (row: typeof rows[0]) => {
   989→    if (row.key === 'visaType') return -1;
   990→    const values = results.map(r => getCellValue(row, r));
   991→    // For scores and safety: higher is better; for costs/fees: lower absolute value is best
   992→    if (row.isScore || row.key === 'confidence' || row.key === 'safety') {
   993→      return values.reduce((best, v, i) => v.value > values[best].value ? i : best, 0);
   994→    }
   995→    return values.reduce((best, v, i) => Math.abs(v.value) < Math.abs(values[best].value) ? i : best, 0);
   996→  };
   997→
   998→  return (
   999→    <Card className="premium-card overflow-x-auto">
  1000→      <CardHeader>
  1001→        <CardTitle className="text-base flex items-center gap-2">
  1002→          <Gavel className="w-5 h-5 text-amber-500" />
  1003→          Detailed Comparison Table
  1004→        </CardTitle>
  1005→        <CardDescription>Side-by-side analysis with best values highlighted</CardDescription>
  1006→      </CardHeader>
  1007→      <CardContent className="overflow-x-auto">
  1008→        <Table>
  1009→          <TableHeader>
  1010→            <TableRow>
  1011→              <TableHead className="w-36">Metric</TableHead>
  1012→              {results.map(r => (
  1013→                <TableHead key={r.countryCode} className="text-center">
  1014→                  <div className="flex flex-col items-center gap-0.5">
  1015→                    <span className="text-sm font-semibold">{r.country}</span>
  1016→                    <Badge variant="outline" className="text-[10px]">{r.countryCode}</Badge>
  1017→                  </div>
  1018→                </TableHead>
  1019→              ))}
  1020→            </TableRow>
  1021→          </TableHeader>
  1022→          <TableBody>
  1023→            {rows.map(row => {
  1024→              const bestIdx = getBestIdx(row);
  1025→              return (
  1026→                <TableRow key={row.key}>
  1027→                  <TableCell className={`font-medium text-sm ${row.bold ? 'font-bold' : ''}`}>{row.label}</TableCell>
  1028→                  {results.map((r, i) => {
  1029→                    const cell = getCellValue(row, r);
  1030→                    const isBest = i === bestIdx;
  1031→                    return (
  1032→                      <TableCell key={r.countryCode} className={`text-center text-sm ${isBest ? 'print-best bg-amber-50 dark:bg-amber-900/20 font-semibold text-amber-700 dark:text-amber-400' : ''}`}>
  1033→                        {cell.text}
  1034→                        {isBest && <span className="ml-1">✓</span>}
  1035→                      </TableCell>
  1036→                    );
  1037→                  })}
  1038→                </TableRow>
  1039→              );
  1040→            })}
  1041→          </TableBody>
  1042→        </Table>
  1043→      </CardContent>
  1044→    </Card>
  1045→  );
  1046→}
