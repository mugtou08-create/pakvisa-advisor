   497→export function FloatingChatWidget() {
   498→  const { chatOpen, setChatOpen, chatHistory, addChatMessage } = useAppStore();
   499→  const [input, setInput] = useState('');
   500→  const [sending, setSending] = useState(false);
   501→  const messagesEndRef = useRef<HTMLDivElement>(null);
   502→
   503→  useEffect(() => {
   504→    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
   505→  }, [chatHistory, sending]);
   506→
   507→  const sendMessage = async () => {
   508→    if (!input.trim() || sending) return;
   509→    const msg = input.trim();
   510→    setInput('');
   511→    addChatMessage('user', msg);
   512→    setSending(true);
   513→    try {
   514→      const res = await fetch('/api/chat', {
   515→        method: 'POST',
   516→        headers: { 'Content-Type': 'application/json' },
   517→        body: JSON.stringify({ message: msg, history: chatHistory.slice(-6) }),
   518→      });
   519→      const data = await res.json();
   520→      if (data.success && data.data) {
   521→        addChatMessage('assistant', data.data);
   522→      } else {
   523→        addChatMessage('assistant', data.reply || data.message || data.data || 'Sorry, I could not process that.');
   524→      }
   525→    } catch {
   526→      addChatMessage('assistant', 'Network error. Please try again.');
   527→    }
   528→    setSending(false);
   529→  };
   530→
   531→  return (
   532→    <>
   533→      <button
   534→        className={`chat-widget-btn ${!chatOpen ? 'chat-widget-pulse' : ''} print:hidden`}
   535→        onClick={() => setChatOpen(!chatOpen)}
   536→        aria-label="Toggle AI Chat"
   537→      >
   538→        <Bot className="w-6 h-6" />
   539→      </button>
   540→      <AnimatePresence>

   551→                <Bot className="w-4 h-4" />
   552→                <span className="text-sm font-semibold">Visa AI Assistant</span>

   655→export function BackToTopButton() {
   656→  const [visible, setVisible] = useState(false);
   657→
   658→  useEffect(() => {
   659→    const handleScroll = () => {
   660→      setVisible(window.scrollY > 400);
   661→    };
   662→    window.addEventListener('scroll', handleScroll, { passive: true });
   663→    return () => window.removeEventListener('scroll', handleScroll);
   664→  }, []);
   665→
   666→  const scrollToTop = () => {
   667→    window.scrollTo({ top: 0, behavior: 'smooth' });
   668→  };
   669→
   670→  return (
   671→    <AnimatePresence>
   672→      {visible && (
   673→        <motion.div
   674→          className="fixed bottom-20 sm:bottom-6 right-4 sm:right-6 z-50"
   675→          initial={{ opacity: 0, scale: 0.5 }}
   676→          animate={{ opacity: 1, scale: 1 }}
   677→          exit={{ opacity: 0, scale: 0.5 }}
   678→          transition={{ type: 'spring', stiffness: 300, damping: 20 }}
   679→        >
   680→          <Button
   681→            onClick={scrollToTop}
   682→            size="icon"
   683→            className="rounded-full w-10 h-10 bg-amber-600 hover:bg-amber-700 text-white shadow-lg shadow-amber-600/30"
   684→          >
   685→            <ChevronUp className="w-5 h-5" />
   686→          </Button>
   687→        </motion.div>
   688→      )}
   689→    </AnimatePresence>
   690→  );
   691→}
   692→export function NewsletterInput() {
   693→  const [email, setEmail] = useState('');
   694→  const [subscribed, setSubscribed] = useState(false);
   695→
   696→  const handleSubmit = () => {
   697→    if (!email.trim() || !email.includes('@')) {
   698→      toast.error('Please enter a valid email address');
   699→      return;
   700→    }
   701→    try {
   702→      const existing = JSON.parse(localStorage.getItem('pakvisa-newsletter') || '[]');
   703→      if (existing.includes(email)) {
   704→        toast.info('This email is already subscribed');
   705→        return;
   706→      }
   707→      existing.push(email);
   708→      localStorage.setItem('pakvisa-newsletter', JSON.stringify(existing));
   709→      setSubscribed(true);
   710→      toast.success('Subscribed! You will receive visa updates.');
   711→    } catch {
   712→      toast.error('Failed to subscribe');
   713→    }
   714→  };
   715→
   716→  if (subscribed) {
   717→    return (
   718→      <div className="flex items-center gap-2 text-xs text-amber-600 dark:text-amber-400">
   719→        <CheckCircle2 className="w-3.5 h-3.5" />
   720→        <span>Subscribed successfully!</span>
   721→        <button onClick={() => { setSubscribed(false); setEmail(''); }} className="text-muted-foreground hover:text-foreground underline ml-1">Change</button>
   722→      </div>
   723→    );
   724→  }
   725→
   726→  return (
   727→    <>
   728→      <Input
   729→        placeholder="your@email.com"
   730→        className="h-9 text-xs input-glow-focus flex-1"
   731→        value={email}
   732→        onChange={(e) => setEmail(e.target.value)}
   733→        onKeyDown={(e) => e.key === 'Enter' && handleSubmit()}

   735→      <Button
   736→        size="sm"
