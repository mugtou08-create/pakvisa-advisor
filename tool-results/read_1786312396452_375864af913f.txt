   287→                  </div>
   288→                ))}

   293→                </div>

   300→          <ComparisonTable results={results} />

   303→    </div>

   305→}

   307→export function WhatIfSimulator({ scoreResult, userProfile }: { scoreResult: ScoreBreakdown; userProfile: UserProfileData }) {
   308→  const [healthInsurance, setHealthInsurance] = useState(userProfile.hasHealthInsurance);
   309→  const [returnTicket, setReturnTicket] = useState(userProfile.hasReturnTicket);
   310→  const [hotelBooking, setHotelBooking] = useState(userProfile.hasHotelBooking);
   311→  const [sponsor, setSponsor] = useState(userProfile.hasSponsor);
   312→  const [priorTravel, setPriorTravel] = useState(userProfile.hasPriorTravel);
   313→  const [income, setIncome] = useState(userProfile.monthlyIncomeUSD);
   314→  const [savings, setSavings] = useState(userProfile.savingsUSD);
   315→  const [budget, setBudget] = useState(userProfile.budgetUSD);
   316→  const [recalculating, setRecalculating] = useState(false);
   317→  const [afterScore, setAfterScore] = useState<ScoreBreakdown | null>(null);
   318→
   319→  const applyChanges = async () => {
   320→    setRecalculating(true);
   321→    const modifiedProfile = {
   322→      ...userProfile,
   323→      hasHealthInsurance: healthInsurance,
   324→      hasReturnTicket: returnTicket,
   325→      hasHotelBooking: hotelBooking,
   326→      hasSponsor: sponsor,
   327→      hasPriorTravel: priorTravel,
   328→      monthlyIncomeUSD: income,
   329→      savingsUSD: savings,
   330→      budgetUSD: budget,
   331→    };
   332→    try {
   333→      const res = await fetch('/api/score', {
   334→        method: 'POST',
   335→        headers: { 'Content-Type': 'application/json' },
   336→        body: JSON.stringify({ countryCode: scoreResult.countryCode, profile: modifiedProfile }),
   337→      });
   338→      const data = await res.json();
   339→      if (data.data) {
   340→        setAfterScore(data.data);
   341→      } else {
   342→        toast.error(data.error || 'Recalculation failed');
   343→      }
   344→    } catch {
   345→      toast.error('Network error during recalculation');
   346→    }
   347→    setRecalculating(false);
   348→  };
   349→
   350→  const delta = afterScore ? afterScore.finalScore - scoreResult.finalScore : 0;
   351→
   352→  return (
   353→    <div className="space-y-4">
   354→      <div className="grid grid-cols-1 md:grid-cols-2 gap-4 responsive-stack-mobile">
   355→        {/* Toggle Switches */}
   356→        <div className="space-y-3">
   357→          <h5 className="text-xs font-semibold text-muted-foreground uppercase tracking-wider">Quick Actions</h5>
   358→          {[
   359→            { label: 'Add Health Insurance', desc: '+5% health component', checked: healthInsurance, onChange: setHealthInsurance, icon: Shield },
   360→            { label: 'Book Return Ticket', desc: '+15% purpose component', checked: returnTicket, onChange: setReturnTicket, icon: Plane },
   361→            { label: 'Book Hotel', desc: '+10% purpose component', checked: hotelBooking, onChange: setHotelBooking, icon: Home },
   362→            { label: 'Add Financial Sponsor', desc: '+20% funds component', checked: sponsor, onChange: setSponsor, icon: Users },
   363→            { label: 'Have Prior Travel', desc: '+10% travel history', checked: priorTravel, onChange: setPriorTravel, icon: Globe },
   364→          ].map((item) => (
   365→            <div key={item.label} className="flex items-center justify-between p-2.5 rounded-lg border hover:bg-muted/50 transition-colors">
   366→              <div className="flex items-center gap-2.5">
   367→                <item.icon className="w-4 h-4 text-muted-foreground" />
   368→                <div>
   369→                  <div className="text-sm font-medium">{item.label}</div>
   370→                  <div className="text-[10px] text-amber-600 dark:text-amber-400">{item.desc}</div>
   371→                </div>
   372→              </div>
   373→              <Switch checked={item.checked} onCheckedChange={item.onChange} />
   374→            </div>
   375→          ))}
   376→        </div>
   377→
   378→        {/* Sliders */}
   379→        <div className="space-y-4">
   380→          <h5 className="text-xs font-semibold text-muted-foreground uppercase tracking-wider">Financial Adjustments</h5>
   381→          <div className="space-y-2">
   382→            <div className="flex items-center justify-between text-sm">
   383→              <Label className="text-muted-foreground">Monthly Income</Label>
   384→              <span className="font-mono text-xs font-medium">${income.toLocaleString()}</span>
   385→            </div>
   386→            <Slider value={[income]} onValueChange={(v) => setIncome(v[0])} min={0} max={10000} step={100} />
   387→          </div>
   388→          <div className="space-y-2">
   389→            <div className="flex items-center justify-between text-sm">
   390→              <Label className="text-muted-foreground">Savings</Label>
   391→              <span className="font-mono text-xs font-medium">${savings.toLocaleString()}</span>
   392→            </div>
   393→            <Slider value={[savings]} onValueChange={(v) => setSavings(v[0])} min={0} max={100000} step={1000} />
   394→          </div>
   395→          <div className="space-y-2">
   396→            <div className="flex items-center justify-between text-sm">
   397→              <Label className="text-muted-foreground">Travel Budget</Label>
   398→              <span className="font-mono text-xs font-medium">${budget.toLocaleString()}</span>
   399→            </div>
   400→            <Slider value={[budget]} onValueChange={(v) => setBudget(v[0])} min={0} max={50000} step={500} />
   401→          </div>
   402→        </div>
   403→      </div>
   404→
   405→      {/* Apply Button */}
   406→      <Button
   407→        onClick={applyChanges}
   408→        disabled={recalculating}
   409→        className="w-full bg-amber-600 hover:bg-amber-700 text-white"
   410→        size="lg"
   411→      >
   412→        {recalculating ? (
   413→          <><RotateCcw className="w-4 h-4 mr-2 animate-spin" /> Recalculating...</>
   414→        ) : (
   415→          <><Zap className="w-4 h-4 mr-2" /> Apply Changes & Recalculate</>
   416→        )}
   417→      </Button>
   418→
   419→      {/* Comparison Result */}
   420→      {afterScore && (
   421→        <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }}>
   422→          <div className="p-4 rounded-lg border bg-muted/30 space-y-3">
   423→            <h5 className="text-sm font-semibold">Score Comparison</h5>
   424→            <div className="flex items-center justify-center gap-4">
   425→              <div className="text-center">
   426→                <div className="text-xs text-muted-foreground mb-1">Before</div>
   427→                <div className="text-3xl font-bold">{Math.round(scoreResult.finalScore)}</div>
   428→              </div>
   429→              <div className="flex flex-col items-center">
   430→                <ArrowRight className={`w-6 h-6 ${delta >= 0 ? 'text-amber-500' : 'text-red-500'}`} />
   431→                <span className={`text-sm font-bold ${delta >= 0 ? 'text-amber-600' : 'text-red-600'}`}>
   432→                  {delta >= 0 ? '+' : ''}{delta.toFixed(1)}
   433→                </span>
   434→              </div>
   435→              <div className="text-center">
   436→                <div className="text-xs text-muted-foreground mb-1">After</div>
   437→                <div className={`text-3xl font-bold ${delta >= 0 ? 'text-amber-600' : 'text-red-600'}`}>
   438→                  {Math.round(afterScore.finalScore)}
   439→                </div>
   440→              </div>
   441→            </div>
   442→            {/* Component-level comparison */}
   443→            <div className="space-y-1.5 pt-2 border-t">
   444→              {[
   445→                { label: 'Eligibility', before: scoreResult.eligibility, after: afterScore.eligibility },
   446→                { label: 'Visa Likelihood', before: scoreResult.visaLikelihood, after: afterScore.visaLikelihood },
   447→                { label: 'Cost Suitability', before: scoreResult.costSuitability, after: afterScore.costSuitability },
   448→              ].map((item) => {
   449→                const diff = item.after - item.before;
   450→                return (
   451→                  <div key={item.label} className="flex items-center gap-2 text-sm">
   452→                    <span className="w-28 text-muted-foreground text-xs">{item.label}</span>
   453→                    <span className="w-12 text-right font-mono text-xs">{Math.round(item.before)}</span>
   454→                    <span className={`text-xs font-medium ${diff >= 0 ? 'text-amber-600' : 'text-red-600'} w-12 text-center`}>
   455→                      {diff >= 0 ? '+' : ''}{diff.toFixed(1)}
   456→                    </span>
   457→                    <span className="w-12 text-right font-mono text-xs font-medium">{Math.round(item.after)}</span>
   458→                  </div>
   459→                );
   460→              })}
   461→            </div>
   462→            {delta >= 0 ? (
   463→              <p className="text-xs text-amber-600 text-center">
   464→                <CheckCircle2 className="w-3 h-3 inline mr-1" />
   465→                Changes improved your score! Consider applying these in your application.
   466→              </p>
   467→            ) : (
   468→              <p className="text-xs text-red-600 text-center">
   469→                <AlertTriangle className="w-3 h-3 inline mr-1" />
   470→                These changes would lower your score. Reconsider before proceeding.
   471→              </p>
   472→            )}
   473→          </div>
   474→        </motion.div>
   475→      )}
   476→    </div>
   477→  );
   478→}

   481→export function SmartChecklist({ scoreResult }: { scoreResult: ScoreBreakdown }) {
   482→  const [checked, setChecked] = useState<Set<string>>(new Set());
   483→  const checklistRef = useRef<HTMLDivElement>(null);
   484→
   485→  const items = useMemo(() => {
   486→    const generated: ChecklistItem[] = [];
   487→
   488→    // From hard filter failures - high priority documents
   489→    scoreResult.hardFilters.filter(f => !f.passed).forEach((f) => {
   490→      generated.push({
   491→        task: `Resolve: ${f.filter} - ${f.message}`,
   492→        priority: 'high',
   493→        effort: f.severity === 'critical' ? 'time-consuming' : 'moderate',
   494→        completed: false,
   495→        category: 'Documents',
   496→        notes: f.severity,
   497→      });
   498→    });
   499→
   500→    // From missing items
   501→    scoreResult.missingItems.forEach((item, i) => {
   502→      generated.push({
   503→        task: item,
   504→        priority: i < 3 ? 'high' : i < 6 ? 'medium' : 'low',
   505→        effort: i < 2 ? 'quick' : i < 5 ? 'moderate' : 'time-consuming',
   506→        completed: false,
   507→        category: 'Documents',
   508→      });
   509→    });
   510→
   511→    // Financial items from tips
   512→    const financialTips = scoreResult.tips.filter(t => /income|fund|bank|financial|money|salary|savings/i.test(t));
   513→    financialTips.forEach((tip) => {
   514→      generated.push({
   515→        task: tip,
   516→        priority: 'high',
   517→        effort: 'moderate',
   518→        completed: false,
   519→        category: 'Financial',
   520→      });
   521→    });
   522→
   523→    // Travel items from tips
   524→    const travelTips = scoreResult.tips.filter(t => /ticket|book|flight|hotel|travel|itinerary/i.test(t));
   525→    travelTips.forEach((tip) => {
   526→      generated.push({
   527→        task: tip,
   528→        priority: 'medium',
   529→        effort: 'quick',
   530→        completed: false,
   531→        category: 'Travel',
   532→      });
   533→    });
   534→
   535→    // Health items
   536→    if (scoreResult.components.find(c => /health|insurance/i.test(c.name))?.score < 70) {
   537→      generated.push({
   538→        task: 'Purchase travel health insurance covering your entire stay duration',
   539→        priority: 'high',
   540→        effort: 'quick',
   541→        completed: false,
   542→        category: 'Health',
   543→      });
   544→    }
   545→
   546→    // Remaining tips
   547→    const usedTips = new Set([...financialTips, ...travelTips]);
   548→    scoreResult.tips.filter(t => !usedTips.has(t)).forEach((tip) => {
   549→      generated.push({
   550→        task: tip,
   551→        priority: 'low',
   552→        effort: 'moderate',
   553→        completed: false,
   554→        category: 'Tips',
   555→      });
   556→    });
   557→
   558→    // General documents
   559→    generated.push(
   560→      { task: 'Ensure passport validity extends 6+ months beyond travel dates', priority: 'high', effort: 'quick', completed: false, category: 'Documents' },
   561→      { task: 'Get passport-sized photographs (as per destination requirements)', priority: 'medium', effort: 'quick', completed: false, category: 'Documents' },
   562→      { task: 'Prepare bank statements for last 6 months', priority: 'high', effort: 'moderate', completed: false, category: 'Financial' },
   563→      { task: 'Book refundable accommodation for visa application', priority: 'medium', effort: 'quick', completed: false, category: 'Travel' },
   564→      { task: 'Create a day-by-day travel itinerary', priority: 'low', effort: 'time-consuming', completed: false, category: 'Travel' },
   565→    );
   566→
   567→    return generated;
   568→  }, [scoreResult]);
   569→
   570→  const toggleCheck = (idx: number) => {
   571→    setChecked(prev => {
   572→      const next = new Set(prev);
   573→      if (next.has(String(idx))) next.delete(String(idx));
   574→      else next.add(String(idx));
   575→      return next;
   576→    });
   577→  };
   578→
   579→  const progress = items.length > 0 ? (checked.size / items.length) * 100 : 0;
   580→
   581→  const exportChecklist = () => {
   582→    const printWin = window.open('', '_blank');
   583→    if (!printWin) return;
   584→    const categories = ['Documents', 'Financial', 'Travel', 'Health', 'Tips'];
   585→    let html = `<html><head><title>Visa Checklist - ${scoreResult.country}</title>
   586→      <style>body{font-family:system-ui;max-width:700px;margin:40px auto;padding:20px;color:#1a1a1a}
   587→      h1{color:#d97706}h2{color:#374151;margin-top:24px;border-bottom:2px solid #e5e7eb;padding-bottom:4px}
   588→      .item{display:flex;align-items:center;gap:10px;padding:8px 0;border-bottom:1px solid #f3f4f6}
   589→      .item input{width:18px;height:18px}.priority{font-size:11px;padding:2px 8px;border-radius:9999px;font-weight:600}
   590→      .high{background:#fef2f2;color:#dc2626}.medium{background:#fffbeb;color:#d97706}.low{background:#f0fdf4;color:#16a34a}
   591→      .effort{font-size:10px;color:#6b7280;margin-left:8px}</style></head><body>`;
   592→    html += `<h1>Visa Checklist: ${scoreResult.country}</h1>`;
   593→    html += `<p>Generated by PakVisa Advisor · Score: ${Math.round(scoreResult.finalScore)}/100</p>`;
   594→    categories.forEach(cat => {
   595→      const catItems = items.filter(it => it.category === cat);
   596→      if (catItems.length === 0) return;
   597→      html += `<h2>${cat}</h2>`;
   598→      catItems.forEach((it, i) => {
   599→        const realIdx = items.indexOf(it);

   606→    printWin.print();
   607→  };

   609→  const categoryIcons: Record<string, React.ReactNode> = {

   611→    Financial: <DollarSign className="w-4 h-4" />,

   614→    Tips: <Lightbulb className="w-4 h-4" />,
   615→  };
