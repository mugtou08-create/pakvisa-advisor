   118→function getVisaLabel(country: { visaFree: boolean; visaOnArrival: boolean; etaAvailable: boolean }): string {
   119→  if (country.visaFree) return 'Visa Free';
   120→  if (country.visaOnArrival) return 'Visa on Arrival';
   121→  if (country.etaAvailable) return 'e-Visa';
   122→  return 'Embassy Required';

   125→function getVisaBadgeClass(country: { visaFree: boolean; visaOnArrival: boolean; etaAvailable: boolean }): string {
   126→  if (country.visaFree) return 'bg-emerald-100 text-emerald-700';
   127→  if (country.visaOnArrival) return 'bg-amber-100 text-amber-700';
   128→  if (country.etaAvailable) return 'bg-sky-100 text-sky-700';
   129→  return 'bg-gray-100 text-gray-600';
   130→}

   132→function getVisaDotClass(country: { visaFree: boolean; visaOnArrival: boolean; etaAvailable: boolean }): string {

   134→  if (country.visaOnArrival) return 'bg-amber-500';

   139→function usdToPkr(usd: number): string {
   140→  return Math.round(usd * 278.5).toLocaleString();

   143→function safetyStars(rating: number): string {
   144→  const clamped = Math.min(Math.max(rating, 0), 5);
   145→  return '\u2605'.repeat(clamped) + '\u2606'.repeat(5 - clamped);

   201→export async function generateMetadata({ params }: { params: Promise<{ slug: string }> }): Promise<Metadata> {
   202→  const { slug } = await params;
   203→  const code = SLUG_TO_CODE[slug];
   204→  if (!code) return { title: 'Country Not Found | PakVisa Advisor' };
   205→
   206→  const country = await db.country.findUnique({
   207→    where: { code },
   208→    include: { costProfiles: true, visaTypes: true },
   209→  });
   210→
   211→  if (!country) return { title: 'Country Not Found | PakVisa Advisor' };
   212→
   213→  const visaLabel = getVisaLabel(country);
   214→  const costProfile = country.costProfiles[0];
   215→  const visaFee = costProfile ? `$${costProfile.visaFeeUSD} (\u2248 PKR ${usdToPkr(costProfile.visaFeeUSD)})` : 'Free';
   216→  const processing = country.visaFree ? 'No processing required' : `${country.processingDaysMin}-${country.processingDaysMax} business days`;
   217→  const title = `${country.name} Visa for Pakistani Citizens - Requirements, Fees & Guide 2026`;
   218→  const description = `Complete ${country.name} visa guide for Pakistani passport holders. ${visaLabel} access, ${visaFee} fee, ${processing} processing. Updated for 2026 with requirements, costs, and embassy details.`;

   220→  return {
   221→    title,
   222→    description,
   223→    keywords: [

   226→      `${country.name} ${visaLabel.toLowerCase()} Pakistan`,

   234→      type: 'article',

   798→function generateFAQs(
   799→  country: {
   800→    name: string; code: string; visaFree: boolean; visaOnArrival: boolean; etaAvailable: boolean;
   801→    processingDaysMin: number; processingDaysMax: number; safetyRating: number; safetySummary: string;
   802→    bestTravelMonths: string; avgTempC: string; currency: string; currencyCode: string;
   803→  },
   804→  costProfile: CostProfileData | undefined,
   805→  visaLabel: string,
   806→  isEmbassyRequired: boolean,
   807→): { q: string; a: string }[] {
   808→  const faqs: { q: string; a: string }[] = [];
   809→
   810→  // Q1: Do Pakistani citizens need a visa?
   811→  faqs.push({
   812→    q: `Do Pakistani citizens need a visa for ${country.name}?`,
   813→    a: country.visaFree
   814→      ? `No, Pakistani citizens do not need a visa to visit ${country.name}. You can travel visa-free with a valid Pakistani passport. Make sure your passport has at least 6 months of validity remaining.`
   815→      : country.visaOnArrival
   816→        ? `Pakistani citizens can get a visa on arrival at ${country.name}. Simply present your valid Pakistani passport at the immigration counter upon arrival. Processing is usually instant.`
   817→        : country.etaAvailable
   818→          ? `Yes, Pakistani citizens need an e-Visa to visit ${country.name}. The e-Visa can be applied for online through the official government portal. Processing typically takes ${country.processingDaysMin}-${country.processingDaysMax} business days.`
   819→          : `Yes, Pakistani citizens need a visa to visit ${country.name}. You must apply at the embassy or consulate. The application process typically takes ${country.processingDaysMin}-${country.processingDaysMax} business days.`,
   820→  });
   821→
   822→  // Q2: What is the visa fee?
   823→  const feeText = costProfile && costProfile.visaFeeUSD > 0
   824→    ? `The visa fee for ${country.name} is approximately $${costProfile.visaFeeUSD} USD (\u2248 PKR ${usdToPkr(costProfile.visaFeeUSD)}). Additional service fees of $${costProfile.serviceFeeUSD} may apply.`
   825→    : `There is no visa fee for Pakistani citizens visiting ${country.name} as it is ${visaLabel.toLowerCase()}.`
   826→  ;
   827→  faqs.push({
   828→    q: `What is the visa fee for ${country.name} for Pakistani citizens?`,
   829→    a: feeText,
   830→  });
   831→
   832→  // Q3: Processing time
   833→  faqs.push({
   834→    q: `How long does it take to get a ${country.name} visa?`,
   835→    a: country.visaFree
   836→      ? `No processing time is needed as ${country.name} offers visa-free entry to Pakistani passport holders. You can travel immediately.`
   837→      : country.visaOnArrival
   838→        ? `The visa on arrival for ${country.name} is processed instantly at the airport or border crossing. The entire process typically takes 5-15 minutes.`
   839→        : `${country.name} visa processing typically takes ${country.processingDaysMin}-${country.processingDaysMax} business days for Pakistani citizens. It is recommended to apply at least 2-3 weeks before your planned travel date.`,
   840→  });
   841→
   842→  // Q4: Safety
   843→  faqs.push({
   844→    q: `Is ${country.name} safe for Pakistani tourists?`,
   845→    a: country.safetySummary
   846→      ? `${country.name} has a safety rating of ${Math.min(country.safetyRating, 5)}/5. ${country.safetySummary}`
   847→      : `${country.name} has a safety rating of ${Math.min(country.safetyRating, 5)}/5. As with any travel destination, exercise standard precautions, keep your belongings secure, and follow local laws and customs.`,
   848→  });
   849→
   850→  // Q5: Best time to visit or documents required
   851→  if (country.bestTravelMonths) {
   852→    faqs.push({
   853→      q: `What is the best time to visit ${country.name} from Pakistan?`,
   854→      a: `The best months to visit ${country.name} are ${country.bestTravelMonths}. The average temperature is around ${country.avgTempC}\u00B0C. The local currency is ${country.currency} (${country.currencyCode}). Plan your trip during these months for the most comfortable weather conditions.`,
   855→    });
   856→  } else {
   857→    faqs.push({
   858→      q: `What documents do I need for a ${country.name} visa?`,
   859→      a: `Common requirements for a ${country.name} visa include: a valid Pakistani passport with 6+ months validity, passport-sized photographs, proof of accommodation, return flight ticket, travel insurance, and proof of sufficient funds. For the complete and most up-to-date list, check the official embassy website or use PakVisa Advisor.`,
   860→    });
   861→  }
   862→
   863→  return faqs;
   864→}
