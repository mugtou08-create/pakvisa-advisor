  1622→// ============ APPLICATION TIMELINE TRACKER ============
  1623→export function ApplicationTimelineTracker() {
  1624→  const [checkedStages, setCheckedStages] = useState<Set<string>>(new Set());
  1625→  const [checkedItems, setCheckedItems] = useState<Set<string>>(new Set());
  1626→
  1627→  const toggleStage = (stageId: string) => {
  1628→    setCheckedStages(prev => {
  1629→      const next = new Set(prev);
  1630→      if (next.has(stageId)) next.delete(stageId);
  1631→      else next.add(stageId);
  1632→      return next;
  1633→    });
  1634→  };
  1635→
  1636→  const toggleItem = (key: string) => {
  1637→    setCheckedItems(prev => {
  1638→      const next = new Set(prev);
  1639→      if (next.has(key)) next.delete(key);
  1640→      else next.add(key);
  1641→      return next;
  1642→    });
  1643→  };
  1644→
  1645→  const totalItems = TIMELINE_STAGES.reduce((sum, s) => sum + s.items.length, 0);
  1646→  const completedItems = checkedItems.size;
  1647→  const progress = totalItems > 0 ? (completedItems / totalItems) * 100 : 0;
  1648→
  1649→  return (
