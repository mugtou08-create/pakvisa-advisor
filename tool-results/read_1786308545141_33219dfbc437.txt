   247→export function WhatIfSimulator({ scoreResult, userProfile }: { scoreResult: ScoreBreakdown; userProfile: UserProfileData }) {
   248→  const [healthInsurance, setHealthInsurance] = useState(userProfile.hasHealthInsurance);
   249→  const [returnTicket, setReturnTicket] = useState(userProfile.hasReturnTicket);
   250→  const [hotelBooking, setHotelBooking] = useState(userProfile.hasHotelBooking);
   251→  const [sponsor, setSponsor] = useState(userProfile.hasSponsor);
   252→  const [priorTravel, setPriorTravel] = useState(userProfile.hasPriorTravel);
   253→  const [income, setIncome] = useState(userProfile.monthlyIncomeUSD);
   254→  const [savings, setSavings] = useState(userProfile.savingsUSD);
   255→  const [budget, setBudget] = useState(userProfile.budgetUSD);
   256→  const [recalculating, setRecalculating] = useState(false);
   257→  const [afterScore, setAfterScore] = useState<ScoreBreakdown | null>(null);
   258→
   259→  const applyChanges = async () => {
   260→    setRecalculating(true);
   261→    const modifiedProfile = {
   262→      ...userProfile,
   263→      hasHealthInsurance: healthInsurance,
   264→      hasReturnTicket: returnTicket,
   265→      hasHotelBooking: hotelBooking,
   266→      hasSponsor: sponsor,
   267→      hasPriorTravel: priorTravel,
   268→      monthlyIncomeUSD: income,
   269→      savingsUSD: savings,
   270→      budgetUSD: budget,
   271→    };
   272→    try {
   273→      const res = await fetch('/api/score', {
   274→        method: 'POST',
   275→        headers: { 'Content-Type': 'application/json' },
   276→        body: JSON.stringify({ countryCode: scoreResult.countryCode, profile: modifiedProfile }),
   277→      });
   278→      const data = await res.json();
   279→      if (data.data) {
   280→        setAfterScore(data.data);
   281→      } else {
   282→        toast.error(data.error || 'Recalculation failed');
   283→      }
   284→    } catch {
   285→      toast.error('Network error during recalculation');
   286→    }
   287→    setRecalculating(false);
   288→  };
   289→
   290→  const delta = afterScore ? afterScore.finalScore - scoreResult.finalScore : 0;
   291→
   292→  return (
   293→    <div className="space-y-4">
   294→      <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   295→        {/* Toggle Switches */}
   296→        <div className="space-y-3">
   297→          <h5 className="text-xs font-semibold text-muted-foreground uppercase tracking-wider">Quick Actions</h5>
   298→          {[
   299→            { label: 'Add Health Insurance', desc: '+5% health component', checked: healthInsurance, onChange: setHealthInsurance, icon: Shield },
   300→            { label: 'Book Return Ticket', desc: '+15% purpose component', checked: returnTicket, onChange: setReturnTicket, icon: Plane },
   301→            { label: 'Book Hotel', desc: '+10% purpose component', checked: hotelBooking, onChange: setHotelBooking, icon: Home },
   302→            { label: 'Add Financial Sponsor', desc: '+20% funds component', checked: sponsor, onChange: setSponsor, icon: Users },
   303→            { label: 'Have Prior Travel', desc: '+10% travel history', checked: priorTravel, onChange: setPriorTravel, icon: Globe },
   304→          ].map((item) => (
   305→            <div key={item.label} className="flex items-center justify-between p-2.5 rounded-lg border hover:bg-muted/50 transition-colors">
   306→              <div className="flex items-center gap-2.5">
   307→                <item.icon className="w-4 h-4 text-muted-foreground" />
   308→                <div>
   309→                  <div className="text-sm font-medium">{item.label}</div>
   310→                  <div className="text-[10px] text-amber-600 dark:text-amber-400">{item.desc}</div>
   311→                </div>
   312→              </div>
   313→              <Switch checked={item.checked} onCheckedChange={item.onChange} />
   314→            </div>
   315→          ))}
   316→        </div>
   317→
   318→        {/* Sliders */}
   319→        <div className="space-y-4">
   320→          <h5 className="text-xs font-semibold text-muted-foreground uppercase tracking-wider">Financial Adjustments</h5>
   321→          <div className="space-y-2">
   322→            <div className="flex items-center justify-between text-sm">
   323→              <Label className="text-muted-foreground">Monthly Income</Label>
   324→              <span className="font-mono text-xs font-medium">${income.toLocaleString()}</span>
   325→            </div>
   326→            <Slider value={[income]} onValueChange={(v) => setIncome(v[0])} min={0} max={10000} step={100} />
   327→          </div>
   328→          <div className="space-y-2">
   329→            <div className="flex items-center justify-between text-sm">
   330→              <Label className="text-muted-foreground">Savings</Label>
   331→              <span className="font-mono text-xs font-medium">${savings.toLocaleString()}</span>

   334→          </div>

   336→            <div className="flex items-center justify-between text-sm">

   351→      >

   353→          <><RotateCcw className="w-4 h-4 mr-2 animate-spin" /> Recalculating...</>
   354→        ) : (
   355→          <><Zap className="w-4 h-4 mr-2" /> Apply Changes & Recalculate</>
   356→        )}
   357→      </Button>
   358→
   359→      {/* Comparison Result */}
   360→      {afterScore && (
   361→        <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }}>
   362→          <div className="p-4 rounded-lg border bg-muted/30 space-y-3">
   363→            <h5 className="text-sm font-semibold">Score Comparison</h5>
   364→            <div className="flex items-center justify-center gap-4">
   365→              <div className="text-center">
   366→                <div className="text-xs text-muted-foreground mb-1">Before</div>
   367→                <div className="text-3xl font-bold">{Math.round(scoreResult.finalScore)}</div>
   368→              </div>
   369→              <div className="flex flex-col items-center">
   370→                <ArrowRight className={`w-6 h-6 ${delta >= 0 ? 'text-amber-500' : 'text-red-500'}`} />
   371→                <span className={`text-sm font-bold ${delta >= 0 ? 'text-amber-600' : 'text-red-600'}`}>

   378→                  {Math.round(afterScore.finalScore)}
   379→                </div>
   380→              </div>
   381→            </div>
   382→            {/* Component-level comparison */}
   383→            <div className="space-y-1.5 pt-2 border-t">
   384→              {[
   385→                { label: 'Eligibility', before: scoreResult.eligibility, after: afterScore.eligibility },
   386→                { label: 'Visa Likelihood', before: scoreResult.visaLikelihood, after: afterScore.visaLikelihood },
   387→                { label: 'Cost Suitability', before: scoreResult.costSuitability, after: afterScore.costSuitability },
   388→              ].map((item) => {
   389→                const diff = item.after - item.before;
   390→                return (
   391→                  <div key={item.label} className="flex items-center gap-2 text-sm">
   392→                    <span className="w-28 text-muted-foreground text-xs">{item.label}</span>
   393→                    <span className="w-12 text-right font-mono text-xs">{Math.round(item.before)}</span>
   394→                    <span className={`text-xs font-medium ${diff >= 0 ? 'text-amber-600' : 'text-red-600'} w-12 text-center`}>
   395→                      {diff >= 0 ? '+' : ''}{diff.toFixed(1)}
   396→                    </span>
   397→                    <span className="w-12 text-right font-mono text-xs font-medium">{Math.round(item.after)}</span>
   398→                  </div>
   399→                );
   400→              })}
   401→            </div>
   402→            {delta >= 0 ? (
   403→              <p className="text-xs text-amber-600 text-center">
   404→                <CheckCircle2 className="w-3 h-3 inline mr-1" />
   405→                Changes improved your score! Consider applying these in your application.
   406→              </p>
   407→            ) : (
   408→              <p className="text-xs text-red-600 text-center">
   409→                <AlertTriangle className="w-3 h-3 inline mr-1" />
   410→                These changes would lower your score. Reconsider before proceeding.
   411→              </p>
   412→            )}
   413→          </div>
   414→        </motion.div>
   415→      )}
   416→    </div>
   417→  );
   418→}

   420→// ============ SMART CHECKLIST ============
   421→export function SmartChecklist({ scoreResult }: { scoreResult: ScoreBreakdown }) {
   422→  const [checked, setChecked] = useState<Set<string>>(new Set());
   423→  const checklistRef = useRef<HTMLDivElement>(null);
   424→
   425→  const items = useMemo(() => {
   426→    const generated: ChecklistItem[] = [];
   427→
   428→    // From hard filter failures - high priority documents
   429→    scoreResult.hardFilters.filter(f => !f.passed).forEach((f) => {
   430→      generated.push({
   431→        task: `Resolve: ${f.filter} - ${f.message}`,
   432→        priority: 'high',
   433→        effort: f.severity === 'critical' ? 'time-consuming' : 'moderate',
   434→        completed: false,
   435→        category: 'Documents',
   436→        notes: f.severity,
   437→      });
   438→    });
   439→
   440→    // From missing items
   441→    scoreResult.missingItems.forEach((item, i) => {
   442→      generated.push({
   443→        task: item,
   444→        priority: i < 3 ? 'high' : i < 6 ? 'medium' : 'low',
   445→        effort: i < 2 ? 'quick' : i < 5 ? 'moderate' : 'time-consuming',
   446→        completed: false,
   447→        category: 'Documents',
   448→      });
   449→    });
   450→
   451→    // Financial items from tips
