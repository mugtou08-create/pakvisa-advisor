    28→function getVisaStatus(c: CountryData) {
    29→  if (c.visaFree) return { label: 'Visa Free', color: 'bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-400' };
    30→  if (c.visaOnArrival) return { label: 'Visa on Arrival', color: 'bg-orange-100 text-orange-700 dark:bg-orange-900/40 dark:text-orange-400' };
    31→  if (c.etaAvailable) return { label: 'e-Visa', color: 'bg-amber-50 text-amber-600 dark:bg-amber-900/30 dark:text-amber-400' };
    32→  return { label: 'Embassy Required', color: 'bg-orange-100 text-orange-700 dark:bg-orange-900/40 dark:text-orange-400' };
    33→}

    48→export function CountryComparisonSwiper({ countries, onSelectCountry }: {
    49→  countries: CountryData[];
    50→  onSelectCountry: (code: string) => void;
    51→}) {
    52→  const [currentIndex, setCurrentIndex] = useState(0);
    53→  const [isPaused, setIsPaused] = useState(false);
    54→  const [direction, setDirection] = useState(0);
    55→  const dragRef = useRef<HTMLDivElement>(null);
    56→  const startXRef = useRef(0);
    57→
    58→  // Take top 8 countries for the swiper
    59→  const swiperCountries = countries.slice(0, 8);
    60→  const total = swiperCountries.length;
    61→
    62→  const goTo = useCallback((newIndex: number, dir: number) => {
    63→    setDirection(dir);
    64→    setCurrentIndex((newIndex + total) % total);
    65→  }, [total]);
    66→
    67→  const goNext = useCallback(() => goTo(currentIndex + 1, 1), [currentIndex, goTo]);
    68→  const goPrev = useCallback(() => goTo(currentIndex - 1, -1), [currentIndex, goTo]);
    69→
    70→  // Auto-rotate every 5 seconds
    71→  useEffect(() => {
    72→    if (isPaused || total === 0) return;
    73→    const timer = setInterval(goNext, 5000);
    74→    return () => clearInterval(timer);
    75→  }, [isPaused, goNext, total]);
    76→
    77→  // Touch/drag handlers
    78→  const handleDragStart = (e: React.MouseEvent | React.TouchEvent) => {
    79→    const clientX = 'touches' in e ? e.touches[0].clientX : e.clientX;
    80→    startXRef.current = clientX;
    81→  };
    82→
    83→  const handleDragEnd = (e: React.MouseEvent | React.TouchEvent) => {
    84→    const clientX = 'changedTouches' in e ? e.changedTouches[0].clientX : e.clientX;
    85→    const diff = startXRef.current - clientX;
    86→    if (Math.abs(diff) > 50) {
    87→      if (diff > 0) goNext();
    88→      else goPrev();
    89→    }
    90→  };
    91→
    92→  if (swiperCountries.length === 0) return null;
    93→
    94→  // Get visible indices for 3-card desktop, 1-card mobile
    95→  const getVisibleIndices = () => {
    96→    const indices: number[] = [];
    97→    for (let offset = -1; offset <= 1; offset++) {
    98→      indices.push((currentIndex + offset + total) % total);
    99→    }
   100→    return indices;
   101→  };
   102→
   103→  const visibleIndices = getVisibleIndices();
   104→  const centerIdx = visibleIndices[1]; // center card
   105→  const centerCountry = swiperCountries[centerIdx];
   106→
   107→  return (
   108→    <div
   109→      className="relative overflow-hidden"
   110→      onMouseEnter={() => setIsPaused(true)}
   111→      onMouseLeave={() => setIsPaused(false)}
   112→    >
   113→      {/* Desktop: 3-card 3D stack */}
   114→      <div className="hidden md:flex items-center justify-center gap-4 py-4">
   115→        {/* Left card */}
   116→        <motion.div
   117→          key={visibleIndices[0]}
   118→          initial={{ opacity: 0, x: direction > 0 ? -80 : -40, scale: 0.85 }}
   119→          animate={{ opacity: 0.5, x: 0, scale: 0.85 }}
   120→          exit={{ opacity: 0, x: direction > 0 ? 40 : -80, scale: 0.85 }}
   121→          transition={{ duration: 0.4, ease: 'easeInOut' }}
   122→          className="w-64 shrink-0 cursor-pointer"
   123→          style={{ perspective: '1000px', transform: 'rotateY(8deg)' }}
   124→          onClick={() => goTo(visibleIndices[0], -1)}
   125→        >
   126→          <SwipeCard country={swiperCountries[visibleIndices[0]]} onSelect={onSelectCountry} isSide />
   127→        </motion.div>
   128→
   129→        {/* Center card */}
   130→        <motion.div
   131→          key={centerIdx}
   132→          initial={{ opacity: 0, x: direction * 120, scale: 0.9 }}
   133→          animate={{ opacity: 1, x: 0, scale: 1 }}
   134→          exit={{ opacity: 0, x: -direction * 120, scale: 0.9 }}
   135→          transition={{ duration: 0.4, ease: 'easeInOut' }}
   136→          className="w-72 shrink-0 z-10"
   137→        >
   138→          <SwipeCard country={centerCountry} onSelect={onSelectCountry} isCenter />
   139→        </motion.div>
   140→
   141→        {/* Right card */}
   142→        <motion.div
   143→          key={visibleIndices[2]}
   144→          initial={{ opacity: 0, x: direction > 0 ? 80 : 40, scale: 0.85 }}
   145→          animate={{ opacity: 0.5, x: 0, scale: 0.85 }}
   146→          exit={{ opacity: 0, x: direction > 0 ? -40 : 80, scale: 0.85 }}
   147→          transition={{ duration: 0.4, ease: 'easeInOut' }}
   148→          className="w-64 shrink-0 cursor-pointer"
   149→          style={{ perspective: '1000px', transform: 'rotateY(-8deg)' }}
   150→          onClick={() => goTo(visibleIndices[2], 1)}
   151→        >
   152→          <SwipeCard country={swiperCountries[visibleIndices[2]]} onSelect={onSelectCountry} isSide />
   153→        </motion.div>
   154→      </div>
   155→
   156→      {/* Mobile: single card with swipe */}
   157→      <div
   158→        className="md:hidden overflow-hidden px-4"
   159→        ref={dragRef}
   160→        onMouseDown={handleDragStart}
   161→        onMouseUp={handleDragEnd}
   162→        onTouchStart={handleDragStart}
   163→        onTouchEnd={handleDragEnd}
   164→      >
   165→        <AnimatePresence mode="wait" custom={direction}>
   166→          <motion.div
   167→            key={currentIndex}
   168→            custom={direction}
   169→            initial={{ opacity: 0, x: direction * 200 }}
   170→            animate={{ opacity: 1, x: 0 }}
   171→            exit={{ opacity: 0, x: -direction * 200 }}
   172→            transition={{ duration: 0.35, ease: 'easeInOut' }}
   173→            drag="x"
   174→            dragConstraints={{ left: 0, right: 0 }}
   175→            dragElastic={0.2}
   176→            onDragEnd={(_, info) => {
   177→              if (info.offset.x > 60) goPrev();
   178→              else if (info.offset.x < -60) goNext();
   179→            }}
   180→          >
   181→            <SwipeCard country={swiperCountries[currentIndex]} onSelect={onSelectCountry} isCenter />
   182→          </motion.div>
   183→        </AnimatePresence>
   184→      </div>
   185→
   186→      {/* Navigation buttons */}
   187→      <div className="flex items-center justify-center gap-3 mt-3">
   188→        <Button
   189→          variant="outline"
   190→          size="icon"
   191→          className="h-8 w-8 rounded-full border-amber-300 hover:bg-amber-50 dark:border-amber-700 dark:hover:bg-amber-900/30"
   192→          onClick={goPrev}
   193→          aria-label="Previous country"
   194→        >
   195→          <ChevronLeft className="w-4 h-4 text-amber-700 dark:text-amber-400" />
   196→        </Button>
   197→
   198→        {/* Dot indicators */}
   199→        <div className="flex items-center gap-1.5">
   200→          {swiperCountries.map((_, i) => (
   201→            <button
   202→              key={i}
   203→              onClick={() => goTo(i, i > currentIndex ? 1 : -1)}
   204→              className={`rounded-full transition-all duration-300 ${
   205→                i === currentIndex
   206→                  ? 'w-6 h-2 bg-amber-500'
   207→                  : 'w-2 h-2 bg-amber-300 dark:bg-amber-700 hover:bg-amber-400'
   208→              }`}
   209→              aria-label={`Go to slide ${i + 1}`}
   210→            />
   211→          ))}
   212→        </div>
   213→
   214→        <Button
   215→          variant="outline"
   216→          size="icon"
   217→          className="h-8 w-8 rounded-full border-amber-300 hover:bg-amber-50 dark:border-amber-700 dark:hover:bg-amber-900/30"
   218→          onClick={goNext}
   219→          aria-label="Next country"
   220→        >
   221→          <ChevronRight className="w-4 h-4 text-amber-700 dark:text-amber-400" />
   222→        </Button>
   223→      </div>
   224→    </div>
   225→  );
   226→}

   300→const ACCENT_PRESETS = [
   301→  { name: 'Amber', color: '#F59E0B', primary: '245, 158, 11' },
   302→  { name: 'Orange', color: '#F97316', primary: '249, 115, 22' },

   304→  { name: 'Gold', color: '#EAB308', primary: '234, 179, 8' },
   305→  { name: 'Warm Brown', color: '#A16207', primary: '161, 98, 7' },
   306→  { name: 'Rose Gold', color: '#E11D48', primary: '225, 29, 72' },
   307→] as const;
   308→
   309→const STORAGE_KEY = 'pakvisa-accent-color';
   310→
   311→function applyAccent(preset: typeof ACCENT_PRESETS[number]) {
   312→  if (typeof document === 'undefined') return;
   313→  const root = document.documentElement;
   314→  const [r, g, b] = preset.primary.split(', ');
   315→  root.style.setProperty('--accent-primary', preset.color);
   316→  root.style.setProperty('--accent-rgb', `${r}, ${g}, ${b}`);
   317→  root.style.setProperty('--accent-foreground', '#FFFFFF');
   318→  root.style.setProperty('--accent-light', `rgba(${r}, ${g}, ${b}, 0.15)`);
   319→  root.style.setProperty('--accent-medium', `rgba(${r}, ${g}, ${b}, 0.3)`);
   320→  localStorage.setItem(STORAGE_KEY, preset.name);
   321→}
   322→
   323→export function ThemeAccentCustomizer({ defaultAccent }: { defaultAccent?: string }) {
   324→  const [selected, setSelected] = useState(defaultAccent || 'Amber');
   325→
   326→  useEffect(() => {
   327→    const saved = localStorage.getItem(STORAGE_KEY);
   328→    if (saved) {
   329→      const preset = ACCENT_PRESETS.find(p => p.name === saved);
   330→      if (preset) {
   331→        applyAccent(preset);
   332→      }
   333→    }
   334→  }, []);
   335→
   336→  const handleSelect = (preset: typeof ACCENT_PRESETS[number]) => {
   337→    setSelected(preset.name);
   338→    applyAccent(preset);
   339→  };
   340→
   341→  const handleReset = () => {
   342→    const defaultPreset = ACCENT_PRESETS[0];
   343→    setSelected(defaultPreset.name);
   344→    applyAccent(defaultPreset);
   345→  };
   346→
   347→  const activePreset = ACCENT_PRESETS.find(p => p.name === selected) || ACCENT_PRESETS[0];
   348→
   349→  return (
   350→    <Card className="card-elevated-1 glass-card">
   351→      <CardHeader className="pb-3">
   352→        <CardTitle className="text-sm font-semibold flex items-center gap-2">
   353→          <Settings className="w-4 h-4 text-amber-500" />
   354→          Accent Color
   355→        </CardTitle>
   356→      </CardHeader>
   357→      <CardContent className="space-y-3">
   358→        {/* Color preset buttons */}
   359→        <div className="flex items-center justify-center gap-3">
   360→          {ACCENT_PRESETS.map((preset) => (
   361→            <button
   362→              key={preset.name}
   363→              onClick={() => handleSelect(preset)}
   364→              className={`press-effect relative w-8 h-8 rounded-full border-2 transition-all duration-200 hover:scale-110 ${
   365→                selected === preset.name
   366→                  ? 'border-amber-900 dark:border-amber-100 ring-2 ring-offset-2 ring-amber-400 dark:ring-offset-background'
   367→                  : 'border-transparent hover:border-amber-400/50'
   368→              }`}
   369→              style={{ backgroundColor: preset.color }}
   370→              aria-label={`Select ${preset.name} accent color`}
   371→              title={preset.name}
   372→            >
   373→              {selected === preset.name && (
   374→                <Check className="w-4 h-4 text-white absolute inset-0 m-auto drop-shadow-sm" strokeWidth={3} />
   375→              )}
   376→            </button>
   377→          ))}
   378→        </div>
   379→
   380→        {/* Current accent name */}
   381→        <p className="text-center text-xs text-muted-foreground">
   382→          Current: <span className="font-medium text-amber-700 dark:text-amber-400">{activePreset.name}</span>
   383→        </p>
   384→
   385→        {/* Reset button */}
   386→        <Button
   387→          variant="ghost"
   388→          size="sm"
   389→          className="w-full text-xs text-muted-foreground hover:text-amber-700 dark:hover:text-amber-400"
   390→          onClick={handleReset}
   391→        >
   392→          <RotateCcw className="w-3 h-3 mr-1" />
   393→          Reset to Default
   394→        </Button>
   395→      </CardContent>
   396→    </Card>
   397→  );
   398→}

  1093→  const avgFee = countriesWithFee.length > 0

  1102→      .map(([name, count]) => ({ name, count }))
  1103→      .sort((a, b) => b.count - a.count);

  1109→    { label: 'Visa-Free', count: visaFreeCount, color: '#f59e0b' },

  1577→function AnimatedNumber({ value, suffix }: { value: number; suffix: string }) {
  1578→  const [display, setDisplay] = useState(0);
  1579→  useEffect(() => {
  1580→    const dur = 1500;
  1581→    const start = Date.now();
  1582→    const step = () => {
  1583→      const elapsed = Date.now() - start;
  1584→      const progress = Math.min(elapsed / dur, 1);
  1585→      const eased = 1 - Math.pow(1 - progress, 3);
  1586→      setDisplay(Math.round(value * eased));
  1587→      if (progress < 1) requestAnimationFrame(step);
  1588→    };
  1589→    requestAnimationFrame(step);
  1590→  }, [value]);
  1591→  return <>{display.toLocaleString()}{suffix}</>;
  1592→}
