    39→// Animated Counter Hook
    40→// ============================================================
    41→function useAnimatedCounter(target: number, duration: number = 1200, enabled: boolean = true) {
    42→  const [count, setCount] = useState(0);
    43→  useEffect(() => {
    44→    if (!enabled || target === 0) return;
    45→    let start = 0;
    46→    let startTime: number | null = null;
    47→    let raf: number;
    48→    const step = (timestamp: number) => {
    49→      if (!startTime) startTime = timestamp;
    50→      const progress = Math.min((timestamp - startTime) / duration, 1);
    51→      // Ease-out cubic
    52→      const eased = 1 - Math.pow(1 - progress, 3);
    53→      const current = Math.round(eased * target);
    54→      setCount(current);
    55→      if (progress < 1) {
    56→        raf = requestAnimationFrame(step);
    57→      }
    58→    };
    59→    raf = requestAnimationFrame(step);
    60→    return () => cancelAnimationFrame(raf);
    61→  }, [target, duration, enabled]);
    62→  return count;
    63→}

   133→// Pagination Helper
   134→// ============================================================
   135→function generatePageNumbers(current: number, total: number): (number | 'ellipsis')[] {
   136→  if (total <= 7) {
   137→    return Array.from({ length: total }, (_, i) => i + 1);
   138→  }
   139→  const pages: (number | 'ellipsis')[] = [1];
   140→  if (current > 3) pages.push('ellipsis');
   141→  const start = Math.max(2, current - 1);
   142→  const end = Math.min(total - 1, current + 1);
   143→  for (let i = start; i <= end; i++) pages.push(i);
   144→  if (current < total - 2) pages.push('ellipsis');
   145→  if (total > 1) pages.push(total);
   146→  return pages;
   147→}

   149→// ============================================================
   150→// Visa Type Helpers (single source of truth for access type)
   151→// Priority: visaFree > visaOnArrival > etaAvailable > embassy
   152→// ============================================================
   153→function getPrimaryAccess(c: CountryData): string {
   154→  if (c.visaFree) return 'visa-free';
   155→  if (c.visaOnArrival) return 'visa-on-arrival';
   156→  if (c.etaAvailable) return 'e-visa';
   157→  return 'embassy';
   158→}
   159→function getVisaType(c: CountryData) {
   160→  const access = getPrimaryAccess(c);
   161→  switch (access) {
   162→    case 'visa-free':     return { label: 'Visa Free', color: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-400', dot: 'bg-emerald-500' };
   163→    case 'visa-on-arrival': return { label: 'Visa on Arrival', color: 'bg-amber-100 text-amber-700 dark:bg-amber-900/30 dark:text-amber-400', dot: 'bg-amber-500' };
   164→    case 'e-visa':       return { label: 'e-Visa', color: 'bg-sky-100 text-sky-700 dark:bg-sky-900/30 dark:text-sky-400', dot: 'bg-sky-500' };
   165→    default:             return { label: 'Embassy Required', color: 'bg-gray-100 text-gray-600 dark:bg-gray-800 dark:text-gray-400', dot: 'bg-gray-400' };
   166→  }
   167→}
