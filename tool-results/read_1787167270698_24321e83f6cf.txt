    73→    73→  const [analyticsLoading, setAnalyticsLoading] = useState(false);
    74→    74→  const [maintenanceMode, setMaintenanceMode] = useState(false);
    75→    75→  const [togglingAi, setTogglingAi] = useState(false);
    76→    76→  const [togglingMaintenance, setTogglingMaintenance] = useState(false);
    77→    77→  const [refreshingAnalytics, setRefreshingAnalytics] = useState(false);
    78→    78→
    79→    79→  // Messages state
    80→    80→  const [messages, setMessages] = useState<ContactMessage[]>([]);
    81→    81→  const [messagesTotal, setMessagesTotal] = useState(0);
    82→    82→  const [messagesUnread, setMessagesUnread] = useState(0);

    84→    84→  const [messagesLoading, setMessagesLoading] = useState(false);
    85→    85→  const [replyingTo, setReplyingTo] = useState<string | null>(null);

   100→   100→      setIsLoggedIn(true);
   101→   101→    }

   106→   106→      toast.error('Please enter username and password');

   112→   112→        method: 'POST',

   114→   114→        body: JSON.stringify({ username: username.trim(), password }),
   115→   115→      });
   116→   116→      const data = await res.json();
   117→   117→      if (data.success) {
   118→   118→        localStorage.setItem('pakvisa-admin-token', data.data.token);
   119→   119→        setToken(data.data.token);
   120→   120→        setIsLoggedIn(true);

   123→   123→        toast.error(data.error || 'Login failed');
   124→   124→      }

   169→   169→        setMessagesTotal(data.data.total);

   172→   172→      }
   173→   173→    } catch { toast.error('Failed to fetch messages'); }

   178→   178→    if (!token) return;
   179→   179→    setSubscribersLoading(true);

   181→   181→      const res = await fetch(`/api/admin/newsletter?page=${page}&limit=15`, {
   182→   182→        headers: { 'Authorization': `Bearer ${token}` },
   183→   183→      });
   184→   184→      const data = await res.json();

   186→   186→        setSubscribers(data.data.subscribers);
   187→   187→        setSubscribersTotal(data.data.total);
   188→   188→        setSubscribersActive(data.data.activeCount);
   189→   189→        setSubscribersPage(data.data.page);
   190→   190→      }

   198→   198→    if (activeSection === 'overview' || activeSection === 'analytics') {
   199→   199→      if (!analytics) fetchAnalytics();
   200→   200→    }
   201→   201→    if (activeSection === 'overview') {
   202→   202→      fetchMessages();
   203→   203→      fetchSubscribers();
   204→   204→    }
   205→   205→    if (activeSection === 'messages') fetchMessages();
   206→   206→    if (activeSection === 'newsletter') fetchSubscribers();
   207→   207→  }, [isLoggedIn, token, activeSection]);
   208→   208→
   209→   209→  const toggleAiFeature = useCallback(async (enabled: boolean) => {

   213→   213→      const res = await fetch('/api/admin/settings', {
   214→   214→        method: 'PUT',
   215→   215→        headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` },
   216→   216→        body: JSON.stringify({ key: 'ai_enabled', value: String(enabled) }),

   221→   221→    } catch { toast.error('Connection error'); }
   222→   222→    finally { setTogglingAi(false); }
   223→   223→  }, [token, setAiEnabled]);
   224→   224→
   225→   225→  const toggleMaintenance = useCallback(async (enabled: boolean) => {
   226→   226→    if (!token) return;

   228→   228→    try {
   229→   229→      const res = await fetch('/api/admin/settings', {
   230→   230→        method: 'PUT',
   231→   231→        headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` },
   232→   232→        body: JSON.stringify({ key: 'maintenance_mode', value: String(enabled) }),
   233→   233→      });
   234→   234→      const data = await res.json();

   236→   236→      else toast.error('Failed to update setting');
   237→   237→    } catch { toast.error('Connection error'); }

   244→   244→      await fetch('/api/admin/messages', {
   245→   245→        method: 'PUT',

   258→   258→        method: 'PUT',
   259→   259→        headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` },
   260→   260→        body: JSON.stringify({ id, action: 'reply', reply: replyText }),
   261→   261→      });
   262→   262→      const data = await res.json();
   263→   263→      if (data.success) {
   264→   264→        setMessages(prev => prev.map(m => m.id === id ? { ...m, isReplied: true, reply: replyText.trim(), isRead: true } : m));
   265→   265→        setReplyingTo(null); setReplyText('');

   267→   267→      }

   275→   275→        method: 'PUT',

   288→   288→      await fetch(`/api/admin/newsletter?id=${id}`, {
   289→   289→        method: 'DELETE',
   290→   290→        headers: { 'Authorization': `Bearer ${token}` },

   292→   292→      setSubscribers(prev => prev.filter(s => s.id !== id));
   293→   293→      setSubscribersTotal(prev => prev - 1);
   294→   294→      setSubscribersActive(prev => prev - 1);

   297→   297→  };
   298→   298→

   300→   300→    { key: 'overview', label: 'Overview', icon: <TrendingUp className="w-4 h-4" /> },
   301→   301→    { key: 'messages', label: 'Messages', icon: <Inbox className="w-4 h-4" />, badge: messagesUnread || undefined },

   310→   310→    const mins = Math.floor(diff / 60000);

   314→   314→    return `${Math.floor(hours / 24)}d ago`;
   315→   315→  };

   336→   336→            )}
   337→   337→          </div>
   338→   338→        </DialogHeader>
   339→   339→
   340→   340→        {!isLoggedIn ? (
   341→   341→          <div className="p-8 flex-1 flex items-center justify-center">
   342→   342→            <Card className="w-full max-w-sm border-emerald-200 dark:border-emerald-800">
   343→   343→              <CardHeader className="text-center">
   344→   344→                <div className="w-16 h-16 rounded-full bg-emerald-100 dark:bg-emerald-900/30 flex items-center justify-center mx-auto mb-3">
   345→   345→                  <Lock className="w-8 h-8 text-emerald-600" />
   346→   346→                </div>
   347→   347→                <CardTitle className="text-lg">Admin Login</CardTitle>
   348→   348→                <CardDescription>Enter credentials to access dashboard</CardDescription>

   350→   350→              <CardContent className="space-y-4">
