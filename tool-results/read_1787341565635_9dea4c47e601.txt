   118→function safetyStars(rating: number): string {
   119→  const clamped = Math.min(Math.max(rating, 0), 5);
   120→  return '\u2605'.repeat(clamped) + '\u2606'.repeat(5 - clamped);

   755→function generateFAQs(
   756→  country: {
   757→    name: string; code: string; visaFree: boolean; visaOnArrival: boolean; etaAvailable: boolean;
   758→    processingDaysMin: number; processingDaysMax: number; safetyRating: number; safetySummary: string;
   759→    bestTravelMonths: string; avgTempC: string; currency: string; currencyCode: string;
   760→  },
   761→  costProfile: CostProfileData | undefined,
   762→  visaLabel: string,
   763→  isEmbassyRequired: boolean,
   764→): { q: string; a: string }[] {
   765→  const faqs: { q: string; a: string }[] = [];
   766→
   767→  // Q1: Do Pakistani citizens need a visa?
   768→  faqs.push({
   769→    q: `Do Pakistani citizens need a visa for ${country.name}?`,
   770→    a: country.visaFree
   771→      ? `No, Pakistani citizens do not need a visa to visit ${country.name}. You can travel visa-free with a valid Pakistani passport. Make sure your passport has at least 6 months of validity remaining.`
   772→      : country.visaOnArrival
   773→        ? `Pakistani citizens can get a visa on arrival at ${country.name}. Simply present your valid Pakistani passport at the immigration counter upon arrival. Processing is usually instant.`
   774→        : country.etaAvailable
   775→          ? `Yes, Pakistani citizens need an e-Visa to visit ${country.name}. The e-Visa can be applied for online through the official government portal. Processing typically takes ${country.processingDaysMin}-${country.processingDaysMax} business days.`
   776→          : `Yes, Pakistani citizens need a visa to visit ${country.name}. You must apply at the embassy or consulate. The application process typically takes ${country.processingDaysMin}-${country.processingDaysMax} business days.`,
   777→  });
   778→
   779→  // Q2: What is the visa fee?
   780→  const feeText = costProfile && costProfile.visaFeeUSD > 0
   781→    ? `The visa fee for ${country.name} is approximately $${costProfile.visaFeeUSD} USD (\u2248 PKR ${usdToPkr(costProfile.visaFeeUSD)}). Additional service fees of $${costProfile.serviceFeeUSD} may apply.`
   782→    : `There is no visa fee for Pakistani citizens visiting ${country.name} as it is ${visaLabel.toLowerCase()}.`
   783→  ;
   784→  faqs.push({
   785→    q: `What is the visa fee for ${country.name} for Pakistani citizens?`,
   786→    a: feeText,
   787→  });
   788→
   789→  // Q3: Processing time
   790→  faqs.push({
   791→    q: `How long does it take to get a ${country.name} visa?`,
   792→    a: country.visaFree
   793→      ? `No processing time is needed as ${country.name} offers visa-free entry to Pakistani passport holders. You can travel immediately.`
   794→      : country.visaOnArrival
   795→        ? `The visa on arrival for ${country.name} is processed instantly at the airport or border crossing. The entire process typically takes 5-15 minutes.`
   796→        : `${country.name} visa processing typically takes ${country.processingDaysMin}-${country.processingDaysMax} business days for Pakistani citizens. It is recommended to apply at least 2-3 weeks before your planned travel date.`,
   797→  });
   798→
   799→  // Q4: Safety
   800→  faqs.push({
   801→    q: `Is ${country.name} safe for Pakistani tourists?`,
   802→    a: country.safetySummary
   803→      ? `${country.name} has a safety rating of ${Math.min(country.safetyRating, 5)}/5. ${country.safetySummary}`
   804→      : `${country.name} has a safety rating of ${Math.min(country.safetyRating, 5)}/5. As with any travel destination, exercise standard precautions, keep your belongings secure, and follow local laws and customs.`,
   805→  });
   806→
   807→  // Q5: Best time to visit or documents required
   808→  if (country.bestTravelMonths) {
   809→    faqs.push({
   810→      q: `What is the best time to visit ${country.name} from Pakistan?`,
   811→      a: `The best months to visit ${country.name} are ${country.bestTravelMonths}. The average temperature is around ${country.avgTempC}\u00B0C. The local currency is ${country.currency} (${country.currencyCode}). Plan your trip during these months for the most comfortable weather conditions.`,
   812→    });
   813→  } else {
   814→    faqs.push({
   815→      q: `What documents do I need for a ${country.name} visa?`,
   816→      a: `Common requirements for a ${country.name} visa include: a valid Pakistani passport with 6+ months validity, passport-sized photographs, proof of accommodation, return flight ticket, travel insurance, and proof of sufficient funds. For the complete and most up-to-date list, check the official embassy website or use PakVisa Advisor.`,
   817→    });
   818→  }
   819→
   820→  return faqs;
   821→}
