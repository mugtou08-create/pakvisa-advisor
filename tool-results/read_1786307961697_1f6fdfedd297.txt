   174→// ============ SCORE HISTORY CHART (SVG) ============
   175→export function ScoreHistoryChart() {
   176→  // eslint-disable-next-line react-hooks/preserve-manual-memoization
   177→  const history = useMemo(() => {
   178→    try {
   179→      const raw = localStorage.getItem(SCORING_HISTORY_KEY);
   180→      if (raw) {
   181→        const entries = JSON.parse(raw);
   182→        return entries.slice(-8).map((e: any) => ({
   183→          date: new Date(e.date).toLocaleDateString('en-US', { month: 'short', day: 'numeric' }),
   184→          topScore: e.topScore || (e.scores && e.scores[0] ? e.scores[0].finalScore : 0),
   185→          countryCount: e.countryCount || (e.scores ? e.scores.length : 0),
   186→        }));
   187→      }
   188→    } catch {}
   189→    return [];
   190→  }, []);
   191→  
   192→  if (history.length < 2) return null;
   193→  
   194→  const chartW = 400, chartH = 120, pad = 30;
   195→  const maxScore = Math.max(...history.map(e => e.topScore), 1);
   196→  const pts = history.map((e, i) => ({
   197→    x: pad + (i / (history.length - 1)) * (chartW - 2 * pad),
   198→    y: chartH - pad - (e.topScore / maxScore) * (chartH - 2 * pad),
   199→  }));
   200→  
   201→  return (
   202→    <Card className="p-4">
   203→      <h4 className="text-sm font-semibold mb-3 flex items-center gap-1.5">
   204→        <TrendingUp className="w-4 h-4 text-amber-500" />
   205→        Score Trend
   206→      </h4>
   207→      <svg viewBox={`0 0 ${chartW} ${chartH}`} className="w-full">
   208→        {/* Grid lines */}
   209→        {[0, 0.5, 1].map(v => (
   210→          <line key={v} x1={pad} y1={chartH - pad - v * (chartH - 2 * pad)} x2={chartW - pad} y2={chartH - pad - v * (chartH - 2 * pad)} stroke="var(--border)" strokeWidth="0.5" strokeDasharray="4 4" />
   211→        ))}
   212→        {/* Y-axis labels */}
   213→        <text x={pad - 5} y={chartH - pad + 4} textAnchor="end" fill="var(--muted-foreground)" fontSize="9">0</text>
   214→        <text x={pad - 5} y={pad + 4} textAnchor="end" fill="var(--muted-foreground)" fontSize="9">{maxScore}</text>
   215→        {/* Area fill */}
   216→        <path d={`M ${pts[0].x} ${chartH - pad} L ${pts.map(p => `${p.x} ${p.y}`).join(' L')} L ${pts[pts.length - 1].x} ${chartH - pad} Z`} fill="url(#chartGradient)" opacity="0.3" />
   217→        {/* Line */}
   218→        <polyline points={pts.map(p => `${p.x},${p.y}`).join(' ')} fill="none" stroke="#f59e0b" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" />
   219→        {/* Data points */}
   220→        {pts.map((p, i) => (
   221→          <g key={i}>
   222→            <circle cx={p.x} cy={p.y} r="4" fill="white" stroke="#f59e0b" strokeWidth="2" />
   223→            <text x={p.x} y={chartH - pad + 14} textAnchor="middle" fill="var(--muted-foreground)" fontSize="8">{history[i].date}</text>
   224→          </g>
   225→        ))}
   226→        <defs>
   227→          <linearGradient id="chartGradient" x1="0" y1="0" x2="0" y2="1">
   228→            <stop offset="0%" stopColor="#f59e0b" />
   229→            <stop offset="100%" stopColor="#f59e0b" stopOpacity="0" />
   230→          </linearGradient>
   231→        </defs>
   232→      </svg>
   233→    </Card>
   234→  );
   235→}

   237→// ============ PASSPORT POWER INDEX WIDGET ============
   238→export function PassportPowerIndex({ countries, stats }: { countries: CountryData[]; stats: { totalCountries: number; visaFreeCount: number; visaOnArrivalCount: number; eVisaCount: number; embassyRequiredCount: number; avgCostUSD?: number; cheapestCountry?: { name: string; code: string; flagEmoji: string; visaFeeUSD: number }; fastestProcessing?: { name: string; code: string; flagEmoji: string; processingDaysMin: number } } | null }) {
   239→  // Exclusive classification: visaFree > voa > evisa > embassy (same logic as API stats)
   240→  const visaFreeCount = stats?.visaFreeCount || countries.filter(c => c.visaFree).length;
   241→  const voaCount = stats?.visaOnArrivalCount || countries.filter(c => !c.visaFree && c.visaOnArrival).length;
   242→  const eVisaCount = stats?.eVisaCount || countries.filter(c => !c.visaFree && !c.visaOnArrival && c.etaAvailable).length;
   243→  const embassyCount = stats?.embassyRequiredCount || countries.filter(c => !c.visaFree && !c.visaOnArrival && !c.etaAvailable).length;
   244→  const total = 199;
   245→  const rank = 106;
   246→  const accessible = visaFreeCount + voaCount + eVisaCount;
   247→  const powerScore = Math.round((accessible / total) * 100);
   248→  const segments = [
   249→    { label: 'Visa Free', count: visaFreeCount, color: '#f59e0b' },
   250→    { label: 'Visa on Arrival', count: voaCount, color: '#f59e0b' },
   251→    { label: 'e-Visa', count: eVisaCount, color: '#f97316' },
   252→    { label: 'Embassy Required', count: embassyCount, color: '#ef4444' },
   253→  ];
   254→
   255→  // Circular gauge animation state
   256→  const [animatedScore, setAnimatedScore] = useState(0);
   257→  const [animatedRank, setAnimatedRank] = useState(0);
   258→  const [animatedAccessible, setAnimatedAccessible] = useState(0);
   259→  const [showCalcInfo, setShowCalcInfo] = useState(false);
   260→  const gaugeRef = useRef<HTMLDivElement>(null);

   262→

   264→    const el = gaugeRef.current;
   265→    if (!el) return;
   266→    const observer = new IntersectionObserver(
   267→      ([entry]) => {
   268→        if (entry.isIntersecting && !hasAnimated.current) {
   269→          hasAnimated.current = true;
   270→          const duration = 1500;
   271→          const start = performance.now();
   272→          const animate = (currentTime: number) => {
   273→            const elapsed = currentTime - start;
   274→            const progress = Math.min(elapsed / duration, 1);
   275→            const eased = 1 - Math.pow(1 - progress, 3);
   276→            setAnimatedScore(Math.round(eased * powerScore));
   277→            setAnimatedRank(Math.round(eased * rank));
   278→            setAnimatedAccessible(Math.round(eased * accessible));
   279→            if (progress < 1) requestAnimationFrame(animate);
   280→          };
   281→          requestAnimationFrame(animate);
   282→          observer.disconnect();
   283→        }
   284→      },
   285→      { threshold: 0.2 }
   286→    );
   287→    observer.observe(el);
   288→    return () => observer.disconnect();
   289→  }, [powerScore, rank, accessible]);
   290→
   291→  // SVG circular gauge params
   292→  const radius = 54;
   293→  const strokeWidth = 8;
   294→  const circumference = 2 * Math.PI * radius;
   295→  const gaugeProgress = (animatedScore / 100) * circumference;

   310→    { flag: '\u{1F1E7}\u{1F1E9}', name: 'Bangladesh', rank: 101 },
   311→    { flag: '\u{1F1E6}\u{1F1EB}', name: 'Afghanistan', rank: 111 },
   312→  ];
   313→
   314→  return (
   315→    <div className="glass-section p-5" ref={gaugeRef}>
   316→      <div className="flex items-center justify-between mb-4">
   317→        <h3 className="text-sm font-semibold flex items-center gap-2">
   318→          <Globe className="w-4 h-4 text-amber-500" />
   319→          Passport Power Index
   320→        </h3>
   321→        <Badge variant="secondary" className="text-[10px] bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-400">
   322→          {animatedAccessible} accessible
   323→        </Badge>

   326→      <div className="flex flex-col sm:flex-row items-center gap-6 mb-4">
   327→        {/* Circular Gauge */}
   328→        <div className="relative shrink-0">
   329→          <svg width={140} height={140} className="-rotate-90">
   330→            {/* Background track */}
   331→            <circle
   332→              cx={70} cy={70} r={radius}

   352→            {/* Animated progress arc */}
   353→            <circle
   354→              cx={70} cy={70} r={radius}
   355→              fill="none"
   356→              stroke={gaugeColor}
   357→              strokeWidth={strokeWidth}
   358→              strokeLinecap="round"
