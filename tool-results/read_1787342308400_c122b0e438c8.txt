    20→// Live Clock Component
    21→// ============================================================
    22→function LiveClock({ timezone, cityName }: { timezone: string; cityName: string }) {
    23→  const [time, setTime] = useState('');
    24→  const [date, setDate] = useState('');
    25→
    26→  useEffect(() => {
    27→    function updateTime() {
    28→      try {
    29→        const now = new Date();
    30→        const formatter = new Intl.DateTimeFormat('en-US', {
    31→          timeZone: timezone,
    32→          hour: '2-digit',
    33→          minute: '2-digit',
    34→          second: '2-digit',
    35→          hour12: true,
    36→        });
    37→        const dateFormatter = new Intl.DateTimeFormat('en-US', {
    38→          timeZone: timezone,
    39→          weekday: 'short',
    40→          month: 'short',
    41→          day: 'numeric',
    42→        });
    43→        setTime(formatter.format(now));
    44→        setDate(dateFormatter.format(now));
    45→      } catch {
    46→        setTime('—');
    47→        setDate('—');
    48→      }
    49→    }
    50→    updateTime();
    51→    const interval = setInterval(updateTime, 1000);
    52→    return () => clearInterval(interval);
    53→  }, [timezone]);
    54→
    55→  // Calculate time difference from PKT (Asia/Karachi)
    56→  const timeDiff = useMemo(() => {
    57→    try {
    58→      const now = new Date();
    59→      // Get UTC offsets in minutes for both timezones
    60→      const targetParts = new Intl.DateTimeFormat('en-US', {
    61→        timeZone: timezone,
    62→        timeZoneName: 'shortOffset',
    63→      }).formatToParts(now);
    64→      const pktParts = new Intl.DateTimeFormat('en-US', {
    65→        timeZone: 'Asia/Karachi',
    66→        timeZoneName: 'shortOffset',
    67→      }).formatToParts(now);
    68→
    69→      const parseOffset = (parts: Intl.DateTimeFormatPart[]) => {
    70→        const tzPart = parts.find(p => p.type === 'timeZoneName');
    71→        if (!tzPart) return 0;
    72→        const match = tzPart.value.match(/GMT([+-]?)(\d{1,2})(?::(\d{2}))?/);
    73→        if (!match) return 0;
    74→        const sign = match[1] === '-' ? -1 : 1;
    75→        const hours = parseInt(match[2]);
    76→        const minutes = match[3] ? parseInt(match[3]) : 0;
    77→        return sign * (hours * 60 + minutes);
    78→      };
    79→
    80→      const targetOffset = parseOffset(targetParts);
    81→      const pktOffset = parseOffset(pktParts);
    82→      const diffHours = (targetOffset - pktOffset) / 60;
    83→
    84→      if (diffHours === 0) return 'Same as PKT';
    85→      if (diffHours > 0) return `+${diffHours}h from PKT`;
    86→      return `${diffHours}h from PKT`;
    87→    } catch {
    88→      return '';
    89→    }
    90→  }, [timezone]);
    91→
    92→  return (
    93→    <div className="flex items-center gap-3 rounded-lg bg-sky-50 dark:bg-sky-950/20 border border-sky-200 dark:border-sky-800 p-3">
    94→      <Clock className="w-5 h-5 text-sky-600 shrink-0" />
    95→      <div>
    96→        <p className="text-sm font-semibold text-sky-900 dark:text-sky-200">{time}</p>
    97→        <p className="text-xs text-sky-700 dark:text-sky-400">{date} {timeDiff && `· ${timeDiff}`}</p>
    98→      </div>
    99→    </div>
   100→  );
   101→}

   106→function CurrencyConverter({ currencyCode, currencyName }: { currencyCode: string; currencyName: string }) {
   107→  const [amount, setAmount] = useState<string>('100');
   108→  const [direction, setDirection] = useState<'topkr' | 'frompkr'>('topkr');
   109→
   110→  const rate = EXCHANGE_RATES[currencyCode];
   111→  const [topkr, frompkr] = useMemo(() => {
   112→    if (!rate) return ['—', '—'];
   113→    return [
   114→      (parseFloat(amount || '0') * rate).toLocaleString('en-PK', { maximumFractionDigits: 2 }),
   115→      (parseFloat(amount || '0') / rate).toLocaleString(undefined, { maximumFractionDigits: 2 }),
   116→    ];
   117→  }, [amount, rate]);
   118→
   119→  return (
   120→    <div className="rounded-lg bg-emerald-50 dark:bg-emerald-950/20 border border-emerald-200 dark:border-emerald-800 p-3 space-y-2">
   121→      <div className="flex items-center justify-between">
   122→        <div className="flex items-center gap-2">
   123→          <Banknote className="w-4 h-4 text-emerald-600" />
   124→          <span className="text-xs font-semibold text-emerald-900 dark:text-emerald-200">Currency Converter</span>
   125→        </div>
   126→        <Badge variant="outline" className="text-[10px] border-emerald-300 dark:border-emerald-700 text-emerald-700 dark:text-emerald-400">
   127→          {currencyCode} · {currencyName}
   128→        </Badge>
   129→      </div>
   130→      <div className="flex items-center gap-2">
   131→        <input
   132→          type="number"
   133→          value={amount}
   134→          onChange={(e) => setAmount(e.target.value)}
   135→          className="w-full h-8 px-2 rounded-md border bg-white dark:bg-background text-sm focus:outline-none focus:ring-1 focus:ring-emerald-500"
   136→          min="0"
   137→        />
   138→        <button
   139→          onClick={() => setDirection((d) => d === 'topkr' ? 'frompkr' : 'topkr')}
   140→          className="p-1.5 rounded-md border bg-white dark:bg-background hover:bg-muted transition-colors"
   141→          title="Swap direction"
   142→        >
   143→          <ArrowRightLeft className="w-3.5 h-3.5 text-emerald-600" />
   144→        </button>
   145→      </div>
   146→      {rate ? (
   147→        <div className="flex items-center justify-between text-xs">
   148→          <span className="text-muted-foreground">
   149→            {direction === 'topkr' ? `${amount || '0'} ${currencyCode} =` : `PKR ${amount || '0'} =`}
   150→          </span>
   151→          <span className="font-semibold text-emerald-800 dark:text-emerald-300">
   152→            {direction === 'topkr' ? `PKR ${topkr}` : `${frompkr} ${currencyCode}`}
   153→          </span>
   154→        </div>
   155→      ) : (
   156→        <p className="text-xs text-muted-foreground">Exchange rate not available</p>
   157→      )}
   158→      <p className="text-[10px] text-emerald-700/70 dark:text-emerald-400/70">
   159→        Rate: 1 {currencyCode} ≈ PKR {rate?.toFixed(2) || '—'} · Approximate, verify before transaction
   160→      </p>
   161→    </div>
   162→  );
   163→}

   386→// ============================================================
   387→// Weather Info Section
   388→// ============================================================
   389→function WeatherInfo({ avgTemp, monthlyTemps }: { avgTemp: string; monthlyTemps: Record<string, number> }) {
   390→  const temps = Object.entries(monthlyTemps);
   391→  if (temps.length === 0 && !avgTemp) return null;
   392→
   393→  const maxTemp = temps.length > 0 ? Math.max(...temps.map(([, v]) => v)) : parseInt(avgTemp) || 0;
   394→  const minTemp = temps.length > 0 ? Math.min(...temps.map(([, v]) => v)) : parseInt(avgTemp) || 0;
   395→
   396→  return (
   397→    <div>
   398→      <h5 className="text-xs font-semibold text-muted-foreground uppercase tracking-wide mb-2">Climate & Weather</h5>
   399→      <div className="rounded-lg bg-muted/50 p-3 space-y-3">
   400→        <div className="flex items-center justify-between text-sm">
   401→          <span className="flex items-center gap-1.5">
   402→            <Thermometer className="w-4 h-4 text-red-500" />
   403→            <span className="font-medium">Avg Temperature</span>
   404→          </span>
   405→          <span className="font-semibold">{avgTemp}°C</span>
   406→        </div>
   407→        <div className="flex items-center gap-3 text-xs text-muted-foreground">
   408→          <span>🔥 High: {maxTemp}°C</span>
   409→          <span>❄️ Low: {minTemp}°C</span>
   410→        </div>
   411→        {temps.length > 0 && (
   412→          <div className="flex items-end gap-0.5 h-8">
   413→            {temps.map(([month, temp], idx) => {
   414→              const height = maxTemp > minTemp ? ((temp - minTemp) / (maxTemp - minTemp)) * 100 : 50;
   415→              return (
   416→                <div key={month} className="flex-1 flex flex-col items-center gap-0.5">
   417→                  <div
   418→                    className="w-full rounded-t-sm min-h-[2px] transition-all"
   419→                    style={{
   420→                      height: 'height: ' + Math.max(height, 5) + '%' === 'height: ' + Math.max(height, 5) + '%' ? Math.max(height, 5) + '%' : Math.max(height, 5) + '%',
   421→                      backgroundColor: temp > 30 ? '#ef4444' : temp > 20 ? '#f59e0b' : temp > 10 ? '#3b82f6' : '#6366f1',
   422→                    }}
   423→                  />
   424→                  <span className="text-[8px] text-muted-foreground">{MONTH_NAMES[idx]?.[0] || ''}</span>
   425→                </div>
   426→              );
   427→            })}
   428→          </div>
   429→        )}
   430→      </div>
   431→    </div>
   432→  );
   433→}

   739→function FullDocumentChecklist({
   740→  requirements,
   741→  countryName,
   742→}: {
   743→  requirements: { id: string; category: string; requirement: string; mandatory: boolean; description?: string }[];
   744→  countryName: string;
   745→}) {
   746→  const { user, isAuthenticated } = useAuthStore();

   750→  // Group requirements by category

   752→    const map = new Map<string, typeof requirements>();
