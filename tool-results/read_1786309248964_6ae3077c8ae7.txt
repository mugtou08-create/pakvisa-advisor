   789→// ============ TYPING TEXT ANIMATION ============
   790→export function TypingText({ phrases, className = '' }: { phrases: string[]; className?: string }) {
   791→  const [current, setCurrent] = useState(0);
   792→  const [text, setText] = useState('');
   793→  const [isDeleting, setIsDeleting] = useState(false);
   794→
   795→  useEffect(() => {
   796→    const phrase = phrases[current];
   797→    const timeout = setTimeout(() => {
   798→      if (!isDeleting) {
   799→        setText(phrase.substring(0, text.length + 1));
   800→        if (text.length === phrase.length) {
   801→          setTimeout(() => setIsDeleting(true), 2200);
   802→        }
   803→      } else {
   804→        setText(phrase.substring(0, text.length - 1));
   805→        if (text.length === 0) {
   806→          setIsDeleting(false);
   807→          setCurrent((current + 1) % phrases.length);
   808→        }
   809→      }
   810→    }, isDeleting ? 25 : 55);
   811→    return () => clearTimeout(timeout);
   812→  }, [text, isDeleting, current, phrases]);
   813→
   814→  return (
   815→    <span className={className}>
   816→      {text}
   817→      <span className="inline-block w-0 h-[1.1em] ml-0.5 align-middle typing-text-cursor" />
   818→    </span>
   819→  );
   820→}

   872→// ============ MINI RADAR CHART (3-axis for country cards) ============
   873→export function MiniRadarChart({ safety, cost, ease }: { safety: number; cost: number; ease: number }) {
   874→  const cx = 28, cy = 28, r = 22;
   875→  const axes = 3;
   876→  const angleStep = (2 * Math.PI) / axes;
   877→  const startAngle = -Math.PI / 2;
   878→  const s = safety / 10;
   879→  const c = Math.max(0.05, 1 - cost / 3000);
   880→  const e = ease;
   881→  const values = [s, c, e];
   882→  const points = values.map((val, i) => {
   883→    const angle = startAngle + i * angleStep;
   884→    return `${cx + val * r * Math.cos(angle)},${cy + val * r * Math.sin(angle)}`;
   885→  }).join(' ');
   886→  const gridLines = [0.33, 0.66, 1].map(level => {
   887→    const pts = Array.from({ length: axes }, (_, i) => {
   888→      const angle = startAngle + i * angleStep;
   889→      return `${cx + level * r * Math.cos(angle)},${cy + level * r * Math.sin(angle)}`;
   890→    }).join(' ');
   891→    return <polygon key={level} points={pts} fill="none" stroke="currentColor" strokeWidth="0.3" className="text-white/20" />;
   892→  });
   893→
   894→  return (
   895→    <svg width="56" height="56" viewBox="0 0 56 56" className="inline-block opacity-80">
   896→      {gridLines}
   897→      {Array.from({ length: axes }, (_, i) => {
   898→        const angle = startAngle + i * angleStep;
   899→        return <line key={i} x1={cx} y1={cy} x2={cx + r * Math.cos(angle)} y2={cy + r * Math.sin(angle)} stroke="currentColor" strokeWidth="0.3" className="text-white/20" />;
   900→      })}
   901→      <polygon points={points} fill="rgba(249,115,22,0.15)" stroke="#fbbf24" strokeWidth="1" />
   902→      {values.map((val, i) => {
   903→        const angle = startAngle + i * angleStep;
   904→        return <circle key={i} cx={cx + val * r * Math.cos(angle)} cy={cy + val * r * Math.sin(angle)} r="1.5" fill="#fbbf24" />;
   905→      })}
   906→    </svg>

   940→// ============ ANIMATED SCORE NUMBER ============
   941→export function AnimatedScoreNumber({ value, delay = 0 }: { value: number; delay?: number }) {
   942→  const [displayed, setDisplayed] = useState(0);
   943→  const hasAnimated = useRef(false);
   944→
   945→  useEffect(() => {
   946→    if (hasAnimated.current) return;
   947→    hasAnimated.current = true;
   948→    const start = performance.now() + delay;
   949→    const duration = 1200;
   950→    const animate = (now: number) => {
   951→      const elapsed = now - start;
   952→      if (elapsed < 0) { requestAnimationFrame(animate); return; }
   953→      const progress = Math.min(elapsed / duration, 1);
   954→      const eased = 1 - Math.pow(1 - progress, 3);
   955→      setDisplayed(Math.round(eased * value));
   956→      if (progress < 1) requestAnimationFrame(animate);
   957→    };
   958→    requestAnimationFrame(animate);
   959→  }, [value, delay]);
   960→
   961→  return <span>{displayed}</span>;
   962→}

  1421→// ============ MICRO SPARKLINE ============
  1422→export function MicroSparkline({ data, color, width = 60, height = 20 }: { data: number[]; color?: string; width?: number; height?: number }) {
  1423→  if (!data || data.length === 0) return null;
  1424→  const maxVal = Math.max(...data);
  1425→  const minVal = Math.min(...data);
  1426→  const range = maxVal - minVal || 1;
  1427→  const points = data.map((v, i) => {
  1428→    const x = (i / (data.length - 1)) * width;
  1429→    const y = height - ((v - minVal) / range) * (height - 4) - 2;
  1430→    return `${x},${y}`;
  1431→  }).join(' ');
  1432→  const lineColor = color || (data[data.length - 1] >= data[0] ? '#f59e0b' : '#ef4444');
  1433→  return (
  1434→    <svg width={width} height={height} className="inline-block" aria-hidden="true">
  1435→      <polyline points={points} fill="none" stroke={lineColor} strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
  1436→    </svg>
  1437→  );
  1438→}
