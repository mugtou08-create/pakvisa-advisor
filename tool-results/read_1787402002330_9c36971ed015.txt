    24→// ============================================================
    25→// Live Clock Component
    26→// ============================================================
    27→function LiveClock({ timezone, cityName }: { timezone: string; cityName: string }) {
    28→  const [time, setTime] = useState('');
    29→  const [date, setDate] = useState('');
    30→
    31→  useEffect(() => {
    32→    function updateTime() {
    33→      try {
    34→        const now = new Date();
    35→        const formatter = new Intl.DateTimeFormat('en-US', {
    36→          timeZone: timezone,
    37→          hour: '2-digit',
    38→          minute: '2-digit',
    39→          second: '2-digit',
    40→          hour12: true,
    41→        });
    42→        const dateFormatter = new Intl.DateTimeFormat('en-US', {
    43→          timeZone: timezone,
    44→          weekday: 'short',
    45→          month: 'short',
    46→          day: 'numeric',
    47→        });
    48→        setTime(formatter.format(now));
    49→        setDate(dateFormatter.format(now));
    50→      } catch {
    51→        setTime('—');
    52→        setDate('—');
    53→      }
    54→    }
    55→    updateTime();
    56→    const interval = setInterval(updateTime, 1000);
    57→    return () => clearInterval(interval);
    58→  }, [timezone]);
    59→
    60→  // Calculate time difference from PKT (Asia/Karachi)
    61→  const timeDiff = useMemo(() => {
    62→    try {
    63→      const now = new Date();
    64→      // Get UTC offsets in minutes for both timezones
    65→      const targetParts = new Intl.DateTimeFormat('en-US', {
    66→        timeZone: timezone,
    67→        timeZoneName: 'shortOffset',
    68→      }).formatToParts(now);
    69→      const pktParts = new Intl.DateTimeFormat('en-US', {
    70→        timeZone: 'Asia/Karachi',
    71→        timeZoneName: 'shortOffset',
    72→      }).formatToParts(now);
    73→
    74→      const parseOffset = (parts: Intl.DateTimeFormatPart[]) => {
    75→        const tzPart = parts.find(p => p.type === 'timeZoneName');
    76→        if (!tzPart) return 0;
    77→        const match = tzPart.value.match(/GMT([+-]?)(\d{1,2})(?::(\d{2}))?/);
    78→        if (!match) return 0;
    79→        const sign = match[1] === '-' ? -1 : 1;
    80→        const hours = parseInt(match[2]);
    81→        const minutes = match[3] ? parseInt(match[3]) : 0;
    82→        return sign * (hours * 60 + minutes);
    83→      };
    84→
    85→      const targetOffset = parseOffset(targetParts);
    86→      const pktOffset = parseOffset(pktParts);
    87→      const diffHours = (targetOffset - pktOffset) / 60;
    88→
    89→      if (diffHours === 0) return 'Same as PKT';
    90→      if (diffHours > 0) return `+${diffHours}h from PKT`;
    91→      return `${diffHours}h from PKT`;
    92→    } catch {
    93→      return '';
    94→    }
    95→  }, [timezone]);
    96→
    97→  return (
    98→    <div className="flex items-center gap-3 rounded-lg bg-sky-50 dark:bg-sky-950/20 border border-sky-200 dark:border-sky-800 p-3">
    99→      <Clock className="w-5 h-5 text-sky-600 shrink-0" />
   100→      <div>
   101→        <p className="text-sm font-semibold text-sky-900 dark:text-sky-200">{time}</p>
   102→        <p className="text-xs text-sky-700 dark:text-sky-400">{date} {timeDiff && `· ${timeDiff}`}</p>
   103→      </div>
   104→    </div>
   105→  );
   106→}

   111→function CurrencyConverter({ currencyCode, currencyName }: { currencyCode: string; currencyName: string }) {
   112→  const [amount, setAmount] = useState<string>('100');
   113→  const [direction, setDirection] = useState<'topkr' | 'frompkr'>('topkr');
   114→
   115→  const rate = EXCHANGE_RATES[currencyCode];
   116→  const [topkr, frompkr] = useMemo(() => {
   117→    if (!rate) return ['—', '—'];
   118→    return [
   119→      (parseFloat(amount || '0') * rate).toLocaleString('en-PK', { maximumFractionDigits: 2 }),
   120→      (parseFloat(amount || '0') / rate).toLocaleString(undefined, { maximumFractionDigits: 2 }),
   121→    ];
   122→  }, [amount, rate]);
   123→
   124→  return (
   125→    <div className="rounded-lg bg-emerald-50 dark:bg-emerald-950/20 border border-emerald-200 dark:border-emerald-800 p-3 space-y-2">
   126→      <div className="flex items-center justify-between">
   127→        <div className="flex items-center gap-2">
   128→          <Banknote className="w-4 h-4 text-emerald-600" />
   129→          <span className="text-xs font-semibold text-emerald-900 dark:text-emerald-200">Currency Converter</span>
   130→        </div>
   131→        <Badge variant="outline" className="text-[10px] border-emerald-300 dark:border-emerald-700 text-emerald-700 dark:text-emerald-400">
   132→          {currencyCode} · {currencyName}
   133→        </Badge>
   134→      </div>
   135→      <div className="flex items-center gap-2">
   136→        <input
   137→          type="number"
   138→          value={amount}
   139→          onChange={(e) => setAmount(e.target.value)}
   140→          className="w-full h-8 px-2 rounded-md border bg-white dark:bg-background text-sm focus:outline-none focus:ring-1 focus:ring-emerald-500"
   141→          min="0"
   142→        />
   143→        <button
   144→          onClick={() => setDirection((d) => d === 'topkr' ? 'frompkr' : 'topkr')}
   145→          className="p-1.5 rounded-md border bg-white dark:bg-background hover:bg-muted transition-colors"
   146→          title="Swap direction"
   147→        >
   148→          <ArrowRightLeft className="w-3.5 h-3.5 text-emerald-600" />
   149→        </button>
   150→      </div>
   151→      {rate ? (
   152→        <div className="flex items-center justify-between text-xs">
   153→          <span className="text-muted-foreground">
   154→            {direction === 'topkr' ? `${amount || '0'} ${currencyCode} =` : `PKR ${amount || '0'} =`}
   155→          </span>
   156→          <span className="font-semibold text-emerald-800 dark:text-emerald-300">
   157→            {direction === 'topkr' ? `PKR ${topkr}` : `${frompkr} ${currencyCode}`}
   158→          </span>
   159→        </div>
   160→      ) : (
   161→        <p className="text-xs text-muted-foreground">Exchange rate not available</p>
   162→      )}
   163→      <p className="text-[10px] text-emerald-700/70 dark:text-emerald-400/70">
   164→        Rate: 1 {currencyCode} ≈ PKR {rate?.toFixed(2) || '—'} · Approximate, verify before transaction
   165→      </p>
   166→    </div>
   167→  );
   168→}

   391→// ============================================================
   392→// Weather Info Section
   393→// ============================================================
   394→function WeatherInfo({ avgTemp, monthlyTemps }: { avgTemp: string; monthlyTemps: Record<string, number> }) {
   395→  const temps = Object.entries(monthlyTemps);
   396→  if (temps.length === 0 && !avgTemp) return null;
   397→
   398→  const maxTemp = temps.length > 0 ? Math.max(...temps.map(([, v]) => v)) : parseInt(avgTemp) || 0;
   399→  const minTemp = temps.length > 0 ? Math.min(...temps.map(([, v]) => v)) : parseInt(avgTemp) || 0;
   400→
   401→  return (
   402→    <div>
   403→      <h5 className="text-xs font-semibold text-muted-foreground uppercase tracking-wide mb-2">Climate & Weather</h5>
   404→      <div className="rounded-lg bg-muted/50 p-3 space-y-3">
   405→        <div className="flex items-center justify-between text-sm">
   406→          <span className="flex items-center gap-1.5">
   407→            <Thermometer className="w-4 h-4 text-red-500" />
   408→            <span className="font-medium">Avg Temperature</span>
   409→          </span>
   410→          <span className="font-semibold">{avgTemp}°C</span>
   411→        </div>
   412→        <div className="flex items-center gap-3 text-xs text-muted-foreground">
   413→          <span>🔥 High: {maxTemp}°C</span>
   414→          <span>❄️ Low: {minTemp}°C</span>
   415→        </div>
   416→        {temps.length > 0 && (
   417→          <div className="flex items-end gap-0.5 h-8">
   418→            {temps.map(([month, temp], idx) => {
   419→              const height = maxTemp > minTemp ? ((temp - minTemp) / (maxTemp - minTemp)) * 100 : 50;
   420→              return (
   421→                <div key={month} className="flex-1 flex flex-col items-center gap-0.5">
   422→                  <div
   423→                    className="w-full rounded-t-sm min-h-[2px] transition-all"
   424→                    style={{
   425→                      height: 'height: ' + Math.max(height, 5) + '%' === 'height: ' + Math.max(height, 5) + '%' ? Math.max(height, 5) + '%' : Math.max(height, 5) + '%',
   426→                      backgroundColor: temp > 30 ? '#ef4444' : temp > 20 ? '#f59e0b' : temp > 10 ? '#3b82f6' : '#6366f1',
   427→                    }}
   428→                  />
   429→                  <span className="text-[8px] text-muted-foreground">{MONTH_NAMES[idx]?.[0] || ''}</span>
   430→                </div>
   431→              );
   432→            })}
   433→          </div>
   434→        )}
   435→      </div>
   436→    </div>
   437→  );
   438→}

   745→  requirements,
