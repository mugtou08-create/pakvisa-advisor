   283→                    </div>
   284→                  </div>
   285→                ))}

   290→                </div>
   291→              </CardContent>
   292→            </Card>
   293→          ))}
   294→        </div>
   295→
   296→          {/* Comparison Table - Task 10 B4 */}
   297→          <ComparisonTable results={results} />
   298→        </div>
   299→      )}
   300→    </div>
   301→  );
   302→}
   303→// ============ WHAT-IF SIMULATOR ============
   304→export function WhatIfSimulator({ scoreResult, userProfile }: { scoreResult: ScoreBreakdown; userProfile: UserProfileData }) {
   305→  const [healthInsurance, setHealthInsurance] = useState(userProfile.hasHealthInsurance);
   306→  const [returnTicket, setReturnTicket] = useState(userProfile.hasReturnTicket);
   307→  const [hotelBooking, setHotelBooking] = useState(userProfile.hasHotelBooking);
   308→  const [sponsor, setSponsor] = useState(userProfile.hasSponsor);
   309→  const [priorTravel, setPriorTravel] = useState(userProfile.hasPriorTravel);
   310→  const [income, setIncome] = useState(userProfile.monthlyIncomeUSD);
   311→  const [savings, setSavings] = useState(userProfile.savingsUSD);
   312→  const [budget, setBudget] = useState(userProfile.budgetUSD);
   313→  const [recalculating, setRecalculating] = useState(false);
   314→  const [afterScore, setAfterScore] = useState<ScoreBreakdown | null>(null);
   315→
   316→  const applyChanges = async () => {
   317→    setRecalculating(true);
   318→    const modifiedProfile = {
   319→      ...userProfile,
   320→      hasHealthInsurance: healthInsurance,
   321→      hasReturnTicket: returnTicket,
   322→      hasHotelBooking: hotelBooking,
   323→      hasSponsor: sponsor,
   324→      hasPriorTravel: priorTravel,
   325→      monthlyIncomeUSD: income,
   326→      savingsUSD: savings,
   327→      budgetUSD: budget,
   328→    };
   329→    try {
   330→      const res = await fetch('/api/score', {
   331→        method: 'POST',
   332→        headers: { 'Content-Type': 'application/json' },
   333→        body: JSON.stringify({ countryCode: scoreResult.countryCode, profile: modifiedProfile }),
   334→      });
   335→      const data = await res.json();
   336→      if (data.data) {
   337→        setAfterScore(data.data);
   338→      } else {
   339→        toast.error(data.error || 'Recalculation failed');
   340→      }
   341→    } catch {
   342→      toast.error('Network error during recalculation');
   343→    }
   344→    setRecalculating(false);
   345→  };
   346→
   347→  const delta = afterScore ? afterScore.finalScore - scoreResult.finalScore : 0;
   348→
   349→  return (
   350→    <div className="space-y-4">
   351→      <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   352→        {/* Toggle Switches */}
   353→        <div className="space-y-3">
   354→          <h5 className="text-xs font-semibold text-muted-foreground uppercase tracking-wider">Quick Actions</h5>
   355→          {[
   356→            { label: 'Add Health Insurance', desc: '+5% health component', checked: healthInsurance, onChange: setHealthInsurance, icon: Shield },
   357→            { label: 'Book Return Ticket', desc: '+15% purpose component', checked: returnTicket, onChange: setReturnTicket, icon: Plane },
   358→            { label: 'Book Hotel', desc: '+10% purpose component', checked: hotelBooking, onChange: setHotelBooking, icon: Home },
   359→            { label: 'Add Financial Sponsor', desc: '+20% funds component', checked: sponsor, onChange: setSponsor, icon: Users },
   360→            { label: 'Have Prior Travel', desc: '+10% travel history', checked: priorTravel, onChange: setPriorTravel, icon: Globe },
   361→          ].map((item) => (
   362→            <div key={item.label} className="flex items-center justify-between p-2.5 rounded-lg border hover:bg-muted/50 transition-colors">
   363→              <div className="flex items-center gap-2.5">
   364→                <item.icon className="w-4 h-4 text-muted-foreground" />
   365→                <div>
   366→                  <div className="text-sm font-medium">{item.label}</div>
   367→                  <div className="text-[10px] text-amber-600 dark:text-amber-400">{item.desc}</div>
   368→                </div>
   369→              </div>
   370→              <Switch checked={item.checked} onCheckedChange={item.onChange} />
   371→            </div>
   372→          ))}
   373→        </div>
   374→
   375→        {/* Sliders */}
   376→        <div className="space-y-4">
   377→          <h5 className="text-xs font-semibold text-muted-foreground uppercase tracking-wider">Financial Adjustments</h5>
   378→          <div className="space-y-2">
   379→            <div className="flex items-center justify-between text-sm">
   380→              <Label className="text-muted-foreground">Monthly Income</Label>
   381→              <span className="font-mono text-xs font-medium">${income.toLocaleString()}</span>
   382→            </div>
   383→            <Slider value={[income]} onValueChange={(v) => setIncome(v[0])} min={0} max={10000} step={100} />
   384→          </div>
   385→          <div className="space-y-2">
   386→            <div className="flex items-center justify-between text-sm">
   387→              <Label className="text-muted-foreground">Savings</Label>
   388→              <span className="font-mono text-xs font-medium">${savings.toLocaleString()}</span>
   389→            </div>
   390→            <Slider value={[savings]} onValueChange={(v) => setSavings(v[0])} min={0} max={100000} step={1000} />
   391→          </div>
   392→          <div className="space-y-2">
   393→            <div className="flex items-center justify-between text-sm">
   394→              <Label className="text-muted-foreground">Travel Budget</Label>
   395→              <span className="font-mono text-xs font-medium">${budget.toLocaleString()}</span>
   396→            </div>
   397→            <Slider value={[budget]} onValueChange={(v) => setBudget(v[0])} min={0} max={50000} step={500} />
   398→          </div>
   399→        </div>
   400→      </div>
   401→
   402→      {/* Apply Button */}
   403→      <Button
   404→        onClick={applyChanges}
   405→        disabled={recalculating}
   406→        className="w-full bg-amber-600 hover:bg-amber-700 text-white"
   407→        size="lg"
   408→      >
   409→        {recalculating ? (
   410→          <><RotateCcw className="w-4 h-4 mr-2 animate-spin" /> Recalculating...</>
   411→        ) : (
   412→          <><Zap className="w-4 h-4 mr-2" /> Apply Changes & Recalculate</>
   413→        )}
   414→      </Button>
   415→
   416→      {/* Comparison Result */}
   417→      {afterScore && (
   418→        <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }}>
   419→          <div className="p-4 rounded-lg border bg-muted/30 space-y-3">
   420→            <h5 className="text-sm font-semibold">Score Comparison</h5>
   421→            <div className="flex items-center justify-center gap-4">
   422→              <div className="text-center">
   423→                <div className="text-xs text-muted-foreground mb-1">Before</div>
   424→                <div className="text-3xl font-bold">{Math.round(scoreResult.finalScore)}</div>
   425→              </div>
   426→              <div className="flex flex-col items-center">
   427→                <ArrowRight className={`w-6 h-6 ${delta >= 0 ? 'text-amber-500' : 'text-red-500'}`} />
   428→                <span className={`text-sm font-bold ${delta >= 0 ? 'text-amber-600' : 'text-red-600'}`}>
   429→                  {delta >= 0 ? '+' : ''}{delta.toFixed(1)}
   430→                </span>
   431→              </div>
   432→              <div className="text-center">
   433→                <div className="text-xs text-muted-foreground mb-1">After</div>
   434→                <div className={`text-3xl font-bold ${delta >= 0 ? 'text-amber-600' : 'text-red-600'}`}>
   435→                  {Math.round(afterScore.finalScore)}
   436→                </div>
   437→              </div>
   438→            </div>
   439→            {/* Component-level comparison */}
   440→            <div className="space-y-1.5 pt-2 border-t">
   441→              {[
   442→                { label: 'Eligibility', before: scoreResult.eligibility, after: afterScore.eligibility },
   443→                { label: 'Visa Likelihood', before: scoreResult.visaLikelihood, after: afterScore.visaLikelihood },
   444→                { label: 'Cost Suitability', before: scoreResult.costSuitability, after: afterScore.costSuitability },
   445→              ].map((item) => {
   446→                const diff = item.after - item.before;
   447→                return (
   448→                  <div key={item.label} className="flex items-center gap-2 text-sm">
   449→                    <span className="w-28 text-muted-foreground text-xs">{item.label}</span>
   450→                    <span className="w-12 text-right font-mono text-xs">{Math.round(item.before)}</span>
   451→                    <span className={`text-xs font-medium ${diff >= 0 ? 'text-amber-600' : 'text-red-600'} w-12 text-center`}>
   452→                      {diff >= 0 ? '+' : ''}{diff.toFixed(1)}
   453→                    </span>
   454→                    <span className="w-12 text-right font-mono text-xs font-medium">{Math.round(item.after)}</span>
   455→                  </div>
   456→                );
   457→              })}
   458→            </div>
   459→            {delta >= 0 ? (
   460→              <p className="text-xs text-amber-600 text-center">
   461→                <CheckCircle2 className="w-3 h-3 inline mr-1" />
   462→                Changes improved your score! Consider applying these in your application.
   463→              </p>
   464→            ) : (
   465→              <p className="text-xs text-red-600 text-center">
   466→                <AlertTriangle className="w-3 h-3 inline mr-1" />
   467→                These changes would lower your score. Reconsider before proceeding.
   468→              </p>
   469→            )}
   470→          </div>
   471→        </motion.div>
   472→      )}
   473→    </div>
   474→  );
   475→}

   504→        category: 'Documents',
   505→      });

   515→        completed: false,
   516→        category: 'Financial',
   517→      });
   518→    });

   524→        task: tip,
   525→        priority: 'medium',
   526→        effort: 'quick',
   527→        completed: false,
   528→        category: 'Travel',

   539→        category: 'Health',
   540→      });
   541→    }

   543→    // Remaining tips
