   728→// ============ TYPING TEXT ANIMATION ============
   729→export function TypingText({ phrases, className = '' }: { phrases: string[]; className?: string }) {
   730→  const [current, setCurrent] = useState(0);
   731→  const [text, setText] = useState('');
   732→  const [isDeleting, setIsDeleting] = useState(false);
   733→
   734→  useEffect(() => {
   735→    const phrase = phrases[current];
   736→    const timeout = setTimeout(() => {
   737→      if (!isDeleting) {
   738→        setText(phrase.substring(0, text.length + 1));
   739→        if (text.length === phrase.length) {
   740→          setTimeout(() => setIsDeleting(true), 2200);
   741→        }
   742→      } else {
   743→        setText(phrase.substring(0, text.length - 1));
   744→        if (text.length === 0) {
   745→          setIsDeleting(false);
   746→          setCurrent((current + 1) % phrases.length);
   747→        }
   748→      }
   749→    }, isDeleting ? 25 : 55);
   750→    return () => clearTimeout(timeout);
   751→  }, [text, isDeleting, current, phrases]);
   752→
   753→  return (
   754→    <span className={className}>
   755→      {text}
   756→      <span className="inline-block w-[2px] h-[1em] bg-white/80 ml-0.5 align-middle typing-cursor-blink" />
   757→    </span>

   810→// ============ MINI RADAR CHART (3-axis for country cards) ============
   811→export function MiniRadarChart({ safety, cost, ease }: { safety: number; cost: number; ease: number }) {
   812→  const cx = 28, cy = 28, r = 22;
   813→  const axes = 3;
   814→  const angleStep = (2 * Math.PI) / axes;
   815→  const startAngle = -Math.PI / 2;
   816→  const s = safety / 10;
   817→  const c = Math.max(0.05, 1 - cost / 3000);
   818→  const e = ease;
   819→  const values = [s, c, e];
   820→  const points = values.map((val, i) => {
   821→    const angle = startAngle + i * angleStep;
   822→    return `${cx + val * r * Math.cos(angle)},${cy + val * r * Math.sin(angle)}`;
   823→  }).join(' ');
   824→  const gridLines = [0.33, 0.66, 1].map(level => {
   825→    const pts = Array.from({ length: axes }, (_, i) => {
   826→      const angle = startAngle + i * angleStep;
   827→      return `${cx + level * r * Math.cos(angle)},${cy + level * r * Math.sin(angle)}`;
   828→    }).join(' ');
   829→    return <polygon key={level} points={pts} fill="none" stroke="currentColor" strokeWidth="0.3" className="text-white/20" />;
   830→  });
   831→
   832→  return (
   833→    <svg width="56" height="56" viewBox="0 0 56 56" className="inline-block opacity-80">
   834→      {gridLines}
   835→      {Array.from({ length: axes }, (_, i) => {
   836→        const angle = startAngle + i * angleStep;
   837→        return <line key={i} x1={cx} y1={cy} x2={cx + r * Math.cos(angle)} y2={cy + r * Math.sin(angle)} stroke="currentColor" strokeWidth="0.3" className="text-white/20" />;
   838→      })}
   839→      <polygon points={points} fill="rgba(249,115,22,0.15)" stroke="#fbbf24" strokeWidth="1" />
   840→      {values.map((val, i) => {
   841→        const angle = startAngle + i * angleStep;
   842→        return <circle key={i} cx={cx + val * r * Math.cos(angle)} cy={cy + val * r * Math.sin(angle)} r="1.5" fill="#fbbf24" />;
   843→      })}
   844→    </svg>
   845→  );
   846→}

   878→// ============ ANIMATED SCORE NUMBER ============
   879→export function AnimatedScoreNumber({ value, delay = 0 }: { value: number; delay?: number }) {
   880→  const [displayed, setDisplayed] = useState(0);
   881→  const hasAnimated = useRef(false);
   882→
   883→  useEffect(() => {
   884→    if (hasAnimated.current) return;
   885→    hasAnimated.current = true;
   886→    const start = performance.now() + delay;
   887→    const duration = 1200;
   888→    const animate = (now: number) => {
   889→      const elapsed = now - start;
   890→      if (elapsed < 0) { requestAnimationFrame(animate); return; }
   891→      const progress = Math.min(elapsed / duration, 1);
   892→      const eased = 1 - Math.pow(1 - progress, 3);
   893→      setDisplayed(Math.round(eased * value));
   894→      if (progress < 1) requestAnimationFrame(animate);
   895→    };
   896→    requestAnimationFrame(animate);
   897→  }, [value, delay]);
   898→
   899→  return <span>{displayed}</span>;
   900→}

  1356→// ============ MICRO SPARKLINE ============
  1357→export function MicroSparkline({ data, color, width = 60, height = 20 }: { data: number[]; color?: string; width?: number; height?: number }) {
  1358→  if (!data || data.length === 0) return null;
  1359→  const maxVal = Math.max(...data);
  1360→  const minVal = Math.min(...data);
  1361→  const range = maxVal - minVal || 1;
  1362→  const points = data.map((v, i) => {
  1363→    const x = (i / (data.length - 1)) * width;
  1364→    const y = height - ((v - minVal) / range) * (height - 4) - 2;
  1365→    return `${x},${y}`;
  1366→  }).join(' ');
  1367→  const lineColor = color || (data[data.length - 1] >= data[0] ? '#f59e0b' : '#ef4444');
  1368→  return (
  1369→    <svg width={width} height={height} className="inline-block" aria-hidden="true">
  1370→      <polyline points={points} fill="none" stroke={lineColor} strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
  1371→    </svg>
  1372→  );
  1373→}
