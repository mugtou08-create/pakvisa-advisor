   762→function ContactFormEmbedded() {
   763→  const [name, setName] = useState('');
   764→  const [email, setEmail] = useState('');
   765→  const [subject, setSubject] = useState('');
   766→  const [message, setMessage] = useState('');
   767→  const [sending, setSending] = useState(false);
   768→  const [sent, setSent] = useState(false);
   769→
   770→  const handleSubmit = async (e: React.FormEvent) => {
   771→    e.preventDefault();
   772→    if (!name.trim() || !email.trim() || !message.trim()) {
   773→      toast.error('Please fill in name, email, and message');
   774→      return;
   775→    }
   776→    setSending(true);
   777→    try {
   778→      const res = await fetch('/api/contact', {
   779→        method: 'POST',
   780→        headers: { 'Content-Type': 'application/json' },
   781→        body: JSON.stringify({ name, email, subject, message }),
   782→      });
   783→      const data = await res.json();
   784→      if (data.success) {
   785→        toast.success(data.message);
   786→        setSent(true);
   787→        setName(''); setEmail(''); setSubject(''); setMessage('');
   788→        setTimeout(() => setSent(false), 5000);
   789→      } else {
   790→        toast.error(data.message || 'Failed to send message');
   791→      }
   792→    } catch {
   793→      toast.error('Connection error. Please try again.');
   794→    } finally {
   795→      setSending(false);
   796→    }
   797→  };
   798→
   799→  if (sent) {
   800→    return (
   801→      <div className="flex flex-col items-center justify-center py-8 text-center">
   802→        <div className="w-14 h-14 rounded-full bg-emerald-100 dark:bg-emerald-900/30 flex items-center justify-center mb-3">
   803→          <CheckCircle2 className="w-7 h-7 text-emerald-600" />
   804→        </div>
   805→        <h4 className="font-semibold text-lg">Message Sent!</h4>
   806→        <p className="text-sm text-muted-foreground mt-1">Thank you for reaching out. We will respond shortly.</p>
   807→      </div>
   808→    );
   809→  }
   810→
   811→  return (
   812→    <form onSubmit={handleSubmit} className="space-y-4">
   813→      <p className="text-sm text-muted-foreground">Have a question? We will get back to you within 24 hours.</p>
   814→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   815→        <div className="space-y-2">
   816→          <Label htmlFor="modal-contact-name" className="text-sm font-medium">
   817→            <User className="w-3.5 h-3.5 inline mr-1" /> Name <span className="text-red-500">*</span>
   818→          </Label>
   819→          <Input id="modal-contact-name" placeholder="Your full name" value={name} onChange={(e) => setName(e.target.value)} maxLength={100} />
   820→        </div>
   821→        <div className="space-y-2">
