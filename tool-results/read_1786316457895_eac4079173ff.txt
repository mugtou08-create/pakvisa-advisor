    26→function getVisaStatus(c: CountryData) {
    27→  if (c.visaFree) return { label: 'Visa Free', color: 'bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-400' };
    28→  if (c.visaOnArrival) return { label: 'Visa on Arrival', color: 'bg-orange-100 text-orange-700 dark:bg-orange-900/40 dark:text-orange-400' };
    29→  if (c.etaAvailable) return { label: 'e-Visa', color: 'bg-amber-50 text-amber-600 dark:bg-amber-900/30 dark:text-amber-400' };
    30→  return { label: 'Embassy Required', color: 'bg-orange-100 text-orange-700 dark:bg-orange-900/40 dark:text-orange-400' };
    31→}

    46→export function CountryComparisonSwiper({ countries, onSelectCountry }: {
    47→  countries: CountryData[];
    48→  onSelectCountry: (code: string) => void;
    49→}) {
    50→  const [currentIndex, setCurrentIndex] = useState(0);
    51→  const [isPaused, setIsPaused] = useState(false);
    52→  const [direction, setDirection] = useState(0);
    53→  const dragRef = useRef<HTMLDivElement>(null);
    54→  const startXRef = useRef(0);
    55→
    56→  // Take top 8 countries for the swiper
    57→  const swiperCountries = countries.slice(0, 8);
    58→  const total = swiperCountries.length;
    59→
    60→  const goTo = useCallback((newIndex: number, dir: number) => {
    61→    setDirection(dir);
    62→    setCurrentIndex((newIndex + total) % total);
    63→  }, [total]);
    64→
    65→  const goNext = useCallback(() => goTo(currentIndex + 1, 1), [currentIndex, goTo]);
    66→  const goPrev = useCallback(() => goTo(currentIndex - 1, -1), [currentIndex, goTo]);
    67→
    68→  // Auto-rotate every 5 seconds
    69→  useEffect(() => {
    70→    if (isPaused || total === 0) return;
    71→    const timer = setInterval(goNext, 5000);
    72→    return () => clearInterval(timer);
    73→  }, [isPaused, goNext, total]);
    74→
    75→  // Touch/drag handlers
    76→  const handleDragStart = (e: React.MouseEvent | React.TouchEvent) => {
    77→    const clientX = 'touches' in e ? e.touches[0].clientX : e.clientX;
    78→    startXRef.current = clientX;
    79→  };
    80→
    81→  const handleDragEnd = (e: React.MouseEvent | React.TouchEvent) => {
    82→    const clientX = 'changedTouches' in e ? e.changedTouches[0].clientX : e.clientX;
    83→    const diff = startXRef.current - clientX;
    84→    if (Math.abs(diff) > 50) {
    85→      if (diff > 0) goNext();
    86→      else goPrev();
    87→    }
    88→  };
    89→
    90→  if (swiperCountries.length === 0) return null;
    91→
    92→  // Get visible indices for 3-card desktop, 1-card mobile
    93→  const getVisibleIndices = () => {
    94→    const indices: number[] = [];
    95→    for (let offset = -1; offset <= 1; offset++) {
    96→      indices.push((currentIndex + offset + total) % total);
    97→    }
    98→    return indices;
    99→  };
   100→
   101→  const visibleIndices = getVisibleIndices();
   102→  const centerIdx = visibleIndices[1]; // center card
   103→  const centerCountry = swiperCountries[centerIdx];
   104→
   105→  return (
   106→    <div
   107→      className="relative"
   108→      onMouseEnter={() => setIsPaused(true)}
   109→      onMouseLeave={() => setIsPaused(false)}
   110→    >
   111→      {/* Desktop: 3-card 3D stack */}
   112→      <div className="hidden md:flex items-center justify-center gap-4 py-4">
   113→        {/* Left card */}
   114→        <motion.div
   115→          key={visibleIndices[0]}
   116→          initial={{ opacity: 0, x: direction > 0 ? -80 : -40, scale: 0.85 }}
   117→          animate={{ opacity: 0.5, x: 0, scale: 0.85 }}
   118→          exit={{ opacity: 0, x: direction > 0 ? 40 : -80, scale: 0.85 }}
   119→          transition={{ duration: 0.4, ease: 'easeInOut' }}
   120→          className="w-64 shrink-0 cursor-pointer"
   121→          style={{ perspective: '1000px', transform: 'rotateY(8deg)' }}
   122→          onClick={() => goTo(visibleIndices[0], -1)}
   123→        >
   124→          <SwipeCard country={swiperCountries[visibleIndices[0]]} onSelect={onSelectCountry} isSide />
   125→        </motion.div>
   126→
   127→        {/* Center card */}
   128→        <motion.div
   129→          key={centerIdx}
   130→          initial={{ opacity: 0, x: direction * 120, scale: 0.9 }}
   131→          animate={{ opacity: 1, x: 0, scale: 1 }}
   132→          exit={{ opacity: 0, x: -direction * 120, scale: 0.9 }}
   133→          transition={{ duration: 0.4, ease: 'easeInOut' }}
   134→          className="w-72 shrink-0 z-10"
   135→        >
   136→          <SwipeCard country={centerCountry} onSelect={onSelectCountry} isCenter />
   137→        </motion.div>
   138→
   139→        {/* Right card */}
   140→        <motion.div
   141→          key={visibleIndices[2]}
   142→          initial={{ opacity: 0, x: direction > 0 ? 80 : 40, scale: 0.85 }}
   143→          animate={{ opacity: 0.5, x: 0, scale: 0.85 }}
   144→          exit={{ opacity: 0, x: direction > 0 ? -40 : 80, scale: 0.85 }}
   145→          transition={{ duration: 0.4, ease: 'easeInOut' }}
   146→          className="w-64 shrink-0 cursor-pointer"
   147→          style={{ perspective: '1000px', transform: 'rotateY(-8deg)' }}
   148→          onClick={() => goTo(visibleIndices[2], 1)}
   149→        >
   150→          <SwipeCard country={swiperCountries[visibleIndices[2]]} onSelect={onSelectCountry} isSide />
   151→        </motion.div>
   152→      </div>
   153→
   154→      {/* Mobile: single card with swipe */}
   155→      <div
   156→        className="md:hidden overflow-hidden px-4"
   157→        ref={dragRef}
   158→        onMouseDown={handleDragStart}
   159→        onMouseUp={handleDragEnd}
   160→        onTouchStart={handleDragStart}
   161→        onTouchEnd={handleDragEnd}
   162→      >
   163→        <AnimatePresence mode="wait" custom={direction}>
   164→          <motion.div
   165→            key={currentIndex}
   166→            custom={direction}
   167→            initial={{ opacity: 0, x: direction * 200 }}
   168→            animate={{ opacity: 1, x: 0 }}
   169→            exit={{ opacity: 0, x: -direction * 200 }}
   170→            transition={{ duration: 0.35, ease: 'easeInOut' }}
   171→            drag="x"
   172→            dragConstraints={{ left: 0, right: 0 }}
   173→            dragElastic={0.2}
   174→            onDragEnd={(_, info) => {
   175→              if (info.offset.x > 60) goPrev();
   176→              else if (info.offset.x < -60) goNext();
   177→            }}
   178→          >
   179→            <SwipeCard country={swiperCountries[currentIndex]} onSelect={onSelectCountry} isCenter />
   180→          </motion.div>
   181→        </AnimatePresence>
   182→      </div>
   183→
   184→      {/* Navigation buttons */}
   185→      <div className="flex items-center justify-center gap-3 mt-3">
   186→        <Button
   187→          variant="outline"
   188→          size="icon"
   189→          className="h-8 w-8 rounded-full border-amber-300 hover:bg-amber-50 dark:border-amber-700 dark:hover:bg-amber-900/30"
   190→          onClick={goPrev}
   191→          aria-label="Previous country"
   192→        >
   193→          <ChevronLeft className="w-4 h-4 text-amber-700 dark:text-amber-400" />
   194→        </Button>
   195→
   196→        {/* Dot indicators */}
   197→        <div className="flex items-center gap-1.5">
   198→          {swiperCountries.map((_, i) => (
   199→            <button
   200→              key={i}
   201→              onClick={() => goTo(i, i > currentIndex ? 1 : -1)}
   202→              className={`rounded-full transition-all duration-300 ${
   203→                i === currentIndex
   204→                  ? 'w-6 h-2 bg-amber-500'
   205→                  : 'w-2 h-2 bg-amber-300 dark:bg-amber-700 hover:bg-amber-400'
   206→              }`}
   207→              aria-label={`Go to slide ${i + 1}`}
   208→            />
   209→          ))}
   210→        </div>
   211→
   212→        <Button
   213→          variant="outline"
   214→          size="icon"
   215→          className="h-8 w-8 rounded-full border-amber-300 hover:bg-amber-50 dark:border-amber-700 dark:hover:bg-amber-900/30"
   216→          onClick={goNext}
   217→          aria-label="Next country"
   218→        >
   219→          <ChevronRight className="w-4 h-4 text-amber-700 dark:text-amber-400" />
   220→        </Button>
   221→      </div>
   222→    </div>
   223→  );
   224→}

   298→const ACCENT_PRESETS = [
   299→  { name: 'Amber', color: '#F59E0B', primary: '245, 158, 11' },
   300→  { name: 'Orange', color: '#F97316', primary: '249, 115, 22' },

   305→] as const;
   306→
   307→const STORAGE_KEY = 'pakvisa-accent-color';
   308→
   309→function applyAccent(preset: typeof ACCENT_PRESETS[number]) {
   310→  if (typeof document === 'undefined') return;
   311→  const root = document.documentElement;
   312→  const [r, g, b] = preset.primary.split(', ');
   313→  root.style.setProperty('--accent-primary', preset.color);
   314→  root.style.setProperty('--accent-rgb', `${r}, ${g}, ${b}`);
   315→  root.style.setProperty('--accent-foreground', '#FFFFFF');
   316→  root.style.setProperty('--accent-light', `rgba(${r}, ${g}, ${b}, 0.15)`);
   317→  root.style.setProperty('--accent-medium', `rgba(${r}, ${g}, ${b}, 0.3)`);
   318→  localStorage.setItem(STORAGE_KEY, preset.name);
   319→}
   320→
   321→export function ThemeAccentCustomizer({ defaultAccent }: { defaultAccent?: string }) {
   322→  const [selected, setSelected] = useState(defaultAccent || 'Amber');
   323→
   324→  useEffect(() => {
   325→    const saved = localStorage.getItem(STORAGE_KEY);
   326→    if (saved) {
   327→      const preset = ACCENT_PRESETS.find(p => p.name === saved);
   328→      if (preset) {
   329→        applyAccent(preset);
   330→      }
   331→    }
   332→  }, []);
   333→
   334→  const handleSelect = (preset: typeof ACCENT_PRESETS[number]) => {
   335→    setSelected(preset.name);
   336→    applyAccent(preset);
   337→  };
   338→
   339→  const handleReset = () => {
   340→    const defaultPreset = ACCENT_PRESETS[0];
   341→    setSelected(defaultPreset.name);
   342→    applyAccent(defaultPreset);
   343→  };
   344→
   345→  const activePreset = ACCENT_PRESETS.find(p => p.name === selected) || ACCENT_PRESETS[0];
   346→
   347→  return (
   348→    <Card className="card-elevated-1 glass-card">
   349→      <CardHeader className="pb-3">
   350→        <CardTitle className="text-sm font-semibold flex items-center gap-2">
   351→          <Settings className="w-4 h-4 text-amber-500" />
   352→          Accent Color
   353→        </CardTitle>
   354→      </CardHeader>
   355→      <CardContent className="space-y-3">
   356→        {/* Color preset buttons */}
   357→        <div className="flex items-center justify-center gap-3">
   358→          {ACCENT_PRESETS.map((preset) => (
   359→            <button
   360→              key={preset.name}
   361→              onClick={() => handleSelect(preset)}
   362→              className={`press-effect relative w-8 h-8 rounded-full border-2 transition-all duration-200 hover:scale-110 ${
   363→                selected === preset.name
   364→                  ? 'border-amber-900 dark:border-amber-100 ring-2 ring-offset-2 ring-amber-400 dark:ring-offset-background'
   365→                  : 'border-transparent hover:border-amber-400/50'
   366→              }`}
   367→              style={{ backgroundColor: preset.color }}
   368→              aria-label={`Select ${preset.name} accent color`}
   369→              title={preset.name}
   370→            >
   371→              {selected === preset.name && (
   372→                <Check className="w-4 h-4 text-white absolute inset-0 m-auto drop-shadow-sm" strokeWidth={3} />
   373→              )}
   374→            </button>
   375→          ))}
   376→        </div>
   377→
   378→        {/* Current accent name */}
   379→        <p className="text-center text-xs text-muted-foreground">
   380→          Current: <span className="font-medium text-amber-700 dark:text-amber-400">{activePreset.name}</span>
   381→        </p>
   382→
   383→        {/* Reset button */}
   384→        <Button
   385→          variant="ghost"
   386→          size="sm"
   387→          className="w-full text-xs text-muted-foreground hover:text-amber-700 dark:hover:text-amber-400"
   388→          onClick={handleReset}
   389→        >
   390→          <RotateCcw className="w-3 h-3 mr-1" />
   391→          Reset to Default
   392→        </Button>
   393→      </CardContent>
   394→    </Card>
   395→  );
   396→}

   579→
   580→function isFieldFilled(profile: UserProfileData, field: string): boolean {
   581→  const val = (profile as Record<string, unknown>)[field];
   582→  if (val === undefined || val === null || val === '') return false;
   583→  if (typeof val === 'number') return val > 0;
   584→  if (Array.isArray(val)) return val.length > 0;
   585→  if (typeof val === 'boolean') return true;
   586→  return true;
   587→}

   703→  return `${days}d ago`;
