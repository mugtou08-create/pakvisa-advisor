    58→function relativeTime(daysAgo: number): string {
    59→  if (daysAgo === 0) return 'Today';
    60→  if (daysAgo === 1) return 'Yesterday';
    61→  if (daysAgo < 7) return `${daysAgo} days ago`;
    62→  if (daysAgo < 14) return '1 week ago';
    63→  if (daysAgo < 21) return `${Math.floor(daysAgo / 7)} weeks ago`;
    64→  if (daysAgo < 30) return '3 weeks ago';
    65→  if (daysAgo < 45) return '1 month ago';
    66→  return `${Math.floor(daysAgo / 30)} months ago`;
    67→}

    69→function generatePolicyChanges(countries: CountryData[]): PolicyChangeEntry[] {
    70→  if (!countries || countries.length === 0) return [];
    71→
    72→  const entries: PolicyChangeEntry[] = [];
    73→  const used = new Set<string>();
    74→  const timestamps = [0, 1, 2, 3, 5, 7, 10, 14, 21, 28, 35, 42, 49, 56];
    75→
    76→  // Pick visa-free countries
    77→  const vfCountries = countries.filter(c => c.visaFree);
    78→  const voaCountries = countries.filter(c => c.visaOnArrival && !c.visaFree);
    79→  const etaCountries = countries.filter(c => c.etaAvailable && !c.visaFree && !c.visaOnArrival);
    80→  const embassyCountries = countries.filter(c => !c.visaFree && !c.visaOnArrival && !c.etaAvailable);
    81→
    82→  const pick = (list: CountryData[]) => {
    83→    for (const c of list) {
    84→      if (!used.has(c.code) && entries.length < 10) return c;
    85→    }
    86→    return null;
    87→  };
    88→
    89→  // Visa-free entries (2-3)
    90→  for (let i = 0; i < 3; i++) {
    91→    const c = pick(vfCountries);
    92→    if (!c) break;
    93→    used.add(c.code);
    94→    entries.push({
    95→      id: `vf-${c.code}`, countryCode: c.code, countryName: c.name,
    96→      flagEmoji: c.flagEmoji, flagUrl: c.flagUrl,
    97→      changeType: 'visa-free',
    98→      title: 'Visa-free access confirmed',
    99→      description: `Pakistan passport holders enjoy visa-free entry to ${c.name} for up to ${c.visaTypes?.[0]?.maxDuration || '30 days'}.`,
   100→      detail: `This visa-free arrangement allows Pakistani citizens to enter ${c.name} without any prior visa application. Ensure your passport has at least 6 months validity. Best travel months: ${c.bestTravelMonths}. Average temperature: ${c.avgTempC}°C.`,
   101→      daysAgo: timestamps[entries.length] ?? 10,
   102→    });
   103→  }
   104→
   105→  // VOA entries (1-2)
   106→  for (let i = 0; i < 2; i++) {
   107→    const c = pick(voaCountries);
   108→    if (!c) break;
   109→    used.add(c.code);
   110→    entries.push({
   111→      id: `voa-${c.code}`, countryCode: c.code, countryName: c.name,
   112→      flagEmoji: c.flagEmoji, flagUrl: c.flagUrl,
   113→      changeType: 'processing',
   114→      title: 'Visa on arrival policy updated',
   115→      description: `${c.name} has streamlined its VOA process. Processing now takes approximately ${c.processingDaysMin}-${c.processingDaysMax} business days.`,
   116→      detail: `Pakistan passport holders can obtain a visa on arrival at ${c.name} ports of entry. The fee is approximately $${c.costProfile?.visaFeeUSD || 'varies'}. Maximum stay: ${c.visaTypes?.[0]?.maxDuration || '30 days'}. Extensions may be available.`,
   117→      daysAgo: timestamps[entries.length] ?? 15,
   118→    });
   119→  }
   120→
   121→  // e-Visa entries (2)
   122→  for (let i = 0; i < 2; i++) {
   123→    const c = pick(etaCountries);
   124→    if (!c) break;
   125→    used.add(c.code);
   126→    entries.push({
   127→      id: `eta-${c.code}`, countryCode: c.code, countryName: c.name,
   128→      flagEmoji: c.flagEmoji, flagUrl: c.flagUrl,
   129→      changeType: 'evisa',
   130→      title: 'e-Visa/eTA system launched',
   131→      description: `${c.name} has launched an electronic visa system for Pakistani passport holders. Apply online before travel.`,
   132→      detail: `The new e-Visa portal allows Pakistanis to apply for a ${c.name} visa entirely online. Processing time: ${c.processingDaysMin}-${c.processingDaysMax} days. Fee: $${c.costProfile?.visaFeeUSD || 'TBD'}. Required documents: valid passport, photographs, and travel itinerary.`,
   133→      daysAgo: timestamps[entries.length] ?? 20,
   134→    });
   135→  }
   136→
   137→  // Embassy/fee entries (2-3)
   138→  for (let i = 0; i < 3; i++) {
   139→    const c = pick([...embassyCountries, ...etaCountries, ...voaCountries]);
   140→    if (!c) break;
   141→    used.add(c.code);
   142→    const isFee = i % 2 === 0;
   143→    entries.push({
   144→      id: `emb-${c.code}`, countryCode: c.code, countryName: c.name,
   145→      flagEmoji: c.flagEmoji, flagUrl: c.flagUrl,
   146→      changeType: isFee ? 'fee-change' : 'requirement',
   147→      title: isFee ? 'Visa fee adjustment' : 'New document requirement added',
   148→      description: isFee
   149→        ? `${c.name} has adjusted its visa fee to $${c.costProfile?.visaFeeUSD || 'updated rate'} (previously $${Math.max(1, (c.costProfile?.visaFeeUSD || 50) - 10)}). Service fee: $${c.costProfile?.serviceFeeUSD || 'varies'}.`
   150→        : `${c.name} now requires additional documentation: ${c.requirements?.[0]?.requirement || 'updated proof of funds'} for all visa applications.`,
   151→      detail: isFee
   152→        ? `The revised fee structure for ${c.name} visas: Visa fee $${c.costProfile?.visaFeeUSD || 'TBD'}, Service fee $${c.costProfile?.serviceFeeUSD || 'TBD'}. Health insurance: $${c.costProfile?.healthInsuranceUSD || 'N/A'}. Budget estimate: $${c.costProfile?.totalMonthlyUSD || 'TBD'}/month for living expenses. `
   153→        : `Updated requirements for ${c.name}: ${c.requirements?.slice(0, 3).map(r => r.requirement).join(', ') || 'Standard documentation'}. Processing time: ${c.processingDaysMin}-${c.processingDaysMax} business days. Check official sources for the most current information.`,
   154→      daysAgo: timestamps[entries.length] ?? 25,
   155→    });
   156→  }
   157→
   158→  return entries.slice(0, 10);
   159→}

   340→interface ChecklistSection {

   348→function buildChecklistSections(country: CountryData | null, userProfile: UserProfileData | null): ChecklistSection[] {
   349→  const sections: ChecklistSection[] = [];
   350→
   351→  // Section 1: Pre-Departure
   352→  const preDepartureItems: { id: string; label: string }[] = [
   353→    { id: 'pre-passport', label: 'Check passport validity (6+ months)' },
   354→    { id: 'pre-insurance', label: 'Purchase travel insurance' },
   355→    { id: 'pre-flights', label: 'Book flights' },
   356→    { id: 'pre-accommodation', label: 'Book accommodation' },
   357→    { id: 'pre-currency', label: `Exchange currency${country ? ` (${country.currencyCode || 'local currency'})` : ''}` },
   358→    { id: 'pre-maps', label: 'Download offline maps' },
   359→    { id: 'pre-bank', label: 'Notify your bank about travel' },
   360→  ];
   361→
   362→  if (country) {
   363→    if (country.visaFree) {
   364→      preDepartureItems.unshift({ id: 'pre-vf-check', label: 'Confirm visa-free entry requirements' });
   365→    } else if (country.visaOnArrival) {
   366→      preDepartureItems.unshift({ id: 'pre-voa-docs', label: 'Prepare VOA documents (return ticket, hotel booking)' });
   367→      preDepartureItems.splice(1, 0, { id: 'pre-voa-fee', label: `Carry VOA fee in cash (~$${country.costProfile?.visaFeeUSD || 'check amount'})` });
   368→    } else if (country.etaAvailable) {
   369→      preDepartureItems.unshift({ id: 'pre-eta-apply', label: 'Apply for e-Visa/eTA online' });
   370→      preDepartureItems.splice(1, 0, { id: 'pre-eta-print', label: 'Print e-Visa approval letter' });
   371→    } else {
   372→      preDepartureItems.unshift({ id: 'pre-embassy-apply', label: 'Apply for visa at embassy/consulate' });
   373→      preDepartureItems.splice(1, 0, { id: 'pre-embassy-appt', label: 'Book embassy appointment' });
   374→      preDepartureItems.splice(2, 0, { id: 'pre-embassy-docs', label: 'Prepare and organize all documents' });
   375→    }
   376→  }
   377→
   378→  if (userProfile) {
   379→    if (userProfile.occupation && userProfile.occupation !== 'Unemployed' && userProfile.occupation !== 'Student') {
   380→      preDepartureItems.push({ id: 'pre-leave', label: 'Request leave from employer' });
   381→    }
   382→    if (userProfile.maritalStatus === 'Married' || (userProfile.dependents && userProfile.dependents > 0)) {
   383→      preDepartureItems.push({ id: 'pre-family', label: 'Arrange family care during travel' });
   384→    }
   385→    if (userProfile.occupation === 'Student') {
   386→      preDepartureItems.push({ id: 'pre-student-docs', label: 'Get student documents / NOC from institution' });
   387→    }
   388→  }
   389→
   390→  sections.push({
   391→    id: 'pre-departure',
   392→    title: `Pre-Departure Checklist${country ? ` — ${country.name}` : ''}`,
   393→    icon: <PlaneTakeoff className="h-4 w-4 text-amber-500" />,
   394→    colorClass: 'text-amber-600 dark:text-amber-400',
   395→    items: preDepartureItems,
   396→  });
   397→
   398→  // Section 2: Required Documents
   399→  const docItems: { id: string; label: string }[] = [
   400→    { id: 'doc-passport', label: 'Valid passport (6+ months validity)' },
   401→    { id: 'doc-photos', label: 'Passport-sized photographs (2x)' },
   402→    { id: 'doc-bank', label: 'Bank statements (last 6 months)' },
   403→    { id: 'doc-flights', label: 'Flight itinerary / booking confirmation' },
   404→    { id: 'doc-hotel', label: 'Hotel reservation confirmation' },
   405→    { id: 'doc-insurance', label: 'Travel insurance certificate' },
   406→    { id: 'doc-employment', label: 'Employment letter / NOC from employer' },
   407→    { id: 'doc-tax', label: 'Income tax returns' },
   408→  ];
   409→
   410→  if (country && country.requirements && country.requirements.length > 0) {
   411→    country.requirements.forEach((req, i) => {
   412→      if (i < 4 && !docItems.some(d => d.label.toLowerCase().includes(req.requirement.toLowerCase().slice(0, 15)))) {
   413→        docItems.push({ id: `doc-req-${i}`, label: req.requirement });

  1149→export function VisaReadinessDashboard({ countries, userProfile }: { countries: CountryData[]; userProfile: UserProfileData | null }) {
  1150→  const { setActiveTab, setSelectedCountry } = useAppStore();
  1151→
  1152→  const metrics = useMemo(() => {
  1153→    // 1. Profile Completeness
  1154→    const profileFields: (keyof UserProfileData)[] = [
  1155→      'fullName', 'age', 'gender', 'passportNumber', 'passportExpiry',
  1156→      'occupation', 'monthlyIncomeUSD', 'education', 'travelPurpose', 'budgetUSD',
  1157→      'maritalStatus', 'intendedStayDays', 'languages', 'hasReturnTicket', 'hasHotelBooking',
  1158→    ];
  1159→    const filledCount = userProfile
  1160→      ? profileFields.filter(k => {
  1161→          const v = userProfile[k];
  1162→          if (Array.isArray(v)) return v.length > 0;
  1163→          return v !== null && v !== undefined && v !== '' && v !== 0;
  1164→        }).length
  1165→      : 0;
  1166→    const profilePct = userProfile ? Math.round((filledCount / profileFields.length) * 100) : 0;
  1167→
  1168→    // 2. Financial Readiness
  1169→    const countriesWithCost = countries.filter(c => c.costProfile);
  1170→    const avgVisaFee = countriesWithCost.length > 0
  1171→      ? countriesWithCost.reduce((s, c) => s + (c.costProfile!.visaFeeUSD || 0), 0) / countriesWithCost.length
  1172→      : 100;
  1173→    const income = userProfile?.monthlyIncomeUSD || 0;
  1174→    const financialPct = income > 0 ? Math.min(100, Math.round((income / (avgVisaFee * 3)) * 100)) : 0;
  1175→
  1176→    // 3. Document Status (proxy from profile + financial readiness)
  1177→    const docPct = userProfile
  1178→      ? Math.round((profilePct * 0.6) + (financialPct > 50 ? 40 : financialPct > 20 ? 25 : 10))
  1179→      : 0;
  1180→
  1181→    // 4. Passport Health
  1182→    let passportPct = 0;
  1183→    if (userProfile?.passportExpiry) {
  1184→      const diffMs = new Date(userProfile.passportExpiry).getTime() - Date.now();
  1185→      const diffMonths = diffMs / (1000 * 60 * 60 * 24 * 30);
  1186→      passportPct = diffMonths >= 12 ? 100 : diffMonths >= 6 ? Math.round((diffMonths / 12) * 100) : diffMonths > 0 ? Math.round((diffMonths / 6) * 30) : 0;
  1187→    }
  1188→
  1189→    const overall = Math.round((profilePct + financialPct + docPct + passportPct) / 4);

  1301→
  1302→interface RecommendationResult {
  1303→  country: CountryData;
  1304→  matchScore: number;
  1305→  reasons: string[];
  1306→}
  1307→
  1308→export function SmartRecommendations({ countries, userProfile }: { countries: CountryData[]; userProfile: UserProfileData | null }) {
  1309→  const { setSelectedCountry } = useAppStore();
  1310→
  1311→  const recommendations = useMemo<RecommendationResult[]>(() => {
  1312→    if (countries.length === 0) return [];
  1313→
  1314→    const scored = countries.map(c => {
  1315→      let score = 50;
  1316→      const reasons: string[] = [];
  1317→
  1318→      // Visa ease (up to +30)
  1319→      if (c.visaFree) { score += 30; reasons.push('Visa-free entry for Pakistani citizens'); }
  1320→      else if (c.visaOnArrival) { score += 22; reasons.push('Visa on arrival — no prior application needed'); }
  1321→      else if (c.etaAvailable) { score += 15; reasons.push('e-Visa available — apply online'); }
  1322→
  1323→      // Safety (up to +15)
  1324→      if (c.safetyRating >= 8) { score += 15; reasons.push(`High safety rating (${c.safetyRating}/10)`); }
  1325→      else if (c.safetyRating >= 6) { score += 8; }
  1326→
  1327→      // Processing speed (up to +10)
  1328→      const avgProcessing = (c.processingDaysMin + c.processingDaysMax) / 2;
  1329→      if (avgProcessing <= 5) { score += 10; reasons.push(`Fast processing (${c.processingDaysMin}-${c.processingDaysMax} days)`); }
  1330→      else if (avgProcessing <= 15) { score += 5; }
  1331→
  1332→      // Budget alignment (up to +20) — only if profile exists
  1333→      if (userProfile?.budgetUSD && userProfile.budgetUSD > 0 && c.costProfile?.totalMonthlyUSD) {
  1334→        const ratio = userProfile.budgetUSD / c.costProfile.totalMonthlyUSD;
  1335→        if (ratio >= 1.5) { score += 20; reasons.push('Well within your budget'); }
  1336→        else if (ratio >= 1) { score += 12; reasons.push('Fits your budget'); }
  1337→        else if (ratio >= 0.7) { score += 5; }
  1338→      }
  1339→
  1340→      return { country: c, matchScore: Math.min(99, score), reasons: reasons.slice(0, 2) };
  1341→    });
  1342→
  1343→    scored.sort((a, b) => b.matchScore - a.matchScore);
  1344→    return scored.slice(0, 6);
  1345→  }, [countries, userProfile]);
  1346→
  1347→  const getVisaBadge = (c: CountryData) => {
  1348→    if (c.visaFree) return { label: 'Visa-Free', cls: 'bg-emerald-500/15 text-emerald-700 dark:text-emerald-400 border-emerald-500/20' };
  1349→    if (c.visaOnArrival) return { label: 'VOA', cls: 'bg-amber-500/15 text-amber-700 dark:text-amber-400 border-amber-500/20' };
  1350→    if (c.etaAvailable) return { label: 'e-Visa', cls: 'bg-orange-500/15 text-orange-700 dark:text-orange-400 border-orange-500/20' };
  1351→    return { label: 'Embassy', cls: 'bg-red-500/15 text-red-700 dark:text-red-400 border-red-500/20' };
  1352→  };
  1353→
  1354→  if (recommendations.length === 0) return null;
  1355→
  1356→  return (
  1357→    <div className="space-y-4">
  1358→      <div className="flex items-center gap-2">
  1359→        <Sparkles className="w-5 h-5 text-amber-500" />
  1360→        <h2 className="text-lg font-bold">Smart Picks For You</h2>
  1361→      </div>
  1362→      <div className="grid grid-cols-2 lg:grid-cols-3 gap-3">
  1363→        {recommendations.map((rec, i) => {
  1364→          const badge = getVisaBadge(rec.country);
  1365→          return (
  1366→            <motion.div
  1367→              key={rec.country.code}
  1368→              initial={{ opacity: 0, y: 16 }} animate={{ opacity: 1, y: 0 }}
  1369→              transition={{ delay: i * 0.07, duration: 0.4 }}
  1370→              className="card-elevated-1 rounded-xl p-4 group hover:shadow-amber-500/10 hover:shadow-lg transition-all duration-300 cursor-pointer"
  1371→              onClick={() => setSelectedCountry(rec.country)}
  1372→            >
  1373→              <div className="flex items-center gap-2 mb-2">
  1374→                <span className="text-xl">{rec.country.flagEmoji}</span>
  1375→                <div className="min-w-0 flex-1">
  1376→                  <p className="text-sm font-semibold truncate">{rec.country.name}</p>
  1377→                  <Badge variant="outline" className={`text-[9px] px-1.5 py-0 border ${badge.cls}`}>{badge.label}</Badge>
  1378→                </div>
  1379→                <div className="text-right shrink-0">
  1380→                  <span className="text-lg font-bold text-amber-600 dark:text-amber-400">{rec.matchScore}%</span>
  1381→                  <p className="text-[9px] text-muted-foreground">match</p>
  1382→                </div>
  1383→              </div>
  1384→              <div className="space-y-1 mb-3">
  1385→                {rec.reasons.map((r, ri) => (
  1386→                  <div key={ri} className="flex items-start gap-1.5">
  1387→                    <Lightbulb className="w-3 h-3 text-amber-400 mt-0.5 shrink-0" />
  1388→                    <span className="text-[11px] text-muted-foreground leading-tight">{r}</span>
  1389→                  </div>
  1390→                ))}
  1391→              </div>
  1392→              <Button size="sm" variant="ghost" className="w-full text-xs gap-1 text-amber-600 dark:text-amber-400 hover:bg-amber-50 dark:hover:bg-amber-950/30 hover:text-amber-700 dark:hover:text-amber-300">
  1393→                View Details <ArrowRight className="w-3 h-3" />
  1394→              </Button>
  1395→            </motion.div>
  1396→          );
  1397→        })}
  1398→      </div>
  1399→    </div>
  1400→  );
