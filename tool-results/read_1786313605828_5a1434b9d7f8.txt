    49→        try {
    50→          const res = await fetch(url);
    51→          if (res.ok) return await res.json();
    52→          if (i < retries - 1) await new Promise(r => setTimeout(r, delay * (i + 1)));

    54→          if (i < retries - 1) await new Promise(r => setTimeout(r, delay * (i + 1)));

    67→        toast.error('Failed to load countries. Retrying...');

    69→      }

    71→      if (!cancelled && statsData?.data) setStats(statsData.data);
    72→    };

    81→      return true;
    82→    });

    84→    // Apply quick filters (single source of truth for visa type filtering)

    87→    else if (quickFilter === 'e-visa') result = result.filter(c => c.etaAvailable);
    88→    else if (quickFilter === 'embassy') result = result.filter(c => !c.visaFree && !c.visaOnArrival && !c.etaAvailable);

   100→        }
   101→        return true;
   102→      });
   103→    }

   105→    return result;

   195→    const diff = Date.now() - timestamp;

   202→    if (days < 7) return `${days}d ago`;

   214→      case 'safety': sorted.sort((a, b) => b.safetyRating - a.safetyRating); break;
   215→      default: break;
   216→    }

   218→  }, [filtered, sortBy, visibleCount]);

   294→        {/* Gradient mesh overlay */}
   295→        <div className="hero-mesh-bg" />
   296→        {/* Floating decorative orbs */}
   297→        <div className="hero-orb hero-orb-1" />
   298→        <div className="hero-orb hero-orb-2" />
   299→        <div className="hero-orb hero-orb-3" />

   426→                  {recentSearches.map((entry, idx) => (
