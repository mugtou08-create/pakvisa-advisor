    35→// Animated Counter Hook
    36→// ============================================================
    37→function useAnimatedCounter(target: number, duration: number = 1200, enabled: boolean = true) {
    38→  const [count, setCount] = useState(0);
    39→  useEffect(() => {
    40→    if (!enabled || target === 0) return;
    41→    let start = 0;
    42→    let startTime: number | null = null;
    43→    let raf: number;
    44→    const step = (timestamp: number) => {
    45→      if (!startTime) startTime = timestamp;
    46→      const progress = Math.min((timestamp - startTime) / duration, 1);
    47→      // Ease-out cubic
    48→      const eased = 1 - Math.pow(1 - progress, 3);
    49→      const current = Math.round(eased * target);
    50→      setCount(current);
    51→      if (progress < 1) {
    52→        raf = requestAnimationFrame(step);
    53→      }
    54→    };
    55→    raf = requestAnimationFrame(step);
    56→    return () => cancelAnimationFrame(raf);
    57→  }, [target, duration, enabled]);
    58→  return count;
    59→}

   128→// ============================================================
   129→// Pagination Helper
   130→// ============================================================
   131→function generatePageNumbers(current: number, total: number): (number | 'ellipsis')[] {
   132→  if (total <= 7) {
   133→    return Array.from({ length: total }, (_, i) => i + 1);
   134→  }
   135→  const pages: (number | 'ellipsis')[] = [1];
   136→  if (current > 3) pages.push('ellipsis');
   137→  const start = Math.max(2, current - 1);
   138→  const end = Math.min(total - 1, current + 1);
   139→  for (let i = start; i <= end; i++) pages.push(i);
   140→  if (current < total - 2) pages.push('ellipsis');
   141→  if (total > 1) pages.push(total);
   142→  return pages;
   143→}

   145→// ============================================================
   146→// Visa Type Helpers (single source of truth for access type)
   147→// Priority: visaFree > visaOnArrival > etaAvailable > embassy
   148→// ============================================================
   149→function getPrimaryAccess(c: CountryData): string {
   150→  if (c.visaFree) return 'visa-free';
   151→  if (c.visaOnArrival) return 'visa-on-arrival';
   152→  if (c.etaAvailable) return 'e-visa';
   153→  return 'embassy';
   154→}
   155→function getVisaType(c: CountryData) {
   156→  const access = getPrimaryAccess(c);
   157→  switch (access) {
   158→    case 'visa-free':     return { label: 'Visa Free', color: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-400', dot: 'bg-emerald-500' };
   159→    case 'visa-on-arrival': return { label: 'Visa on Arrival', color: 'bg-amber-100 text-amber-700 dark:bg-amber-900/30 dark:text-amber-400', dot: 'bg-amber-500' };
   160→    case 'e-visa':       return { label: 'e-Visa', color: 'bg-sky-100 text-sky-700 dark:bg-sky-900/30 dark:text-sky-400', dot: 'bg-sky-500' };
   161→    default:             return { label: 'Embassy Required', color: 'bg-gray-100 text-gray-600 dark:bg-gray-800 dark:text-gray-400', dot: 'bg-gray-400' };
   162→  }
   163→}
