    28→// Animated Counter Hook
    29→// ============================================================
    30→function useAnimatedCounter(target: number, duration: number = 1200, enabled: boolean = true) {
    31→  const [count, setCount] = useState(0);
    32→  useEffect(() => {
    33→    if (!enabled || target === 0) return;
    34→    let start = 0;
    35→    let startTime: number | null = null;
    36→    let raf: number;
    37→    const step = (timestamp: number) => {
    38→      if (!startTime) startTime = timestamp;
    39→      const progress = Math.min((timestamp - startTime) / duration, 1);
    40→      // Ease-out cubic
    41→      const eased = 1 - Math.pow(1 - progress, 3);
    42→      const current = Math.round(eased * target);
    43→      setCount(current);
    44→      if (progress < 1) {
    45→        raf = requestAnimationFrame(step);
    46→      }
    47→    };
    48→    raf = requestAnimationFrame(step);
    49→    return () => cancelAnimationFrame(raf);
    50→  }, [target, duration, enabled]);
    51→  return count;
    52→}

   124→function generatePageNumbers(current: number, total: number): (number | 'ellipsis')[] {
   125→  if (total <= 7) {
   126→    return Array.from({ length: total }, (_, i) => i + 1);
   127→  }
   128→  const pages: (number | 'ellipsis')[] = [1];
   129→  if (current > 3) pages.push('ellipsis');
   130→  const start = Math.max(2, current - 1);
   131→  const end = Math.min(total - 1, current + 1);
   132→  for (let i = start; i <= end; i++) pages.push(i);
   133→  if (current < total - 2) pages.push('ellipsis');
   134→  if (total > 1) pages.push(total);
   135→  return pages;
   136→}

   138→// ============================================================
   139→// Visa Type Helpers (single source of truth for access type)
   140→// Priority: visaFree > visaOnArrival > etaAvailable > embassy
   141→// ============================================================
   142→function getPrimaryAccess(c: CountryData): string {
   143→  if (c.visaFree) return 'visa-free';
   144→  if (c.visaOnArrival) return 'visa-on-arrival';
   145→  if (c.etaAvailable) return 'e-visa';
   146→  return 'embassy';
   147→}
   148→function getVisaType(c: CountryData) {
   149→  const access = getPrimaryAccess(c);
   150→  switch (access) {
   151→    case 'visa-free':     return { label: 'Visa Free', color: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-400', dot: 'bg-emerald-500' };
   152→    case 'visa-on-arrival': return { label: 'Visa on Arrival', color: 'bg-amber-100 text-amber-700 dark:bg-amber-900/30 dark:text-amber-400', dot: 'bg-amber-500' };
   153→    case 'e-visa':       return { label: 'e-Visa', color: 'bg-sky-100 text-sky-700 dark:bg-sky-900/30 dark:text-sky-400', dot: 'bg-sky-500' };
   154→    default:             return { label: 'Embassy Required', color: 'bg-gray-100 text-gray-600 dark:bg-gray-800 dark:text-gray-400', dot: 'bg-gray-400' };
   155→  }
   156→}
