    38→/** Mini radar chart for country cards (3-axis) */
    39→export function MiniRadarChart({ safety, cost, ease }: { safety: number; cost: number; ease: number }) {
    40→  const cx = 28, cy = 28, r = 22;
    41→  const axes = 3;
    42→  const angleStep = (2 * Math.PI) / axes;
    43→  const startAngle = -Math.PI / 2;
    44→  const s = safety / 10;
    45→  const c = Math.max(0.05, 1 - cost / 3000);
    46→  const e = ease;
    47→  const values = [s, c, e];
    48→  const points = values.map((val, i) => {
    49→    const angle = startAngle + i * angleStep;
    50→    return `${cx + val * r * Math.cos(angle)},${cy + val * r * Math.sin(angle)}`;
    51→  }).join(' ');
    52→  const gridLines = [0.33, 0.66, 1].map(level => {
    53→    const pts = Array.from({ length: axes }, (_, i) => {
    54→      const angle = startAngle + i * angleStep;
    55→      return `${cx + level * r * Math.cos(angle)},${cy + level * r * Math.sin(angle)}`;
    56→    }).join(' ');
    57→    return <polygon key={level} points={pts} fill="none" stroke="currentColor" strokeWidth="0.3" className="text-white/20" />;
    58→  });
    59→
    60→  return (
    61→    <svg width="56" height="56" viewBox="0 0 56 56" className="inline-block opacity-80">
    62→      {gridLines}
    63→      {Array.from({ length: axes }, (_, i) => {
    64→        const angle = startAngle + i * angleStep;
    65→        return <line key={i} x1={cx} y1={cy} x2={cx + r * Math.cos(angle)} y2={cy + r * Math.sin(angle)} stroke="currentColor" strokeWidth="0.3" className="text-white/20" />;
    66→      })}
    67→      <polygon points={points} fill="rgba(16,185,129,0.15)" stroke="#34d399" strokeWidth="1" />
    68→      {values.map((val, i) => {
    69→        const angle = startAngle + i * angleStep;
    70→        return <circle key={i} cx={cx + val * r * Math.cos(angle)} cy={cy + val * r * Math.sin(angle)} r="1.5" fill="#34d399" />;
    71→      })}
    72→    </svg>
    73→  );
    74→}

    82→      src={src}

    86→      className={`object-cover rounded-sm ${className}`}
    87→      unoptimized
    88→    />
    89→  );

    91→export function InteractiveWorldMap({ countries, onSelectCountry }: { countries: CountryData[]; onSelectCountry: (c: CountryData) => void }) {
    92→  const [svgContent, setSvgContent] = useState<string | null>(null);
    93→  const [hoveredCountry, setHoveredCountry] = useState<string | null>(null);
    94→  const [tooltipPos, setTooltipPos] = useState<{ x: number; y: number; name: string; status: { fill: string; label: string } } | null>(null);
    95→  const containerRef = useRef<HTMLDivElement>(null);
    96→
    97→  // Build lookup: SVG country name -> CountryData
    98→  const countryLookup = useMemo(() => {
    99→    const map = new globalThis.Map<string, CountryData>();
   100→    countries.forEach(c => {
   101→      const normalized = c.name.replace(/\s+/g, '').toUpperCase();
   102→      map.set(normalized, c);
   103→      map.set(c.name.toUpperCase(), c);
   104→    });
   105→    Object.entries(COUNTRY_NAME_ALIASES).forEach(([svgName, aliases]) => {
   106→      aliases.forEach(alias => {
   107→        const found = countries.find(co => {
   108→          const coNorm = co.name.replace(/\s+/g, '').toUpperCase();
   109→          const aliasNorm = alias.replace(/\s+/g, '').toUpperCase();
   110→          return coNorm === aliasNorm;
   111→        });
   112→        if (found) map.set(svgName.toUpperCase(), found);
   113→      });
   114→    });
   115→    return map;
   116→  }, [countries]);
   117→
   118→  const getStatus = (countryName: string): { fill: string; label: string } => {
   119→    const c = countryLookup.get(countryName.toUpperCase())
   120→      || countryLookup.get(countryName.replace(/\s+/g, '').toUpperCase());
   121→    if (!c) return VISA_CATEGORY_COLORS.unknown;
   122→    if (c.visaFree) return VISA_CATEGORY_COLORS.visaFree;
   123→    if (c.visaOnArrival) return VISA_CATEGORY_COLORS.visaOnArrival;
   124→    if (c.etaAvailable) return VISA_CATEGORY_COLORS.eVisa;
   125→    return VISA_CATEGORY_COLORS.embassy;
   126→  };
   127→
   128→  // Load SVG and pre-process colors into the string itself
   129→  useEffect(() => {
   130→    fetch('/world-map.svg')
   131→      .then(res => res.text())
   132→      .then(text => {
   133→        let processed = text
   134→          .replace(/<defs>[\s\S]*?<\/defs>/, '')
   135→          .replace(/<rect[^>]*fill="[^"]*"[^>]*\/>/, '');
   136→
   137→        // Inject inline style into path elements with data-map-country
   138→        processed = processed.replace(
   139→          /<path\s+([^>]*?)data-map-country="([^"]+)"([^>]*)\s*\/>/g,
   140→          (_full, before, name, after) => {
   141→            const status = getStatus(name);
   142→            // Inject style with !important fill to override any CSS
   143→            return `<path ${before}data-map-country="${name}"${after} style="fill:${status.fill};stroke:#fff;stroke-width:0.8;cursor:pointer;transition:all 0.2s ease;opacity:0.85" />`;
   144→          }
   145→        );
   146→
   147→        // Style non-country paths (generic landmasses)
   148→        processed = processed.replace(
   149→          /<path\s+(?!([^>]*?)data-map-country)([^>]*?)\s*\/>/g,
   150→          (_full, _before, after) => {
   151→            // Only style paths that don't already have a style from above
   152→            if (_full.includes('style=')) return _full;
   153→            return `<path style="fill:#d4d4d8;stroke:#fff;stroke-width:0.5" ${after.replace(/<\/?path[^>]*>/g, '')} />`;
   154→          }
   155→        );
   156→
   157→        // Add ocean background rect as first child after <svg>
   158→        processed = processed.replace(
   159→          /<svg[^>]*>/,
   160→          (match) => `${match}<rect width="1000" height="500" fill="#e8f4f8" />`
   161→        );
   162→
   163→        setSvgContent(processed);
   164→      })
   165→      .catch(() => setSvgContent(''));
   166→  }, []);
   167→
   168→  const handleMouseMove = useCallback((e: React.MouseEvent) => {
   169→    if (!e.target || !(e.target instanceof SVGElement)) {
   170→      setTooltipPos(null);
   171→      return;
   172→    }
   173→    const path = (e.target as SVGElement).closest('path[data-map-country]') as SVGPathElement | null;
   174→    if (path) {
   175→      const name = path.getAttribute('data-map-country') || '';
   176→      const status = getStatus(name);
   177→      if (hoveredCountry !== name) {
   178→        setHoveredCountry(name);
   179→        const allPaths = containerRef.current?.querySelectorAll('path[data-map-country]');
   180→        allPaths?.forEach(p => {
   181→          if (p === path) {
   182→            p.style.opacity = '1';
   183→            p.style.strokeWidth = '1.5';
   184→          } else {
   185→            p.style.opacity = '0.5';
   186→            p.style.strokeWidth = '0.8';
   187→          }
   188→        });
   189→      }
   190→      const rect = containerRef.current?.getBoundingClientRect();
   191→      if (rect) {
   192→        setTooltipPos({
   193→          x: e.clientX - rect.left,
   194→          y: e.clientY - rect.top,
   195→          name,
   196→          status,
   197→        });
   198→      }
   199→    } else {
   200→      if (hoveredCountry) {
   201→        setHoveredCountry(null);
   202→        const allPaths = containerRef.current?.querySelectorAll('path[data-map-country]');
   203→        allPaths?.forEach(p => {
   204→          p.style.opacity = '0.85';
   205→          p.style.strokeWidth = '0.8';
   206→        });
   207→        setTooltipPos(null);
   208→      }
   209→    }
   210→  }, [hoveredCountry, getStatus]);
   211→
   212→  const handleClick = useCallback((e: React.MouseEvent) => {
   213→    if (!e.target || !(e.target instanceof SVGElement)) return;
   214→    const path = (e.target as SVGElement).closest('path[data-map-country]') as SVGPathElement | null;
   215→    if (!path) return;
   216→    const name = path.getAttribute('data-map-country') || '';
   217→    const c = countryLookup.get(name.toUpperCase())
   218→      || countryLookup.get(name.replace(/\s+/g, '').toUpperCase());
   219→    if (c) onSelectCountry(c);
   220→  }, [countryLookup, onSelectCountry]);
   221→
   222→  const handleMouseLeave = useCallback(() => {
   223→    setHoveredCountry(null);
   224→    setTooltipPos(null);
   225→    const allPaths = containerRef.current?.querySelectorAll('path[data-map-country]');
   226→    allPaths?.forEach(p => {
   227→      p.style.opacity = '0.85';
   228→      p.style.strokeWidth = '0.8';
   229→    });
   230→  }, []);
   231→
   232→  // Count stats for the legend
   233→  const legendStats = useMemo(() => {
   234→    let visaFree = 0, voa = 0, eVisa = 0, embassy = 0, noData = 0;
   235→    const svgCountries = svgContent?.match(/data-map-country="([^"]+)"/g) || [];
   236→    svgCountries.forEach(match => {
   237→      const name = match.match(/"([^"]+)"/)?.[1] || '';
   238→      const status = getStatus(name);
   239→      if (status.fill === VISA_CATEGORY_COLORS.visaFree.fill) visaFree++;
   240→      else if (status.fill === VISA_CATEGORY_COLORS.visaOnArrival.fill) voa++;
   241→      else if (status.fill === VISA_CATEGORY_COLORS.eVisa.fill) eVisa++;
   242→      else if (status.fill === VISA_CATEGORY_COLORS.embassy.fill) embassy++;
   243→      else noData++;
   244→    });
   245→    return { visaFree, voa, eVisa, embassy, noData };
   246→  }, [svgContent, getStatus]);
   247→
   248→  return (
   249→    <div className="bg-gradient-to-br from-slate-50 to-slate-100 dark:from-slate-900 dark:to-slate-800 rounded-2xl p-4 md:p-6 border border-border/50">
   250→      <div className="flex items-center justify-between mb-4">
   251→        <h3 className="text-sm font-semibold flex items-center gap-2">
   252→          <Globe className="w-4 h-4 text-emerald-500" />
   253→          Visa Requirements World Map
   254→        </h3>
   255→        <span className="text-[10px] text-muted-foreground bg-muted px-2 py-0.5 rounded-full">Hover &amp; click countries</span>
   256→      </div>
   257→
   258→      <div
   259→        ref={containerRef}
   260→        className="relative w-full rounded-lg overflow-hidden bg-[#e8f4f8] dark:bg-slate-800"
   261→        style={{ paddingBottom: '50%' }}
   262→        onMouseMove={handleMouseMove}
   263→        onClick={handleClick}
   264→        onMouseLeave={handleMouseLeave}
   265→      >
   266→        {svgContent ? (
   267→          <svg
   268→            viewBox="0 0 1000 500"
   269→            className="absolute inset-0 w-full h-full"
   270→            preserveAspectRatio="xMidYMid meet"
   271→            dangerouslySetInnerHTML={{ __html: svgContent }}
   272→          />
   273→        ) : (
   274→          <div className="absolute inset-0 flex items-center justify-center">
   275→            <div className="flex flex-col items-center gap-3">
   276→              <div className="w-8 h-8 border-2 border-emerald-500 border-t-transparent rounded-full animate-spin" />
   277→              <span className="text-xs text-muted-foreground">Loading world map...</span>
   278→            </div>
   279→          </div>
   280→        )}
   281→
   282→        {/* Floating tooltip */}
   283→        {tooltipPos && (
   284→          <div
   285→            className="absolute z-50 pointer-events-none px-3 py-2 bg-white dark:bg-slate-900 rounded-lg shadow-xl border border-border/50 text-xs whitespace-nowrap"
   286→            style={{
   287→              left: `${Math.min(tooltipPos.x + 12, 90)}%`,
   288→              top: `${Math.max(tooltipPos.y - 10, 2)}%`,
   289→            }}
   290→          >
   291→            <div className="font-semibold">{tooltipPos.name}</div>
   292→            <div className="flex items-center gap-1.5 mt-0.5">
   293→              <span className="inline-block w-2 h-2 rounded-full" style={{ backgroundColor: tooltipPos.status.fill }} />
   294→              <span style={{ color: tooltipPos.status.fill }}>{tooltipPos.status.label}</span>
   295→            </div>
   296→          </div>
   297→        )}
   298→      </div>
   299→
   300→      {/* Legend with counts */}
   301→      <div className="flex flex-wrap gap-x-4 gap-y-2 mt-4 justify-center">
   302→        {Object.entries(VISA_CATEGORY_COLORS).map(([key, item]) => {
   303→          const count = key === 'visaFree' ? legendStats.visaFree
   304→            : key === 'visaOnArrival' ? legendStats.voa
   305→            : key === 'eVisa' ? legendStats.eVisa
   306→            : key === 'embassy' ? legendStats.embassy
   307→            : legendStats.noData;
   308→          return (
   309→            <div key={key} className="flex items-center gap-1.5 text-xs text-muted-foreground">
   310→              <div className="w-3 h-3 rounded-sm shrink-0" style={{ backgroundColor: item.fill }} />

   312→              <span className="text-[10px] bg-muted/50 px-1.5 py-0.5 rounded-full font-medium">{count}</span>
   313→            </div>
   314→          );
   315→        })}
   316→      </div>
   317→    </div>
   318→  );
   319→}

   366→export function BestMatchRecommendations({ countries, scoreResults, onSelectCountry }: { countries: CountryData[]; scoreResults: ScoreBreakdown[]; onSelectCountry: (c: CountryData) => void }) {
   367→  const [loaded, setLoaded] = useState(false);
   368→  const { userProfile } = useAppStore();
   369→
   370→  useEffect(() => {
   371→    const timer = setTimeout(() => setLoaded(true), 2000);
   372→    return () => clearTimeout(timer);
   373→  }, []);
   374→
   375→  const recommendations = useMemo(() => {
   376→    if (scoreResults.length > 0) {
   377→      return scoreResults
   378→        .sort((a, b) => b.finalScore - a.finalScore)
   379→        .slice(0, 3)
   380→        .map((s) => {
   381→          const c = countries.find((co) => co.code === s.countryCode);
   382→          if (!c) return null;
   383→          const reason = s.finalScore >= 70 ? 'Excellent match based on your profile' : s.finalScore >= 40 ? 'Good match with minor improvements needed' : 'Consider improving your profile first';
   384→          return { country: c, matchPct: Math.round(s.finalScore), reason };
   385→        })
   386→        .filter(Boolean) as { country: CountryData; matchPct: number; reason: string }[];
   387→    }
   388→    // Default recommendations
   389→    const defaults = [
   390→      { code: 'MY', reason: 'Visa-free entry for 30 days with beautiful destinations' },
   391→      { code: 'AE', reason: 'Visa on arrival with world-class tourism infrastructure' },
   392→      { code: 'TR', reason: 'Easy e-Visa process with rich cultural experiences' },
   393→    ];
   394→    return defaults.map((d) => {
   395→      const c = countries.find((co) => co.code === d.code);
   396→      return c ? { country: c, matchPct: c.visaFree ? 95 : c.visaOnArrival ? 88 : 75, reason: d.reason } : null;
   397→    }).filter(Boolean) as { country: CountryData; matchPct: number; reason: string }[];
   398→  }, [scoreResults, countries]);
   399→
   400→  if (!loaded) {
   401→    return (
   402→      <div>
   403→        <div className="scroll-section-title">
   404→          <h3 className="text-sm font-semibold flex items-center gap-2 whitespace-nowrap">
   405→            <Target className="w-4 h-4 text-emerald-500" />
   406→            Recommended For You
   407→          </h3>
   408→        </div>
   409→        <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
   410→          {[1, 2, 3].map((i) => (
   411→            <Skeleton key={i} className="h-28 rounded-xl" />
   412→          ))}
   413→        </div>
   414→      </div>
   415→    );
   416→  }
   417→
   418→  return (
   419→    <div>
   420→      <div className="scroll-section-title">
   421→        <h3 className="text-sm font-semibold flex items-center gap-2 whitespace-nowrap">
   422→          <Target className="w-4 h-4 text-emerald-500" />
   423→          Recommended For You
   424→          {!userProfile && <Badge variant="secondary" className="text-[10px]">Based on popularity</Badge>}
   425→          {userProfile && <Badge variant="secondary" className="text-[10px] bg-emerald-100 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-400">Personalized</Badge>}
   426→        </h3>
   427→      </div>
   428→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
   429→        {recommendations.map((rec, idx) => (
   430→          <motion.div
   431→            key={rec.country.code}
   432→            initial={{ opacity: 0, y: 20 }}
   433→            animate={{ opacity: 1, y: 0 }}
   434→            transition={{ delay: idx * 0.1 }}
   435→            className="glass-section p-4 hover:border-emerald-500/30 transition-all cursor-pointer group"
   436→            onClick={() => onSelectCountry(rec.country)}
   437→          >
   438→            <div className="flex items-center gap-2 mb-2">

   479→// ============ ANIMATED COUNTER ============
   480→export function AnimatedCounter({ target, suffix = '', prefix = '' }: { target: number; suffix?: string; prefix?: string }) {
   481→  const [count, setCount] = useState(0);
   482→  const hasAnimated = useRef(false);
   483→
   484→  useEffect(() => {
   485→    if (hasAnimated.current) return;
   486→    hasAnimated.current = true;
   487→    const duration = 2000;
   488→    const start = performance.now();
   489→    const animate = (now: number) => {
   490→      const elapsed = now - start;
   491→      const progress = Math.min(elapsed / duration, 1);
   492→      const eased = 1 - Math.pow(1 - progress, 3);
   493→      setCount(Math.round(eased * target));
   494→      if (progress < 1) requestAnimationFrame(animate);
   495→    };
   496→    requestAnimationFrame(animate);
   497→  }, [target]);
   498→
   499→  return <span>{prefix}{count}{suffix}</span>;
   500→}

   575→export function getTimeOfDay(timezone: string): { label: string; icon: React.ReactNode; color: string } {
   576→  try {
   577→    const now = new Date();
   578→    const formatter = new Intl.DateTimeFormat('en-US', { timeZone: timezone, hour: 'numeric', hour12: false });
   579→    const hour = parseInt(formatter.format(now));
   580→    if (hour >= 6 && hour < 18) return { label: 'Day', icon: <Sun className="w-3 h-3" />, color: 'text-amber-500' };
   581→    if (hour >= 18 && hour < 22) return { label: 'Evening', icon: <Sunset className="w-3 h-3" />, color: 'text-orange-500' };
   582→    return { label: 'Night', icon: <Moon className="w-3 h-3" />, color: 'text-slate-400' };
   583→  } catch {
   584→    return { label: 'Unknown', icon: <Clock className="w-3 h-3" />, color: 'text-muted-foreground' };
   585→  }

   588→/** Hook: returns the current time string (e.g. "3:45 PM") in the given timezone, updates every 60s */
   589→export function useCountryTime(timezone: string) {
   590→  const [time, setTime] = useState('');
   591→  useEffect(() => {
   592→    const tick = () => {
   593→      try {
   594→        const now = new Date();
   595→        setTime(now.toLocaleTimeString('en-US', { timeZone: timezone, hour: 'numeric', minute: '2-digit', hour12: true }));
   596→      } catch { setTime(''); }
   597→    };
   598→    tick();
   599→    const id = setInterval(tick, 30000);
   600→    return () => clearInterval(id);
   601→  }, [timezone]);
   602→  return time;
   603→}

   924→// ============ VISA REQUIREMENT COMPARISON TABLE (TASK 10 - B4) ============
   925→export function ComparisonTable({ results }: { results: ScoreBreakdown[] }) {
   926→  const [countryDetails, setCountryDetails] = useState<Record<string, CountryData>>({});
   927→  const [loading, setLoading] = useState(false);
   928→
   929→  useEffect(() => {
   930→    if (results.length === 0) return;
   931→    let cancelled = false;
   932→    setLoading(true);
   933→    void (async () => {
   934→      try {
   935→        const details = await Promise.all(results.map(r =>
   936→          fetch(`/api/countries/${r.countryCode}`).then(res => res.json()).then(data => data.data).catch(() => null)
   937→        ));
   938→        if (cancelled) return;
   939→        const map: Record<string, CountryData> = {};
   940→        details.forEach((d: CountryData | null, i: number) => {
   941→          if (d) map[results[i].countryCode] = d;
   942→        });
   943→        setCountryDetails(map);
   944→      } finally {
   945→        if (!cancelled) setLoading(false);
   946→      }
   947→    })();
   948→    return () => { cancelled = true; };
   949→  }, [results]);
   950→
   951→  if (results.length < 2 || loading) return null;
   952→
   953→  const rows = [
   954→    { label: 'Visa Type', key: 'visaType' },
   955→    { label: 'Eligibility', key: 'eligibility', isScore: true },
   956→    { label: 'Visa Likelihood', key: 'visaLikelihood', isScore: true },
   957→    { label: 'Cost Suitability', key: 'costSuitability', isScore: true },
   958→    { label: 'Final Score', key: 'finalScore', isScore: true, bold: true },
   959→    { label: 'Confidence', key: 'confidence', isPercent: true },
   960→    { label: 'Processing Time', key: 'processing', isText: true },
   961→    { label: 'Visa Fee', key: 'fee', isText: true },
   962→    { label: 'Monthly Cost', key: 'monthly', isText: true },
   963→    { label: 'Safety Rating', key: 'safety', isText: true },
   964→  ];
   965→
   966→  const getCellValue = (row: typeof rows[0], result: ScoreBreakdown) => {
   967→    if (row.isScore) {
   968→      const val = result[row.key as keyof ScoreBreakdown] as number;
   969→      return { text: `${Math.round(val)}%`, value: val };
   970→    }
   971→    if (row.isPercent) {
   972→      return { text: `${Math.round(result.confidence * 100)}%`, value: result.confidence };
   973→    }
   974→    if (row.isText) {
   975→      const detail = countryDetails[result.countryCode];
   976→      switch (row.key) {
   977→        case 'processing': return { text: `${detail?.processingDaysMin || '?'}-${detail?.processingDaysMax || '?'} days`, value: 0 };
   978→        case 'fee': return { text: `$${detail?.costProfile?.visaFeeUSD || 0}`, value: -(detail?.costProfile?.visaFeeUSD || 0) };
   979→        case 'monthly': return { text: `$${detail?.costProfile?.totalMonthlyUSD || 0}/mo`, value: -(detail?.costProfile?.totalMonthlyUSD || 0) };
   980→        case 'safety': return { text: `${detail?.safetyRating || '?'}/10`, value: detail?.safetyRating || 0 };
   981→        default: return { text: 'N/A', value: 0 };
   982→      }
   983→    }
   984→    return { text: String(result[row.key as keyof ScoreBreakdown] || 'N/A'), value: 0 };
   985→  };
   986→
   987→  const getBestIdx = (row: typeof rows[0]) => {
   988→    if (row.key === 'visaType') return -1;
   989→    const values = results.map(r => getCellValue(row, r));
   990→    // For scores and safety: higher is better; for costs/fees: lower absolute value is best
   991→    if (row.isScore || row.key === 'confidence' || row.key === 'safety') {
   992→      return values.reduce((best, v, i) => v.value > values[best].value ? i : best, 0);
   993→    }
   994→    return values.reduce((best, v, i) => Math.abs(v.value) < Math.abs(values[best].value) ? i : best, 0);
   995→  };
   996→
   997→  return (
   998→    <Card className="premium-card overflow-x-auto">
   999→      <CardHeader>
  1000→        <CardTitle className="text-base flex items-center gap-2">
  1001→          <Gavel className="w-5 h-5 text-emerald-500" />
  1002→          Detailed Comparison Table
  1003→        </CardTitle>
  1004→        <CardDescription>Side-by-side analysis with best values highlighted</CardDescription>
  1005→      </CardHeader>
  1006→      <CardContent className="overflow-x-auto">
  1007→        <Table>
  1008→          <TableHeader>
  1009→            <TableRow>
  1010→              <TableHead className="w-36">Metric</TableHead>
  1011→              {results.map(r => (
  1012→                <TableHead key={r.countryCode} className="text-center">
  1013→                  <div className="flex flex-col items-center gap-0.5">
  1014→                    <span className="text-sm font-semibold">{r.country}</span>
  1015→                    <Badge variant="outline" className="text-[10px]">{r.countryCode}</Badge>
  1016→                  </div>
  1017→                </TableHead>
  1018→              ))}
  1019→            </TableRow>
  1020→          </TableHeader>
  1021→          <TableBody>
  1022→            {rows.map(row => {
  1023→              const bestIdx = getBestIdx(row);
  1024→              return (
  1025→                <TableRow key={row.key}>
  1026→                  <TableCell className={`font-medium text-sm ${row.bold ? 'font-bold' : ''}`}>{row.label}</TableCell>
  1027→                  {results.map((r, i) => {
  1028→                    const cell = getCellValue(row, r);
  1029→                    const isBest = i === bestIdx;
  1030→                    return (
  1031→                      <TableCell key={r.countryCode} className={`text-center text-sm ${isBest ? 'print-best bg-emerald-50 dark:bg-emerald-900/20 font-semibold text-emerald-700 dark:text-emerald-400' : ''}`}>
  1032→                        {cell.text}
  1033→                        {isBest && <span className="ml-1">✓</span>}
  1034→                      </TableCell>
  1035→                    );
  1036→                  })}
  1037→                </TableRow>
  1038→              );
  1039→            })}
  1040→          </TableBody>
  1041→        </Table>
  1042→      </CardContent>
  1043→    </Card>
  1044→  );
  1045→}
