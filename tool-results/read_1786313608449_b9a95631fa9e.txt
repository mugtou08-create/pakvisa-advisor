    49→function BudgetDonutChart({ budgetResult, budgetTripDays, budgetViewMode, budgetChartActiveIndex, setBudgetChartActiveIndex, formatNum }: {
    50→  budgetResult: { visaFee: number; flightEst: number; accommodation: number; food: number; transport: number; insurance: number; misc: number; totalUSD: number; };
    51→  budgetTripDays: number;
    52→  budgetViewMode: 'total' | 'daily';
    53→  budgetChartActiveIndex: number | null;
    54→  setBudgetChartActiveIndex: (i: number | null) => void;
    55→  formatNum: (n: number) => string;
    56→}) {
    57→  const divisor = budgetViewMode === 'daily' ? budgetTripDays : 1;
    58→  const chartItems = [
    59→    { name: 'Visa Fee', value: Math.round(budgetResult.visaFee / divisor), color: CHART_COLORS[0] },
    60→    { name: 'Flights', value: Math.round(budgetResult.flightEst / divisor), color: CHART_COLORS[1] },
    61→    { name: 'Accommodation', value: Math.round(budgetResult.accommodation / divisor), color: CHART_COLORS[2] },
    62→    { name: 'Food', value: Math.round(budgetResult.food / divisor), color: CHART_COLORS[3] },
    63→    { name: 'Transport', value: Math.round(budgetResult.transport / divisor), color: CHART_COLORS[4] },
    64→    { name: 'Insurance', value: Math.round(budgetResult.insurance / divisor), color: CHART_COLORS[5] },
    65→    { name: 'Misc', value: Math.round(budgetResult.misc / divisor), color: CHART_COLORS[6] },
    66→  ].filter(d => d.value > 0);
    67→  const chartIcons = [FileText, PlaneTakeoff, Home, UtensilsCrossed, Map, Shield, MoreHorizontal];
    68→  const chartNames = ['Visa Fee', 'Flights', 'Accommodation', 'Food', 'Transport', 'Insurance', 'Misc'];
    69→  const totalVal = chartItems.reduce((s, d) => s + d.value, 0);
    70→
    71→  const StepIcon = ({ icon: StepIconProp, className }: { icon: React.ElementType; className?: string }) => React.createElement(StepIconProp, { className });
    72→
    73→  return (
    74→    <div className="p-4 rounded-lg border bg-muted/20">
    75→      <div className="flex items-center justify-between mb-3">
    76→        <div className="flex items-center gap-2">
    77→          <PieChartIcon className="w-4 h-4 text-amber-500" />
    78→          <p className="text-xs font-semibold">Budget Breakdown</p>
    79→        </div>
    80→        <div className="flex items-center gap-1 bg-muted rounded-lg p-0.5">
    81→          <button
    82→            className={`px-2 py-0.5 rounded-md text-[10px] font-medium transition-all ${budgetViewMode === 'total' ? 'bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-400' : 'text-muted-foreground'}`}
    83→            onClick={() => setBudgetViewMode('total')}
    84→          >Total</button>
    85→          <button
    86→            className={`px-2 py-0.5 rounded-md text-[10px] font-medium transition-all ${budgetViewMode === 'daily' ? 'bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-400' : 'text-muted-foreground'}`}
    87→            onClick={() => setBudgetViewMode('daily')}
    88→          >Daily</button>
    89→        </div>
    90→      </div>
    91→      <div className="flex items-center gap-4">
    92→        <div className="w-36 h-36 shrink-0">
    93→          <ResponsiveContainer width="100%" height="100%">
    94→            <PieChart>
    95→              <Pie
    96→                data={chartItems}
    97→                cx="50%"
    98→                cy="50%"
    99→                innerRadius={35}
   100→                outerRadius={58}
   101→                paddingAngle={2}
   102→                dataKey="value"
   103→                onMouseEnter={(_, index) => setBudgetChartActiveIndex(index)}
   104→                onMouseLeave={() => setBudgetChartActiveIndex(null)}
   105→                stroke="none"
   106→              >
   107→                {chartItems.map((entry, index) => (
   108→                  <Cell
   109→                    key={`cell-${index}`}
   110→                    fill={entry.color}
   111→                    opacity={budgetChartActiveIndex === null || budgetChartActiveIndex === index ? 1 : 0.4}
   112→                    stroke="none"
   113→                  />
   114→                ))}
   115→              </Pie>
   116→              <RechartsTooltip
   117→                contentStyle={{ borderRadius: '8px', border: '1px solid var(--border)', fontSize: '11px', padding: '6px 10px' }}
   118→                formatter={(value: number) => [`$${formatNum(value)}`, undefined]}
   119→              />
   120→            </PieChart>
   121→          </ResponsiveContainer>
   122→        </div>
   123→        <div className="flex-1 space-y-1.5">
   124→          {chartItems.map((item, idx) => {
   125→            const pct = Math.round((item.value / totalVal) * 100);
   126→            return (
   127→              <div
   128→                key={item.name}
   129→                className={`flex items-center gap-2 text-[11px] px-1.5 py-0.5 rounded transition-all ${budgetChartActiveIndex === idx ? 'bg-muted font-semibold' : ''}`}
   130→                onMouseEnter={() => setBudgetChartActiveIndex(idx)}
   131→                onMouseLeave={() => setBudgetChartActiveIndex(null)}
   132→              >
   133→                <div className="w-2.5 h-2.5 rounded-sm shrink-0" style={{ backgroundColor: item.color }} />
   134→                <StepIcon icon={chartIcons[idx]} className="w-3 h-3 text-muted-foreground shrink-0" />
   135→                <span className="flex-1 truncate">{chartNames[idx]}</span>
   136→                <span className="font-medium tabular-nums">${formatNum(item.value)}</span>
   137→                <span className="text-muted-foreground w-8 text-right">{pct}%</span>
   138→              </div>
   139→            );
   140→          })}
   141→        </div>
   142→      </div>
   143→    </div>
   144→  );
   145→}
