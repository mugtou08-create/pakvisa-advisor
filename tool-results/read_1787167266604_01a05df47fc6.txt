     1→'use client';
     2→
     3→import React, { useState, useEffect, useCallback } from 'react';
     4→import {
     5→  Settings, Lock, Eye, EyeOff, Shield, Globe, Users, Activity,
     6→  ToggleLeft, ToggleRight, BarChart3, LogOut, Database, Zap, AlertTriangle,
     7→  RefreshCw, Server, FileCheck, MessageSquare, Mail, Send, Trash2,
     8→  ChevronLeft, ChevronRight, Check, Clock, User, Inbox, TrendingUp,
     9→  Phone, ExternalLink, Reply,
    10→} from 'lucide-react';
    11→import { Button } from '@/components/ui/button';
    12→import { Input } from '@/components/ui/input';
    13→import { Textarea } from '@/components/ui/textarea';
    14→import { Card, CardContent, CardDescription, CardHeader, CardTitle } from '@/components/ui/card';
    15→import { Badge } from '@/components/ui/badge';
    16→import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription } from '@/components/ui/dialog';
    17→import { ScrollArea } from '@/components/ui/scroll-area';
    18→import { Separator } from '@/components/ui/separator';
    19→import { Switch } from '@/components/ui/switch';
    20→import { toast } from 'sonner';
    21→
    22→interface AdminDialogProps {
    23→  open: boolean;
    24→  onClose: () => void;
    25→  aiEnabled: boolean;
    26→  setAiEnabled: (enabled: boolean) => void;
    27→}
    28→
    29→interface AnalyticsData {
    30→  countries: { total: number; visaFree: number; visaOnArrival: number; etaAvailable: number };
    31→  visaCategories: { visaFree: number; visaOnArrival: number; etaAvailable: number; regularVisa: number };
    32→  continents: Array<{ continent: string; count: number }>;
    33→  dataFreshness: string | null;
    34→  requirements: number;
    35→  costProfiles: number;
    36→  users: number;
    37→  sessions: number;
    38→  adminUsers: Array<{ username: string; lastLogin: string | null; isOnline: boolean; createdAt: string }>;
    39→  settings: Record<string, string>;
    40→}
    41→
    42→interface ContactMessage {
    43→  id: string;
    44→  name: string;
    45→  email: string;
    46→  subject: string;
    47→  message: string;
    48→  isRead: boolean;
    49→  isReplied: boolean;
    50→  reply: string;
    51→  ip: string;
    52→  createdAt: string;
    53→}
    54→
    55→interface NewsletterSubscriber {
    56→  id: string;
    57→  email: string;
    58→  subscribedAt: string;
    59→  isActive: boolean;
    60→}
    61→
    62→type AdminSection = 'overview' | 'messages' | 'newsletter' | 'analytics' | 'settings';
    63→
    64→export function AdminDialog({ open, onClose, aiEnabled, setAiEnabled }: AdminDialogProps) {
    65→  const [isLoggedIn, setIsLoggedIn] = useState(false);
    66→  const [token, setToken] = useState<string | null>(null);
    67→  const [username, setUsername] = useState('');
    68→  const [password, setPassword] = useState('');
    69→  const [showPassword, setShowPassword] = useState(false);
    70→  const [loginLoading, setLoginLoading] = useState(false);
    71→  const [activeSection, setActiveSection] = useState<AdminSection>('overview');
    72→  const [analytics, setAnalytics] = useState<AnalyticsData | null>(null);
    73→  const [analyticsLoading, setAnalyticsLoading] = useState(false);
    74→  const [maintenanceMode, setMaintenanceMode] = useState(false);
    75→  const [togglingAi, setTogglingAi] = useState(false);
    76→  const [togglingMaintenance, setTogglingMaintenance] = useState(false);
    77→  const [refreshingAnalytics, setRefreshingAnalytics] = useState(false);
    78→
    79→  // Messages state
    80→  const [messages, setMessages] = useState<ContactMessage[]>([]);
    81→  const [messagesTotal, setMessagesTotal] = useState(0);
    82→  const [messagesUnread, setMessagesUnread] = useState(0);
    83→  const [messagesPage, setMessagesPage] = useState(1);
    84→  const [messagesLoading, setMessagesLoading] = useState(false);
    85→  const [replyingTo, setReplyingTo] = useState<string | null>(null);
    86→  const [replyText, setReplyText] = useState('');
    87→  const [expandedMessage, setExpandedMessage] = useState<string | null>(null);
    88→
    89→  // Newsletter state
    90→  const [subscribers, setSubscribers] = useState<NewsletterSubscriber[]>([]);
    91→  const [subscribersTotal, setSubscribersTotal] = useState(0);
    92→  const [subscribersActive, setSubscribersActive] = useState(0);
    93→  const [subscribersPage, setSubscribersPage] = useState(1);
    94→  const [subscribersLoading, setSubscribersLoading] = useState(false);
    95→
    96→  useEffect(() => {
    97→    const savedToken = localStorage.getItem('pakvisa-admin-token');
    98→    if (savedToken) {
    99→      setToken(savedToken);
   100→      setIsLoggedIn(true);
   101→    }
   102→  }, []);
   103→
   104→  const handleLogin = useCallback(async () => {
   105→    if (!username.trim() || !password.trim()) {
   106→      toast.error('Please enter username and password');
   107→      return;
   108→    }
   109→    setLoginLoading(true);
   110→    try {
   111→      const res = await fetch('/api/admin', {
   112→        method: 'POST',
   113→        headers: { 'Content-Type': 'application/json' },
   114→        body: JSON.stringify({ username: username.trim(), password }),
   115→      });
   116→      const data = await res.json();
   117→      if (data.success) {
   118→        localStorage.setItem('pakvisa-admin-token', data.data.token);
   119→        setToken(data.data.token);
   120→        setIsLoggedIn(true);
   121→        toast.success('Logged in successfully');
   122→      } else {
   123→        toast.error(data.error || 'Login failed');
   124→      }
   125→    } catch {
   126→      toast.error('Connection error');
   127→    } finally {
   128→      setLoginLoading(false);
   129→    }
   130→  }, [username, password]);
   131→
   132→  const handleLogout = useCallback(() => {
   133→    localStorage.removeItem('pakvisa-admin-token');
   134→    setToken(null);
   135→    setIsLoggedIn(false);
   136→    setUsername(''); setPassword('');
   137→    setAnalytics(null);
   138→    setMessages([]);
   139→    setSubscribers([]);
   140→    setActiveSection('overview');
   141→    toast.success('Logged out');
   142→  }, []);
   143→
   144→  const fetchAnalytics = useCallback(async (showLoading = true) => {
   145→    if (!token) return;
   146→    if (showLoading) setAnalyticsLoading(true);
   147→    setRefreshingAnalytics(true);
   148→    try {
   149→      const res = await fetch('/api/admin/analytics', { headers: { 'Authorization': `Bearer ${token}` } });
   150→      const data = await res.json();
   151→      if (data.success) {
   152→        setAnalytics(data.data);
   153→        setMaintenanceMode(data.data.settings?.maintenance_mode === 'true');
   154→      }
   155→    } catch { toast.error('Failed to fetch analytics'); }
   156→    finally { setAnalyticsLoading(false); setRefreshingAnalytics(false); }
   157→  }, [token]);
   158→
   159→  const fetchMessages = useCallback(async (page = 1) => {
   160→    if (!token) return;
   161→    setMessagesLoading(true);
   162→    try {
   163→      const res = await fetch(`/api/admin/messages?page=${page}&limit=15`, {
   164→        headers: { 'Authorization': `Bearer ${token}` },
   165→      });
   166→      const data = await res.json();
   167→      if (data.success) {
   168→        setMessages(data.data.messages);
   169→        setMessagesTotal(data.data.total);
   170→        setMessagesUnread(data.data.unreadCount);
   171→        setMessagesPage(data.data.page);
   172→      }
   173→    } catch { toast.error('Failed to fetch messages'); }
   174→    finally { setMessagesLoading(false); }
   175→  }, [token]);
   176→
   177→  const fetchSubscribers = useCallback(async (page = 1) => {
   178→    if (!token) return;
   179→    setSubscribersLoading(true);
   180→    try {
   181→      const res = await fetch(`/api/admin/newsletter?page=${page}&limit=15`, {
   182→        headers: { 'Authorization': `Bearer ${token}` },
   183→      });
   184→      const data = await res.json();
   185→      if (data.success) {
   186→        setSubscribers(data.data.subscribers);
   187→        setSubscribersTotal(data.data.total);
   188→        setSubscribersActive(data.data.activeCount);
   189→        setSubscribersPage(data.data.page);
   190→      }
   191→    } catch { toast.error('Failed to fetch subscribers'); }
   192→    finally { setSubscribersLoading(false); }
   193→  }, [token]);
   194→
   195→  // Fetch data on section change
   196→  useEffect(() => {
   197→    if (!isLoggedIn || !token) return;
   198→    if (activeSection === 'overview' || activeSection === 'analytics') {
   199→      if (!analytics) fetchAnalytics();
   200→    }
   201→    if (activeSection === 'overview') {
   202→      fetchMessages();
   203→      fetchSubscribers();
   204→    }
   205→    if (activeSection === 'messages') fetchMessages();
   206→    if (activeSection === 'newsletter') fetchSubscribers();
   207→  }, [isLoggedIn, token, activeSection]);
   208→
   209→  const toggleAiFeature = useCallback(async (enabled: boolean) => {
   210→    if (!token) return;
   211→    setTogglingAi(true);
   212→    try {
   213→      const res = await fetch('/api/admin/settings', {
   214→        method: 'PUT',
   215→        headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` },
   216→        body: JSON.stringify({ key: 'ai_enabled', value: String(enabled) }),
   217→      });
   218→      const data = await res.json();
   219→      if (data.success) { setAiEnabled(enabled); toast.success(`AI features ${enabled ? 'enabled' : 'disabled'}`); }
   220→      else toast.error('Failed to update setting');
   221→    } catch { toast.error('Connection error'); }
   222→    finally { setTogglingAi(false); }
   223→  }, [token, setAiEnabled]);
   224→
   225→  const toggleMaintenance = useCallback(async (enabled: boolean) => {
   226→    if (!token) return;
   227→    setTogglingMaintenance(true);
   228→    try {
   229→      const res = await fetch('/api/admin/settings', {
   230→        method: 'PUT',
   231→        headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` },
   232→        body: JSON.stringify({ key: 'maintenance_mode', value: String(enabled) }),
   233→      });
   234→      const data = await res.json();
   235→      if (data.success) { setMaintenanceMode(enabled); toast.success(`Maintenance mode ${enabled ? 'enabled' : 'disabled'}`); }
   236→      else toast.error('Failed to update setting');
   237→    } catch { toast.error('Connection error'); }
   238→    finally { setTogglingMaintenance(false); }
   239→  }, [token]);
   240→
   241→  const markMessageRead = async (id: string) => {
   242→    if (!token) return;
   243→    try {
   244→      await fetch('/api/admin/messages', {
   245→        method: 'PUT',
   246→        headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` },
   247→        body: JSON.stringify({ id, action: 'mark_read' }),
   248→      });
   249→      setMessages(prev => prev.map(m => m.id === id ? { ...m, isRead: true } : m));
   250→      setMessagesUnread(prev => Math.max(0, prev - 1));
   251→    } catch { toast.error('Failed to mark as read'); }
   252→  };
   253→
   254→  const sendMessageReply = async (id: string) => {
   255→    if (!token || !replyText.trim()) return;
   256→    try {
   257→      const res = await fetch('/api/admin/messages', {
   258→        method: 'PUT',
   259→        headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` },
   260→        body: JSON.stringify({ id, action: 'reply', reply: replyText }),
   261→      });
   262→      const data = await res.json();
   263→      if (data.success) {
   264→        setMessages(prev => prev.map(m => m.id === id ? { ...m, isReplied: true, reply: replyText.trim(), isRead: true } : m));
   265→        setReplyingTo(null); setReplyText('');
   266→        toast.success('Reply saved');
   267→      }
   268→    } catch { toast.error('Failed to send reply'); }
   269→  };
   270→
   271→  const deleteMessage = async (id: string) => {
   272→    if (!token) return;
   273→    try {
   274→      await fetch('/api/admin/messages', {
   275→        method: 'PUT',
   276→        headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` },
   277→        body: JSON.stringify({ id, action: 'delete' }),
   278→      });
   279→      setMessages(prev => prev.filter(m => m.id !== id));
   280→      setMessagesTotal(prev => prev - 1);
   281→      toast.success('Message deleted');
   282→    } catch { toast.error('Failed to delete'); }
   283→  };
   284→
   285→  const deleteSubscriber = async (id: string) => {
   286→    if (!token) return;
   287→    try {
   288→      await fetch(`/api/admin/newsletter?id=${id}`, {
   289→        method: 'DELETE',
   290→        headers: { 'Authorization': `Bearer ${token}` },
   291→      });
   292→      setSubscribers(prev => prev.filter(s => s.id !== id));
   293→      setSubscribersTotal(prev => prev - 1);
   294→      setSubscribersActive(prev => prev - 1);
   295→      toast.success('Subscriber removed');
   296→    } catch { toast.error('Failed to remove'); }
   297→  };
   298→
   299→  const navItems: { key: AdminSection; label: string; icon: React.ReactNode; badge?: number }[] = [
   300→    { key: 'overview', label: 'Overview', icon: <TrendingUp className="w-4 h-4" /> },
   301→    { key: 'messages', label: 'Messages', icon: <Inbox className="w-4 h-4" />, badge: messagesUnread || undefined },
   302→    { key: 'newsletter', label: 'Newsletter', icon: <Mail className="w-4 h-4" /> },
   303→    { key: 'analytics', label: 'Analytics', icon: <BarChart3 className="w-4 h-4" /> },
   304→    { key: 'settings', label: 'Settings', icon: <Settings className="w-4 h-4" /> },
   305→  ];
   306→
   307→  const formatDate = (d: string) => new Date(d).toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric', hour: '2-digit', minute: '2-digit' });
   308→  const timeAgo = (d: string) => {
   309→    const diff = Date.now() - new Date(d).getTime();
   310→    const mins = Math.floor(diff / 60000);
   311→    if (mins < 60) return `${mins}m ago`;
   312→    const hours = Math.floor(mins / 60);
   313→    if (hours < 24) return `${hours}h ago`;
   314→    return `${Math.floor(hours / 24)}d ago`;
   315→  };
   316→
   317→  // ===== RENDER =====
   318→  return (
   319→    <Dialog open={open} onOpenChange={(isOpen) => { if (!isOpen) onClose(); }}>
   320→      <DialogContent className="max-w-4xl max-h-[90vh] p-0 overflow-hidden flex flex-col">
   321→        <DialogHeader className="p-5 pb-0 border-b">
   322→          <div className="flex items-center justify-between">
   323→            <div className="flex items-center gap-3">
   324→              <div className="w-9 h-9 rounded-lg bg-emerald-600 flex items-center justify-center">
   325→                <Shield className="w-4.5 h-4.5 text-white" />
   326→              </div>
   327→              <div>
   328→                <DialogTitle className="text-lg">Admin Dashboard</DialogTitle>
   329→                <DialogDescription className="text-xs">Manage your PakVisa Advisor</DialogDescription>
   330→              </div>
   331→            </div>
   332→            {isLoggedIn && (
   333→              <Button variant="ghost" size="sm" className="text-red-500 hover:text-red-600 hover:bg-red-50 dark:hover:bg-red-900/20" onClick={handleLogout}>
   334→                <LogOut className="w-4 h-4 mr-1.5" /> Logout
   335→              </Button>
   336→            )}
   337→          </div>
   338→        </DialogHeader>
   339→
   340→        {!isLoggedIn ? (
   341→          <div className="p-8 flex-1 flex items-center justify-center">
   342→            <Card className="w-full max-w-sm border-emerald-200 dark:border-emerald-800">
   343→              <CardHeader className="text-center">
   344→                <div className="w-16 h-16 rounded-full bg-emerald-100 dark:bg-emerald-900/30 flex items-center justify-center mx-auto mb-3">
   345→                  <Lock className="w-8 h-8 text-emerald-600" />
   346→                </div>
   347→                <CardTitle className="text-lg">Admin Login</CardTitle>
   348→                <CardDescription>Enter credentials to access dashboard</CardDescription>
   349→              </CardHeader>
   350→              <CardContent className="space-y-4">
   351→                <div className="space-y-2">
   352→                  <label className="text-sm font-medium" htmlFor="admin-username">Username</label>
   353→                  <Input id="admin-username" placeholder="admin" value={username} onChange={(e) => setUsername(e.target.value)} onKeyDown={(e) => e.key === 'Enter' && handleLogin()} autoComplete="username" />
   354→                </div>
   355→                <div className="space-y-2">
   356→                  <label className="text-sm font-medium" htmlFor="admin-password">Password</label>
   357→                  <div className="relative">
   358→                    <Input id="admin-password" type={showPassword ? 'text' : 'password'} placeholder="••••••••" value={password} onChange={(e) => setPassword(e.target.value)} onKeyDown={(e) => e.key === 'Enter' && handleLogin()} autoComplete="current-password" />
   359→                    <Button variant="ghost" size="icon" className="absolute right-1 top-1/2 -translate-y-1/2 h-7 w-7" onClick={() => setShowPassword(!showPassword)}>
   360→                      {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
   361→                    </Button>
   362→                  </div>
   363→                </div>
   364→                <Button className="w-full bg-emerald-600 hover:bg-emerald-700" onClick={handleLogin} disabled={loginLoading}>
   365→                  {loginLoading ? (<><span className="inline-block w-4 h-4 border-2 border-white border-t-transparent rounded-full animate-spin mr-2" />Logging in...</>) : (<><Lock className="w-4 h-4 mr-2" />Sign In</>)}
   366→                </Button>
   367→              </CardContent>
   368→            </Card>
   369→          </div>
   370→        ) : (
   371→          <div className="flex flex-1 overflow-hidden">
   372→            {/* Sidebar Navigation */}
   373→            <div className="w-52 border-r bg-muted/20 p-3 space-y-1 hidden md:flex flex-col">
   374→              {navItems.map((item) => (
   375→                <button
   376→                  key={item.key}
   377→                  className={`w-full flex items-center gap-2.5 px-3 py-2.5 rounded-lg text-sm transition-colors relative ${
   378→                    activeSection === item.key
   379→                      ? 'bg-emerald-100 dark:bg-emerald-900/30 text-emerald-700 dark:text-emerald-400 font-medium'
   380→                      : 'hover:bg-muted text-muted-foreground hover:text-foreground'
   381→                  }`}
   382→                  onClick={() => setActiveSection(item.key)}
   383→                >
   384→                  {item.icon}
   385→                  <span>{item.label}</span>
   386→                  {item.badge ? (
   387→                    <span className="ml-auto bg-red-500 text-white text-[10px] font-bold rounded-full w-5 h-5 flex items-center justify-center">
   388→                      {item.badge > 99 ? '99+' : item.badge}
   389→                    </span>
   390→                  ) : null}
   391→                </button>
   392→              ))}
   393→            </div>
   394→
   395→            {/* Mobile Nav */}
   396→            <div className="md:hidden p-3 border-b flex gap-1.5 overflow-x-auto">
   397→              {navItems.map((item) => (
   398→                <Button
   399→                  key={item.key}
   400→                  variant={activeSection === item.key ? 'default' : 'outline'}
   401→                  size="sm"
   402→                  className={`shrink-0 gap-1.5 text-xs ${activeSection === item.key ? 'bg-emerald-600 hover:bg-emerald-700' : ''}`}
   403→                  onClick={() => setActiveSection(item.key)}
   404→                >
   405→                  {item.icon}
   406→                  {item.label}
   407→                  {item.badge ? <span className="bg-red-500 text-white text-[9px] rounded-full w-4 h-4 flex items-center justify-center">{item.badge}</span> : null}
   408→                </Button>
   409→              ))}
   410→            </div>
   411→
   412→            {/* Content Area */}
   413→            <ScrollArea className="flex-1">
   414→              <div className="p-5 space-y-5">
   415→                {/* ====== OVERVIEW TAB ====== */}
   416→                {activeSection === 'overview' && (
   417→                  <>
   418→                    {/* Quick Stats Row */}
   419→                    <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
   420→                      <StatCard icon={<Inbox className="w-4 h-4" />} label="Unread Messages" value={messagesUnread} color="red" />
   421→                      <StatCard icon={<Mail className="w-4 h-4" />} label="Subscribers" value={subscribersActive} color="blue" />
   422→                      <StatCard icon={<Globe className="w-4 h-4" />} label="Countries" value={analytics?.countries.total ?? '—'} color="emerald" />
   423→                      <StatCard icon={<Zap className="w-4 h-4" />} label="AI Status" value={aiEnabled ? 'Online' : 'Offline'} color={aiEnabled ? 'green' : 'amber'} />
   424→                    </div>
   425→
   426→                    {/* Two Column */}
   427→                    <div className="grid grid-cols-1 lg:grid-cols-2 gap-5">
   428→                      {/* Recent Messages */}
   429→                      <Card>
   430→                        <CardHeader className="pb-3">
   431→                          <div className="flex items-center justify-between">
   432→                            <CardTitle className="text-sm font-semibold flex items-center gap-2">
   433→                              <MessageSquare className="w-4 h-4 text-emerald-500" /> Recent Messages
   434→                            </CardTitle>
   435→                            <Button variant="ghost" size="sm" className="text-xs text-emerald-600" onClick={() => setActiveSection('messages')}>
   436→                              View All
   437→                            </Button>
   438→                          </div>
   439→                        </CardHeader>
   440→                        <CardContent>
   441→                          {messagesLoading ? (
   442→                            <div className="flex justify-center py-6"><div className="w-6 h-6 border-2 border-emerald-400 border-t-transparent rounded-full animate-spin" /></div>
   443→                          ) : messages.length === 0 ? (
   444→                            <p className="text-sm text-muted-foreground text-center py-6">No messages yet</p>
   445→                          ) : (
   446→                            <div className="space-y-2">
   447→                              {messages.slice(0, 5).map((m) => (
   448→                                <div key={m.id} className={`flex items-start gap-3 p-2.5 rounded-lg transition-colors ${!m.isRead ? 'bg-emerald-50 dark:bg-emerald-900/10 border border-emerald-200/50 dark:border-emerald-800/30' : 'hover:bg-muted/50'}`}>
   449→                                  <div className={`w-8 h-8 rounded-full flex items-center justify-center shrink-0 text-xs font-bold ${!m.isRead ? 'bg-emerald-200 dark:bg-emerald-800 text-emerald-700 dark:text-emerald-300' : 'bg-muted text-muted-foreground'}`}>
   450→                                    {m.name.charAt(0).toUpperCase()}
   451→                                  </div>
   452→                                  <div className="min-w-0 flex-1">
   453→                                    <div className="flex items-center gap-2">
   454→                                      <span className={`text-sm ${!m.isRead ? 'font-semibold' : 'font-medium'}`}>{m.name}</span>
   455→                                      {!m.isRead && <span className="w-2 h-2 rounded-full bg-emerald-500 shrink-0" />}
   456→                                      <span className="text-xs text-muted-foreground ml-auto shrink-0">{timeAgo(m.createdAt)}</span>
   457→                                    </div>
