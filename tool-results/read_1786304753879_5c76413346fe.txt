   742→// ============ TYPING TEXT ANIMATION ============
   743→export function TypingText({ phrases, className = '' }: { phrases: string[]; className?: string }) {
   744→  const [current, setCurrent] = useState(0);
   745→  const [text, setText] = useState('');
   746→  const [isDeleting, setIsDeleting] = useState(false);
   747→
   748→  useEffect(() => {
   749→    const phrase = phrases[current];
   750→    const timeout = setTimeout(() => {
   751→      if (!isDeleting) {
   752→        setText(phrase.substring(0, text.length + 1));
   753→        if (text.length === phrase.length) {
   754→          setTimeout(() => setIsDeleting(true), 2200);
   755→        }
   756→      } else {
   757→        setText(phrase.substring(0, text.length - 1));
   758→        if (text.length === 0) {
   759→          setIsDeleting(false);
   760→          setCurrent((current + 1) % phrases.length);
   761→        }
   762→      }
   763→    }, isDeleting ? 25 : 55);
   764→    return () => clearTimeout(timeout);
   765→  }, [text, isDeleting, current, phrases]);
   766→
   767→  return (
   768→    <span className={className}>
   769→      {text}
   770→      <span className="inline-block w-0 h-[1.1em] ml-0.5 align-middle typing-text-cursor" />
   771→    </span>
   772→  );
   773→}

   825→// ============ MINI RADAR CHART (3-axis for country cards) ============
   826→export function MiniRadarChart({ safety, cost, ease }: { safety: number; cost: number; ease: number }) {
   827→  const cx = 28, cy = 28, r = 22;
   828→  const axes = 3;
   829→  const angleStep = (2 * Math.PI) / axes;
   830→  const startAngle = -Math.PI / 2;
   831→  const s = safety / 10;
   832→  const c = Math.max(0.05, 1 - cost / 3000);
   833→  const e = ease;
   834→  const values = [s, c, e];
   835→  const points = values.map((val, i) => {
   836→    const angle = startAngle + i * angleStep;
   837→    return `${cx + val * r * Math.cos(angle)},${cy + val * r * Math.sin(angle)}`;
   838→  }).join(' ');
   839→  const gridLines = [0.33, 0.66, 1].map(level => {
   840→    const pts = Array.from({ length: axes }, (_, i) => {
   841→      const angle = startAngle + i * angleStep;
   842→      return `${cx + level * r * Math.cos(angle)},${cy + level * r * Math.sin(angle)}`;
   843→    }).join(' ');
   844→    return <polygon key={level} points={pts} fill="none" stroke="currentColor" strokeWidth="0.3" className="text-white/20" />;
   845→  });
   846→
   847→  return (
   848→    <svg width="56" height="56" viewBox="0 0 56 56" className="inline-block opacity-80">
   849→      {gridLines}
   850→      {Array.from({ length: axes }, (_, i) => {
   851→        const angle = startAngle + i * angleStep;
   852→        return <line key={i} x1={cx} y1={cy} x2={cx + r * Math.cos(angle)} y2={cy + r * Math.sin(angle)} stroke="currentColor" strokeWidth="0.3" className="text-white/20" />;
   853→      })}
   854→      <polygon points={points} fill="rgba(249,115,22,0.15)" stroke="#fbbf24" strokeWidth="1" />
   855→      {values.map((val, i) => {
   856→        const angle = startAngle + i * angleStep;
   857→        return <circle key={i} cx={cx + val * r * Math.cos(angle)} cy={cy + val * r * Math.sin(angle)} r="1.5" fill="#fbbf24" />;
   858→      })}
   859→    </svg>
   860→  );
   861→}

   863→// ============ RADIAL GAUGE (SEMICIRCLE) ============
   864→export function RadialGauge({ score, size = 140 }: { score: number; size?: number }) {
   865→  const strokeWidth = 12;
   866→  const r = (size - strokeWidth) / 2;
   867→  const cx = size / 2;
   868→  const cy = size - 15;
   869→  const startX = cx - r;
   870→  const endX = cx + r;
   871→  const arcLength = Math.PI * r;
   872→  const filledLength = (score / 100) * arcLength;
   873→  const color = score >= 70 ? '#f59e0b' : score >= 40 ? '#f59e0b' : '#ef4444';
   874→
   875→  return (
   876→    <div className="flex flex-col items-center">
   877→      <svg width={size} height={size / 2 + 25} viewBox={`0 0 ${size} ${size / 2 + 25}`}>
   878→        <path d={`M ${startX} ${cy} A ${r} ${r} 0 0 1 ${endX} ${cy}`} fill="none" stroke="currentColor" strokeWidth={strokeWidth} className="text-muted/20" strokeLinecap="round" />
   879→        <path
   880→          d={`M ${startX} ${cy} A ${r} ${r} 0 0 1 ${endX} ${cy}`}
   881→          fill="none" stroke={color} strokeWidth={strokeWidth} strokeLinecap="round"

   894→export function AnimatedScoreNumber({ value, delay = 0 }: { value: number; delay?: number }) {
   895→  const [displayed, setDisplayed] = useState(0);
   896→  const hasAnimated = useRef(false);
   897→
   898→  useEffect(() => {
   899→    if (hasAnimated.current) return;
   900→    hasAnimated.current = true;
   901→    const start = performance.now() + delay;
   902→    const duration = 1200;
   903→    const animate = (now: number) => {
   904→      const elapsed = now - start;
   905→      if (elapsed < 0) { requestAnimationFrame(animate); return; }
   906→      const progress = Math.min(elapsed / duration, 1);
   907→      const eased = 1 - Math.pow(1 - progress, 3);
   908→      setDisplayed(Math.round(eased * value));
   909→      if (progress < 1) requestAnimationFrame(animate);
   910→    };
   911→    requestAnimationFrame(animate);
   912→  }, [value, delay]);
   913→
   914→  return <span>{displayed}</span>;
   915→}

  1374→// ============ MICRO SPARKLINE ============
  1375→export function MicroSparkline({ data, color, width = 60, height = 20 }: { data: number[]; color?: string; width?: number; height?: number }) {
  1376→  if (!data || data.length === 0) return null;
  1377→  const maxVal = Math.max(...data);
  1378→  const minVal = Math.min(...data);
  1379→  const range = maxVal - minVal || 1;
  1380→  const points = data.map((v, i) => {
  1381→    const x = (i / (data.length - 1)) * width;
  1382→    const y = height - ((v - minVal) / range) * (height - 4) - 2;
  1383→    return `${x},${y}`;
  1384→  }).join(' ');
  1385→  const lineColor = color || (data[data.length - 1] >= data[0] ? '#f59e0b' : '#ef4444');
  1386→  return (
  1387→    <svg width={width} height={height} className="inline-block" aria-hidden="true">
  1388→      <polyline points={points} fill="none" stroke={lineColor} strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
  1389→    </svg>
  1390→  );
  1391→}
