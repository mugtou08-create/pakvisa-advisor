   367→export function WhatIfSimulator({ scoreResult, userProfile }: { scoreResult: ScoreBreakdown; userProfile: UserProfileData }) {
   368→  const [healthInsurance, setHealthInsurance] = useState(userProfile.hasHealthInsurance);
   369→  const [returnTicket, setReturnTicket] = useState(userProfile.hasReturnTicket);

   376→  const [recalculating, setRecalculating] = useState(false);
   377→  const [afterScore, setAfterScore] = useState<ScoreBreakdown | null>(null);
   378→
   379→  const applyChanges = async () => {
   380→    setRecalculating(true);
   381→    const modifiedProfile = {
   382→      ...userProfile,
   383→      hasHealthInsurance: healthInsurance,
   384→      hasReturnTicket: returnTicket,
   385→      hasHotelBooking: hotelBooking,

   387→      hasPriorTravel: priorTravel,
   388→      monthlyIncomeUSD: income,
   389→      savingsUSD: savings,
   390→      budgetUSD: budget,
   391→    };
   392→    try {
   393→      const res = await fetch('/api/score', {
   394→        method: 'POST',
   395→        headers: { 'Content-Type': 'application/json' },
   396→        body: JSON.stringify({ countryCode: scoreResult.countryCode, profile: modifiedProfile }),
   397→      });
   398→      const data = await res.json();
   399→      if (data.data) {
   400→        setAfterScore(data.data);
   401→      } else {
   402→        toast.error(data.error || 'Recalculation failed');
   403→      }
   404→    } catch {
   405→      toast.error('Network error during recalculation');
   406→    }
   407→    setRecalculating(false);
   408→  };
   409→
   410→  const delta = afterScore ? afterScore.finalScore - scoreResult.finalScore : 0;
   411→
   412→  return (
   413→    <div className="space-y-4">
   414→      <div className="grid grid-cols-1 md:grid-cols-2 gap-4 responsive-stack-mobile">
   415→        {/* Toggle Switches */}

   532→            )}
   533→          </div>
   534→        </motion.div>
