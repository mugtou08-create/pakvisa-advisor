    32→// ============================================================
    33→function useAnimatedCounter(target: number, duration: number = 1200, enabled: boolean = true) {
    34→  const [count, setCount] = useState(0);
    35→  useEffect(() => {
    36→    if (!enabled || target === 0) return;
    37→    let start = 0;
    38→    let startTime: number | null = null;
    39→    let raf: number;
    40→    const step = (timestamp: number) => {
    41→      if (!startTime) startTime = timestamp;
    42→      const progress = Math.min((timestamp - startTime) / duration, 1);
    43→      // Ease-out cubic
    44→      const eased = 1 - Math.pow(1 - progress, 3);
    45→      const current = Math.round(eased * target);
    46→      setCount(current);
    47→      if (progress < 1) {
    48→        raf = requestAnimationFrame(step);
    49→      }
    50→    };
    51→    raf = requestAnimationFrame(step);
    52→    return () => cancelAnimationFrame(raf);
    53→  }, [target, duration, enabled]);
    54→  return count;
    55→}

   125→// Pagination Helper
   126→// ============================================================
   127→function generatePageNumbers(current: number, total: number): (number | 'ellipsis')[] {
   128→  if (total <= 7) {
   129→    return Array.from({ length: total }, (_, i) => i + 1);
   130→  }
   131→  const pages: (number | 'ellipsis')[] = [1];
   132→  if (current > 3) pages.push('ellipsis');
   133→  const start = Math.max(2, current - 1);
   134→  const end = Math.min(total - 1, current + 1);
   135→  for (let i = start; i <= end; i++) pages.push(i);
   136→  if (current < total - 2) pages.push('ellipsis');
   137→  if (total > 1) pages.push(total);
   138→  return pages;
   139→}

   142→// Visa Type Helpers (single source of truth for access type)
   143→// Priority: visaFree > visaOnArrival > etaAvailable > embassy
   144→// ============================================================
   145→function getPrimaryAccess(c: CountryData): string {
   146→  if (c.visaFree) return 'visa-free';
   147→  if (c.visaOnArrival) return 'visa-on-arrival';
   148→  if (c.etaAvailable) return 'e-visa';
   149→  return 'embassy';
   150→}
   151→function getVisaType(c: CountryData) {
   152→  const access = getPrimaryAccess(c);
   153→  switch (access) {
   154→    case 'visa-free':     return { label: 'Visa Free', color: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-400', dot: 'bg-emerald-500' };

   156→    case 'e-visa':       return { label: 'e-Visa', color: 'bg-sky-100 text-sky-700 dark:bg-sky-900/30 dark:text-sky-400', dot: 'bg-sky-500' };
   157→    default:             return { label: 'Embassy Required', color: 'bg-gray-100 text-gray-600 dark:bg-gray-800 dark:text-gray-400', dot: 'bg-gray-400' };
