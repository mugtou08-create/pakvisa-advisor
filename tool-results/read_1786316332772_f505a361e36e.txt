   369→export function WhatIfSimulator({ scoreResult, userProfile }: { scoreResult: ScoreBreakdown; userProfile: UserProfileData }) {
   370→  const [healthInsurance, setHealthInsurance] = useState(userProfile.hasHealthInsurance);
   371→  const [returnTicket, setReturnTicket] = useState(userProfile.hasReturnTicket);
   372→  const [hotelBooking, setHotelBooking] = useState(userProfile.hasHotelBooking);
   373→  const [sponsor, setSponsor] = useState(userProfile.hasSponsor);
   374→  const [priorTravel, setPriorTravel] = useState(userProfile.hasPriorTravel);
   375→  const [income, setIncome] = useState(userProfile.monthlyIncomeUSD);
   376→  const [savings, setSavings] = useState(userProfile.savingsUSD);
   377→  const [budget, setBudget] = useState(userProfile.budgetUSD);
   378→  const [recalculating, setRecalculating] = useState(false);
   379→  const [afterScore, setAfterScore] = useState<ScoreBreakdown | null>(null);
   380→
   381→  const applyChanges = async () => {
   382→    setRecalculating(true);
   383→    const modifiedProfile = {
   384→      ...userProfile,
   385→      hasHealthInsurance: healthInsurance,
   386→      hasReturnTicket: returnTicket,
   387→      hasHotelBooking: hotelBooking,
   388→      hasSponsor: sponsor,
   389→      hasPriorTravel: priorTravel,
   390→      monthlyIncomeUSD: income,
   391→      savingsUSD: savings,
   392→      budgetUSD: budget,
   393→    };
   394→    try {
   395→      const res = await fetch('/api/score', {
   396→        method: 'POST',
   397→        headers: { 'Content-Type': 'application/json' },
   398→        body: JSON.stringify({ countryCode: scoreResult.countryCode, profile: modifiedProfile }),
   399→      });
   400→      const data = await res.json();
   401→      if (data.data) {
   402→        setAfterScore(data.data);
   403→      } else {
   404→        toast.error(data.error || 'Recalculation failed');
   405→      }
   406→    } catch {
   407→      toast.error('Network error during recalculation');
   408→    }
   409→    setRecalculating(false);
   410→  };
   411→
   412→  const delta = afterScore ? afterScore.finalScore - scoreResult.finalScore : 0;
   413→
   414→  return (
   415→    <div className="space-y-4">
   416→      <div className="grid grid-cols-1 md:grid-cols-2 gap-4 responsive-stack-mobile">
