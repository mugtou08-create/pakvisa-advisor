  1624→export function VisaCountdownTimer() {
  1625→  const { targetTravelDate, setTargetTravelDate } = useAppStore();
  1626→  const [now, setNow] = useState(new Date());
  1627→  const [showPicker, setShowPicker] = useState(false);
  1628→
  1629→  useEffect(() => {
  1630→    const interval = setInterval(() => setNow(new Date()), 1000);
  1631→    return () => clearInterval(interval);
  1632→  }, []);
  1633→
  1634→  const target = targetTravelDate ? new Date(targetTravelDate + 'T00:00:00') : null;
  1635→  const diffMs = target ? target.getTime() - now.getTime() : 0;
  1636→  const totalDays = target ? Math.max(0, Math.ceil(diffMs / 86400000)) : 0;
  1637→  const hours = target ? Math.max(0, Math.floor((diffMs % 86400000) / 3600000)) : 0;
  1638→  const mins = target ? Math.max(0, Math.floor((diffMs % 3600000) / 60000)) : 0;
  1639→
  1640→  // Determine color based on urgency
  1641→  const getColorClass = () => {
  1642→    if (!target) return 'text-amber-600 dark:text-amber-400';

  1651→  if (!target) {

  1667→            initial={{ opacity: 0, scale: 0.95 }}

  1669→            className="relative"
