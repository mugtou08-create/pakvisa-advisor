    31→// ============================================================
    32→function getVisaType(c: CountryData) {
    33→  if (c.visaFree) return { label: 'Visa Free', color: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-400', dot: 'bg-emerald-500' };
    34→  if (c.visaOnArrival) return { label: 'Visa on Arrival', color: 'bg-amber-100 text-amber-700 dark:bg-amber-900/30 dark:text-amber-400', dot: 'bg-amber-500' };
    35→  if (c.etaAvailable) return { label: 'e-Visa', color: 'bg-sky-100 text-sky-700 dark:bg-sky-900/30 dark:text-sky-400', dot: 'bg-sky-500' };
    36→  return { label: 'Embassy Required', color: 'bg-gray-100 text-gray-600 dark:bg-gray-800 dark:text-gray-400', dot: 'bg-gray-400' };
    37→}

   238→  const mounted = useSyncExternalStore(() => () => {}, () => true, () => false);
   239→  const searchRef = useRef<HTMLInputElement>(null);
   240→
   241→  // Fetch all countries on mount
   242→  useEffect(() => {
   243→    let cancelled = false;
   244→    const load = async () => {
   245→      try {
   246→        const res = await fetch('/api/countries');
   247→        if (res.ok && !cancelled) {
   248→          const data = await res.json();
   249→          setCountries(data.data || []);
   250→        }
   251→      } catch {}
   252→      if (!cancelled) setLoading(false);
   253→    };
   254→    load();
   255→    return () => { cancelled = true; };
   256→  }, []);
   257→
   258→  // Filter countries by search query
   259→  const filtered = useMemo(() => {
   260→    if (!search.trim()) return [];
   261→    const q = search.toLowerCase();
   262→    return countries
   263→      .filter(c =>
   264→        c.name.toLowerCase().includes(q) ||
   265→        c.code.toLowerCase().includes(q) ||
   266→        c.continent.toLowerCase().includes(q)
   267→      )
   268→      .slice(0, 12);
   269→  }, [search, countries]);
   270→

   282→      .map(c => {
   283→        const p = priority.find(p => c.name.includes(p.match));
   284→        return p ? { ...c, _rank: p.rank } : null;
   285→      })
   286→      .filter((c): c is CountryData & { _rank: number } => c !== null)
   287→      .sort((a, b) => a._rank - b._rank)
   288→      .slice(0, 8);
   289→  }, [countries]);
   290→
   291→  // Stats
   292→  const stats = useMemo(() => ({
   293→    total: countries.length,
   294→    visaFree: countries.filter(c => c.visaFree).length,

   300→    <div className="min-h-screen flex flex-col bg-background">
   301→      {/* Skip to content */}
   302→      <a href="#main-content" className="skip-to-content">Skip to main content</a>

   304→      {/* ===== HEADER ===== */}

   306→        <div className="max-w-5xl mx-auto px-4 h-14 flex items-center justify-between">
   307→          <button onClick={() => { setSearch(''); setExpandedCountry(null); }} className="flex items-center gap-2.5 hover:opacity-80 transition-opacity">
   308→            <div className="w-8 h-8 rounded-lg bg-primary flex items-center justify-center shadow-sm">
   309→              <Globe className="w-4 h-4 text-primary-foreground" />
   310→            </div>
   311→            <div className="hidden sm:block">
   312→              <p className="font-bold text-sm leading-none">PakVisa</p>
   313→              <p className="text-[10px] text-muted-foreground leading-tight">Free Visa Checker</p>
   314→            </div>
   315→          </button>
   316→          <div className="flex items-center gap-0.5">
   317→            {mounted && (
   318→              <Button
   319→                variant="ghost"
   320→                size="icon"
   321→                className="h-9 w-9"
   322→                onClick={() => setTheme(theme === 'dark' ? 'light' : 'dark')}
   323→                aria-label="Toggle theme"
   324→              >
   325→                {theme === 'dark' ? <Sun className="w-4 h-4" /> : <Moon className="w-4 h-4" />}
   326→              </Button>
