    52→function relativeTime(daysAgo: number): string {
    53→  if (daysAgo === 0) return 'Today';
    54→  if (daysAgo === 1) return 'Yesterday';
    55→  if (daysAgo < 7) return `${daysAgo} days ago`;
    56→  if (daysAgo < 14) return '1 week ago';
    57→  if (daysAgo < 21) return `${Math.floor(daysAgo / 7)} weeks ago`;
    58→  if (daysAgo < 30) return '3 weeks ago';
    59→  if (daysAgo < 45) return '1 month ago';
    60→  return `${Math.floor(daysAgo / 30)} months ago`;
    61→}

    63→function generatePolicyChanges(countries: CountryData[]): PolicyChangeEntry[] {
    64→  if (!countries || countries.length === 0) return [];
    65→
    66→  const entries: PolicyChangeEntry[] = [];
    67→  const used = new Set<string>();
    68→  const timestamps = [0, 1, 2, 3, 5, 7, 10, 14, 21, 28, 35, 42, 49, 56];
    69→
    70→  // Pick visa-free countries
    71→  const vfCountries = countries.filter(c => c.visaFree);
    72→  const voaCountries = countries.filter(c => c.visaOnArrival && !c.visaFree);
    73→  const etaCountries = countries.filter(c => c.etaAvailable && !c.visaFree && !c.visaOnArrival);
    74→  const embassyCountries = countries.filter(c => !c.visaFree && !c.visaOnArrival && !c.etaAvailable);
    75→
    76→  const pick = (list: CountryData[]) => {
    77→    for (const c of list) {
    78→      if (!used.has(c.code) && entries.length < 10) return c;
    79→    }
    80→    return null;
    81→  };
    82→
    83→  // Visa-free entries (2-3)
    84→  for (let i = 0; i < 3; i++) {
    85→    const c = pick(vfCountries);
    86→    if (!c) break;
    87→    used.add(c.code);
    88→    entries.push({
    89→      id: `vf-${c.code}`, countryCode: c.code, countryName: c.name,
    90→      flagEmoji: c.flagEmoji, flagUrl: c.flagUrl,
    91→      changeType: 'visa-free',
    92→      title: 'Visa-free access confirmed',
    93→      description: `Pakistan passport holders enjoy visa-free entry to ${c.name} for up to ${c.visaTypes?.[0]?.maxDuration || '30 days'}.`,
    94→      detail: `This visa-free arrangement allows Pakistani citizens to enter ${c.name} without any prior visa application. Ensure your passport has at least 6 months validity. Best travel months: ${c.bestTravelMonths}. Average temperature: ${c.avgTempC}°C.`,
    95→      daysAgo: timestamps[entries.length] ?? 10,
    96→    });
    97→  }
    98→
    99→  // VOA entries (1-2)
   100→  for (let i = 0; i < 2; i++) {
   101→    const c = pick(voaCountries);
   102→    if (!c) break;
   103→    used.add(c.code);
   104→    entries.push({
   105→      id: `voa-${c.code}`, countryCode: c.code, countryName: c.name,
   106→      flagEmoji: c.flagEmoji, flagUrl: c.flagUrl,
   107→      changeType: 'processing',
   108→      title: 'Visa on arrival policy updated',
   109→      description: `${c.name} has streamlined its VOA process. Processing now takes approximately ${c.processingDaysMin}-${c.processingDaysMax} business days.`,
   110→      detail: `Pakistan passport holders can obtain a visa on arrival at ${c.name} ports of entry. The fee is approximately $${c.costProfile?.visaFeeUSD || 'varies'}. Maximum stay: ${c.visaTypes?.[0]?.maxDuration || '30 days'}. Extensions may be available.`,
   111→      daysAgo: timestamps[entries.length] ?? 15,
   112→    });
   113→  }
   114→
   115→  // e-Visa entries (2)
   116→  for (let i = 0; i < 2; i++) {
   117→    const c = pick(etaCountries);
   118→    if (!c) break;
   119→    used.add(c.code);
   120→    entries.push({
   121→      id: `eta-${c.code}`, countryCode: c.code, countryName: c.name,
   122→      flagEmoji: c.flagEmoji, flagUrl: c.flagUrl,
   123→      changeType: 'evisa',
   124→      title: 'e-Visa/eTA system launched',
   125→      description: `${c.name} has launched an electronic visa system for Pakistani passport holders. Apply online before travel.`,
   126→      detail: `The new e-Visa portal allows Pakistanis to apply for a ${c.name} visa entirely online. Processing time: ${c.processingDaysMin}-${c.processingDaysMax} days. Fee: $${c.costProfile?.visaFeeUSD || 'TBD'}. Required documents: valid passport, photographs, and travel itinerary.`,
   127→      daysAgo: timestamps[entries.length] ?? 20,
   128→    });
   129→  }
   130→
   131→  // Embassy/fee entries (2-3)
   132→  for (let i = 0; i < 3; i++) {
   133→    const c = pick([...embassyCountries, ...etaCountries, ...voaCountries]);
   134→    if (!c) break;
   135→    used.add(c.code);
   136→    const isFee = i % 2 === 0;
   137→    entries.push({
   138→      id: `emb-${c.code}`, countryCode: c.code, countryName: c.name,
   139→      flagEmoji: c.flagEmoji, flagUrl: c.flagUrl,
   140→      changeType: isFee ? 'fee-change' : 'requirement',
   141→      title: isFee ? 'Visa fee adjustment' : 'New document requirement added',
   142→      description: isFee
   143→        ? `${c.name} has adjusted its visa fee to $${c.costProfile?.visaFeeUSD || 'updated rate'} (previously $${Math.max(1, (c.costProfile?.visaFeeUSD || 50) - 10)}). Service fee: $${c.costProfile?.serviceFeeUSD || 'varies'}.`
   144→        : `${c.name} now requires additional documentation: ${c.requirements?.[0]?.requirement || 'updated proof of funds'} for all visa applications.`,
   145→      detail: isFee
   146→        ? `The revised fee structure for ${c.name} visas: Visa fee $${c.costProfile?.visaFeeUSD || 'TBD'}, Service fee $${c.costProfile?.serviceFeeUSD || 'TBD'}. Health insurance: $${c.costProfile?.healthInsuranceUSD || 'N/A'}. Budget estimate: $${c.costProfile?.totalMonthlyUSD || 'TBD'}/month for living expenses. `
   147→        : `Updated requirements for ${c.name}: ${c.requirements?.slice(0, 3).map(r => r.requirement).join(', ') || 'Standard documentation'}. Processing time: ${c.processingDaysMin}-${c.processingDaysMax} business days. Check official sources for the most current information.`,
   148→      daysAgo: timestamps[entries.length] ?? 25,
   149→    });
   150→  }
   151→
   152→  return entries.slice(0, 10);
   153→}

   334→interface ChecklistSection {
   335→  id: string;
   336→  title: string;

   342→function buildChecklistSections(country: CountryData | null, userProfile: UserProfileData | null): ChecklistSection[] {
   343→  const sections: ChecklistSection[] = [];
   344→
   345→  // Section 1: Pre-Departure
   346→  const preDepartureItems: { id: string; label: string }[] = [
   347→    { id: 'pre-passport', label: 'Check passport validity (6+ months)' },
   348→    { id: 'pre-insurance', label: 'Purchase travel insurance' },
   349→    { id: 'pre-flights', label: 'Book flights' },
   350→    { id: 'pre-accommodation', label: 'Book accommodation' },
   351→    { id: 'pre-currency', label: `Exchange currency${country ? ` (${country.currencyCode || 'local currency'})` : ''}` },
   352→    { id: 'pre-maps', label: 'Download offline maps' },
   353→    { id: 'pre-bank', label: 'Notify your bank about travel' },
   354→  ];
   355→
   356→  if (country) {
   357→    if (country.visaFree) {
   358→      preDepartureItems.unshift({ id: 'pre-vf-check', label: 'Confirm visa-free entry requirements' });
   359→    } else if (country.visaOnArrival) {
   360→      preDepartureItems.unshift({ id: 'pre-voa-docs', label: 'Prepare VOA documents (return ticket, hotel booking)' });
   361→      preDepartureItems.splice(1, 0, { id: 'pre-voa-fee', label: `Carry VOA fee in cash (~$${country.costProfile?.visaFeeUSD || 'check amount'})` });
   362→    } else if (country.etaAvailable) {
   363→      preDepartureItems.unshift({ id: 'pre-eta-apply', label: 'Apply for e-Visa/eTA online' });
   364→      preDepartureItems.splice(1, 0, { id: 'pre-eta-print', label: 'Print e-Visa approval letter' });
   365→    } else {
   366→      preDepartureItems.unshift({ id: 'pre-embassy-apply', label: 'Apply for visa at embassy/consulate' });
   367→      preDepartureItems.splice(1, 0, { id: 'pre-embassy-appt', label: 'Book embassy appointment' });
   368→      preDepartureItems.splice(2, 0, { id: 'pre-embassy-docs', label: 'Prepare and organize all documents' });
   369→    }
   370→  }
   371→
   372→  if (userProfile) {
   373→    if (userProfile.occupation && userProfile.occupation !== 'Unemployed' && userProfile.occupation !== 'Student') {
   374→      preDepartureItems.push({ id: 'pre-leave', label: 'Request leave from employer' });
   375→    }
   376→    if (userProfile.maritalStatus === 'Married' || (userProfile.dependents && userProfile.dependents > 0)) {
   377→      preDepartureItems.push({ id: 'pre-family', label: 'Arrange family care during travel' });
   378→    }
   379→    if (userProfile.occupation === 'Student') {
   380→      preDepartureItems.push({ id: 'pre-student-docs', label: 'Get student documents / NOC from institution' });
   381→    }
   382→  }
   383→
   384→  sections.push({
   385→    id: 'pre-departure',
   386→    title: `Pre-Departure Checklist${country ? ` — ${country.name}` : ''}`,
   387→    icon: <PlaneTakeoff className="h-4 w-4 text-amber-500" />,
   388→    colorClass: 'text-amber-600 dark:text-amber-400',
   389→    items: preDepartureItems,
   390→  });
   391→
   392→  // Section 2: Required Documents
   393→  const docItems: { id: string; label: string }[] = [
   394→    { id: 'doc-passport', label: 'Valid passport (6+ months validity)' },
   395→    { id: 'doc-photos', label: 'Passport-sized photographs (2x)' },
   396→    { id: 'doc-bank', label: 'Bank statements (last 6 months)' },
   397→    { id: 'doc-flights', label: 'Flight itinerary / booking confirmation' },
   398→    { id: 'doc-hotel', label: 'Hotel reservation confirmation' },
   399→    { id: 'doc-insurance', label: 'Travel insurance certificate' },
   400→    { id: 'doc-employment', label: 'Employment letter / NOC from employer' },
   401→    { id: 'doc-tax', label: 'Income tax returns' },
   402→  ];
   403→
   404→  if (country && country.requirements && country.requirements.length > 0) {
   405→    country.requirements.forEach((req, i) => {
   406→      if (i < 4 && !docItems.some(d => d.label.toLowerCase().includes(req.requirement.toLowerCase().slice(0, 15)))) {
   407→        docItems.push({ id: `doc-req-${i}`, label: req.requirement });
   408→      }
   409→    });
   410→  }
   411→
   412→  sections.push({
   413→    id: 'documents',
   414→    title: 'Required Documents',
   415→    icon: <FileText className="h-4 w-4 text-orange-500" />,
   416→    colorClass: 'text-orange-600 dark:text-orange-400',
   417→    items: docItems,
   418→  });
   419→

   629→function getVisaTypeLabel(c: CountryData): string {
   630→  if (c.visaFree) return 'Visa-Free';
   631→  if (c.visaOnArrival) return 'VOA';
   632→  if (c.etaAvailable) return 'eTA';
   633→  return 'Embassy';

   636→export const ContinentQuickStats = React.memo(function ContinentQuickStats({ countries }: { countries: CountryData[] }) {
   637→  const stats = useMemo<ContinentStat[]>(() => {
   638→    if (!countries.length) return [];
   639→
   640→    const byContinent = new Map<string, CountryData[]>();
   641→    for (const c of countries) {
   642→      const cont = c.continent || 'Other';
   643→      if (!byContinent.has(cont)) byContinent.set(cont, []);
   644→      byContinent.get(cont)!.push(c);
   645→    }
   646→
   647→    return CONTINENT_ORDER
   648→      .filter((name) => {
   649→        const list = byContinent.get(name);
   650→        return list && list.length > 0;
   651→      })
   652→      .map((name) => {
   653→        const list = byContinent.get(name)!;
   654→        const fees = list.map((c) => c.costProfile?.visaFeeUSD ?? 0).filter((f) => f > 0);
   655→        const avgFee = fees.length > 0 ? Math.round(fees.reduce((a, b) => a + b, 0) / fees.length) : 0;
   656→        const avgSafety = list.reduce((a, c) => a + (c.safetyRating || 5), 0) / list.length;
   657→
   658→        const typeCounts: Record<string, number> = {};
   659→        for (const c of list) {
   660→          const label = getVisaTypeLabel(c);
   661→          typeCounts[label] = (typeCounts[label] || 0) + 1;
   662→        }
   663→        const topVisaType = Object.entries(typeCounts).sort((a, b) => b[1] - a[1])[0]?.[0] || 'N/A';
   664→
   665→        return { name, count: list.length, avgFee, topVisaType, avgSafety: Math.round(avgSafety * 10) / 10 };
   666→      });
   667→  }, [countries]);
   668→
   669→  if (stats.length === 0) return null;
   670→
   671→  return (

   676→        <Badge variant="secondary" className="text-[10px] bg-amber-100 text-amber-700 dark:bg-amber-900/30 dark:text-amber-400">
   677→          {stats.length} regions
   678→        </Badge>
   679→      </div>
   680→      <div className="flex gap-3 overflow-x-auto custom-scrollbar-thin pb-2 -mx-1 px-1">
   681→        {stats.map((s, idx) => {
   682→          const colors = CONTINENT_COLORS[s.name] || CONTINENT_COLORS['Asia'];
   683→          return (
   684→            <motion.div
   685→              key={s.name}
   686→              initial={{ opacity: 0, y: 12 }}
   687→              animate={{ opacity: 1, y: 0 }}
