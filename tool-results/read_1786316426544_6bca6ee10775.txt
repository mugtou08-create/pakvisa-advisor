    21→function getVisaStatus(c: CountryData) {
    22→  if (c.visaFree) return { label: 'Visa Free', color: 'bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-400' };
    23→  if (c.visaOnArrival) return { label: 'Visa on Arrival', color: 'bg-orange-100 text-orange-700 dark:bg-orange-900/40 dark:text-orange-400' };
    24→  if (c.etaAvailable) return { label: 'e-Visa', color: 'bg-amber-50 text-amber-600 dark:bg-amber-900/30 dark:text-amber-400' };
    25→  return { label: 'Embassy Required', color: 'bg-orange-100 text-orange-700 dark:bg-orange-900/40 dark:text-orange-400' };
    26→}

    41→export function CountryComparisonSwiper({ countries, onSelectCountry }: {
    42→  countries: CountryData[];
    43→  onSelectCountry: (code: string) => void;
    44→}) {
    45→  const [currentIndex, setCurrentIndex] = useState(0);
    46→  const [isPaused, setIsPaused] = useState(false);
    47→  const [direction, setDirection] = useState(0);
    48→  const dragRef = useRef<HTMLDivElement>(null);
    49→  const startXRef = useRef(0);
    50→
    51→  // Take top 8 countries for the swiper
    52→  const swiperCountries = countries.slice(0, 8);
    53→  const total = swiperCountries.length;
    54→
    55→  const goTo = useCallback((newIndex: number, dir: number) => {
    56→    setDirection(dir);
    57→    setCurrentIndex((newIndex + total) % total);
    58→  }, [total]);
    59→
    60→  const goNext = useCallback(() => goTo(currentIndex + 1, 1), [currentIndex, goTo]);
    61→  const goPrev = useCallback(() => goTo(currentIndex - 1, -1), [currentIndex, goTo]);
    62→
    63→  // Auto-rotate every 5 seconds
    64→  useEffect(() => {
    65→    if (isPaused || total === 0) return;
    66→    const timer = setInterval(goNext, 5000);
    67→    return () => clearInterval(timer);
    68→  }, [isPaused, goNext, total]);
    69→
    70→  // Touch/drag handlers
    71→  const handleDragStart = (e: React.MouseEvent | React.TouchEvent) => {
    72→    const clientX = 'touches' in e ? e.touches[0].clientX : e.clientX;
    73→    startXRef.current = clientX;
    74→  };
    75→
    76→  const handleDragEnd = (e: React.MouseEvent | React.TouchEvent) => {
    77→    const clientX = 'changedTouches' in e ? e.changedTouches[0].clientX : e.clientX;
    78→    const diff = startXRef.current - clientX;
    79→    if (Math.abs(diff) > 50) {
    80→      if (diff > 0) goNext();
    81→      else goPrev();
    82→    }
    83→  };
    84→
    85→  if (swiperCountries.length === 0) return null;
    86→
    87→  // Get visible indices for 3-card desktop, 1-card mobile
    88→  const getVisibleIndices = () => {
    89→    const indices: number[] = [];
    90→    for (let offset = -1; offset <= 1; offset++) {
    91→      indices.push((currentIndex + offset + total) % total);
    92→    }
    93→    return indices;
    94→  };
    95→
    96→  const visibleIndices = getVisibleIndices();
    97→  const centerIdx = visibleIndices[1]; // center card
    98→  const centerCountry = swiperCountries[centerIdx];
    99→
   100→  return (
   101→    <div
   102→      className="relative"
   103→      onMouseEnter={() => setIsPaused(true)}
   104→      onMouseLeave={() => setIsPaused(false)}
   105→    >
   106→      {/* Desktop: 3-card 3D stack */}
   107→      <div className="hidden md:flex items-center justify-center gap-4 py-4">
   108→        {/* Left card */}
   109→        <motion.div
   110→          key={visibleIndices[0]}
   111→          initial={{ opacity: 0, x: direction > 0 ? -80 : -40, scale: 0.85 }}
   112→          animate={{ opacity: 0.5, x: 0, scale: 0.85 }}
   113→          exit={{ opacity: 0, x: direction > 0 ? 40 : -80, scale: 0.85 }}
   114→          transition={{ duration: 0.4, ease: 'easeInOut' }}
   115→          className="w-64 shrink-0 cursor-pointer"
   116→          style={{ perspective: '1000px', transform: 'rotateY(8deg)' }}
   117→          onClick={() => goTo(visibleIndices[0], -1)}
   118→        >
   119→          <SwipeCard country={swiperCountries[visibleIndices[0]]} onSelect={onSelectCountry} isSide />
   120→        </motion.div>
   121→
   122→        {/* Center card */}
   123→        <motion.div
   124→          key={centerIdx}
   125→          initial={{ opacity: 0, x: direction * 120, scale: 0.9 }}
   126→          animate={{ opacity: 1, x: 0, scale: 1 }}
   127→          exit={{ opacity: 0, x: -direction * 120, scale: 0.9 }}
   128→          transition={{ duration: 0.4, ease: 'easeInOut' }}
   129→          className="w-72 shrink-0 z-10"
   130→        >
   131→          <SwipeCard country={centerCountry} onSelect={onSelectCountry} isCenter />
   132→        </motion.div>
   133→
   134→        {/* Right card */}
   135→        <motion.div
   136→          key={visibleIndices[2]}
   137→          initial={{ opacity: 0, x: direction > 0 ? 80 : 40, scale: 0.85 }}
   138→          animate={{ opacity: 0.5, x: 0, scale: 0.85 }}
   139→          exit={{ opacity: 0, x: direction > 0 ? -40 : 80, scale: 0.85 }}
   140→          transition={{ duration: 0.4, ease: 'easeInOut' }}
   141→          className="w-64 shrink-0 cursor-pointer"
   142→          style={{ perspective: '1000px', transform: 'rotateY(-8deg)' }}
   143→          onClick={() => goTo(visibleIndices[2], 1)}
   144→        >
   145→          <SwipeCard country={swiperCountries[visibleIndices[2]]} onSelect={onSelectCountry} isSide />
   146→        </motion.div>
   147→      </div>
   148→
   149→      {/* Mobile: single card with swipe */}
   150→      <div
   151→        className="md:hidden overflow-hidden px-4"
   152→        ref={dragRef}
   153→        onMouseDown={handleDragStart}
   154→        onMouseUp={handleDragEnd}
   155→        onTouchStart={handleDragStart}
   156→        onTouchEnd={handleDragEnd}
   157→      >
   158→        <AnimatePresence mode="wait" custom={direction}>
   159→          <motion.div
   160→            key={currentIndex}
   161→            custom={direction}
   162→            initial={{ opacity: 0, x: direction * 200 }}
   163→            animate={{ opacity: 1, x: 0 }}
   164→            exit={{ opacity: 0, x: -direction * 200 }}
   165→            transition={{ duration: 0.35, ease: 'easeInOut' }}
   166→            drag="x"
   167→            dragConstraints={{ left: 0, right: 0 }}
   168→            dragElastic={0.2}
   169→            onDragEnd={(_, info) => {
   170→              if (info.offset.x > 60) goPrev();
   171→              else if (info.offset.x < -60) goNext();
   172→            }}
   173→          >
   174→            <SwipeCard country={swiperCountries[currentIndex]} onSelect={onSelectCountry} isCenter />
   175→          </motion.div>
   176→        </AnimatePresence>
   177→      </div>
   178→
   179→      {/* Navigation buttons */}
   180→      <div className="flex items-center justify-center gap-3 mt-3">
   181→        <Button
   182→          variant="outline"
   183→          size="icon"
   184→          className="h-8 w-8 rounded-full border-amber-300 hover:bg-amber-50 dark:border-amber-700 dark:hover:bg-amber-900/30"
   185→          onClick={goPrev}
   186→          aria-label="Previous country"
   187→        >
   188→          <ChevronLeft className="w-4 h-4 text-amber-700 dark:text-amber-400" />
   189→        </Button>
   190→
   191→        {/* Dot indicators */}
   192→        <div className="flex items-center gap-1.5">
   193→          {swiperCountries.map((_, i) => (
   194→            <button
   195→              key={i}
   196→              onClick={() => goTo(i, i > currentIndex ? 1 : -1)}
   197→              className={`rounded-full transition-all duration-300 ${
   198→                i === currentIndex
   199→                  ? 'w-6 h-2 bg-amber-500'
   200→                  : 'w-2 h-2 bg-amber-300 dark:bg-amber-700 hover:bg-amber-400'
   201→              }`}
   202→              aria-label={`Go to slide ${i + 1}`}
   203→            />
   204→          ))}
   205→        </div>
   206→
   207→        <Button
   208→          variant="outline"
   209→          size="icon"
   210→          className="h-8 w-8 rounded-full border-amber-300 hover:bg-amber-50 dark:border-amber-700 dark:hover:bg-amber-900/30"
   211→          onClick={goNext}
   212→          aria-label="Next country"
   213→        >
   214→          <ChevronRight className="w-4 h-4 text-amber-700 dark:text-amber-400" />
   215→        </Button>
   216→      </div>
   217→    </div>
   218→  );
   219→}

   293→const ACCENT_PRESETS = [
   294→  { name: 'Amber', color: '#F59E0B', primary: '245, 158, 11' },
   295→  { name: 'Orange', color: '#F97316', primary: '249, 115, 22' },
   296→  { name: 'Mango', color: '#FB923C', primary: '251, 146, 60' },
   297→  { name: 'Gold', color: '#EAB308', primary: '234, 179, 8' },
   298→  { name: 'Warm Brown', color: '#A16207', primary: '161, 98, 7' },
   299→  { name: 'Rose Gold', color: '#E11D48', primary: '225, 29, 72' },
   300→] as const;
   301→
   302→const STORAGE_KEY = 'pakvisa-accent-color';
   303→
   304→function applyAccent(preset: typeof ACCENT_PRESETS[number]) {
   305→  if (typeof document === 'undefined') return;
   306→  const root = document.documentElement;
   307→  const [r, g, b] = preset.primary.split(', ');
   308→  root.style.setProperty('--accent-primary', preset.color);
   309→  root.style.setProperty('--accent-rgb', `${r}, ${g}, ${b}`);
   310→  root.style.setProperty('--accent-foreground', '#FFFFFF');
   311→  root.style.setProperty('--accent-light', `rgba(${r}, ${g}, ${b}, 0.15)`);
   312→  root.style.setProperty('--accent-medium', `rgba(${r}, ${g}, ${b}, 0.3)`);
   313→  localStorage.setItem(STORAGE_KEY, preset.name);
   314→}
   315→
   316→export function ThemeAccentCustomizer({ defaultAccent }: { defaultAccent?: string }) {
   317→  const [selected, setSelected] = useState(defaultAccent || 'Amber');
   318→
   319→  useEffect(() => {
   320→    const saved = localStorage.getItem(STORAGE_KEY);
   321→    if (saved) {
   322→      const preset = ACCENT_PRESETS.find(p => p.name === saved);
   323→      if (preset) {
   324→        applyAccent(preset);
   325→      }
   326→    }
   327→  }, []);
   328→
   329→  const handleSelect = (preset: typeof ACCENT_PRESETS[number]) => {
   330→    setSelected(preset.name);
   331→    applyAccent(preset);
   332→  };
   333→
   334→  const handleReset = () => {
   335→    const defaultPreset = ACCENT_PRESETS[0];
   336→    setSelected(defaultPreset.name);
   337→    applyAccent(defaultPreset);
   338→  };
   339→
   340→  const activePreset = ACCENT_PRESETS.find(p => p.name === selected) || ACCENT_PRESETS[0];
   341→
   342→  return (
   343→    <Card className="card-elevated-1 glass-card">
   344→      <CardHeader className="pb-3">
   345→        <CardTitle className="text-sm font-semibold flex items-center gap-2">
   346→          <Settings className="w-4 h-4 text-amber-500" />
   347→          Accent Color
   348→        </CardTitle>
   349→      </CardHeader>
   350→      <CardContent className="space-y-3">
   351→        {/* Color preset buttons */}
   352→        <div className="flex items-center justify-center gap-3">
   353→          {ACCENT_PRESETS.map((preset) => (
   354→            <button
   355→              key={preset.name}
   356→              onClick={() => handleSelect(preset)}
   357→              className={`press-effect relative w-8 h-8 rounded-full border-2 transition-all duration-200 hover:scale-110 ${
   358→                selected === preset.name
   359→                  ? 'border-amber-900 dark:border-amber-100 ring-2 ring-offset-2 ring-amber-400 dark:ring-offset-background'
   360→                  : 'border-transparent hover:border-amber-400/50'
   361→              }`}
   362→              style={{ backgroundColor: preset.color }}
   363→              aria-label={`Select ${preset.name} accent color`}
   364→              title={preset.name}
   365→            >
   366→              {selected === preset.name && (
   367→                <Check className="w-4 h-4 text-white absolute inset-0 m-auto drop-shadow-sm" strokeWidth={3} />
   368→              )}
   369→            </button>
   370→          ))}
   371→        </div>
   372→
   373→        {/* Current accent name */}
   374→        <p className="text-center text-xs text-muted-foreground">
   375→          Current: <span className="font-medium text-amber-700 dark:text-amber-400">{activePreset.name}</span>
   376→        </p>
   377→
   378→        {/* Reset button */}
   379→        <Button
   380→          variant="ghost"
   381→          size="sm"
   382→          className="w-full text-xs text-muted-foreground hover:text-amber-700 dark:hover:text-amber-400"
   383→          onClick={handleReset}
   384→        >
   385→          <RotateCcw className="w-3 h-3 mr-1" />
   386→          Reset to Default
   387→        </Button>
   388→      </CardContent>
   389→    </Card>
   390→  );
   391→}

   574→
   575→function isFieldFilled(profile: UserProfileData, field: string): boolean {
   576→  const val = (profile as Record<string, unknown>)[field];
   577→  if (val === undefined || val === null || val === '') return false;
   578→  if (typeof val === 'number') return val > 0;
   579→  if (Array.isArray(val)) return val.length > 0;
   580→  if (typeof val === 'boolean') return true;
   581→  return true;
   582→}
   583→
   584→export function UserProfileProgressWidget({ profile }: { profile: UserProfileData | null }) {
   585→  const allFields = PROFILE_CATEGORIES.flatMap(c => c.fields);
   586→  const filledCount = profile ? allFields.filter(f => isFieldFilled(profile, f)).length : 0;
   587→  const totalFields = allFields.length;
   588→  const pct = totalFields > 0 ? Math.round((filledCount / totalFields) * 100) : 0;
   589→  const radius = 34;
   590→  const circumference = 2 * Math.PI * radius;
   591→  const offset = circumference * (1 - pct / 100);
   592→
   593→  const categoryStatuses = PROFILE_CATEGORIES.map(cat => {
   594→    const filled = cat.fields.filter(f => profile && isFieldFilled(profile, f)).length;
   595→    return { ...cat, filled, total: cat.fields.length, complete: filled === cat.fields.length };
   596→  });
   597→
   598→  return (
   599→    <motion.div
   600→      initial={{ opacity: 0, y: -10 }}
   601→      animate={{ opacity: 1, y: 0 }}
   602→      transition={{ duration: 0.4 }}
   603→      className="card-elevated-1 stat-card-compact rounded-xl p-4"
   604→    >
   605→      <div className="flex items-center gap-4">
   606→        {/* Circular progress ring */}
   607→        <div className="relative w-20 h-20 shrink-0">
   608→          <svg className="w-20 h-20 -rotate-90" viewBox="0 0 80 80">
   609→            <circle cx="40" cy="40" r={radius} fill="none" stroke="currentColor" strokeWidth="5" className="text-muted/20" />
   610→            <circle
   611→              cx="40" cy="40" r={radius} fill="none"
   612→              stroke="#f59e0b" strokeWidth="5" strokeLinecap="round"
   613→              strokeDasharray={circumference}
   614→              strokeDashoffset={offset}
   615→              className="transition-all duration-700 progress-amber"
   616→            />
   617→          </svg>
   618→          <div className="absolute inset-0 flex items-center justify-center">
   619→            <span className="text-sm font-bold text-amber-700 dark:text-amber-400">{pct}%</span>
   620→          </div>
   621→        </div>
   622→
   623→        {/* Text content */}
   624→        <div className="flex-1 min-w-0">
   625→          <p className="text-sm font-semibold mb-1">
   626→            {pct === 100 ? 'Profile Complete! ✓' : 'Complete your profile'}
   627→          </p>

   629→            {filledCount} of {totalFields} fields completed
   630→          </p>
   631→
   632→          {/* Category dots */}
   633→          <div className="flex items-center gap-2 flex-wrap">

   637→                  <TooltipTrigger asChild>

   639→                      className={`w-3 h-3 rounded-full transition-colors ${
   640→                        cat.complete ? 'bg-amber-500' : 'bg-stone-300 dark:bg-stone-600'

   643→                  </TooltipTrigger>

   657→            Fill Now

   691→function getRelativeTime(iso: string) {
   692→  const diff = Date.now() - new Date(iso).getTime();
   693→  const mins = Math.floor(diff / 60000);
   694→  if (mins < 60) return `${mins}m ago`;
   695→  const hrs = Math.floor(mins / 60);
   696→  if (hrs < 24) return `${hrs}h ago`;
   697→  const days = Math.floor(hrs / 24);
   698→  return `${days}d ago`;
   699→}
