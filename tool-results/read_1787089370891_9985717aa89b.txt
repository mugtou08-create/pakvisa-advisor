    17→// Live Clock Component
    18→// ============================================================
    19→function LiveClock({ timezone, cityName }: { timezone: string; cityName: string }) {
    20→  const [time, setTime] = useState('');
    21→  const [date, setDate] = useState('');
    22→
    23→  useEffect(() => {
    24→    function updateTime() {
    25→      try {
    26→        const now = new Date();
    27→        const formatter = new Intl.DateTimeFormat('en-US', {
    28→          timeZone: timezone,
    29→          hour: '2-digit',
    30→          minute: '2-digit',
    31→          second: '2-digit',
    32→          hour12: true,
    33→        });
    34→        const dateFormatter = new Intl.DateTimeFormat('en-US', {
    35→          timeZone: timezone,
    36→          weekday: 'short',
    37→          month: 'short',
    38→          day: 'numeric',
    39→        });
    40→        setTime(formatter.format(now));
    41→        setDate(dateFormatter.format(now));
    42→      } catch {
    43→        setTime('—');
    44→        setDate('—');
    45→      }
    46→    }
    47→    updateTime();
    48→    const interval = setInterval(updateTime, 1000);
    49→    return () => clearInterval(interval);
    50→  }, [timezone]);
    51→
    52→  // Calculate time difference from PKT (Asia/Karachi)
    53→  const timeDiff = useMemo(() => {
    54→    try {
    55→      const now = new Date();
    56→      // Get UTC offsets in minutes for both timezones
    57→      const targetParts = new Intl.DateTimeFormat('en-US', {
    58→        timeZone: timezone,
    59→        timeZoneName: 'shortOffset',
    60→      }).formatToParts(now);
    61→      const pktParts = new Intl.DateTimeFormat('en-US', {
    62→        timeZone: 'Asia/Karachi',
    63→        timeZoneName: 'shortOffset',
    64→      }).formatToParts(now);
    65→
    66→      const parseOffset = (parts: Intl.DateTimeFormatPart[]) => {
    67→        const tzPart = parts.find(p => p.type === 'timeZoneName');
    68→        if (!tzPart) return 0;
    69→        const match = tzPart.value.match(/GMT([+-]?)(\d{1,2})(?::(\d{2}))?/);
    70→        if (!match) return 0;
    71→        const sign = match[1] === '-' ? -1 : 1;
    72→        const hours = parseInt(match[2]);
    73→        const minutes = match[3] ? parseInt(match[3]) : 0;
    74→        return sign * (hours * 60 + minutes);
    75→      };
    76→
    77→      const targetOffset = parseOffset(targetParts);
    78→      const pktOffset = parseOffset(pktParts);
    79→      const diffHours = (targetOffset - pktOffset) / 60;
    80→
    81→      if (diffHours === 0) return 'Same as PKT';
    82→      if (diffHours > 0) return `+${diffHours}h from PKT`;
    83→      return `${diffHours}h from PKT`;
    84→    } catch {
    85→      return '';
    86→    }
    87→  }, [timezone]);
    88→
    89→  return (
    90→    <div className="flex items-center gap-3 rounded-lg bg-sky-50 dark:bg-sky-950/20 border border-sky-200 dark:border-sky-800 p-3">
    91→      <Clock className="w-5 h-5 text-sky-600 shrink-0" />
    92→      <div>
    93→        <p className="text-sm font-semibold text-sky-900 dark:text-sky-200">{time}</p>
    94→        <p className="text-xs text-sky-700 dark:text-sky-400">{date} {timeDiff && `· ${timeDiff}`}</p>
    95→      </div>
    96→    </div>
    97→  );
    98→}

   101→// Currency Converter Widget
   102→// ============================================================
   103→function CurrencyConverter({ currencyCode, currencyName }: { currencyCode: string; currencyName: string }) {
   104→  const [amount, setAmount] = useState<string>('100');
   105→  const [direction, setDirection] = useState<'topkr' | 'frompkr'>('topkr');
   106→
   107→  const rate = EXCHANGE_RATES[currencyCode];
   108→  const [topkr, frompkr] = useMemo(() => {
   109→    if (!rate) return ['—', '—'];
   110→    return [
   111→      (parseFloat(amount || '0') * rate).toLocaleString('en-PK', { maximumFractionDigits: 2 }),
   112→      (parseFloat(amount || '0') / rate).toLocaleString(undefined, { maximumFractionDigits: 2 }),
   113→    ];
   114→  }, [amount, rate]);
   115→
   116→  return (
   117→    <div className="rounded-lg bg-emerald-50 dark:bg-emerald-950/20 border border-emerald-200 dark:border-emerald-800 p-3 space-y-2">
   118→      <div className="flex items-center justify-between">
   119→        <div className="flex items-center gap-2">
   120→          <Banknote className="w-4 h-4 text-emerald-600" />
   121→          <span className="text-xs font-semibold text-emerald-900 dark:text-emerald-200">Currency Converter</span>
   122→        </div>
   123→        <Badge variant="outline" className="text-[10px] border-emerald-300 dark:border-emerald-700 text-emerald-700 dark:text-emerald-400">
   124→          {currencyCode} · {currencyName}
   125→        </Badge>
   126→      </div>
   127→      <div className="flex items-center gap-2">
   128→        <input
   129→          type="number"
   130→          value={amount}
   131→          onChange={(e) => setAmount(e.target.value)}
   132→          className="w-full h-8 px-2 rounded-md border bg-white dark:bg-background text-sm focus:outline-none focus:ring-1 focus:ring-emerald-500"
   133→          min="0"
   134→        />
   135→        <button
   136→          onClick={() => setDirection((d) => d === 'topkr' ? 'frompkr' : 'topkr')}
   137→          className="p-1.5 rounded-md border bg-white dark:bg-background hover:bg-muted transition-colors"
   138→          title="Swap direction"
   139→        >
   140→          <ArrowRightLeft className="w-3.5 h-3.5 text-emerald-600" />
   141→        </button>
   142→      </div>
   143→      {rate ? (
   144→        <div className="flex items-center justify-between text-xs">
   145→          <span className="text-muted-foreground">
   146→            {direction === 'topkr' ? `${amount || '0'} ${currencyCode} =` : `PKR ${amount || '0'} =`}
   147→          </span>
   148→          <span className="font-semibold text-emerald-800 dark:text-emerald-300">
   149→            {direction === 'topkr' ? `PKR ${topkr}` : `${frompkr} ${currencyCode}`}
   150→          </span>
   151→        </div>
   152→      ) : (
   153→        <p className="text-xs text-muted-foreground">Exchange rate not available</p>
   154→      )}
   155→      <p className="text-[10px] text-emerald-700/70 dark:text-emerald-400/70">
   156→        Rate: 1 {currencyCode} ≈ PKR {rate?.toFixed(2) || '—'} · Approximate, verify before transaction
   157→      </p>
   158→    </div>
   159→  );
   160→}

   367→// ============================================================
   368→// Weather Info Section
   369→// ============================================================
   370→function WeatherInfo({ avgTemp, monthlyTemps }: { avgTemp: string; monthlyTemps: Record<string, number> }) {
   371→  const temps = Object.entries(monthlyTemps);
   372→  if (temps.length === 0 && !avgTemp) return null;
   373→
   374→  const maxTemp = temps.length > 0 ? Math.max(...temps.map(([, v]) => v)) : parseInt(avgTemp) || 0;
   375→  const minTemp = temps.length > 0 ? Math.min(...temps.map(([, v]) => v)) : parseInt(avgTemp) || 0;
   376→
   377→  return (
   378→    <div>
   379→      <h5 className="text-xs font-semibold text-muted-foreground uppercase tracking-wide mb-2">Climate & Weather</h5>
   380→      <div className="rounded-lg bg-muted/50 p-3 space-y-3">
   381→        <div className="flex items-center justify-between text-sm">
   382→          <span className="flex items-center gap-1.5">
   383→            <Thermometer className="w-4 h-4 text-red-500" />
   384→            <span className="font-medium">Avg Temperature</span>
   385→          </span>
   386→          <span className="font-semibold">{avgTemp}°C</span>
   387→        </div>
   388→        <div className="flex items-center gap-3 text-xs text-muted-foreground">
   389→          <span>🔥 High: {maxTemp}°C</span>
   390→          <span>❄️ Low: {minTemp}°C</span>
   391→        </div>
   392→        {temps.length > 0 && (
   393→          <div className="flex items-end gap-0.5 h-8">
   394→            {temps.map(([month, temp], idx) => {
   395→              const height = maxTemp > minTemp ? ((temp - minTemp) / (maxTemp - minTemp)) * 100 : 50;
   396→              return (
   397→                <div key={month} className="flex-1 flex flex-col items-center gap-0.5">
   398→                  <div
   399→                    className="w-full rounded-t-sm min-h-[2px] transition-all"
   400→                    style={{
   401→                      height: `${Math.max(height, 5)}%`,
   402→                      backgroundColor: temp > 30 ? '#ef4444' : temp > 20 ? '#f59e0b' : temp > 10 ? '#3b82f6' : '#6366f1',
   403→                    }}
   404→                  />
   405→                  <span className="text-[8px] text-muted-foreground">{MONTH_NAMES[idx]?.[0] || ''}</span>
   406→                </div>
   407→              );
   408→            })}
   409→          </div>
   410→        )}
   411→      </div>
   412→    </div>
   413→  );
   414→}
