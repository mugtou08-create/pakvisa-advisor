   782→// ============ TYPING TEXT ANIMATION ============
   783→export function TypingText({ phrases, className = '' }: { phrases: string[]; className?: string }) {
   784→  const [current, setCurrent] = useState(0);
   785→  const [text, setText] = useState('');
   786→  const [isDeleting, setIsDeleting] = useState(false);
   787→
   788→  useEffect(() => {
   789→    const phrase = phrases[current];
   790→    const timeout = setTimeout(() => {
   791→      if (!isDeleting) {
   792→        setText(phrase.substring(0, text.length + 1));
   793→        if (text.length === phrase.length) {
   794→          setTimeout(() => setIsDeleting(true), 2200);
   795→        }
   796→      } else {
   797→        setText(phrase.substring(0, text.length - 1));
   798→        if (text.length === 0) {
   799→          setIsDeleting(false);
   800→          setCurrent((current + 1) % phrases.length);
   801→        }
   802→      }
   803→    }, isDeleting ? 25 : 55);
   804→    return () => clearTimeout(timeout);
   805→  }, [text, isDeleting, current, phrases]);
   806→
   807→  return (
   808→    <span className={className}>
   809→      {text}
   810→      <span className="inline-block w-0 h-[1.1em] ml-0.5 align-middle typing-text-cursor" />
   811→    </span>
   812→  );
   813→}

   865→// ============ MINI RADAR CHART (3-axis for country cards) ============
   866→export function MiniRadarChart({ safety, cost, ease }: { safety: number; cost: number; ease: number }) {
   867→  const cx = 28, cy = 28, r = 22;
   868→  const axes = 3;
   869→  const angleStep = (2 * Math.PI) / axes;
   870→  const startAngle = -Math.PI / 2;
   871→  const s = safety / 10;
   872→  const c = Math.max(0.05, 1 - cost / 3000);
   873→  const e = ease;
   874→  const values = [s, c, e];
   875→  const points = values.map((val, i) => {
   876→    const angle = startAngle + i * angleStep;
   877→    return `${cx + val * r * Math.cos(angle)},${cy + val * r * Math.sin(angle)}`;
   878→  }).join(' ');
   879→  const gridLines = [0.33, 0.66, 1].map(level => {
   880→    const pts = Array.from({ length: axes }, (_, i) => {
   881→      const angle = startAngle + i * angleStep;
   882→      return `${cx + level * r * Math.cos(angle)},${cy + level * r * Math.sin(angle)}`;
   883→    }).join(' ');
   884→    return <polygon key={level} points={pts} fill="none" stroke="currentColor" strokeWidth="0.3" className="text-white/20" />;
   885→  });
   886→
   887→  return (
   888→    <svg width="56" height="56" viewBox="0 0 56 56" className="inline-block opacity-80">
   889→      {gridLines}
   890→      {Array.from({ length: axes }, (_, i) => {
   891→        const angle = startAngle + i * angleStep;
   892→        return <line key={i} x1={cx} y1={cy} x2={cx + r * Math.cos(angle)} y2={cy + r * Math.sin(angle)} stroke="currentColor" strokeWidth="0.3" className="text-white/20" />;
   893→      })}
   894→      <polygon points={points} fill="rgba(249,115,22,0.15)" stroke="#fbbf24" strokeWidth="1" />
   895→      {values.map((val, i) => {
   896→        const angle = startAngle + i * angleStep;
   897→        return <circle key={i} cx={cx + val * r * Math.cos(angle)} cy={cy + val * r * Math.sin(angle)} r="1.5" fill="#fbbf24" />;
   898→      })}
   899→    </svg>
   900→  );
   901→}

   903→// ============ RADIAL GAUGE (SEMICIRCLE) ============
   904→export function RadialGauge({ score, size = 140 }: { score: number; size?: number }) {

   933→// ============ ANIMATED SCORE NUMBER ============
   934→export function AnimatedScoreNumber({ value, delay = 0 }: { value: number; delay?: number }) {
   935→  const [displayed, setDisplayed] = useState(0);
   936→  const hasAnimated = useRef(false);
   937→
   938→  useEffect(() => {
   939→    if (hasAnimated.current) return;
   940→    hasAnimated.current = true;
   941→    const start = performance.now() + delay;
   942→    const duration = 1200;
   943→    const animate = (now: number) => {
   944→      const elapsed = now - start;
   945→      if (elapsed < 0) { requestAnimationFrame(animate); return; }
   946→      const progress = Math.min(elapsed / duration, 1);
   947→      const eased = 1 - Math.pow(1 - progress, 3);
   948→      setDisplayed(Math.round(eased * value));
   949→      if (progress < 1) requestAnimationFrame(animate);
   950→    };
   951→    requestAnimationFrame(animate);
   952→  }, [value, delay]);
   953→
   954→  return <span>{displayed}</span>;
   955→}

  1414→// ============ MICRO SPARKLINE ============
  1415→export function MicroSparkline({ data, color, width = 60, height = 20 }: { data: number[]; color?: string; width?: number; height?: number }) {
  1416→  if (!data || data.length === 0) return null;
  1417→  const maxVal = Math.max(...data);
  1418→  const minVal = Math.min(...data);
  1419→  const range = maxVal - minVal || 1;
  1420→  const points = data.map((v, i) => {
  1421→    const x = (i / (data.length - 1)) * width;
  1422→    const y = height - ((v - minVal) / range) * (height - 4) - 2;
  1423→    return `${x},${y}`;
  1424→  }).join(' ');
  1425→  const lineColor = color || (data[data.length - 1] >= data[0] ? '#f59e0b' : '#ef4444');
  1426→  return (
  1427→    <svg width={width} height={height} className="inline-block" aria-hidden="true">
  1428→      <polyline points={points} fill="none" stroke={lineColor} strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
  1429→    </svg>
  1430→  );
  1431→}
