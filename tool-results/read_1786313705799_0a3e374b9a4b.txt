    56→function relativeTime(daysAgo: number): string {
    57→  if (daysAgo === 0) return 'Today';
    58→  if (daysAgo === 1) return 'Yesterday';
    59→  if (daysAgo < 7) return `${daysAgo} days ago`;
    60→  if (daysAgo < 14) return '1 week ago';
    61→  if (daysAgo < 21) return `${Math.floor(daysAgo / 7)} weeks ago`;
    62→  if (daysAgo < 30) return '3 weeks ago';
    63→  if (daysAgo < 45) return '1 month ago';
    64→  return `${Math.floor(daysAgo / 30)} months ago`;
    65→}

    67→function generatePolicyChanges(countries: CountryData[]): PolicyChangeEntry[] {
    68→  if (!countries || countries.length === 0) return [];
    69→
    70→  const entries: PolicyChangeEntry[] = [];
    71→  const used = new Set<string>();
    72→  const timestamps = [0, 1, 2, 3, 5, 7, 10, 14, 21, 28, 35, 42, 49, 56];
    73→
    74→  // Pick visa-free countries
    75→  const vfCountries = countries.filter(c => c.visaFree);
    76→  const voaCountries = countries.filter(c => c.visaOnArrival && !c.visaFree);
    77→  const etaCountries = countries.filter(c => c.etaAvailable && !c.visaFree && !c.visaOnArrival);
    78→  const embassyCountries = countries.filter(c => !c.visaFree && !c.visaOnArrival && !c.etaAvailable);
    79→
    80→  const pick = (list: CountryData[]) => {
    81→    for (const c of list) {
    82→      if (!used.has(c.code) && entries.length < 10) return c;
    83→    }
    84→    return null;
    85→  };
    86→
    87→  // Visa-free entries (2-3)
    88→  for (let i = 0; i < 3; i++) {
    89→    const c = pick(vfCountries);
    90→    if (!c) break;
    91→    used.add(c.code);
    92→    entries.push({
    93→      id: `vf-${c.code}`, countryCode: c.code, countryName: c.name,
    94→      flagEmoji: c.flagEmoji, flagUrl: c.flagUrl,
    95→      changeType: 'visa-free',
    96→      title: 'Visa-free access confirmed',
    97→      description: `Pakistan passport holders enjoy visa-free entry to ${c.name} for up to ${c.visaTypes?.[0]?.maxDuration || '30 days'}.`,
    98→      detail: `This visa-free arrangement allows Pakistani citizens to enter ${c.name} without any prior visa application. Ensure your passport has at least 6 months validity. Best travel months: ${c.bestTravelMonths}. Average temperature: ${c.avgTempC}°C.`,
    99→      daysAgo: timestamps[entries.length] ?? 10,
   100→    });
   101→  }
   102→
   103→  // VOA entries (1-2)
   104→  for (let i = 0; i < 2; i++) {
   105→    const c = pick(voaCountries);
   106→    if (!c) break;
   107→    used.add(c.code);
   108→    entries.push({
   109→      id: `voa-${c.code}`, countryCode: c.code, countryName: c.name,
   110→      flagEmoji: c.flagEmoji, flagUrl: c.flagUrl,
   111→      changeType: 'processing',
   112→      title: 'Visa on arrival policy updated',
   113→      description: `${c.name} has streamlined its VOA process. Processing now takes approximately ${c.processingDaysMin}-${c.processingDaysMax} business days.`,
   114→      detail: `Pakistan passport holders can obtain a visa on arrival at ${c.name} ports of entry. The fee is approximately $${c.costProfile?.visaFeeUSD || 'varies'}. Maximum stay: ${c.visaTypes?.[0]?.maxDuration || '30 days'}. Extensions may be available.`,
   115→      daysAgo: timestamps[entries.length] ?? 15,
   116→    });
   117→  }
   118→
   119→  // e-Visa entries (2)
   120→  for (let i = 0; i < 2; i++) {
   121→    const c = pick(etaCountries);
   122→    if (!c) break;
   123→    used.add(c.code);
   124→    entries.push({
   125→      id: `eta-${c.code}`, countryCode: c.code, countryName: c.name,
   126→      flagEmoji: c.flagEmoji, flagUrl: c.flagUrl,
   127→      changeType: 'evisa',
   128→      title: 'e-Visa/eTA system launched',
   129→      description: `${c.name} has launched an electronic visa system for Pakistani passport holders. Apply online before travel.`,
   130→      detail: `The new e-Visa portal allows Pakistanis to apply for a ${c.name} visa entirely online. Processing time: ${c.processingDaysMin}-${c.processingDaysMax} days. Fee: $${c.costProfile?.visaFeeUSD || 'TBD'}. Required documents: valid passport, photographs, and travel itinerary.`,
   131→      daysAgo: timestamps[entries.length] ?? 20,
   132→    });
   133→  }
   134→
   135→  // Embassy/fee entries (2-3)
   136→  for (let i = 0; i < 3; i++) {
   137→    const c = pick([...embassyCountries, ...etaCountries, ...voaCountries]);
   138→    if (!c) break;
   139→    used.add(c.code);
   140→    const isFee = i % 2 === 0;
   141→    entries.push({
   142→      id: `emb-${c.code}`, countryCode: c.code, countryName: c.name,
   143→      flagEmoji: c.flagEmoji, flagUrl: c.flagUrl,
   144→      changeType: isFee ? 'fee-change' : 'requirement',
   145→      title: isFee ? 'Visa fee adjustment' : 'New document requirement added',
   146→      description: isFee
   147→        ? `${c.name} has adjusted its visa fee to $${c.costProfile?.visaFeeUSD || 'updated rate'} (previously $${Math.max(1, (c.costProfile?.visaFeeUSD || 50) - 10)}). Service fee: $${c.costProfile?.serviceFeeUSD || 'varies'}.`
   148→        : `${c.name} now requires additional documentation: ${c.requirements?.[0]?.requirement || 'updated proof of funds'} for all visa applications.`,
   149→      detail: isFee
   150→        ? `The revised fee structure for ${c.name} visas: Visa fee $${c.costProfile?.visaFeeUSD || 'TBD'}, Service fee $${c.costProfile?.serviceFeeUSD || 'TBD'}. Health insurance: $${c.costProfile?.healthInsuranceUSD || 'N/A'}. Budget estimate: $${c.costProfile?.totalMonthlyUSD || 'TBD'}/month for living expenses. `
   151→        : `Updated requirements for ${c.name}: ${c.requirements?.slice(0, 3).map(r => r.requirement).join(', ') || 'Standard documentation'}. Processing time: ${c.processingDaysMin}-${c.processingDaysMax} business days. Check official sources for the most current information.`,
   152→      daysAgo: timestamps[entries.length] ?? 25,
   153→    });
   154→  }
   155→
   156→  return entries.slice(0, 10);
   157→}

   338→interface ChecklistSection {

   346→function buildChecklistSections(country: CountryData | null, userProfile: UserProfileData | null): ChecklistSection[] {
   347→  const sections: ChecklistSection[] = [];
   348→
   349→  // Section 1: Pre-Departure
   350→  const preDepartureItems: { id: string; label: string }[] = [
   351→    { id: 'pre-passport', label: 'Check passport validity (6+ months)' },
   352→    { id: 'pre-insurance', label: 'Purchase travel insurance' },
   353→    { id: 'pre-flights', label: 'Book flights' },
   354→    { id: 'pre-accommodation', label: 'Book accommodation' },
   355→    { id: 'pre-currency', label: `Exchange currency${country ? ` (${country.currencyCode || 'local currency'})` : ''}` },
   356→    { id: 'pre-maps', label: 'Download offline maps' },
   357→    { id: 'pre-bank', label: 'Notify your bank about travel' },
   358→  ];
   359→
   360→  if (country) {
   361→    if (country.visaFree) {
   362→      preDepartureItems.unshift({ id: 'pre-vf-check', label: 'Confirm visa-free entry requirements' });
   363→    } else if (country.visaOnArrival) {
   364→      preDepartureItems.unshift({ id: 'pre-voa-docs', label: 'Prepare VOA documents (return ticket, hotel booking)' });
   365→      preDepartureItems.splice(1, 0, { id: 'pre-voa-fee', label: `Carry VOA fee in cash (~$${country.costProfile?.visaFeeUSD || 'check amount'})` });
   366→    } else if (country.etaAvailable) {
   367→      preDepartureItems.unshift({ id: 'pre-eta-apply', label: 'Apply for e-Visa/eTA online' });
   368→      preDepartureItems.splice(1, 0, { id: 'pre-eta-print', label: 'Print e-Visa approval letter' });
   369→    } else {
   370→      preDepartureItems.unshift({ id: 'pre-embassy-apply', label: 'Apply for visa at embassy/consulate' });
   371→      preDepartureItems.splice(1, 0, { id: 'pre-embassy-appt', label: 'Book embassy appointment' });
   372→      preDepartureItems.splice(2, 0, { id: 'pre-embassy-docs', label: 'Prepare and organize all documents' });
   373→    }
   374→  }
   375→
   376→  if (userProfile) {
   377→    if (userProfile.occupation && userProfile.occupation !== 'Unemployed' && userProfile.occupation !== 'Student') {
   378→      preDepartureItems.push({ id: 'pre-leave', label: 'Request leave from employer' });
   379→    }
   380→    if (userProfile.maritalStatus === 'Married' || (userProfile.dependents && userProfile.dependents > 0)) {
   381→      preDepartureItems.push({ id: 'pre-family', label: 'Arrange family care during travel' });
   382→    }
   383→    if (userProfile.occupation === 'Student') {
   384→      preDepartureItems.push({ id: 'pre-student-docs', label: 'Get student documents / NOC from institution' });
   385→    }
   386→  }
   387→
   388→  sections.push({
   389→    id: 'pre-departure',
   390→    title: `Pre-Departure Checklist${country ? ` — ${country.name}` : ''}`,
   391→    icon: <PlaneTakeoff className="h-4 w-4 text-amber-500" />,
   392→    colorClass: 'text-amber-600 dark:text-amber-400',
   393→    items: preDepartureItems,
   394→  });
   395→
   396→  // Section 2: Required Documents
   397→  const docItems: { id: string; label: string }[] = [
   398→    { id: 'doc-passport', label: 'Valid passport (6+ months validity)' },
   399→    { id: 'doc-photos', label: 'Passport-sized photographs (2x)' },
   400→    { id: 'doc-bank', label: 'Bank statements (last 6 months)' },
   401→    { id: 'doc-flights', label: 'Flight itinerary / booking confirmation' },
   402→    { id: 'doc-hotel', label: 'Hotel reservation confirmation' },
   403→    { id: 'doc-insurance', label: 'Travel insurance certificate' },
   404→    { id: 'doc-employment', label: 'Employment letter / NOC from employer' },
   405→    { id: 'doc-tax', label: 'Income tax returns' },
   406→  ];
   407→
   408→  if (country && country.requirements && country.requirements.length > 0) {
   409→    country.requirements.forEach((req, i) => {
   410→      if (i < 4 && !docItems.some(d => d.label.toLowerCase().includes(req.requirement.toLowerCase().slice(0, 15)))) {
   411→        docItems.push({ id: `doc-req-${i}`, label: req.requirement });
   412→      }
   413→    });
   414→  }
   415→
   416→  sections.push({

   633→function getVisaTypeLabel(c: CountryData): string {
   634→  if (c.visaFree) return 'Visa-Free';
   635→  if (c.visaOnArrival) return 'VOA';
   636→  if (c.etaAvailable) return 'eTA';
   637→  return 'Embassy';

   640→export const ContinentQuickStats = React.memo(function ContinentQuickStats({ countries }: { countries: CountryData[] }) {
   641→  const stats = useMemo<ContinentStat[]>(() => {
   642→    if (!countries.length) return [];
   643→
   644→    const byContinent = new Map<string, CountryData[]>();
   645→    for (const c of countries) {
   646→      const cont = c.continent || 'Other';
   647→      if (!byContinent.has(cont)) byContinent.set(cont, []);
   648→      byContinent.get(cont)!.push(c);
   649→    }
   650→
   651→    return CONTINENT_ORDER
   652→      .filter((name) => {
   653→        const list = byContinent.get(name);
   654→        return list && list.length > 0;
   655→      })
   656→      .map((name) => {
   657→        const list = byContinent.get(name)!;
   658→        const fees = list.map((c) => c.costProfile?.visaFeeUSD ?? 0).filter((f) => f > 0);
   659→        const avgFee = fees.length > 0 ? Math.round(fees.reduce((a, b) => a + b, 0) / fees.length) : 0;
   660→        const avgSafety = list.reduce((a, c) => a + (c.safetyRating || 5), 0) / list.length;
   661→
   662→        const typeCounts: Record<string, number> = {};
   663→        for (const c of list) {
   664→          const label = getVisaTypeLabel(c);
   665→          typeCounts[label] = (typeCounts[label] || 0) + 1;
   666→        }
   667→        const topVisaType = Object.entries(typeCounts).sort((a, b) => b[1] - a[1])[0]?.[0] || 'N/A';
   668→
   669→        return { name, count: list.length, avgFee, topVisaType, avgSafety: Math.round(avgSafety * 10) / 10 };
   670→      });
   671→  }, [countries]);
   672→
   673→  if (stats.length === 0) return null;

   681→          {stats.length} regions
   682→        </Badge>
   683→      </div>
   684→      <div className="flex gap-3 overflow-x-auto custom-scrollbar-thin pb-2 -mx-1 px-1">
   685→        {stats.map((s, idx) => {
   686→          const colors = CONTINENT_COLORS[s.name] || CONTINENT_COLORS['Asia'];
   687→          return (
