  1375→export function MicroSparkline({ data, color, width = 60, height = 20 }: { data: number[]; color?: string; width?: number; height?: number }) {
  1376→  if (!data || data.length === 0) return null;
  1377→  const maxVal = Math.max(...data);
  1378→  const minVal = Math.min(...data);
  1379→  const range = maxVal - minVal || 1;
  1380→  const points = data.map((v, i) => {
  1381→    const x = (i / (data.length - 1)) * width;
  1382→    const y = height - ((v - minVal) / range) * (height - 4) - 2;
  1383→    return `${x},${y}`;
  1384→  }).join(' ');
  1385→  const lineColor = color || (data[data.length - 1] >= data[0] ? '#f59e0b' : '#ef4444');
  1386→  return (
  1387→    <svg width={width} height={height} className="inline-block" aria-hidden="true">
  1388→      <polyline points={points} fill="none" stroke={lineColor} strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
  1389→    </svg>
  1390→  );
  1391→}
