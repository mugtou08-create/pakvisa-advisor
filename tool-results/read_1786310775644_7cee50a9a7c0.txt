    48→function BudgetDonutChart({ budgetResult, budgetTripDays, budgetViewMode, budgetChartActiveIndex, setBudgetChartActiveIndex, formatNum }: {
    49→  budgetResult: { visaFee: number; flightEst: number; accommodation: number; food: number; transport: number; insurance: number; misc: number; totalUSD: number; };
    50→  budgetTripDays: number;
    51→  budgetViewMode: 'total' | 'daily';
    52→  budgetChartActiveIndex: number | null;
    53→  setBudgetChartActiveIndex: (i: number | null) => void;
    54→  formatNum: (n: number) => string;
    55→}) {
    56→  const divisor = budgetViewMode === 'daily' ? budgetTripDays : 1;
    57→  const chartItems = [
    58→    { name: 'Visa Fee', value: Math.round(budgetResult.visaFee / divisor), color: CHART_COLORS[0] },
    59→    { name: 'Flights', value: Math.round(budgetResult.flightEst / divisor), color: CHART_COLORS[1] },
    60→    { name: 'Accommodation', value: Math.round(budgetResult.accommodation / divisor), color: CHART_COLORS[2] },
    61→    { name: 'Food', value: Math.round(budgetResult.food / divisor), color: CHART_COLORS[3] },
    62→    { name: 'Transport', value: Math.round(budgetResult.transport / divisor), color: CHART_COLORS[4] },
    63→    { name: 'Insurance', value: Math.round(budgetResult.insurance / divisor), color: CHART_COLORS[5] },
    64→    { name: 'Misc', value: Math.round(budgetResult.misc / divisor), color: CHART_COLORS[6] },
    65→  ].filter(d => d.value > 0);
    66→  const chartIcons = [FileText, PlaneTakeoff, Home, UtensilsCrossed, Map, Shield, MoreHorizontal];
    67→  const chartNames = ['Visa Fee', 'Flights', 'Accommodation', 'Food', 'Transport', 'Insurance', 'Misc'];
    68→  const totalVal = chartItems.reduce((s, d) => s + d.value, 0);
    69→
    70→  const StepIcon = ({ icon: StepIconProp, className }: { icon: React.ElementType; className?: string }) => React.createElement(StepIconProp, { className });
    71→
    72→  return (
    73→    <div className="p-4 rounded-lg border bg-muted/20">
    74→      <div className="flex items-center justify-between mb-3">
    75→        <div className="flex items-center gap-2">
    76→          <PieChartIcon className="w-4 h-4 text-amber-500" />
    77→          <p className="text-xs font-semibold">Budget Breakdown</p>
    78→        </div>
    79→        <div className="flex items-center gap-1 bg-muted rounded-lg p-0.5">
    80→          <button
    81→            className={`px-2 py-0.5 rounded-md text-[10px] font-medium transition-all ${budgetViewMode === 'total' ? 'bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-400' : 'text-muted-foreground'}`}
    82→            onClick={() => setBudgetViewMode('total')}
    83→          >Total</button>
    84→          <button
    85→            className={`px-2 py-0.5 rounded-md text-[10px] font-medium transition-all ${budgetViewMode === 'daily' ? 'bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-400' : 'text-muted-foreground'}`}
    86→            onClick={() => setBudgetViewMode('daily')}
    87→          >Daily</button>
    88→        </div>
    89→      </div>
    90→      <div className="flex items-center gap-4">
    91→        <div className="w-36 h-36 shrink-0">
    92→          <ResponsiveContainer width="100%" height="100%">
    93→            <PieChart>
    94→              <Pie
    95→                data={chartItems}
    96→                cx="50%"
    97→                cy="50%"
    98→                innerRadius={35}
    99→                outerRadius={58}
   100→                paddingAngle={2}
   101→                dataKey="value"
   102→                onMouseEnter={(_, index) => setBudgetChartActiveIndex(index)}
   103→                onMouseLeave={() => setBudgetChartActiveIndex(null)}
   104→                stroke="none"
   105→              >
   106→                {chartItems.map((entry, index) => (
   107→                  <Cell
   108→                    key={`cell-${index}`}
   109→                    fill={entry.color}
   110→                    opacity={budgetChartActiveIndex === null || budgetChartActiveIndex === index ? 1 : 0.4}
   111→                    stroke="none"
   112→                  />
   113→                ))}
   114→              </Pie>
   115→              <RechartsTooltip
   116→                contentStyle={{ borderRadius: '8px', border: '1px solid var(--border)', fontSize: '11px', padding: '6px 10px' }}
   117→                formatter={(value: number) => [`$${formatNum(value)}`, undefined]}
   118→              />
   119→            </PieChart>
   120→          </ResponsiveContainer>
   121→        </div>
   122→        <div className="flex-1 space-y-1.5">
   123→          {chartItems.map((item, idx) => {
   124→            const pct = Math.round((item.value / totalVal) * 100);
   125→            return (
   126→              <div
   127→                key={item.name}
   128→                className={`flex items-center gap-2 text-[11px] px-1.5 py-0.5 rounded transition-all ${budgetChartActiveIndex === idx ? 'bg-muted font-semibold' : ''}`}
   129→                onMouseEnter={() => setBudgetChartActiveIndex(idx)}
   130→                onMouseLeave={() => setBudgetChartActiveIndex(null)}
   131→              >
   132→                <div className="w-2.5 h-2.5 rounded-sm shrink-0" style={{ backgroundColor: item.color }} />
   133→                <StepIcon icon={chartIcons[idx]} className="w-3 h-3 text-muted-foreground shrink-0" />
   134→                <span className="flex-1 truncate">{chartNames[idx]}</span>
   135→                <span className="font-medium tabular-nums">${formatNum(item.value)}</span>
   136→                <span className="text-muted-foreground w-8 text-right">{pct}%</span>
   137→              </div>
   138→            );
   139→          })}
   140→        </div>
   141→      </div>
   142→    </div>
   143→  );
   144→}
