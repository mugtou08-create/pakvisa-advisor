    96→function generatePageNumbers(current: number, total: number): (number | 'ellipsis')[] {
    97→  if (total <= 7) {
    98→    return Array.from({ length: total }, (_, i) => i + 1);
    99→  }
   100→  const pages: (number | 'ellipsis')[] = [1];
   101→  if (current > 3) pages.push('ellipsis');
   102→  const start = Math.max(2, current - 1);
   103→  const end = Math.min(total - 1, current + 1);
   104→  for (let i = start; i <= end; i++) pages.push(i);
   105→  if (current < total - 2) pages.push('ellipsis');
   106→  if (total > 1) pages.push(total);
   107→  return pages;
   108→}

   110→// ============================================================
   111→// Visa Type Helpers (single source of truth for access type)
   112→// Priority: visaFree > visaOnArrival > etaAvailable > embassy
   113→// ============================================================
   114→function getPrimaryAccess(c: CountryData): string {
   115→  if (c.visaFree) return 'visa-free';
   116→  if (c.visaOnArrival) return 'visa-on-arrival';
   117→  if (c.etaAvailable) return 'e-visa';
   118→  return 'embassy';
   119→}
   120→function getVisaType(c: CountryData) {
   121→  const access = getPrimaryAccess(c);
   122→  switch (access) {
   123→    case 'visa-free':     return { label: 'Visa Free', color: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-400', dot: 'bg-emerald-500' };
   124→    case 'visa-on-arrival': return { label: 'Visa on Arrival', color: 'bg-amber-100 text-amber-700 dark:bg-amber-900/30 dark:text-amber-400', dot: 'bg-amber-500' };
   125→    case 'e-visa':       return { label: 'e-Visa', color: 'bg-sky-100 text-sky-700 dark:bg-sky-900/30 dark:text-sky-400', dot: 'bg-sky-500' };
   126→    default:             return { label: 'Embassy Required', color: 'bg-gray-100 text-gray-600 dark:bg-gray-800 dark:text-gray-400', dot: 'bg-gray-400' };
   127→  }
   128→}
