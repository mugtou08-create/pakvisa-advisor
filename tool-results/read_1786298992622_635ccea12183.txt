   174→// ============ SCORE HISTORY CHART (SVG) ============
   175→export function ScoreHistoryChart() {
   176→  // eslint-disable-next-line react-hooks/preserve-manual-memoization
   177→  const history = useMemo(() => {
   178→    try {
   179→      const raw = localStorage.getItem(SCORING_HISTORY_KEY);
   180→      if (raw) {
   181→        const entries = JSON.parse(raw);
   182→        return entries.slice(-8).map((e: any) => ({
   183→          date: new Date(e.date).toLocaleDateString('en-US', { month: 'short', day: 'numeric' }),
   184→          topScore: e.topScore || (e.scores && e.scores[0] ? e.scores[0].finalScore : 0),
   185→          countryCount: e.countryCount || (e.scores ? e.scores.length : 0),
   186→        }));
   187→      }
   188→    } catch {}
   189→    return [];
   190→  }, []);
   191→  
   192→  if (history.length < 2) return null;
   193→  
   194→  const chartW = 400, chartH = 120, pad = 30;
   195→  const maxScore = Math.max(...history.map(e => e.topScore), 1);
   196→  const pts = history.map((e, i) => ({
   197→    x: pad + (i / (history.length - 1)) * (chartW - 2 * pad),
   198→    y: chartH - pad - (e.topScore / maxScore) * (chartH - 2 * pad),
   199→  }));
   200→  
   201→  return (
   202→    <Card className="p-4">
   203→      <h4 className="text-sm font-semibold mb-3 flex items-center gap-1.5">
   204→        <TrendingUp className="w-4 h-4 text-amber-500" />
   205→        Score Trend
   206→      </h4>
   207→      <svg viewBox={`0 0 ${chartW} ${chartH}`} className="w-full">
   208→        {/* Grid lines */}
   209→        {[0, 0.5, 1].map(v => (
   210→          <line key={v} x1={pad} y1={chartH - pad - v * (chartH - 2 * pad)} x2={chartW - pad} y2={chartH - pad - v * (chartH - 2 * pad)} stroke="var(--border)" strokeWidth="0.5" strokeDasharray="4 4" />
   211→        ))}
   212→        {/* Y-axis labels */}
   213→        <text x={pad - 5} y={chartH - pad + 4} textAnchor="end" fill="var(--muted-foreground)" fontSize="9">0</text>
   214→        <text x={pad - 5} y={pad + 4} textAnchor="end" fill="var(--muted-foreground)" fontSize="9">{maxScore}</text>
   215→        {/* Area fill */}
   216→        <path d={`M ${pts[0].x} ${chartH - pad} L ${pts.map(p => `${p.x} ${p.y}`).join(' L')} L ${pts[pts.length - 1].x} ${chartH - pad} Z`} fill="url(#chartGradient)" opacity="0.3" />
   217→        {/* Line */}
   218→        <polyline points={pts.map(p => `${p.x},${p.y}`).join(' ')} fill="none" stroke="#f59e0b" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" />
   219→        {/* Data points */}
   220→        {pts.map((p, i) => (
   221→          <g key={i}>
   222→            <circle cx={p.x} cy={p.y} r="4" fill="white" stroke="#f59e0b" strokeWidth="2" />
   223→            <text x={p.x} y={chartH - pad + 14} textAnchor="middle" fill="var(--muted-foreground)" fontSize="8">{history[i].date}</text>
   224→          </g>
   225→        ))}
   226→        <defs>
   227→          <linearGradient id="chartGradient" x1="0" y1="0" x2="0" y2="1">
   228→            <stop offset="0%" stopColor="#f59e0b" />
   229→            <stop offset="100%" stopColor="#f59e0b" stopOpacity="0" />
   230→          </linearGradient>
   231→        </defs>
   232→      </svg>
   233→    </Card>
   234→  );
   235→}

   317→// ============ VISA STATISTICS DASHBOARD WIDGET ============
   318→export function VisaStatsDashboard({ countries, stats }: { countries: CountryData[]; stats: { totalCountries: number; visaFreeCount: number; visaOnArrivalCount: number; eVisaCount: number; embassyRequiredCount: number; avgCostUSD?: number; cheapestCountry?: { name: string; code: string; flagEmoji: string; visaFeeUSD: number }; fastestProcessing?: { name: string; code: string; flagEmoji: string; processingDaysMin: number } } | null }) {
   319→  const [visible, setVisible] = useState(false);
   320→  const ref = useRef<HTMLDivElement>(null);
   321→
   322→  useEffect(() => {
   323→    const el = ref.current;
   324→    if (!el) return;
   325→    const observer = new IntersectionObserver(
   326→      ([entry]) => { if (entry.isIntersecting) { setVisible(true); observer.disconnect(); } },
   327→      { threshold: 0.1 }
   328→    );
   329→    observer.observe(el);
   330→    // Trigger initial check in observer callback
   331→    observer.takeRecords().forEach(entry => {
   332→      if (entry.isIntersecting) { setVisible(true); observer.disconnect(); }
   333→    });
   334→    return () => observer.disconnect();
   335→  }, [countries.length]);
   336→
   337→  const cheapest = stats?.cheapestCountry;
   338→  const fastest = stats?.fastestProcessing;
   339→  const avgFee = stats?.avgCostUSD ? Math.round(stats.avgCostUSD) : Math.round(countries.reduce((sum, c) => sum + (c.costProfile?.visaFeeUSD || 0), 0) / (countries.length || 1));
   340→  const highestSafety = useMemo(() => {
   341→    return countries.reduce((best, c) => c.safetyRating > (best?.safetyRating || 0) ? c : best, countries[0]);
   342→  }, [countries]);
   343→
   344→  const statCards = [
   345→    { label: 'Total Countries', value: visible ? String(countries.length) : '0', icon: Globe, color: 'text-amber-600 dark:text-amber-400', bg: 'bg-amber-50 dark:bg-amber-900/20' },

   348→    { label: 'Cheapest Destination', value: visible ? (cheapest ? `$${cheapest.visaFeeUSD}` : '—') : '$0', sub: cheapest?.name, icon: Wallet, color: 'text-purple-600 dark:text-purple-400', bg: 'bg-purple-50 dark:bg-purple-900/20' },
   349→    { label: 'Most Popular', value: 'Turkey', sub: 'Based on searches', icon: Flame, color: 'text-orange-600 dark:text-orange-400', bg: 'bg-orange-50 dark:bg-orange-900/20' },
   350→    { label: 'Highest Safety', value: visible ? `${highestSafety?.safetyRating || 0}/10` : '—', sub: highestSafety?.name, icon: Shield, color: 'text-orange-600 dark:text-orange-400', bg: 'bg-amber-50 dark:bg-orange-900/20' },
   351→  ];
   352→
   353→  return (
   354→    <div ref={ref}>
   355→      <div className="scroll-section-title">
   356→        <h3 className="text-sm font-semibold flex items-center gap-2 whitespace-nowrap">
   357→          <BarChart3 className="w-4 h-4 text-amber-500" />
   358→          Visa Statistics Overview
   359→        </h3>
   360→      </div>
   361→      <div className="flex gap-3 overflow-x-auto pb-2 -mx-4 px-4 filter-scroll">
   362→        {statCards.map((card) => (
   363→          <div key={card.label} className={`mini-stat-hover shrink-0 w-44 p-4 rounded-xl border bg-background ${card.bg}`}>
   364→            <div className="flex items-center justify-between mb-2">
   365→              <card.icon className={`w-4 h-4 ${card.color}`} />
   366→            </div>
   367→            <p className={`text-lg font-bold ${card.color}`}>{card.value}</p>
   368→            <p className="text-[11px] text-muted-foreground mt-0.5">{card.label}</p>
   369→            {card.sub && <p className="text-[10px] text-muted-foreground/70 mt-0.5">{card.sub}</p>}
   370→          </div>
   371→        ))}
   372→      </div>
   373→    </div>
   374→  );
   375→}

   384→    return () => clearInterval(interval);
   385→  }, []);
   386→
   387→  const target = targetTravelDate ? new Date(targetTravelDate) : null;
   388→  const diff = target ? Math.max(0, Math.ceil((target.getTime() - now.getTime()) / 86400000)) : null;
   389→
   390→  return (
   391→    <div className="glass-section p-4 rounded-xl">
   392→      <h3 className="text-sm font-semibold flex items-center gap-2 mb-3">
   393→        <CalendarClock className="w-4 h-4 text-amber-500" />
   394→        Travel Countdown
   395→      </h3>
   396→      {diff !== null && target ? (
   397→        <div className="text-center">
   398→          <div className={`text-4xl font-extrabold text-amber-600 dark:text-amber-400 tabular-nums ${diff < 30 && diff > 0 ? 'countdown-urgent' : ''}`}>
   399→            {diff}
   400→          </div>
   401→          <p className="text-sm text-muted-foreground mt-1">days until departure</p>
   402→          <div className="grid grid-cols-3 gap-2 mt-3">
   403→            {[
   404→              { label: 'Weeks', value: Math.floor(diff / 7) },
   405→              { label: 'Months', value: Math.round(diff / 30) },
   406→              { label: 'Target', value: target.toLocaleDateString('en-PK', { month: 'short', day: 'numeric' }) },
   407→            ].map(item => (
   408→              <div key={item.label} className="rounded-lg bg-muted/50 p-2 text-center">
   409→                <div className="text-sm font-bold">{item.value}</div>
   410→                <div className="text-[10px] text-muted-foreground">{item.label}</div>
   411→              </div>
   412→            ))}
   413→          </div>
   414→          {diff < 30 && diff > 0 && (
   415→            <Badge className="mt-2 bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-400" variant="secondary">
   416→              <AlarmClock className="w-3 h-3 mr-1" /> Visa application recommended soon
   417→            </Badge>
   418→          )}
   419→          {diff === 0 && (
   420→            <Badge className="mt-2 bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-400" variant="secondary">
   421→              <PlaneTakeoff className="w-3 h-3 mr-1" /> Travel day! Have a safe trip!
   422→            </Badge>
   423→          )}
   424→          <button
   425→            onClick={() => setTargetTravelDate('')}
   426→            className="text-[10px] text-muted-foreground hover:text-foreground mt-2 underline"
   427→          >
   428→            Change date
   429→          </button>
   430→        </div>
   431→      ) : (
   432→        <div className="text-center">
   433→          <p className="text-sm text-muted-foreground mb-3">Set your target travel date to see countdown</p>
   434→          <Input
   435→            type="date"
   436→            className="input-glow-focus mx-auto w-auto"
   437→            onChange={(e) => setTargetTravelDate(e.target.value)}
   438→            min={new Date().toISOString().split('T')[0]}
   439→          />
   440→        </div>
   441→      )}
   442→    </div>
   443→  );
   444→}
