   790→// ============ TYPING TEXT ANIMATION ============
   791→export function TypingText({ phrases, className = '' }: { phrases: string[]; className?: string }) {
   792→  const [current, setCurrent] = useState(0);
   793→  const [text, setText] = useState('');
   794→  const [isDeleting, setIsDeleting] = useState(false);
   795→
   796→  useEffect(() => {
   797→    const phrase = phrases[current];
   798→    const timeout = setTimeout(() => {
   799→      if (!isDeleting) {
   800→        setText(phrase.substring(0, text.length + 1));
   801→        if (text.length === phrase.length) {
   802→          setTimeout(() => setIsDeleting(true), 2200);
   803→        }
   804→      } else {
   805→        setText(phrase.substring(0, text.length - 1));
   806→        if (text.length === 0) {
   807→          setIsDeleting(false);
   808→          setCurrent((current + 1) % phrases.length);
   809→        }
   810→      }
   811→    }, isDeleting ? 25 : 55);
   812→    return () => clearTimeout(timeout);
   813→  }, [text, isDeleting, current, phrases]);
   814→
   815→  return (
   816→    <span className={className}>
   817→      {text}
   818→      <span className="inline-block w-0 h-[1.1em] ml-0.5 align-middle typing-text-cursor" />
   819→    </span>
   820→  );
   821→}

   873→// ============ MINI RADAR CHART (3-axis for country cards) ============
   874→export function MiniRadarChart({ safety, cost, ease }: { safety: number; cost: number; ease: number }) {
   875→  const cx = 28, cy = 28, r = 22;
   876→  const axes = 3;
   877→  const angleStep = (2 * Math.PI) / axes;
   878→  const startAngle = -Math.PI / 2;
   879→  const s = safety / 10;
   880→  const c = Math.max(0.05, 1 - cost / 3000);
   881→  const e = ease;
   882→  const values = [s, c, e];
   883→  const points = values.map((val, i) => {
   884→    const angle = startAngle + i * angleStep;
   885→    return `${cx + val * r * Math.cos(angle)},${cy + val * r * Math.sin(angle)}`;
   886→  }).join(' ');
   887→  const gridLines = [0.33, 0.66, 1].map(level => {
   888→    const pts = Array.from({ length: axes }, (_, i) => {
   889→      const angle = startAngle + i * angleStep;
   890→      return `${cx + level * r * Math.cos(angle)},${cy + level * r * Math.sin(angle)}`;
   891→    }).join(' ');
   892→    return <polygon key={level} points={pts} fill="none" stroke="currentColor" strokeWidth="0.3" className="text-white/20" />;
   893→  });
   894→
   895→  return (
   896→    <svg width="56" height="56" viewBox="0 0 56 56" className="inline-block opacity-80">
   897→      {gridLines}
   898→      {Array.from({ length: axes }, (_, i) => {
   899→        const angle = startAngle + i * angleStep;
   900→        return <line key={i} x1={cx} y1={cy} x2={cx + r * Math.cos(angle)} y2={cy + r * Math.sin(angle)} stroke="currentColor" strokeWidth="0.3" className="text-white/20" />;
   901→      })}
   902→      <polygon points={points} fill="rgba(249,115,22,0.15)" stroke="#fbbf24" strokeWidth="1" />
   903→      {values.map((val, i) => {
   904→        const angle = startAngle + i * angleStep;
   905→        return <circle key={i} cx={cx + val * r * Math.cos(angle)} cy={cy + val * r * Math.sin(angle)} r="1.5" fill="#fbbf24" />;
   906→      })}

   941→// ============ ANIMATED SCORE NUMBER ============
   942→export function AnimatedScoreNumber({ value, delay = 0 }: { value: number; delay?: number }) {
   943→  const [displayed, setDisplayed] = useState(0);
   944→  const hasAnimated = useRef(false);
   945→
   946→  useEffect(() => {
   947→    if (hasAnimated.current) return;
   948→    hasAnimated.current = true;
   949→    const start = performance.now() + delay;
   950→    const duration = 1200;
   951→    const animate = (now: number) => {
   952→      const elapsed = now - start;
   953→      if (elapsed < 0) { requestAnimationFrame(animate); return; }
   954→      const progress = Math.min(elapsed / duration, 1);
   955→      const eased = 1 - Math.pow(1 - progress, 3);
   956→      setDisplayed(Math.round(eased * value));
   957→      if (progress < 1) requestAnimationFrame(animate);
   958→    };
   959→    requestAnimationFrame(animate);
   960→  }, [value, delay]);
   961→
   962→  return <span>{displayed}</span>;
   963→}

  1422→// ============ MICRO SPARKLINE ============
  1423→export function MicroSparkline({ data, color, width = 60, height = 20 }: { data: number[]; color?: string; width?: number; height?: number }) {
  1424→  if (!data || data.length === 0) return null;
  1425→  const maxVal = Math.max(...data);
  1426→  const minVal = Math.min(...data);
  1427→  const range = maxVal - minVal || 1;
  1428→  const points = data.map((v, i) => {
  1429→    const x = (i / (data.length - 1)) * width;
  1430→    const y = height - ((v - minVal) / range) * (height - 4) - 2;
  1431→    return `${x},${y}`;
  1432→  }).join(' ');
  1433→  const lineColor = color || (data[data.length - 1] >= data[0] ? '#f59e0b' : '#ef4444');
  1434→  return (
  1435→    <svg width={width} height={height} className="inline-block" aria-hidden="true">
  1436→      <polyline points={points} fill="none" stroke={lineColor} strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
  1437→    </svg>
  1438→  );
  1439→}
