   731→// ============ TYPING TEXT ANIMATION ============
   732→export function TypingText({ phrases, className = '' }: { phrases: string[]; className?: string }) {
   733→  const [current, setCurrent] = useState(0);
   734→  const [text, setText] = useState('');
   735→  const [isDeleting, setIsDeleting] = useState(false);
   736→
   737→  useEffect(() => {
   738→    const phrase = phrases[current];
   739→    const timeout = setTimeout(() => {
   740→      if (!isDeleting) {
   741→        setText(phrase.substring(0, text.length + 1));
   742→        if (text.length === phrase.length) {
   743→          setTimeout(() => setIsDeleting(true), 2200);
   744→        }
   745→      } else {
   746→        setText(phrase.substring(0, text.length - 1));
   747→        if (text.length === 0) {
   748→          setIsDeleting(false);
   749→          setCurrent((current + 1) % phrases.length);
   750→        }
   751→      }
   752→    }, isDeleting ? 25 : 55);
   753→    return () => clearTimeout(timeout);
   754→  }, [text, isDeleting, current, phrases]);
   755→
   756→  return (
   757→    <span className={className}>
   758→      {text}
   759→      <span className="inline-block w-0 h-[1.1em] ml-0.5 align-middle typing-text-cursor" />

   814→// ============ MINI RADAR CHART (3-axis for country cards) ============
   815→export function MiniRadarChart({ safety, cost, ease }: { safety: number; cost: number; ease: number }) {
   816→  const cx = 28, cy = 28, r = 22;
   817→  const axes = 3;
   818→  const angleStep = (2 * Math.PI) / axes;
   819→  const startAngle = -Math.PI / 2;
   820→  const s = safety / 10;
   821→  const c = Math.max(0.05, 1 - cost / 3000);
   822→  const e = ease;
   823→  const values = [s, c, e];
   824→  const points = values.map((val, i) => {
   825→    const angle = startAngle + i * angleStep;
   826→    return `${cx + val * r * Math.cos(angle)},${cy + val * r * Math.sin(angle)}`;
   827→  }).join(' ');
   828→  const gridLines = [0.33, 0.66, 1].map(level => {
   829→    const pts = Array.from({ length: axes }, (_, i) => {
   830→      const angle = startAngle + i * angleStep;
   831→      return `${cx + level * r * Math.cos(angle)},${cy + level * r * Math.sin(angle)}`;
   832→    }).join(' ');
   833→    return <polygon key={level} points={pts} fill="none" stroke="currentColor" strokeWidth="0.3" className="text-white/20" />;
   834→  });
   835→
   836→  return (
   837→    <svg width="56" height="56" viewBox="0 0 56 56" className="inline-block opacity-80">
   838→      {gridLines}
   839→      {Array.from({ length: axes }, (_, i) => {
   840→        const angle = startAngle + i * angleStep;
   841→        return <line key={i} x1={cx} y1={cy} x2={cx + r * Math.cos(angle)} y2={cy + r * Math.sin(angle)} stroke="currentColor" strokeWidth="0.3" className="text-white/20" />;
   842→      })}
   843→      <polygon points={points} fill="rgba(249,115,22,0.15)" stroke="#fbbf24" strokeWidth="1" />
   844→      {values.map((val, i) => {
   845→        const angle = startAngle + i * angleStep;
   846→        return <circle key={i} cx={cx + val * r * Math.cos(angle)} cy={cy + val * r * Math.sin(angle)} r="1.5" fill="#fbbf24" />;
   847→      })}
   848→    </svg>
   849→  );
   850→}

   893→      const elapsed = now - start;
   894→      if (elapsed < 0) { requestAnimationFrame(animate); return; }
   895→      const progress = Math.min(elapsed / duration, 1);
   896→      const eased = 1 - Math.pow(1 - progress, 3);
   897→      setDisplayed(Math.round(eased * value));
   898→      if (progress < 1) requestAnimationFrame(animate);
   899→    };
   900→    requestAnimationFrame(animate);
   901→  }, [value, delay]);
   902→
   903→  return <span>{displayed}</span>;
   904→}

  1364→export function MicroSparkline({ data, color, width = 60, height = 20 }: { data: number[]; color?: string; width?: number; height?: number }) {

  1373→  }).join(' ');
  1374→  const lineColor = color || (data[data.length - 1] >= data[0] ? '#f59e0b' : '#ef4444');
  1375→  return (
  1376→    <svg width={width} height={height} className="inline-block" aria-hidden="true">
  1377→      <polyline points={points} fill="none" stroke={lineColor} strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
