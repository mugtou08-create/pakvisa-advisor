    94→function generatePageNumbers(current: number, total: number): (number | 'ellipsis')[] {
    95→  if (total <= 7) {
    96→    return Array.from({ length: total }, (_, i) => i + 1);
    97→  }
    98→  const pages: (number | 'ellipsis')[] = [1];
    99→  if (current > 3) pages.push('ellipsis');
   100→  const start = Math.max(2, current - 1);
   101→  const end = Math.min(total - 1, current + 1);
   102→  for (let i = start; i <= end; i++) pages.push(i);
   103→  if (current < total - 2) pages.push('ellipsis');
   104→  if (total > 1) pages.push(total);
   105→  return pages;
   106→}

   108→// ============================================================
   109→// Visa Type Helpers (single source of truth for access type)
   110→// Priority: visaFree > visaOnArrival > etaAvailable > embassy
   111→// ============================================================
   112→function getPrimaryAccess(c: CountryData): string {
   113→  if (c.visaFree) return 'visa-free';
   114→  if (c.visaOnArrival) return 'visa-on-arrival';
   115→  if (c.etaAvailable) return 'e-visa';
   116→  return 'embassy';
   117→}
   118→function getVisaType(c: CountryData) {
   119→  const access = getPrimaryAccess(c);
   120→  switch (access) {
   121→    case 'visa-free':     return { label: 'Visa Free', color: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-400', dot: 'bg-emerald-500' };
   122→    case 'visa-on-arrival': return { label: 'Visa on Arrival', color: 'bg-amber-100 text-amber-700 dark:bg-amber-900/30 dark:text-amber-400', dot: 'bg-amber-500' };
   123→    case 'e-visa':       return { label: 'e-Visa', color: 'bg-sky-100 text-sky-700 dark:bg-sky-900/30 dark:text-sky-400', dot: 'bg-sky-500' };
   124→    default:             return { label: 'Embassy Required', color: 'bg-gray-100 text-gray-600 dark:bg-gray-800 dark:text-gray-400', dot: 'bg-gray-400' };
   125→  }
   126→}
