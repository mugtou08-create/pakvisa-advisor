    94→function getVisaLabel(country: { visaFree: boolean; visaOnArrival: boolean; etaAvailable: boolean }): string {
    95→  if (country.visaFree) return 'Visa Free';

   101→function getVisaBadgeClass(country: { visaFree: boolean; visaOnArrival: boolean; etaAvailable: boolean }): string {
   102→  if (country.visaFree) return 'bg-emerald-100 text-emerald-700';
   103→  if (country.visaOnArrival) return 'bg-amber-100 text-amber-700';

   115→function usdToPkr(usd: number): string {
   116→  return Math.round(usd * 278.5).toLocaleString();

   119→function safetyStars(rating: number): string {
   120→  const clamped = Math.min(Math.max(rating, 0), 5);
   121→  return '\u2605'.repeat(clamped) + '\u2606'.repeat(5 - clamped);
   122→}

   774→function generateFAQs(
   775→  country: {
   776→    name: string; code: string; visaFree: boolean; visaOnArrival: boolean; etaAvailable: boolean;
   777→    processingDaysMin: number; processingDaysMax: number; safetyRating: number; safetySummary: string;
   778→    bestTravelMonths: string; avgTempC: string; currency: string; currencyCode: string;
   779→  },
   780→  costProfile: CostProfileData | undefined,
   781→  visaLabel: string,
   782→  isEmbassyRequired: boolean,
   783→): { q: string; a: string }[] {
   784→  const faqs: { q: string; a: string }[] = [];
   785→
   786→  // Q1: Do Pakistani citizens need a visa?
   787→  faqs.push({
   788→    q: `Do Pakistani citizens need a visa for ${country.name}?`,
   789→    a: country.visaFree
   790→      ? `No, Pakistani citizens do not need a visa to visit ${country.name}. You can travel visa-free with a valid Pakistani passport. Make sure your passport has at least 6 months of validity remaining.`
   791→      : country.visaOnArrival
   792→        ? `Pakistani citizens can get a visa on arrival at ${country.name}. Simply present your valid Pakistani passport at the immigration counter upon arrival. Processing is usually instant.`
   793→        : country.etaAvailable
   794→          ? `Yes, Pakistani citizens need an e-Visa to visit ${country.name}. The e-Visa can be applied for online through the official government portal. Processing typically takes ${country.processingDaysMin}-${country.processingDaysMax} business days.`
   795→          : `Yes, Pakistani citizens need a visa to visit ${country.name}. You must apply at the embassy or consulate. The application process typically takes ${country.processingDaysMin}-${country.processingDaysMax} business days.`,
   796→  });
   797→
   798→  // Q2: What is the visa fee?
   799→  const feeText = costProfile && costProfile.visaFeeUSD > 0
   800→    ? `The visa fee for ${country.name} is approximately $${costProfile.visaFeeUSD} USD (\u2248 PKR ${usdToPkr(costProfile.visaFeeUSD)}). Additional service fees of $${costProfile.serviceFeeUSD} may apply.`
   801→    : `There is no visa fee for Pakistani citizens visiting ${country.name} as it is ${visaLabel.toLowerCase()}.`
   802→  ;
   803→  faqs.push({
   804→    q: `What is the visa fee for ${country.name} for Pakistani citizens?`,
   805→    a: feeText,
   806→  });
   807→
   808→  // Q3: Processing time
   809→  faqs.push({
   810→    q: `How long does it take to get a ${country.name} visa?`,
   811→    a: country.visaFree
   812→      ? `No processing time is needed as ${country.name} offers visa-free entry to Pakistani passport holders. You can travel immediately.`
   813→      : country.visaOnArrival
   814→        ? `The visa on arrival for ${country.name} is processed instantly at the airport or border crossing. The entire process typically takes 5-15 minutes.`
   815→        : `${country.name} visa processing typically takes ${country.processingDaysMin}-${country.processingDaysMax} business days for Pakistani citizens. It is recommended to apply at least 2-3 weeks before your planned travel date.`,
   816→  });
   817→
   818→  // Q4: Safety
   819→  faqs.push({
   820→    q: `Is ${country.name} safe for Pakistani tourists?`,
   821→    a: country.safetySummary
   822→      ? `${country.name} has a safety rating of ${Math.min(country.safetyRating, 5)}/5. ${country.safetySummary}`
   823→      : `${country.name} has a safety rating of ${Math.min(country.safetyRating, 5)}/5. As with any travel destination, exercise standard precautions, keep your belongings secure, and follow local laws and customs.`,
   824→  });
   825→
   826→  // Q5: Best time to visit or documents required
   827→  if (country.bestTravelMonths) {
   828→    faqs.push({
   829→      q: `What is the best time to visit ${country.name} from Pakistan?`,
   830→      a: `The best months to visit ${country.name} are ${country.bestTravelMonths}. The average temperature is around ${country.avgTempC}\u00B0C. The local currency is ${country.currency} (${country.currencyCode}). Plan your trip during these months for the most comfortable weather conditions.`,
   831→    });
   832→  } else {
   833→    faqs.push({
   834→      q: `What documents do I need for a ${country.name} visa?`,
   835→      a: `Common requirements for a ${country.name} visa include: a valid Pakistani passport with 6+ months validity, passport-sized photographs, proof of accommodation, return flight ticket, travel insurance, and proof of sufficient funds. For the complete and most up-to-date list, check the official embassy website or use PakVisa Advisor.`,
   836→    });
   837→  }
   838→
   839→  return faqs;
   840→}
