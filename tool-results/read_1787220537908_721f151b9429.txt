   145→        headers: { 'Content-Type': 'application/json' },

   192→        setWhatsappNumber(data.data.settings?.whatsapp_number || '');

   195→        setAnalyticsError(data.error || 'Failed to load analytics');

   199→      if (err instanceof DOMException && err.name === 'AbortError') {
   200→        setAnalyticsError('Request timed out. Please try again.');

   202→        setAnalyticsError('Network error. Please try again.');
   203→        toast.error('Failed to fetch analytics');

   217→        headers: { 'Authorization': `Bearer ${token}` },
   218→      });

   222→        setMessagesTotal(data.data.total);
   223→        setMessagesUnread(data.data.unreadCount);
   224→        setMessagesPage(data.data.page);
   225→      }

   279→    if (!token) return;

   281→    try {
   282→      const res = await fetch('/api/admin/settings', {
   283→        method: 'PUT',

   285→        body: JSON.stringify({ key: 'maintenance_mode', value: String(enabled) }),

   324→    if (!token) return;
   325→    setMarkingAllRead(true);

   327→      const res = await fetch('/api/admin/messages', {
   328→        method: 'PUT',

   419→    link.click();

   476→        </div>
   477→        <div className="flex items-center gap-2">
